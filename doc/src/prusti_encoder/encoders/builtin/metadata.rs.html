13/// The (uninterpreted) function transferring the *referent value* of a `&mut`
14/// across an unsize coercion: `value_cast(old_referent_snap, from_ty, to_ty)` is
15/// the new referent snapshot. Like `metadata_cast`, it is only defined by
16/// axioms; the array->slice axioms are added (for the concrete element types) in
17/// `MirBuiltinUseCastEnc`. The identity axiom handles degenerate `T -> T`
18/// reborrows; every other (e.g. `Struct -> dyn`) coercion is left unconstrained.
19pub struct ValueCastEnc;
20
21impl TaskEncoder for ValueCastEnc {
22    task_encoder::encoder_cache!(ValueCastEnc);
23    const ENCODER_NAME: &'static str = "unsize value cast encoder";
24    type TaskDescription<'vir> = ();
25    type OutputFullDependency<'vir> =
26        FunctionIdn<'vir, (vir::PSnap, vir::TyVal, vir::TyVal), vir::PSnap>;
27    type OutputFullLocal<'vir> = vir::Domain<'vir>;
28
29    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
30        *task
31    }
32
33    fn do_encode_full<'vir>(
34        task_key: &Self::TaskKey<'vir>,
35        deps: &mut TaskEncoderDependencies<'vir, Self>,
36    ) -> EncodeFullResult<'vir, Self> {
37        deps.emit_output_ref(*task_key, ())?;
38        let idn = vir::ViperIdent::new("unsize_value_cast");
39        let fn_idn = vir::FunctionIdn::new(
40            idn,
41            (vir::TYPE_PSNAP, vir::TYPE_TYVAL, vir::TYPE_TYVAL),
42            vir::TYPE_PSNAP,
43        );
44        vir::with_vcx(|vcx| {
45            let domain_fn = vcx.mk_domain_function(fn_idn, false, None);
46            // forall input: s_Param, same: Type :: { value_cast(input, same, same) }
47            //     value_cast(input, same, same) == input
48            let expr = vir::expr! {
49                forall input: PSnap, same: Type :: { [fn_idn](input, same, same) }
50                    ([fn_idn](input, same, same)) == (input)
51            };
52            let axiom = vcx.mk_domain_axiom(vir::ViperIdent::new("unsize_value_cast_eq"), expr);
53            let domain = vcx.mk_domain(
54                vir::ViperIdent::new("UnsizeValueCast"),
55                &[],
56                vcx.alloc_slice(&[axiom]),
57                vcx.alloc_slice(&[domain_fn]),
58                None,
59            );
60            Ok((domain, fn_idn))
61        })
62    }
63
64    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
65        match Self::all_outputs_local(program).as_slice() {
66            [] => (),
67            [domain] => program.add_domain(domain),
68            _ => unreachable!(),
69        }
70    }
71}

73/// The (uninterpreted) function giving a pointer's metadata after a cast from
74/// a pointer to `from_ty` to a pointer to `to_ty`:
75/// `metadata_cast(metadata, from_ty, to_ty)`. It is only defined by axioms:
76/// the ones here, which hold for every pointer cast, and those for particular
77/// types in `MetadataCastAxiomEnc`. Like rustc's `ptr_to_ptr`
78/// (`rustc_const_eval/src/interpret/cast.rs`), a cast keeps the metadata
79/// between pointees with the same metadata type and replaces it with `()` in a
80/// cast to a thin pointer; rustc accepts no other raw-pointer casts (all trait
81/// objects are one Viper type). An unsizing coercion from a sized type changes
82/// the metadata type, so neither axiom applies to it. A coercion between trait
83/// objects falls under `metadata_cast_eq`: it keeps the vtable, also for an
84/// upcast, where rustc replaces it.
85pub struct MetadataCastEnc;
86
87impl TaskEncoder for MetadataCastEnc {
88    task_encoder::encoder_cache!(MetadataCastEnc);
89    const ENCODER_NAME: &'static str = "metadata cast encoder";
90    type TaskDescription<'vir> = ();
91    type OutputFullDependency<'vir> =
92        FunctionIdn<'vir, (vir::PSnap, vir::TyVal, vir::TyVal), vir::PSnap>;
93    type OutputFullLocal<'vir> = vir::Domain<'vir>;
94
95    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
96        *task
97    }
98
99    fn do_encode_full<'vir>(
100        task_key: &Self::TaskKey<'vir>,
101        deps: &mut TaskEncoderDependencies<'vir, Self>,
102    ) -> EncodeFullResult<'vir, Self> {
103        deps.emit_output_ref(*task_key, ())?;
104        let idn = vir::ViperIdent::new("metadata_cast");
105        let fn_idn = vir::FunctionIdn::new(
106            idn,
107            (vir::TYPE_PSNAP, vir::TYPE_TYVAL, vir::TYPE_TYVAL),
108            vir::TYPE_PSNAP,
109        );
110        // The type `()` and its value as an `s_Param`, the snapshot type in
111        // which pointers hold their metadata.
112        let unit = vir::with_vcx(|vcx| {
113            RustTyDecomposition::from_ty(vcx.tcx().types.unit, GParams::empty())
114        });
115        let unit_ty = deps.require_dep::<TyExprEnc>(unit)?;
116        let unit_snap = deps
117            .require_dep::<TyUsePureEnc>(unit)?
118            .zst_to_snap()
119            .unwrap();
120        let unit_caster = deps.require_dep::<GArgsCastEnc<Pure>>(Some(RustTyNormalized {
121            param: RustTyDecomposition::param(),
122            concrete: unit,
123        }))?;
124        let unit_metadata = unit_caster
125            .cast_to_callee_ctx(unit_snap.upcast_ty())
126            .downcast_ty::<vir::PSnap>();
127        let (_, pointee_metadata) = pointee_metadata(deps)?;
128        vir::with_vcx(|vcx| {
129            let domain_fn = vcx.mk_domain_function(fn_idn, false, None);
130            let input_decl = vcx.mk_local_decl("input", vir::TYPE_PSNAP);
131            let input = vcx.mk_local_ex(input_decl);
132            let from_decl = vcx.mk_local_decl("from", vir::TYPE_TYVAL);
133            let from = vcx.mk_local_ex(from_decl);
134            let to_decl = vcx.mk_local_decl("to", vir::TYPE_TYVAL);
135            let to = vcx.mk_local_ex(to_decl);
136            let call = fn_idn.call()(input, from, to);
137            let from_metadata = pointee_metadata(&[from], &[]);
138            let to_metadata = pointee_metadata(&[to], &[]);
139
140            // The two conditions are disjoint: both conclusions together would
141            // equate `input` with `()` for any `input`.
142            let eq = vir::expr! {
143                forall [input_decl], [from_decl], [to_decl] :: { [call] }
144                    ((from_metadata) == (to_metadata)) ==> ((call) == (input))
145            };
146            let axiom_eq = vcx.mk_domain_axiom(vir::ViperIdent::new("metadata_cast_eq"), eq);
147            let thin = vir::expr! {
148                forall [input_decl], [from_decl], [to_decl] :: { [call] }
149                    ((!((from_metadata) == (unit_ty))) && ((to_metadata) == (unit_ty)))
150                        ==> ((call) == (unit_metadata))
151            };
152            let axiom_thin = vcx.mk_domain_axiom(vir::ViperIdent::new("metadata_cast_thin"), thin);
153
154            let domain = vcx.mk_domain(
155                vir::ViperIdent::new("MetadataCast"),
156                &[],
157                vcx.alloc_slice(&[axiom_eq, axiom_thin]),
158                vcx.alloc_slice(&[domain_fn]),
159                None,
160            );
161            Ok((domain, fn_idn))
162        })
163    }
164
165    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
166        match Self::all_outputs_local(program).as_slice() {
167            [] => (),
168            [domain] => program.add_domain(domain),
169            _ => unreachable!(),
170        }
171    }
