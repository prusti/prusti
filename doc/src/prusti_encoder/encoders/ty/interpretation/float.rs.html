40pub(crate) fn ty_pure_float<'vir>(
41    vcx: &'vir VirCtxt<'vir>,
42    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
43    builder: &mut DomainBuilder<'vir>,
44    float: ty::FloatTy,
45    prim_to_snap: FunctionIdn<'vir, vir::Prim, vir::CSnap>,
46) -> Result<FloatDomainData<'vir>, EncodeFullError<'vir, TyPureEnc>> {
47    let backend_type = match float {
48        ty::FloatTy::F16 => vcx.alloc_slice(&[
49            vcx.alloc(BackendInterpretationPair {
50                key: "SMTLIB",
51                value: "(_ FloatingPoint 5 11)",
52            }),
53            vcx.alloc(BackendInterpretationPair {
54                key: ("Boogie"),
55                value: ("float11e5"),
56            }),
57        ]),
58        ty::FloatTy::F32 => vcx.alloc_slice(&[
59            vcx.alloc(BackendInterpretationPair {
60                key: "SMTLIB",
61                value: "(_ FloatingPoint 8 24)",
62            }),
63            vcx.alloc(BackendInterpretationPair {
64                key: ("Boogie"),
65                value: ("float24e8"),
66            }),
67        ]),
68        ty::FloatTy::F64 => vcx.alloc_slice(&[
69            vcx.alloc(BackendInterpretationPair {
70                key: "SMTLIB",
71                value: "(_ FloatingPoint 11 53)",
72            }),
73            vcx.alloc(BackendInterpretationPair {
74                key: ("Boogie"),
75                value: ("float53e11"),
76            }),
77        ]),
78        ty::FloatTy::F128 => vcx.alloc_slice(&[
79            vcx.alloc(BackendInterpretationPair {
80                key: "SMTLIB",
81                value: "(_ FloatingPoint 15 113)",
82            }),
83            vcx.alloc(BackendInterpretationPair {
84                key: ("Boogie"),
85                value: ("float113e15"),
86            }),
87        ]),
88    };
89    builder.set_interpretation(backend_type);
90
91    let self_type = builder.self_type();
92    let fp_eq = wrapped_binary(builder, "eq", vir::TYPE_BOOL, "fp.eq");
93    let fp_add = wrapped_binary(builder, "add", self_type, "fp.add RNE");
94    let fp_sub = wrapped_binary(builder, "sub", self_type, "fp.sub RNE");
95
96    let fp_mul = builder.backend_func(
97        "mul",
98        (builder.self_type(), builder.self_type()),
99        builder.self_type(),
100        Some("fp.mul RNE"),
101    );
102
103    let fp_div = builder.backend_func(
104        "div",
105        (builder.self_type(), builder.self_type()),
106        builder.self_type(),
107        Some("fp.div RNE"),
108    );
109
110    let fp_trunc = builder.backend_func(
111        "trunc",
112        builder.self_type(),
113        builder.self_type(),
114        Some("fp.roundToIntegral RTZ"),
115    );
116
117    let fp_is_nan = builder.backend_func(
118        "is_nan",
119        builder.self_type(),
120        vir::TYPE_BOOL,
121        Some("fp.isNaN"),
122    );
123
124    let fp_is_infinite = builder.backend_func(
125        "is_infinite",
126        builder.self_type(),
127        vir::TYPE_BOOL,
128        Some("fp.isInfinite"),
129    );
130
131    let fp_lt = wrapped_binary(builder, "lt", vir::TYPE_BOOL, "fp.lt");
132    let fp_leq = wrapped_binary(builder, "leq", vir::TYPE_BOOL, "fp.leq");
133    let fp_geq = wrapped_binary(builder, "geq", vir::TYPE_BOOL, "fp.geq");
134    let fp_gt = wrapped_binary(builder, "gt", vir::TYPE_BOOL, "fp.gt");
135    let fp_neg = wrapped_unary(builder, "neg", "fp.neg");
136    let fp_abs = wrapped_unary(builder, "abs", "fp.abs");
137
138    let bit_vec = deps.require_dep::<BitVecEnc>(match float {
139        ty::FloatTy::F16 => BitVecSize::BitVec16,
140        ty::FloatTy::F32 => BitVecSize::BitVec32,
141        ty::FloatTy::F64 => BitVecSize::BitVec64,
142        ty::FloatTy::F128 => BitVecSize::BitVec128,
143    })?;
144
145    let from_bv_name = match float {
146        ty::FloatTy::F16 => "(_ to_fp 5 11)",
147        ty::FloatTy::F32 => "(_ to_fp 8 24)",
148        ty::FloatTy::F64 => "(_ to_fp 11 53)",
149        ty::FloatTy::F128 => "(_ to_fp 15 113)",
150    };
151    let from_bv = builder.backend_func(
152        "from_bv",
153        (bit_vec.domain)(),
154        builder.self_type(),
155        Some(from_bv_name),
156    );
157
158    // Not `fp.to_real`: Z3 cannot reason about it on symbolic floats, and
159    // versions before 5.0 can answer `unsat` for satisfiable queries containing
160    // it (Z3Prover/z3#9022). Its meaning comes from `real_value_axioms` and
161    // the literal values of `FloatLitEnc`. Once the Z3 that Viper ships is 5.0
162    // or later, using `fp.to_real` again can be reconsidered.
163    let fp_to_real = builder.function("to_real", builder.self_type(), vir::TYPE_PERM);
164
165    builder.axiom("prim_to_snap", vir::expr! {
166        forall i: [prim_to_snap.arity()] :: {[prim_to_snap](i)} ([prim_to_snap](i)) == ([from_bv]([bit_vec.from_int](i)))
167    });
168
169    let data = FloatDomainData {
170        prim_to_snap,
171        from_bv,
172        fp_eq,
173        fp_add,
174        fp_sub,
175        fp_mul,
176        fp_div,
177        fp_trunc,
178        fp_is_nan,
179        fp_is_infinite,
180        fp_lt,
181        fp_leq,
182        fp_gt,
183        fp_geq,
184        fp_neg,
185        fp_abs,
186        fp_to_real,
187    };
188    real_value_axioms(vcx, builder, &data, float);
189    Ok(data)
190}

192/// An uninterpreted wrapper around a binary backend operation, so that axioms
193/// can trigger on it (Viper rejects interpreted functions in triggers).
194fn wrapped_binary<'vir, T: vir::CompType>(
195    builder: &mut DomainBuilder<'vir>,
196    name: &str,
197    ret: vir::Type<'vir, T>,
198    interpretation: &'static str,
199) -> FunctionIdn<'vir, (vir::CSnap, vir::CSnap), T> {
200    let ty = builder.self_type();
201    let backend: FunctionIdn<'vir, (vir::CSnap, vir::CSnap), T> =
202        builder.backend_func(&format!("{name}_ieee"), (ty, ty), ret, Some(interpretation));
203    let wrapper = builder.function(name, (ty, ty), ret);
204    builder.axiom(
205        &format!("{name}_def"),
206        vir::expr! {
207            forall x: [ty], y: [ty] :: {[wrapper](x, y)} ([wrapper](x, y)) == ([backend](x, y))
208        },
209    );
210    wrapper
211}

213/// Like [`wrapped_binary`], for a unary operation on the float itself.
214fn wrapped_unary<'vir>(
215    builder: &mut DomainBuilder<'vir>,
216    name: &str,
217    interpretation: &'static str,
218) -> FunctionIdn<'vir, vir::CSnap, vir::CSnap> {
219    let ty = builder.self_type();
220    let backend: FunctionIdn<'vir, vir::CSnap, vir::CSnap> =
221        builder.backend_func(&format!("{name}_ieee"), ty, ty, Some(interpretation));
222    let wrapper = builder.function(name, ty, ty);
223    builder.axiom(
224        &format!("{name}_def"),
225        vir::expr! {
226            forall x: [ty] :: {[wrapper](x)} ([wrapper](x)) == ([backend](x))
227        },
228    );
229    wrapper
230}

232/// The precision `p` (significand bits, including the implicit one) and the
233/// maximum exponent `emax` of an IEEE 754 binary format.
234pub(crate) fn ieee_params(float: ty::FloatTy) -> (u32, u32) {
235    let (ebits, sbits) = float_format(float);
236    (sbits, (1 << (ebits - 1)) - 1)
237}

244/// `2^k`, as a single constant if it fits into one, and otherwise as a product
245/// of `2^127`, the largest power of two that does.
246fn pow2<'vir>(vcx: &'vir VirCtxt<'vir>, k: u32) -> vir::ExprInt<'vir> {
247    if k < 128 {
248        return int_const(vcx, 1 << k);
249    }
250    let mut res = int_const(vcx, 1 << (k % 127));
251    for _ in 0..k / 127 {
252        res = vir::expr! { (res) * ([int_const(vcx, 1 << 127)]) };
253    }
254    res
255}

267/// The absolute value of a real.
268fn perm_abs<'vir>(vcx: &'vir VirCtxt<'vir>, e: vir::ExprPerm<'vir>) -> vir::ExprPerm<'vir> {
269    let zero = rational(vcx, vcx.mk_int::<0>(), vcx.mk_int::<1>());
270    vir::expr! { ((e) >= (zero)) ? (e) : (-(e)) }
271}

273/// The float is neither NaN nor an infinity.
274fn finite<'vir>(data: &FloatDomainData<'vir>, e: vir::ExprCSnap<'vir>) -> vir::ExprBool<'vir> {
275    vir::expr! { (!([data.fp_is_nan](e))) && (!([data.fp_is_infinite](e))) }
276}

278/// The overflow threshold of round-to-nearest, `2^(emax + 1) - 2^(emax - p)`:
279/// halfway between the largest finite value and the next power of two. A value
280/// rounds to an infinity exactly when its magnitude is at least this.
281pub(crate) fn rne_overflow<'vir>(
282    vcx: &'vir VirCtxt<'vir>,
283    float: ty::FloatTy,
284) -> vir::ExprInt<'vir> {
285    let (p, emax) = ieee_params(float);
286    if emax <= 127 {
287        int_const(vcx, (u128::MAX >> (127 - emax)) - (1 << (emax - p)) + 1)
288    } else {
289        vir::expr! { ([pow2(vcx, emax + 1)]) - ([pow2(vcx, emax - p)]) }
290    }
291}

293/// The assertion that the float `app` is finite and within a relative error of
294/// `2^-p` of `exact` (which should be a variable or a short term, since it is
295/// repeated). Round-to-nearest guarantees this when `exact` is below the
296/// overflow threshold and is either representable or outside the subnormal
297/// range. Finiteness is part of it because deriving it from the bit-level
298/// definition stalls under Silicon's `smt.case_split 3` and
299/// `smt.delay_units true`.
300pub(crate) fn rne_rounded<'vir>(
301    vcx: &'vir VirCtxt<'vir>,
302    data: &FloatDomainData<'vir>,
303    float: ty::FloatTy,
304    app: vir::ExprCSnap<'vir>,
305    exact: vir::ExprPerm<'vir>,
306) -> vir::ExprBool<'vir> {
307    let (p, _) = ieee_params(float);
308    let eps = rational(vcx, vcx.mk_int::<1>(), pow2(vcx, p));
309    let error_decl = vcx.mk_local_decl("error", vir::TYPE_PERM);
310    let bound_decl = vcx.mk_local_decl("bound", vir::TYPE_PERM);
311    let (error, bound) = (vcx.mk_local_ex(error_decl), vcx.mk_local_ex(bound_decl));
312    let body =
313        vir::expr! { ([finite(data, app)]) && (((-(bound)) <= (error)) && ((error) <= (bound))) };
314    let body = vcx.mk_let_expr(
315        bound_decl,
316        vir::expr! { (eps) * ([perm_abs(vcx, exact)]) },
317        body,
318    );
319    vcx.mk_let_expr(
320        error_decl,
321        vir::expr! { ([data.fp_to_real](app)) - (exact) },
322        body,
323    )
324}

326/// Axioms giving `to_real` the IEEE 754 semantics of round-to-nearest-even:
327/// comparisons of finite floats are comparisons of their values, and a sum or
328/// difference of finite floats is finite and within a relative error of
329/// `2^-p` of the exact result, unless that exceeds the overflow threshold.
330/// (Subnormal results of additions are exact, so no absolute error term is
331/// needed.) Negation and absolute value of a finite float are exact. This
332/// keeps the reasoning in linear real arithmetic. The values of products,
333/// quotients and truncations are left unconstrained.
334fn real_value_axioms<'vir>(
335    vcx: &'vir VirCtxt<'vir>,
336    builder: &mut DomainBuilder<'vir>,
337    data: &FloatDomainData<'vir>,
338    float: ty::FloatTy,
339) {
340    let ty = builder.self_type();
341    let overflow = rational(vcx, rne_overflow(vcx, float), vcx.mk_int::<1>());
342    let x = vcx.mk_local_decl("x", ty);
343    let y = vcx.mk_local_decl("y", ty);
344    let (xe, ye) = (vcx.mk_local_ex(x), vcx.mk_local_ex(y));
345    let (rx, ry) = ((data.fp_to_real)(xe), (data.fp_to_real)(ye));
346
347    for (name, op, value) in [
348        ("eq", data.fp_eq, vir::expr! { (rx) == (ry) }),
349        ("lt", data.fp_lt, vir::expr! { (rx) < (ry) }),
350        ("leq", data.fp_leq, vir::expr! { (rx) <= (ry) }),
351        ("geq", data.fp_geq, vir::expr! { (rx) >= (ry) }),
352        ("gt", data.fp_gt, vir::expr! { (rx) > (ry) }),
353    ] {
354        let app = op(xe, ye);
355        let body =
356            vir::expr! { (([finite(data, xe)]) && ([finite(data, ye)])) ==> ((app) == (value)) };
357        builder.axiom(
358            &format!("{name}_real"),
359            vir::expr! { forall [x], [y] :: {app} body },
360        );
361    }
362
363    for (name, op, exact) in [
364        ("add", data.fp_add, vir::expr! { (rx) + (ry) }),
365        ("sub", data.fp_sub, vir::expr! { (rx) - (ry) }),
366    ] {
367        let app = op(xe, ye);
368        let exact_decl = vcx.mk_local_decl("exact", vir::TYPE_PERM);
369        let e = vcx.mk_local_ex(exact_decl);
370        let in_range = vir::expr! { ([perm_abs(vcx, e)]) < (overflow) };
371        let body = vir::expr! {
372            (([finite(data, xe)]) && (([finite(data, ye)]) && (in_range)))
373                ==> ([rne_rounded(vcx, data, float, app, e)])
374        };
375        let body = vcx.mk_let_expr(exact_decl, exact, body);
376        builder.axiom(
377            &format!("{name}_real"),
378            vir::expr! { forall [x], [y] :: {app} body },
379        );
380    }
381
382    for (name, op, value) in [
383        ("neg", data.fp_neg, vir::expr! { -(rx) }),
384        ("abs", data.fp_abs, perm_abs(vcx, rx)),
385    ] {
386        let app = op(xe);
387        let body = vir::expr! {
388            ([finite(data, xe)])
389                ==> (([finite(data, app)]) && (([data.fp_to_real](app)) == (value)))
390        };
391        builder.axiom(
392            &format!("{name}_real"),
393            vir::expr! { forall [x] :: {app} body },
394        );
395    }
396}

398/// The exact value of the float with the given bits, or `None` for NaN and the
399/// infinities.
400fn float_value<'vir>(
401    vcx: &'vir VirCtxt<'vir>,
402    float: ty::FloatTy,
403    bits: u128,
404) -> Option<vir::ExprPerm<'vir>> {
405    let (exp_bits, p) = float_format(float);
406    let (_, emax) = ieee_params(float);
407    let width = exp_bits + p;
408    let mant_bits = p - 1;
409    let mant = bits & ((1 << mant_bits) - 1);
410    let exp = (bits >> mant_bits) & ((1 << exp_bits) - 1);
411    if exp == (1 << exp_bits) - 1 {
412        return None;
413    }
414    // The value is `mant * 2^shift`; subnormals have the minimum exponent.
415    let (mant, shift) = if exp == 0 {
416        (mant, 1 - emax as i64 - mant_bits as i64)
417    } else {
418        (
419            mant | (1 << mant_bits),
420            exp as i64 - emax as i64 - mant_bits as i64,
421        )
422    };
423    // Reduce the fraction, which keeps the emitted constants readable.
424    let tz = if mant == 0 { 0 } else { mant.trailing_zeros() };
425    let (mant, shift) = (mant >> tz, shift + i64::from(tz));
426    let shift_abs = shift.unsigned_abs() as u32;
427    let value = if shift >= 0 && shift_abs < mant.leading_zeros() {
428        rational(vcx, int_const(vcx, mant << shift_abs), vcx.mk_int::<1>())
429    } else {
430        let (mant, scale) = (int_const(vcx, mant), pow2(vcx, shift_abs));
431        if shift >= 0 {
432            rational(vcx, vir::expr! { (mant) * (scale) }, vcx.mk_int::<1>())
433        } else {
434            rational(vcx, mant, scale)
435        }
436    };
437    Some(if (bits >> (width - 1)) & 1 == 1 {
438        vir::expr! { -(value) }
439    } else {
440        value
441    })
442}

444/// Emits an axiom stating the exact value `to_real(prim_to_snap(bits))` of a
445/// finite float constant (nothing for NaN and the infinities). The axioms of
446/// all constants of a float type are emitted in one domain.
447pub struct FloatLitEnc;
448
449impl TaskEncoder for FloatLitEnc {
450    task_encoder::encoder_cache!(FloatLitEnc);
451    const ENCODER_NAME: &'static str = "float literal encoder";
452
453    type TaskDescription<'vir> = (ty::FloatTy, u128);
454
455    type OutputFullLocal<'vir> = Option<(ty::FloatTy, vir::DomainAxiom<'vir>)>;
456
457    type OutputFullDependency<'vir> = ();
458
459    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
460        *task
461    }
462
463    fn do_encode_full<'vir>(
464        task_key: &Self::TaskKey<'vir>,
465        deps: &mut TaskEncoderDependencies<'vir, Self>,
466    ) -> task_encoder::EncodeFullResult<'vir, Self> {
467        deps.emit_output_ref(*task_key, ())?;
468        vir::with_vcx(|vcx| {
469            let (float, bits) = *task_key;
470            let Some(value) = float_value(vcx, float, bits) else {
471                return Ok((None, ()));
472            };
473            let ty = ty::Ty::new_float(vcx.tcx(), float);
474            let domain = *deps
475                .require_dep::<TyUsePureEnc>(RustTyDecomposition::from_prim_ty(ty))?
476                .expect_float();
477            let snap = (domain.prim_to_snap)(int_const(vcx, bits).upcast_ty());
478            let axiom = vcx.mk_domain_axiom(
479                vir::ViperIdent::new(vir::vir_format!(
480                    vcx,
481                    "s_Float_{}_lit_{bits:x}",
482                    float.name_str()
483                )),
484                vir::expr! { ([domain.fp_to_real](snap)) == (value) },
485            );
486            Ok((Some((float, axiom)), ()))
487        })
488    }
489
490    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
491        let mut by_type = std::collections::BTreeMap::<_, Vec<_>>::new();
492        for (float, axiom) in Self::all_outputs_local(program).into_iter().flatten() {
493            by_type.entry(float.name_str()).or_default().push(axiom);
494        }
495        vir::with_vcx(|vcx| {
496            for (float, mut axioms) in by_type {
497                axioms.sort_by_key(|axiom| axiom.name);
498                program.add_domain(vcx.mk_domain(
499                    vir::ViperIdent::new(vir::vir_format!(vcx, "s_Float_{float}_lits")),
500                    &[],
501                    vcx.alloc_slice(&axioms),
502                    &[],
503                    None,
504                ));
505            }
506        });
507    }
