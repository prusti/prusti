11#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
12pub struct RustTyDecomposition<'tcx> {
13    pub ty: RustTy<'tcx>,
14    pub args: GArgs<'tcx>,
15}
16
17impl<'tcx> RustTyDecomposition<'tcx> {
18    fn new(ty: RustTy<'tcx>, args: GArgs<'tcx>) -> Self {
19        ty.params.check(args.args());
20        RustTyDecomposition { ty, args }
21    }
22
23    /// Decomposes a rustc `ty::Ty` into the core type used to generate a Viper
24    /// domain/predicate and its type arguments (not used for the Viper
25    /// definition). For example, for the function:
26    /// ```no_run
27    /// struct MyStruct<I: Iterator> {
28    ///     field: I::Item
29    /// }
30    /// fn foo<T: Iterator<Item = i32>>(x: MyStruct<T>) { ... }
31    /// ```
32    /// when encoding the argument type, this should be called as
33    /// ```no_run
34    /// let decomp = from_ty(tcx, "MyStruct<T>", "<T: Iterator<Item = i32>>")
35    /// // which yields
36    /// RustTyDecomposition {
37    ///     ty: TyData { params: "<I: Iterator>", specifics: "MyStruct(I::Item)" }
38    ///     args: GArgs { args: "<T>", context: "<T: Iterator<Item = i32>>" }
39    /// }
40    /// ```
41    /// The `ty` field is agnostic of the client's generic arguments while the
42    /// `args` field captures everything from the client's side. Note that we
43    /// guarantee that `decomp.ty.params.len() == decomp.args.len()`.
44    ///
45    /// To recursively encode the struct itself, one should walk the
46    /// `decomp.ty.specifics` and call `RustFieldData::decompose` with
47    /// `decomp.ty.params`.
48    ///
49    /// To figure out which casts are required from the client side (e.g. when
50    /// unfolding), one should walk the `decomp.ty.specifics` and call
51    /// `RustFieldData::decompose_compare_normalize` with `decomp.ty.params`
52    /// and `decomp.args`.
53    pub fn from_ty(ty: ty::Ty<'tcx>, context: impl Into<GParams<'tcx>>) -> Self {
54        TyData::<'tcx, RustTyDatas>::from_ty(ty, context.into())
55    }
56
57    /// Same as `from_ty` to get a `RustTyDecomposition` for use in encoding,
58    /// but requires fewer arguments when the type is known to be primitive.
59    pub fn from_prim_ty(ty: ty::Ty<'tcx>) -> Self {
60        assert!(ty.is_primitive());
61        TyData::<'tcx, RustTyDatas>::from_prim_ty(ty)
62    }
63
64    /// When you only have a `RustTy<'tcx>` but you want to use one of the
65    /// type-use encoders.
66    pub fn identity(ty: RustTy<'tcx>) -> Self {
67        let args = GArgs::new(ty.params, ty.params.rust_params());
68        Self::new(ty, args)
69    }
70
71    pub fn param() -> RustTy<'tcx> {
72        let gty = TyData::<RustTyDatas>::args_from_tys([TySpecifics::new_param_ty(0)]);
73        let data = RustTyData {
74            name: symbol::Symbol::intern("Param"),
75            params: GParams::empty_env(gty),
76            special: RustTySpecial::None,
77            sizedness: RustTySizedness::None,
78        };
79        let specifics = TySpecifics::Param(RustParamData::Generic);
80        TyData::<RustTyDatas>::new(data, specifics).alloc()
81    }
82}

90/// A to-be decomposed Rust type. We need this since we cannot infinitely
91/// decompose recursive datatypes (instead fields are left as `LazyRustTy` and
92/// decomposed as needed when recursing).
93#[repr(transparent)]
94#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
95pub struct LazyRustTy<'tcx>(ty::Ty<'tcx>);
96
97impl<'tcx> LazyRustTy<'tcx> {
98    pub fn new(ty: ty::Ty<'tcx>) -> Self {
99        Self(ty)
100    }
101
102    pub fn new_slice(tys: &'tcx [ty::Ty<'tcx>]) -> &'tcx [Self] {
103        // SAFETY: `LazyRustTy` is `repr(transparent)` over `ty::Ty`
104        let ptr = tys as *const [ty::Ty<'tcx>] as *const [Self];
105        unsafe { &*ptr }
106    }
107
108    /// The (sized) type parameter at the given index. `decompose`d against a
109    /// `GParams` that declares it, this yields a `Param` decomposition whose
110    /// argument is that type variable. Used to make a builtin method generic
111    /// over one of the reference's type parameters.
112    ///
113    /// You very likely do NOT want to use this: it's a hacky thing to make
114    /// `MirBuiltinCastEnc` work.
115    pub fn new_param_ty(index: u32) -> Self {
116        Self::new(TySpecifics::new_param_ty(index))
117    }
118
119    /// Whether this is a shared reference `&T` (before any substitution).
120    pub fn is_immref(&self) -> bool {
121        matches!(self.0.kind(), ty::TyKind::Ref(_, _, ty::Mutability::Not))
122    }
123
124    /// The pointer metadata type of a pointer to this type.
125    fn pointee_metadata(self) -> Self {
126        vir::with_vcx(|vcx| {
127            let metadata_did = vcx
128                .tcx()
129                .require_lang_item(LangItem::Metadata, prusti_rustc_interface::span::DUMMY_SP);
130            Self::new(ty::Ty::new_projection(
131                vcx.tcx(),
132                ty::IsRigid::No,
133                metadata_did,
134                [self.0],
135            ))
136        })
137    }
138}

140impl<'tcx> LazyRustTy<'tcx> {
141    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
142    /// when recursing over the fields of a containing `RustTy` to construct
143    /// e.g. a predicate - i.e. when the definition of the predicate is
144    /// independent of the context/generic args).
145    /// The passed `params` should be those of the containing `RustTy::params`.
146    ///
147    /// For example a `Foo<i32>` with definition `struct Foo<T>(T);`, then
148    /// decomposing the field of the struct would yield `TySpecifics::Param`
149    /// with arguments `<T>` (i.e. the `i32` from the context is lost).
150    pub fn decompose(&self, params: GParams<'tcx>) -> RustTyDecomposition<'tcx> {
151        RustTyDecomposition::from_ty(self.0, params)
152    }
153
154    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
155    /// when recursing over the fields of a containing `RustTy`
156    /// non-transparently, e.g. when predicates of fields should be added
157    /// directly to a method itself).
158    /// The passed `args` should be those of the containing `RustTyDecomposition::args`.
159    pub fn decompose_context(
160        &self,
161        params: GParams<'tcx>,
162        args: GArgs<'tcx>,
163    ) -> RustTyDecomposition<'tcx> {
164        let mut decomp = self.decompose(params);
165        decomp.args = decomp.args.substitute(args);
166        decomp
167    }
168
169    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
170    /// when recursing over the fields of a containing `RustTy`).
171    /// The passed `args` should be those of the containing `RustTyDecomposition::args`.
172    ///
173    /// This differs from `Self::decompose` in that it substitutes the `args`
174    /// removing definitional generics. For example a `Foo<i32>` with definition
175    /// `struct Foo<T>(T);` would yield `i32` instead of `T` when called on the
176    /// field of `Foo`.
177    pub fn decompose_normalize(&self, args: GArgs<'tcx>) -> RustTyDecomposition<'tcx> {
178        RustTyDecomposition::from_ty(args.normalize(self.0), args.context())
179    }
180
181    /// Similarly to `Self::decompose`, this decomposes the fields type.
182    /// However, it tries to normalize the type first and only returns a
183    /// decomposition if the type was a `TySpecifics::Param` and is now a
184    /// concrete type. For example, when called on the `field: I::Item` of the
185    /// following example:
186    /// ```no_run
187    /// struct MyStruct<I: Iterator> {
188    ///     field: I::Item
189    /// }
190    /// fn foo<T: Iterator<Item = i32>>(x: MyStruct<T>) { ... }
191    /// ```
192    /// For which the initial decomposition of the argument `MyStruct<T>` was
193    /// ```no_run
194    /// let decomp = RustTyDecomposition {
195    ///     ty: TyData { params: "<I: Iterator>", specifics: "MyStruct(I::Item)" }
196    ///     args: GArgs { args: "<T>", context: "<T: Iterator<Item = i32>>" }
197    /// };
198    /// // one would call
199    /// let field = decomp.ty.specifics.expect_struct().fields[0];
200    /// let decomp_field = field.decompose_compare_normalize(decomp.ty.params, decomp.args)
201    /// // where `decomp_field` would be
202    /// Some(RustTyDecomposition {
203    ///     ty: TyData { params: "", specifics: "i32" }
204    ///     args: GArgs { args: "", context: "<T: Iterator<Item = i32>>" }
205    /// });
206    /// ```
207    pub fn decompose_compare_normalize(
208        &self,
209        params: GParams<'tcx>,
210        args: GArgs<'tcx>,
211    ) -> Option<RustTyNormalized<'tcx>> {
212        let param = self.decompose(params).ty;
213        let TySpecifics::Param(..) = &param.specifics else {
214            return None;
215        };
216        let concrete = self.decompose_normalize(args);
217        if let TySpecifics::Param(..) = &concrete.ty.specifics {
218            None
219        } else {
220            Some(RustTyNormalized { param, concrete })
221        }
222    }
223}

276/// Which of the sizedness traits (`Sized`, `MetaSized`, `PointeeSized`) the
277/// constructor's instances implement. These traits have no impls, so their
278/// facts are encoded per constructor instead (see `SizednessEnc`). Mirrors
279/// `Ty::has_trivial_sizedness`.
280#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
281pub enum RustTySizedness<'tcx> {
282    /// Not a type constructor (a type parameter or projection).
283    None,
284    /// `Sized` and `MetaSized` exactly when the respective tail type, over
285    /// the constructor's parameters, is; unconditionally if there is none.
286    Sized {
287        sized_if: Option<ty::Ty<'tcx>>,
288        meta_sized_if: Option<ty::Ty<'tcx>>,
289    },
290    /// `MetaSized` but not `Sized`: `str`, slices and trait objects.
291    MetaSized,
292    /// Only `PointeeSized`: extern types.
293    PointeeSized,
294}
295
296impl<'tcx> RustTySizedness<'tcx> {
297    fn from_ty(ty: ty::Ty<'tcx>) -> Self {
298        let tail = |kind| match *ty.kind() {
299            ty::TyKind::Adt(adt, _) => vir::with_vcx(|vcx| {
300                adt.sizedness_constraint(vcx.tcx(), kind)
301                    .map(ty::EarlyBinder::instantiate_identity)
302                    .map(ty::Unnormalized::skip_normalization)
303            }),
304            ty::TyKind::Tuple(tys) => tys
305                .len()
306                .checked_sub(1)
307                .map(|last| TySpecifics::new_param_ty(last as u32)),
308            _ => None,
309        };
310        match ty.kind() {
311            ty::TyKind::Param(_) | ty::TyKind::Alias(..) => Self::None,
312            ty::TyKind::Str | ty::TyKind::Slice(_) | ty::TyKind::Dynamic(..) => Self::MetaSized,
313            ty::TyKind::Foreign(_) => Self::PointeeSized,
314            _ => Self::Sized {
315                sized_if: tail(ty::SizedTraitKind::Sized),
316                meta_sized_if: tail(ty::SizedTraitKind::MetaSized),
317            },
318        }
319    }
320}

399impl<'tcx> TyData<'tcx, RustTyDatas> {
400    pub fn ref_data(&self) -> Option<RefData<'tcx>> {
401        match &self.specifics {
402            TySpecifics::ImmRef(data) | TySpecifics::MutRef(data) | TySpecifics::Raw(data) => {

405            _ => None,

409    /// The type of the value stored in a `Box` (its hardcoded last field, see
410    /// `TySpecifics::from_adt`).
411    pub fn box_value_ty(&self) -> LazyRustTy<'tcx> {
412        assert_eq!(self.special, RustTySpecial::Box);
413        self.expect_structlike().fields.last().unwrap().ty()

416    fn from_ty(ty: ty::Ty<'tcx>, context: GParams<'tcx>) -> RustTyDecomposition<'tcx> {
417        // We normalize since we may be translating a type such as the field of
418        // `struct MyStruct<T: Iterator<Item = i32>>(T::Item);` where `ty` is
419        // `T::Item` and `context` is `<T: Iterator<Item = i32>>`. In this case
420        // we want to encode the struct as if it had an `i32` field (without any
421        // Param generics).
422        let ty = context.normalize(ty);
423
424        let name = Self::ty_name(ty);
425        let (params, args) = Self::identity_for_ty(ty, context.is_trait_extern_spec());
426        let args = GArgs::new(context, args);
427        let specifics = TySpecifics::from_ty(ty);
428        let data = RustTyData {
429            name: symbol::Symbol::intern(&name),
430            params,
431            special: RustTySpecial::from_ty(ty),
432            sizedness: RustTySizedness::from_ty(ty),
433        };
434        RustTyDecomposition::new(Self::new(data, specifics).alloc(), args)
435    }
436
437    fn from_prim_ty(ty: ty::Ty<'tcx>) -> RustTyDecomposition<'tcx> {
438        let name = Self::prim_ty_name(ty);
439        let (params, args) = Self::identity_for_prim_ty(ty);
440        let args = GArgs::new(params, args);
441        let data = RustTyData {
442            name: symbol::Symbol::intern(&name),
443            params,
444            special: RustTySpecial::None,
445            sizedness: RustTySizedness::from_ty(ty),
446        };
447        let specifics = TySpecifics::from_prim_ty(ty);
448        RustTyDecomposition::new(Self::new(data, specifics).alloc(), args)
449    }

451    fn ty_name(ty: ty::Ty<'tcx>) -> String {
452        let def_id_name = |def_id| {
453            vir::with_vcx(|vcx| {
454                vir::ViperIdent::from_def_id(vcx, def_id)
455                    .to_str()
456                    .to_owned()
457            })
458        };
459        match ty.kind() {
460            _ if ty.is_primitive() => Self::prim_ty_name(ty),
461            ty::TyKind::Str => String::from("Str"),
462            ty::TyKind::Adt(adt, _) => def_id_name(adt.did()),
463            ty::TyKind::Tuple(params) => format!("{}_Tuple", params.len()),
464            ty::TyKind::Never => String::from("Never"),
465            ty::TyKind::Ref(_, _, ty::Mutability::Not) => String::from("Ref_immutable"),
466            ty::TyKind::Ref(_, _, ty::Mutability::Mut) => String::from("Ref_mutable"),
467            ty::TyKind::RawPtr(_, ty::Mutability::Not) => String::from("RawPtr_immutable"),
468            ty::TyKind::RawPtr(_, ty::Mutability::Mut) => String::from("RawPtr_mutable"),
469            ty::TyKind::Param(_) | ty::TyKind::Alias(..) => String::from("Param"),
470            ty::TyKind::Closure(def_id, _) => def_id_name(*def_id),
471            ty::TyKind::FnPtr(..) => String::from("FnPtr"),
472            ty::TyKind::Array(..) => String::from("Array"),
473            ty::TyKind::Slice(..) => String::from("Slice"),
474            ty::TyKind::Dynamic(..) => String::from("Dyn"),
475            // A tuple struct's constructor has the same def path as the struct.
476            ty::TyKind::FnDef(def_id, _) => format!("FnDef_{}", def_id_name(*def_id)),
477            ty::TyKind::Foreign(def_id) => def_id_name(*def_id),
478            ty::TyKind::Pat(ty, _) => Self::ty_name(*ty),
479            other => unimplemented!("ty_name for {:?}", other),
480        }
481    }
482
483    fn prim_ty_name(ty: ty::Ty<'tcx>) -> String {
484        assert!(ty.is_primitive());
485        match ty.kind() {
486            ty::TyKind::Bool => String::from("Bool"),
487            ty::TyKind::Char => String::from("Char"),
488            ty::TyKind::Int(kind) => format!("Int_{}", kind.name_str()),
489            ty::TyKind::Uint(kind) => format!("UInt_{}", kind.name_str()),
490            ty::TyKind::Float(kind) => format!("Float_{}", kind.name_str()),
491            _ => unreachable!(),
492        }
493    }
494
495    /// For the ty `MyStruct<i32>` (with defn
496    /// `struct MyStruct<T: Iterator<Item = i32>> { ... }`), returns
497    /// `([<T: Iterator<Item = i32>>], [i32])`.
498    pub(super) fn identity_for_ty(
499        ty: ty::Ty<'tcx>,
500        is_trait_extern_spec: bool,
501    ) -> (GParams<'tcx>, ty::GenericArgsRef<'tcx>) {
502        let (params, args) = match *ty.kind() {
503            _ if ty.is_primitive() => return Self::identity_for_prim_ty(ty),
504            ty::TyKind::Adt(adt, args) => (GParams::from(adt.did()), args),
505            ty::TyKind::Tuple(tys) => {
506                let gtys = (0..tys.len()).map(|idx| TySpecifics::new_param_ty(idx as u32));
507                (
508                    GParams::empty_env(Self::args_from_tys(gtys)),
509                    Self::args_from_tys(tys),
510                )
511            }
512            ty::TyKind::Array(ty, cst) => vir::with_vcx(|vcx| {
513                let gcst = TySpecifics::new_param_const(0).into();
514                let gty = TySpecifics::new_param_ty(1).into();
515                let gparams = Self::args_from_generics([gcst, gty]);
516                let predicate =
517                    vcx.tcx()
518                        .mk_predicate(ty::Binder::dummy(ty::PredicateKind::Clause(
519                            ty::ClauseKind::ConstArgHasType(
520                                gcst.expect_const(),
521                                vcx.tcx().types.usize,
522                            ),
523                        )));
524                let param_env =
525                    ty::ParamEnv::new(vcx.tcx().mk_clauses(&[predicate.expect_clause()]));
526                (
527                    GParams::new(gparams, param_env, false),
528                    Self::args_from_generics([cst.into(), ty.into()]),
529                )
530            }),
531            ty::TyKind::Slice(ty) => {
532                let gty = Self::args_from_tys([TySpecifics::new_param_ty(0)]);
533                (GParams::empty_env(gty), Self::args_from_tys([ty]))
534            }
535            ty::TyKind::RawPtr(pointee, _) => {
536                // Like a reference, a raw pointer's pointer-metadata type is
537                // derived from the pointee via `<pointee as Pointee>::Metadata`
538                // (see `RefData` below), so it has a single generic: the pointee.
539                let pointee_ty = TySpecifics::new_param_ty(0).into();
540                let gty = Self::args_from_generics([pointee_ty]);
541                (
542                    GParams::empty_env(gty),
543                    Self::args_from_generics([pointee.into()]),
544                )
545            }
546            ty::TyKind::Ref(region, inner, _) => {
547                // The pointer-metadata type is derived from the referent (see
548                // `RefData` below and `LazyRustTy::pointee_metadata`), so a
549                // reference has just two generics: the lifetime and the referent.
550                // TODO: what lifetime should we use here?
551                let param_region = vir::with_vcx(|vcx| vcx.tcx().lifetimes.re_erased.into());
552                let param_ty = TySpecifics::new_param_ty(1).into();
553                let gty = Self::args_from_generics([param_region, param_ty]);
554                (
555                    GParams::empty_env(gty),
556                    Self::args_from_generics([region.into(), inner.into()]),
557                )
558            }
559            ty::TyKind::Alias(..) | ty::TyKind::Param(_) => {
560                let gty = Self::args_from_tys([TySpecifics::new_param_ty(0)]);
561                (GParams::empty_env(gty), Self::args_from_tys([ty]))
562            }
563            ty::TyKind::Closure(did, args) => vir::with_vcx(|vcx| {
564                let identity = ty::List::identity_for_item(vcx.tcx(), did);
565                let gargs = vcx.tcx().mk_args(identity.as_closure().parent_args());
566                let args = vcx.tcx().mk_args(args.as_closure().parent_args());
567                (
568                    GParams::new(gargs, vcx.tcx().param_env(did), is_trait_extern_spec),
569                    args,
570                )
571            }),
572            ty::TyKind::FnDef(did, args) => vir::with_vcx(|vcx| {
573                let gargs = identity_params(vcx.tcx(), did);
574                (
575                    GParams::new(gargs, vcx.tcx().param_env(did), is_trait_extern_spec),
576                    args.skip_binder(),
577                )
578            }),
579            ty::TyKind::Pat(ty, _) => Self::identity_for_ty(ty, is_trait_extern_spec),
580            ty::TyKind::Never
581            | ty::TyKind::Str
582            | ty::TyKind::FnPtr(..)
583            | ty::TyKind::Dynamic(..)
584            | ty::TyKind::Foreign(_) => (GParams::empty(), ty::GenericArgs::empty()),
585            _ => todo!("instantiate_identity_for_type for {:?}", ty),
586        };
587        params.check(args);
588        (params, args)
589    }
590
591    fn identity_for_prim_ty(ty: ty::Ty<'tcx>) -> (GParams<'tcx>, ty::GenericArgsRef<'tcx>) {
592        assert!(ty.is_primitive());
593        (GParams::empty(), ty::GenericArgs::empty())
594    }
595
596    fn args_from_tys(tys: impl IntoIterator<Item = ty::Ty<'tcx>>) -> ty::GenericArgsRef<'tcx> {
597        Self::args_from_generics(tys.into_iter().map(ty::GenericArg::from))
598    }
599
600    fn args_from_generics(
601        tys: impl IntoIterator<Item = ty::GenericArg<'tcx>>,
602    ) -> ty::GenericArgsRef<'tcx> {
603        vir::with_vcx(|vcx| vcx.tcx().mk_args_from_iter(tys.into_iter()))
604    }
605}

607/// For a `RustTy` zipped with another (encoded) type data.
608impl<'tcx, D: TyDatas<'tcx>> TyData<'tcx, (RustTyDatas, D)> {
609    pub fn box_metadata_ty(&self) -> LazyRustTy<'tcx> {
610        assert_eq!(self.0.special, RustTySpecial::Box);
611        let box_value_ty = self.expect_structlike().fields.last().unwrap().0.ty();
612        box_value_ty.pointee_metadata()
613    }
614
615    /// The type of the `Unique` pointer of a `Box` (its first field).
616    pub fn box_unique_ty(&self) -> LazyRustTy<'tcx> {
617        assert_eq!(self.0.special, RustTySpecial::Box);
618        self.expect_structlike().fields[0].0.ty()
619    }
620
621    /// The type of the allocator of a `Box` (its second field).
622    pub fn box_alloc_ty(&self) -> LazyRustTy<'tcx> {
623        assert_eq!(self.0.special, RustTySpecial::Box);
624        self.expect_structlike().fields[1].0.ty()
625    }
626}

628impl RustTySpecial {
629    fn from_ty(ty: ty::Ty<'_>) -> Self {
630        match ty.kind() {
631            ty::TyKind::Adt(adt, _) if adt.is_box() => RustTySpecial::Box,
632            _ => RustTySpecial::None,
633        }
634    }
635}
636
637impl<'tcx> TySpecifics<'tcx, RustTyDatas> {
638    fn from_ty(ty: ty::Ty<'tcx>) -> Self {
639        if ty.is_primitive() {
640            return Self::from_prim_ty(ty);
641        }
642        if crate::encoders::is_type_trusted(ty) {
643            return TySpecifics::mk_opaque(());
644        }
645
646        match ty.kind() {
647            ty::TyKind::Adt(adt, _) => Self::from_adt(*adt),
648            ty::TyKind::Tuple(args) => {
649                let fields = args
650                    .iter()
651                    .enumerate()
652                    .map(|(i, _)| RustFieldData {
653                        name: symbol::Symbol::intern(&format!("_{i}")),
654                        fid: abi::FieldIdx::from_usize(i),
655                        ty: LazyRustTy(Self::new_param_ty(i as u32)),
656                        address: RustFieldAddress::Constant,
657                    })
658                    .collect::<Vec<_>>();
659                TySpecifics::mk_structlike((), fields)
660            }
661            ty::TyKind::Array(_, _) => TySpecifics::ArrayLike(ArrayData {
662                slice: false,
663                data: LazyRustTy(Self::new_param_ty(1)),
664            }),
665            ty::TyKind::Slice(_) => TySpecifics::ArrayLike(ArrayData {
666                slice: true,
667                data: LazyRustTy(Self::new_param_ty(0)),
668            }),
669            ty::TyKind::Ref(_, _, mutability) => {
670                // The referent is generic parameter 1; its pointer metadata is
671                // derived as `<referent as Pointee>::Metadata` rather than being
672                // a separate generic parameter.
673                let referent = LazyRustTy(Self::new_param_ty(1));
674                let data = RefData {
675                    metadata: referent.pointee_metadata(),
676                    referent,
677                };
678                match mutability {
679                    ty::Mutability::Mut => TySpecifics::mk_mutref(data),
680                    ty::Mutability::Not => TySpecifics::mk_immref(data),
681                }
682            }
683            // Raw pointers get their own conservative `Raw` kind: we don't
684            // reason about the pointee through them, but (like a reference) the
685            // snapshot carries the address and pointer metadata so `PtrMetadata`
686            // (e.g. a slice's length) can be read out. The pointee is generic
687            // parameter 0; the metadata is derived from it as
688            // `<pointee as Pointee>::Metadata`.
689            ty::TyKind::RawPtr(..) => {
690                let referent = LazyRustTy(Self::new_param_ty(0));
691                TySpecifics::mk_raw(RefData {
692                    metadata: referent.pointee_metadata(),
693                    referent,
694                })
695            }
696            ty::TyKind::Alias(..) | ty::TyKind::Param(_) => {
697                TySpecifics::mk_param(RustParamData::Generic)
698            }
699            ty::TyKind::Closure(_, args) => {
700                let captured = args.as_closure().upvar_tys();
701                let fields = vir::with_vcx(|vcx| {
702                    captured
703                        .iter()
704                        .enumerate()
705                        .map(|(i, ty)| RustFieldData {
706                            name: symbol::Symbol::intern(&format!("c{i}")),
707                            fid: abi::FieldIdx::from_usize(i),
708                            ty: LazyRustTy(vcx.tcx().erase_and_anonymize_regions(ty)),
709                            address: RustFieldAddress::Constant,
710                        })
711                        .collect::<Vec<_>>()
712                });
713                TySpecifics::mk_structlike((), fields)
714            }
715            ty::TyKind::Never => {
716                let data = vir::with_vcx(|vcx| RustEnumData {
717                    discr: vcx.tcx().types.isize,
718                });
719                TySpecifics::mk_enumlike(data, Vec::new())
720            }
721            // TODO: add str support
722            ty::TyKind::Str => TySpecifics::mk_opaque(()),
723            // TODO: give dyn Trait a type witness parameter (the concrete type behind the
724            // pointer), enabling virtual dispatch and distinguishing dyn TraitA from dyn TraitB.
725            ty::TyKind::Dynamic(..) => TySpecifics::mk_param(RustParamData::Dyn),
726            // TODO: encode the pattern as narrowed bounds on the base type, so that e.g.
727            // the `u32 is 1..` inside a `NonZeroU32` is known to be non-zero. Doing so
728            // also needs a `ty_name` that distinguishes the pattern from its base type,
729            // since the name and the specifics together are the interning key.
730            ty::TyKind::Pat(ty, _) => Self::from_ty(*ty),
731            _ => TySpecifics::mk_opaque(()),
732        }
733    }
734
735    fn from_prim_ty(ty: ty::Ty<'tcx>) -> Self {
736        assert!(ty.is_primitive());
737        TySpecifics::mk_primitive(ty)
738    }
739
740    fn from_adt(adt: ty::AdtDef<'tcx>) -> Self {
741        if vir::with_vcx(|vcx| vcx.tcx().lang_items().get(LangItem::DynMetadata) == Some(adt.did()))
742        {
743            // `DynMetadata<dyn Trait>` is the metadata of a `&dyn`/`*dyn`
744            // wide pointer. We never reason about vtable contents, so encode it
745            // as an opaque snapshot rather than recursing into its `Foreign`
746            // `VTable` field (which has no `ty_name`).
747            TySpecifics::mk_opaque(())
748        } else if vir::with_vcx(|vcx| {
749            EnvQuery::new(vcx.tcx()).is_adt_in_crate(adt, "prusti_contracts")
750        }) {
751            match adt.non_enum_variant().name.to_string().as_str() {
752                "Int" => Self::Builtin(RustBuiltinData::Int),
753                "Real" => Self::Builtin(RustBuiltinData::Real),
754                "Set" => Self::Builtin(RustBuiltinData::Set(LazyRustTy(Self::new_param_ty(0)))),
755                "Multiset" => {
756                    Self::Builtin(RustBuiltinData::Multiset(LazyRustTy(Self::new_param_ty(0))))
757                }
758                "Seq" => Self::Builtin(RustBuiltinData::Seq(LazyRustTy(Self::new_param_ty(0)))),
759                "Map" => Self::Builtin(RustBuiltinData::Map(
760                    LazyRustTy(Self::new_param_ty(0)),
761                    LazyRustTy(Self::new_param_ty(1)),
762                )),
763                // `Ghost<T>` is encoded as if it were `struct Ghost<T>(T)`:
764                // the snapshot wraps the value of `T`.
765                "Ghost" => {
766                    let fields = vec![RustFieldData {
767                        name: symbol::Symbol::intern("val"),
768                        fid: abi::FieldIdx::from_usize(0),
769                        ty: LazyRustTy(Self::new_param_ty(0)),
770                        address: RustFieldAddress::Constant,
771                    }];
772                    TySpecifics::mk_structlike((), fields)
773                }
774                // TODO: support other builtins (e.g. `Seq`, `Map`, `Set`, etc.)
775                s => todo!("Unimplemented builtin {s}"),
776            }
777        } else {
778            match adt.adt_kind() {
779                ty::AdtKind::Struct => {
780                    let mut data = Self::from_struct(adt.non_enum_variant());
781                    if adt.is_box() {
782                        // The hardcoded value slot for the boxed `T` is
783                        // appended as a regular field; only its (heap-dependent)
784                        // field accessor and the pointer metadata are
785                        // special-cased (see the structlike encoder).
786                        data.fields.push(RustFieldData {
787                            name: symbol::Symbol::intern("value"),
788                            fid: abi::FieldIdx::from_usize(data.fields.len()),
789                            ty: LazyRustTy(Self::new_param_ty(0)),
790                            address: RustFieldAddress::Dynamic,
791                        });
792                    }
793                    Self::StructLike(data)
794                }
795                ty::AdtKind::Enum => {
796                    let data = Self::from_enum(adt);
797                    Self::EnumLike(data)
798                }
799                ty::AdtKind::Union => {
800                    // TODO: add union support
801                    Self::mk_opaque(())
802                }
803            }
804        }
805    }
806
807    fn from_struct(variant: &ty::VariantDef) -> StructData<'tcx, RustTyDatas> {
808        let fields = Self::from_fields(&variant.fields);
809        StructData::new((), fields)
810    }
811
812    fn from_enum(adt: ty::AdtDef<'tcx>) -> EnumData<'tcx, RustTyDatas> {
813        vir::with_vcx(|vcx| {
814            use ty::util::IntTypeExt;
815            let discr = adt.repr().discr_type().to_ty(vcx.tcx());
816            let data = RustEnumData { discr };
817            let variants = adt
818                .discriminants(vcx.tcx())
819                .map(|(vid, discr)| {
820                    let variant = adt.variant(vid);
821                    let fields = Self::from_fields(&variant.fields);
822                    VariantData::new(
823                        RustVariantData {
824                            name: variant.name,
825                            vid,
826                            discr_val: discr.val,
827                        },
828                        StructData::new((), fields),
829                    )
830                })
831                .collect::<Vec<_>>();
832            EnumData::new(data, variants)
833        })
834    }
835
836    fn from_fields(
837        fields: &index::IndexVec<abi::FieldIdx, ty::FieldDef>,
838    ) -> Vec<RustFieldData<'tcx>> {
839        fields
840            .iter_enumerated()
841            .map(|(fid, field)| {
842                let ty = vir::with_vcx(|vcx| vcx.tcx().type_of(field.did).instantiate_identity());
843                RustFieldData {
844                    name: field.name,
845                    fid,
846                    ty: LazyRustTy(ty.skip_normalization()),
847                    address: RustFieldAddress::Constant,
848                }
849            })
850            .collect::<Vec<_>>()
851    }

853    fn new_param_ty(index: u32) -> ty::Ty<'tcx> {
854        let name = match index {
855            0 => symbol::Symbol::intern("T"),
856            1 => symbol::Symbol::intern("U"),
857            2 => symbol::Symbol::intern("V"),
858            other => symbol::Symbol::intern(&format!("T{other}")),
859        };
860        vir::with_vcx(|vcx| ty::Ty::new_param(vcx.tcx(), index, name))
861    }
862
863    fn new_param_const(index: u32) -> ty::Const<'tcx> {
864        let name = match index {
865            0 => symbol::Symbol::intern("M"),
866            1 => symbol::Symbol::intern("N"),
867            other => symbol::Symbol::intern(&format!("N{other}")),
868        };
869        let param = ty::ParamConst { index, name };
870        vir::with_vcx(|vcx| ty::Const::new_param(vcx.tcx(), param))
871    }
