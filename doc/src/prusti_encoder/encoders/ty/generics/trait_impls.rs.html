26/// Encodes the behavioral-subtyping proof obligations of a trait impl:
27/// methods checking that each impl fn weakens the trait fn's precondition and
28/// strengthens its postcondition. Only run on local impls (foreign impls are
29/// trusted to conform, their conditions and axioms are encoded by
30/// [`TraitImplConditionEnc`]).
31pub struct TraitImplEnc;
32
33impl TaskEncoder for TraitImplEnc {
34    task_encoder::encoder_cache!(TraitImplEnc);
35    const ENCODER_NAME: &'static str = "trait impl encoder";
36
37    type TaskDescription<'vir> = DefId;
38    type OutputFullLocal<'vir> = Vec<Method<'vir>>;
39
40    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
41        *task
42    }
43
44    fn do_encode_full<'vir>(
45        task_key: &Self::TaskKey<'vir>,
46        deps: &mut TaskEncoderDependencies<'vir, Self>,
47    ) -> EncodeFullResult<'vir, Self> {
48        deps.emit_output_ref(*task_key, ())?;
49
50        vir::with_vcx(|vcx| {
51            let tcx = vcx.tcx();
52
53            let all_impls = tcx.trait_impls_in_crate(task_key.krate);
54            let idx = all_impls.iter().position(|did| did == task_key).unwrap();
55
56            let impl_context = GParams::from(*task_key);
57
58            let trait_ref = tcx.impl_trait_ref(*task_key).instantiate_identity();
59            let trait_did = trait_ref.skip_normalization().def_id;
60            let trait_data = deps.require_ref::<TraitEnc>(trait_did)?;
61            let trait_name = trait_data.trait_name;
62
63            let mut methods = Vec::new();
64
65            let implementing_ty = tcx
66                .type_of(*task_key)
67                .instantiate_identity()
68                .skip_normalization();
69            let implementing_ty = RustTyDecomposition::from_ty(implementing_ty, impl_context);
70            let implementing_ty = implementing_ty.ty.name();
71
72            for impl_item in tcx.associated_items(*task_key).in_definition_order() {
73                let ty::AssocKind::Fn { .. } = impl_item.kind else {
74                    continue;
75                };
76                let trait_item_def_id = impl_item.trait_item_def_id().unwrap();
77                let impl_item_def_id = impl_item.def_id;
78                let impl_span = vcx.tcx().def_span(impl_item_def_id);
79                let item_name = ViperIdent::from_def_id(vcx, impl_item_def_id);
80
81                let impl_item_context = GParams::from(impl_item_def_id);
82                let impl_item_params = deps.require_dep::<GenericParamsEnc>(impl_item_context)?;
83                let trait_ty_decls = impl_item_params.ty_decls();
84                let trait_const_decls = impl_item_params.const_decls();
85
86                let local_defs = deps.require_dep::<MirLocalDefEnc>(MirLocalDefEncTask::Local {
87                    def_id: impl_item_def_id,
88                    all_locals: false,
89                })?;
90                let arg_count = local_defs.arg_count + 1;
91                let ref_args = vcx.alloc_slice(&vec![vir::TYPE_REF; arg_count]);
92
93                let impl_item_is_pure = crate::encoders::is_function_pure(
94                    impl_item_def_id,
95                    GArgs::new(impl_item_context, impl_item_context.rust_params()),
96                );
97
98                let trait_item_spec = deps.require_dep_spanned::<MirSpecEnc>(
99                    (trait_item_def_id, impl_item_def_id, MirSpecEncMode::Impure),
100                    impl_span,
101                )?;
102                let impl_item_spec = deps.require_dep_spanned::<MirSpecEnc>(
103                    (impl_item_def_id, impl_item_def_id, MirSpecEncMode::Impure),
104                    impl_span,
105                )?;
106
107                let mut impure_arg_preds = Vec::new();
108                let mut ref_arg_decls = Vec::with_capacity(arg_count);
109                for arg_idx in (0..arg_count).map(mir::Local::from) {
110                    let name_p = local_defs[arg_idx].local.name;
111                    ref_arg_decls.push(vir::vir_local_decl! { vcx; [name_p] : Ref });
112                    if arg_idx != mir::RETURN_PLACE {
113                        impure_arg_preds.push(local_defs[arg_idx].impure_pred);
114                    }
115                }
116                // TODO: wands
117
118                let mut pre_weaken_pres = impure_arg_preds.clone();
119                pre_weaken_pres.extend(trait_item_spec.pre_exprs());
120
121                // Spans of the trait method's preconditions, to point the
122                // behavioral-subtyping error's hint at them.
123                let trait_pre_spans: Vec<_> =
124                    trait_item_spec.pres.iter().map(|(_, s)| *s).collect();
125                // If the trait method's preconditions were skipped, the impl's
126                // are not checked against them. The trait axioms then leave
127                // the trait's preconditions out too, so calls through the
128                // trait rely only on the impl's preconditions. This assumes
129                // that a trait precondition that cannot be encoded here (in
130                // impure mode) cannot be encoded for the trait axioms (in a
131                // pure mode) either, as the pure modes support less.
132                let impl_pres_to_check = if trait_item_spec.skipped {
133                    &[][..]
134                } else {
135                    &impl_item_spec.pres[..]
136                };
137
138                // Both checks are only needed where the impl applies.
139                let bounds = Self::assume_context_bounds(vcx, deps, impl_item_context)?;
140
141                methods.push(vcx.mk_method(
142                    MethodIdn::<(vir::ManyRef, vir::ManyTyVal, vir::ManyCSnap)>::new(
143                        vir_format_identifier!(vcx, "trait_{trait_name}_impl_{implementing_ty}_{idx}_fn_pre_weaken_{item_name}"),
144                        (ref_args, impl_item_params.ty_args(), impl_item_params.const_args()),
145                    ),
146                    (ref_arg_decls.as_slice(), trait_ty_decls, trait_const_decls),
147                    &[],
148                    vcx.alloc_slice(&pre_weaken_pres),
149                    &[],
150                    Some(vcx.alloc_slice(&[
151                        vcx.mk_cfg_block(
152                            &vir::CfgBlockLabelData::Start,
153                            &[],
154                            vcx.alloc_slice(&bounds.iter().copied().chain(impl_pres_to_check.iter().copied()
155                                .map(|(pre, pre_span)| vcx.with_span(pre_span, |vcx| {
156                                    let trait_pre_spans = trait_pre_spans.clone();
157                                    vcx.handle_error("exhale.failed:assertion.false", move |_| {
158                                        let mut err = PrustiError::verification(
159                                            "the implementation's precondition may be stronger than the trait method's",
160                                            pre_span.into(),
161                                        );
162                                        for trait_span in &trait_pre_spans {
163                                            err = err.add_note(
164                                                "the trait method's precondition is declared here",
165                                                Some(*trait_span),
166                                            );
167                                        }
168                                        Some(vec![err])
169                                    });
170                                    vcx.mk_exhale_stmt(pre)
171                                })))
172                                .collect::<Vec<_>>()),
173                            vcx.alloc(vir::TerminatorStmtData::Exit),
174                        )
175                    ])),
176                ));
177
178                let mut post_strengthen_pres = impure_arg_preds;
179                post_strengthen_pres.extend(trait_item_spec.pre_exprs());
180
181                // exceptionally, we also put the allocated result in the precondition
182                post_strengthen_pres.push(local_defs[mir::RETURN_PLACE].impure_pred);
183
184                // here we inhale the impl postconditions, since they
185                // can contain "old" variables
186                let mut stmts = bounds;
187                for post in impl_item_spec.post_exprs() {
188                    stmts.push(vcx.mk_inhale_stmt(post));
189                }
190                if impl_item_is_pure {
191                    let pure_func = deps.require_dep::<FunctionCallEnc>(
192                        CallTaskDescription::new(
193                            impl_item_def_id,
194                            impl_item_context.rust_params(),
195                            impl_item_def_id,
196                        )
197                        .resolve_trait_calls(false),
198                    )?;
199                    let pure_func_app = pure_func.call_pure(
200                        local_defs
201                            .args()
202                            .map(|arg| arg.impure_snap)
203                            .collect::<Vec<_>>(),
204                    );
205                    stmts.push(vcx.mk_inhale_stmt(vir::expr! {
206                        ([local_defs[mir::RETURN_PLACE].impure_snap]) == ([pure_func_app])
207                    }));
208                }
209                // The failing exhale is a trait postcondition, but the problem
210                // is the impl's postcondition being too weak: point the error at
211                // the impl's postcondition(s) and hint at the trait's.
212                let impl_post_spans: Vec<_> =
213                    impl_item_spec.posts.iter().map(|(_, s)| *s).collect();
214                for &(post, trait_post_span) in &trait_item_spec.posts {
215                    let impl_post_spans = impl_post_spans.clone();
216                    vcx.with_span(impl_span, |vcx| {
217                        vcx.handle_error("exhale.failed:assertion.false", move |_| {
218                            let primary = if impl_post_spans.is_empty() {
219                                impl_span.into()
220                            } else {
221                                MultiSpan::from_spans(impl_post_spans.clone())
222                            };
223                            let err = PrustiError::verification(
224                                "the implementation's postcondition may be weaker than the trait method's",
225                                primary,
226                            )
227                            .add_note(
228                                "the trait method's (stronger) postcondition is declared here",
229                                Some(trait_post_span),
230                            );
231                            Some(vec![err])
232                        });
233                        stmts.push(vcx.mk_exhale_stmt(post));
234                    });
235                }
236
237                methods.push(vcx.mk_method(
238                    MethodIdn::<(vir::ManyRef, vir::ManyTyVal, vir::ManyCSnap)>::new(
239                        vir_format_identifier!(vcx, "trait_{trait_name}_impl_{implementing_ty}_{idx}_fn_post_strengthen_{item_name}"),
240                        (ref_args, impl_item_params.ty_args(), impl_item_params.const_args()),
241                    ),
242                    (ref_arg_decls.as_slice(), trait_ty_decls, trait_const_decls),
243                    &[],
244                    vcx.alloc_slice(&post_strengthen_pres),
245                    &[],
246                    Some(vcx.alloc_slice(&[
247                        vcx.mk_cfg_block(
248                            &vir::CfgBlockLabelData::Start,
249                            &[],
250                            vcx.alloc_slice(&stmts),
251                            vcx.alloc(vir::TerminatorStmtData::Exit),
252                        )
253                    ])),
254                ));
255            }
256
257            // Make the impl visible to the trait's `impl_fun`.
258            if implements_trait(tcx, *task_key) {
259                deps.require_dep::<TraitImplConditionEnc>(*task_key)?;
260            }
261
262            Ok((methods, ()))
263        })
264    }
265
266    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
267        for methods in Self::all_outputs_local(program) {
268            for method in methods {
269                program.add_method(method);
270            }
271        }
272    }
273}

275/// Encodes the applicability condition of a trait impl: the axiom that the
276/// trait's `impl_fun` holds at the impl's trait ref wherever the impl's
277/// where-clauses do. The per-item assumable content (associated type
278/// resolution, fn-spec axioms) is encoded individually by
279/// [`TraitImplItemEnc`], guarded by the same where-clauses. Unlike
280/// [`TraitImplEnc`], this is safe to run on foreign impls: it produces no
281/// proof obligations, so foreign impls are assumed - not re-verified - to be
282/// behavioural subtypes.
283pub struct TraitImplConditionEnc;
284
285impl TaskEncoder for TraitImplConditionEnc {
286    task_encoder::encoder_cache!(TraitImplConditionEnc);
287    const ENCODER_NAME: &'static str = "trait impl condition encoder";
288
289    type TaskDescription<'vir> = DefId;
290    type OutputFullLocal<'vir> = Domain<'vir>;
291
292    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
293        *task
294    }
295
296    fn do_encode_full<'vir>(
297        task_key: &Self::TaskKey<'vir>,
298        deps: &mut TaskEncoderDependencies<'vir, Self>,
299    ) -> EncodeFullResult<'vir, Self> {
300        deps.emit_output_ref(*task_key, ())?;
301
302        vir::with_vcx(|vcx| {
303            let tcx = vcx.tcx();
304            let impl_name = impl_name(vcx, *task_key);
305
306            let impl_context = GParams::from(*task_key);
307            let trait_ref = tcx
308                .impl_trait_ref(*task_key)
309                .instantiate_identity()
310                .skip_normalization();
311            let trait_ = deps.require_ref::<TraitEnc>(trait_ref.def_id)?;
312            let args = deps.require_dep::<GArgsTyEnc>(GArgs::new(impl_context, trait_ref.args))?;
313            let impl_check = (trait_.impl_fun)(args.get_ty(), args.get_const());
314
315            // Triggering on the application at the impl's trait ref (rather
316            // than quantifying over the trait's parameters) keeps an impl's
317            // axiom from firing for applications it cannot match.
318            let axiom = TraitImplEnc::guarded_forall(
319                vcx,
320                deps,
321                *task_key,
322                *task_key,
323                &[],
324                impl_check.upcast_ty(),
325                impl_check,
326            )?;
327            let domain = vcx.mk_domain(
328                vir_format_identifier!(vcx, "trait_{impl_name}_condition"),
329                &[],
330                vcx.alloc_slice(&[vcx
331                    .mk_domain_axiom(vir_format_identifier!(vcx, "{impl_name}_condition"), axiom)]),
332                &[],
333                None,
334            );
335            Ok((domain, ()))
336        })
337    }
338
339    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
340        for domain in Self::all_outputs_local(program) {
341            program.add_domain(domain);
342        }
343    }
344}

346impl TraitImplEnc {
347    /// Whether every generic that `projection`'s left-hand side mentions is
348    /// already in `bound`.
349    fn projection_ready<'vir>(
350        projection: ty::ProjectionPredicate<'vir>,
351        bound: &FxIndexMap<u32, vir::ExprDyn<'vir>>,
352    ) -> bool {
353        projection
354            .projection_term
355            .args
356            .iter()
357            .flat_map(|arg| arg.walk())
358            .all(|arg| match arg.kind() {
359                ty::GenericArgKind::Type(ty) if let ty::TyKind::Param(p) = ty.kind() => {
360                    bound.contains_key(&p.index)
361                }
362                ty::GenericArgKind::Const(const_)
363                    if let ty::ConstKind::Param(p) = const_.kind() =>
364                {
365                    bound.contains_key(&p.index)
366                }
367                _ => true,
368            })
369    }
370
371    fn discover_bind_points<'vir, E: TaskEncoder + 'vir + ?Sized>(
372        deps: &mut TaskEncoderDependencies<'vir, E>,
373        generic_map: &mut FxIndexMap<u32, vir::ExprDyn<'vir>>,
374        ctx: GParams<'vir>,
375        expr: vir::ExprTyVal<'vir>,
376        ty: ty::Ty<'vir>,
377    ) -> Result<(), EncodeFullError<'vir, E>> {
378        if let ty::TyKind::Param(p) = ty.kind() {
379            generic_map.entry(p.index).or_insert(expr.upcast_ty());
380            return Ok(());
381        }
382        if matches!(ty.kind(), ty::TyKind::Alias(..)) {
383            // Nothing can be bound through an alias, e.g. an unresolved
384            // associated-type projection whose self-type is still generic.
385            return Ok(());
386        }
387
388        let decomp = RustTyDecomposition::from_ty(ty, ctx);
389        let ty_enc = deps.require_ref::<TyConstructorEnc>(decomp.ty)?;
390
391        let args = decomp.args.args();
392        let inner_types = args.iter().filter_map(|arg| arg.as_type());
393        for (i, inner_ty) in inner_types.enumerate() {
394            let accessor = ty_enc.ty_param_accessors[i];
395            let inner_expr = accessor.call()(expr);
396
397            Self::discover_bind_points(deps, generic_map, ctx, inner_expr, inner_ty)?;
398        }
399
400        let inner_consts = args.iter().filter_map(|arg| arg.as_const());
401        for (i, inner_const) in inner_consts.enumerate() {
402            let accessor = ty_enc.const_param_accessors[i];
403            let inner_expr = accessor.call()(expr);
404
405            if let ty::ConstKind::Param(p) = inner_const.kind() {
406                generic_map.entry(p.index).or_insert(inner_expr.upcast_ty());
407            }
408        }
409        Ok(())
410    }

412    /// Assumes the where-clauses in force in `ctx`. rustc guarantees them at
413    /// every instantiation, and resolving through an impl (its associated
414    /// types and fn specs are guarded by its where-clauses) may depend on
415    /// them. Assumed rather than required, since callers could not always
416    /// discharge them (e.g. `Fn*` bounds, which have no encoded impls).
417    pub(crate) fn assume_context_bounds<'vir, E: TaskEncoder + 'vir + ?Sized>(
418        vcx: &'vir vir::VirCtxt<'vir>,
419        deps: &mut TaskEncoderDependencies<'vir, E>,
420        ctx: GParams<'vir>,
421    ) -> Result<Vec<vir::Stmt<'vir>>, EncodeFullError<'vir, E>> {
422        Ok(Self::context_bounds(vcx, deps, ctx, None)?
423            .into_iter()
424            .map(|bound| vcx.mk_inhale_stmt(bound))
425            .collect())
426    }
427
428    /// The where-clauses in force in `ctx`, stated over its parameters.
429    ///
430    /// With `bind_points`, the projection bounds are processed in an order in
431    /// which every generic that a projection's left-hand side mentions is
432    /// already bound; its right-hand side (e.g. `<J as B>::Y`) may mention
433    /// generics bound later. Any parameter a projection determines is added to
434    /// `bind_points` (the parameters initially in `bind_points` count as bound).
435    pub(crate) fn context_bounds<'vir, E: TaskEncoder + 'vir + ?Sized>(
436        vcx: &'vir vir::VirCtxt<'vir>,
437        deps: &mut TaskEncoderDependencies<'vir, E>,
438        ctx: GParams<'vir>,
439        bind_points: Option<&mut FxIndexMap<u32, vir::ExprDyn<'vir>>>,
440    ) -> Result<Vec<vir::ExprBool<'vir>>, EncodeFullError<'vir, E>> {
441        let caller_bounds = ctx.typing_env().param_env.caller_bounds();
442        Self::clause_bounds(vcx, deps, ctx, caller_bounds, bind_points)
443    }
444
445    /// Like [`Self::context_bounds`], for the given clauses.
446    pub(crate) fn clause_bounds<'vir, E: TaskEncoder + 'vir + ?Sized>(
447        vcx: &'vir vir::VirCtxt<'vir>,
448        deps: &mut TaskEncoderDependencies<'vir, E>,
449        ctx: GParams<'vir>,
450        clauses: ty::Clauses<'vir>,
451        mut bind_points: Option<&mut FxIndexMap<u32, vir::ExprDyn<'vir>>>,
452    ) -> Result<Vec<vir::ExprBool<'vir>>, EncodeFullError<'vir, E>> {
453        let tcx = vcx.tcx();
454        let params = deps.require_dep::<GenericParamsEnc>(ctx)?;
455        let mut checks = Vec::new();
456
457        let trait_preds = clauses
458            .iter()
459            .filter_map(ty::Clause::as_trait_clause)
460            .map(ty::Binder::skip_binder)
461            .filter(|pred| pred.polarity == ty::PredicatePolarity::Positive)
462            // Holds of every type (see `BuiltinImplEnc`).
463            .filter(|pred| Some(pred.def_id()) != tcx.lang_items().pointee_sized_trait());
464        for trait_pred in trait_preds {
465            let trait_ = deps.require_ref::<TraitEnc>(trait_pred.def_id())?;
466            let gargs = GArgs::new(ctx, trait_pred.trait_ref.args);
467            let gargs = deps.require_dep::<GArgsTyEnc>(gargs)?;
468            checks.push((trait_.impl_fun)(gargs.get_ty(), gargs.get_const()));
469        }
470
471        let mut proj_preds: Vec<_> = clauses
472            .iter()
473            .filter_map(ty::Clause::as_projection_clause)
474            .map(ty::Binder::skip_binder)
475            .collect();
476        while !proj_preds.is_empty() {
477            let next = match &bind_points {
478                Some(bound) => proj_preds
479                    .iter()
480                    .position(|&p| Self::projection_ready(p, bound)),
481                None => Some(0),
482            };
483            let Some(next) = next else {
484                return Err(EncodeFullError::unsupported(
485                    format!(
486                        "these projection bounds mention generics that are never bound: {}",
487                        proj_preds
488                            .iter()
489                            .map(ToString::to_string)
490                            .collect::<Vec<_>>()
491                            .join(", ")
492                    ),
493                    vec![],
494                ));
495            };
496            let proj_pred = proj_preds.remove(next);
497            let trait_did = proj_pred.trait_def_id(tcx);
498            let trait_ = deps.require_ref::<TraitEnc>(trait_did)?;
499            let gargs = GArgs::new(ctx, proj_pred.projection_term.args);
500            let gargs = deps.require_dep::<GArgsTyEnc>(gargs)?;
501
502            let (projection, expr): (vir::ExprDyn, vir::ExprDyn) = match proj_pred.term.kind() {
503                ty::TermKind::Ty(ty) => {
504                    let projection =
505                        trait_.assoc_types[&proj_pred.def_id()](gargs.get_ty(), gargs.get_const());
506                    let decomp = RustTyDecomposition::from_ty(ty, ctx);
507                    let ty_expr = params.ty_expr(deps, decomp);
508                    if let Some(bound) = bind_points.as_deref_mut() {
509                        Self::discover_bind_points(deps, bound, ctx, projection, ty)?;
510                    }
511                    (projection.upcast_ty(), ty_expr?.upcast_ty())
512                }
513                ty::TermKind::Const(const_) => {
514                    let projection =
515                        trait_.assoc_consts[&proj_pred.def_id()](gargs.get_ty(), gargs.get_const());
516                    let ty = tcx
517                        .type_of(proj_pred.def_id())
518                        .instantiate_identity()
519                        .skip_normalization();
520                    let const_task = ConstEncTask::Ty {
521                        const_,
522                        ty,
523                        context: ctx,
524                    };
525                    let const_expr = deps.require_dep::<ConstEnc>(const_task)?;
526                    if let Some(bound) = bind_points.as_deref_mut()
527                        && let ty::ConstKind::Param(p) = const_.kind()
528                    {
529                        bound.entry(p.index).or_insert(const_expr.upcast_ty());
530                    }
531                    (projection.upcast_ty(), const_expr.upcast_ty())
532                }
533            };
534
535            checks.push(vcx.mk_eq_expr(projection, expr));
536        }
537
538        Ok(checks)
539    }

541    /// `forall params :: {trigger} impl_bounds ==> body`: an assertion about
542    /// the impl `impl_did` that only holds where the impl applies, i.e. where
543    /// its where-clauses hold.
544    ///
545    /// Quantifies over the parameters of `item_did` (the impl itself or one
546    /// of its items, whose parameters extend the impl's) and `extra`. The
547    /// trigger is applied to the impl's trait ref followed by the item's own
548    /// parameters; an impl parameter not occurring in those is instead
549    /// let-bound to the projection that determines it: rustc only accepts
550    /// such an impl parameter if a projection bound constrains it (E0207),
551    /// and binding it keeps every quantified variable covered by the
552    /// trigger. If the trigger mentions it nonetheless (through `extra` or
553    /// `trigger`, e.g. in an item's signature), it is quantified and
554    /// constrained to equal the projection instead.
555    pub(super) fn guarded_forall<'vir, E: TaskEncoder + 'vir + ?Sized>(
556        vcx: &'vir vir::VirCtxt<'vir>,
557        deps: &mut TaskEncoderDependencies<'vir, E>,
558        impl_did: DefId,
559        item_did: DefId,
560        extra: &[vir::LocalDeclDyn<'vir>],
561        trigger: vir::ExprDyn<'vir>,
562        body: vir::ExprBool<'vir>,
563    ) -> Result<vir::ExprBool<'vir>, EncodeFullError<'vir, E>> {
564        Self::guarded_forall_in(
565            vcx,
566            deps,
567            impl_did,
568            GParams::from(impl_did),
569            GParams::from(item_did),
570            extra,
571            trigger,
572            body,
573        )
574    }
575
576    /// [`Self::guarded_forall`] with the contexts of the impl and the item
577    /// given explicitly, e.g. suffixed so that their parameters cannot capture
578    /// variables of `body`.
579    #[allow(clippy::too_many_arguments)]
580    pub(super) fn guarded_forall_in<'vir, E: TaskEncoder + 'vir + ?Sized>(
581        vcx: &'vir vir::VirCtxt<'vir>,
582        deps: &mut TaskEncoderDependencies<'vir, E>,
583        impl_did: DefId,
584        impl_ctx: GParams<'vir>,
585        item_ctx: GParams<'vir>,
586        extra: &[vir::LocalDeclDyn<'vir>],
587        trigger: vir::ExprDyn<'vir>,
588        body: vir::ExprBool<'vir>,
589    ) -> Result<vir::ExprBool<'vir>, EncodeFullError<'vir, E>> {
590        let item_params = deps.require_dep::<GenericParamsEnc>(item_ctx)?;
591        let decl = |idx: u32| match item_params.map_idx(idx) {
592            Ok(idx) => item_params.ty_decls()[idx].upcast_ty(),
593            Err(idx) => item_params.const_decls()[idx].upcast_ty(),
594        };
595
596        let trait_ref = vcx
597            .tcx()
598            .impl_trait_ref(impl_did)
599            .instantiate_identity()
600            .skip_normalization();
601        let pinned = item_ctx
602            .rust_params()
603            .rebase_onto(vcx.tcx(), impl_did, trait_ref.args);
604
605        let mut bind_points = FxIndexMap::default();
606        for arg in pinned.iter().flat_map(|arg| arg.walk()) {
607            let idx = match arg.kind() {
608                ty::GenericArgKind::Type(ty) if let ty::TyKind::Param(p) = ty.kind() => p.index,
609                ty::GenericArgKind::Const(c) if let ty::ConstKind::Param(p) = c.kind() => p.index,
610                _ => continue,
611            };
612            bind_points
613                .entry(idx)
614                .or_insert_with(|| vcx.mk_local_ex(decl(idx)).upcast_ty());
615        }
616        let pinned_count = bind_points.len();
617
618        let bounds = Self::context_bounds(vcx, deps, impl_ctx, Some(&mut bind_points))?;
619        let body = if bounds.is_empty() {
620            body
621        } else {
622            let guard = vcx.mk_conj(&bounds);
623            vir::expr! { (guard) ==> (body) }
624        };
625
626        // A parameter determined by a projection that the trigger mentions
627        // (e.g. `U` of `F: FnMut(T) -> U` in an item's signature) cannot be
628        // let-bound, since the trigger would refer to a variable bound only
629        // in the body. It is quantified instead, constrained to equal the
630        // projection inside the remaining lets.
631        let mut trigger_locals = FxHashSet::default();
632        vir::collect_locals(trigger, &mut trigger_locals);
633        let (constrained, lets): (Vec<_>, Vec<_>) = bind_points
634            .split_off(pinned_count)
635            .into_iter()
636            .partition(|&(idx, _)| trigger_locals.contains(decl(idx).name));
637        let body = if constrained.is_empty() {
638            body
639        } else {
640            let eqs = constrained
641                .iter()
642                .map(|&(idx, expr)| vcx.mk_eq_expr(vcx.mk_local_ex(decl(idx)).upcast_ty(), expr))
643                .collect::<Vec<_>>();
644            let eqs = vcx.mk_conj(&eqs);
645            vir::expr! { (eqs) ==> (body) }
646        };
647        let body = lets.iter().rfold(body, |acc, &(idx, expr)| {
648            vcx.mk_let_expr(decl(idx), expr, acc)
649        });
650
651        let qvars = (0..item_ctx.rust_params().len() as u32)
652            .filter(|&idx| item_ctx.rust_params()[idx as usize].as_region().is_none())
653            .filter(|idx| !lets.iter().any(|(let_idx, _)| let_idx == idx))
654            .map(decl)
655            .chain(extra.iter().copied())
656            .collect::<Vec<_>>();
657        // Known matching loop: an impl forwarding to a projection of its own
658        // parameter, such as rand_core's
659        //
660        //     impl<R: DerefMut> TryRng for R where R::Target: TryRng {
661        //         type Error = <R::Target as TryRng>::Error;
662        //     }
663        //
664        // gets `{TryRng_impl(R)}` with the guard `TryRng_impl(Target(R))`,
665        // and `{Error(R)}` with the equation `Error(R) == Error(Target(R))`.
666        // Each instance creates a term matching its trigger again with
667        // `R := Target(R)`, and `Target` never resolves at a type without a
668        // `Deref` impl. Z3 defers such deep instances. Requiring `Target(R)`
669        // in the trigger would stop the chain, but also stop resolving
670        // `<&mut S as TryRng>::Error` where only `Error(&mut S)` occurs.
671        Ok(vcx.mk_forall_expr(
672            vcx.alloc_slice(&qvars),
673            vcx.alloc_slice(&[vcx.mk_trigger(&[trigger])]),
674            body,
675        ))
676    }
677
678    /// Whether the impl applies at the trait ref with the given arguments:
679    /// they are the impl's trait ref at some instantiation of its parameters,
680    /// and its where-clauses hold there. Unlike [`Self::guarded_forall`],
681    /// which quantifies over the impl's parameters, this is stated over the
682    /// arguments; the impl's parameters are let-bound to the subterms of the
683    /// arguments they occur in (in a suffixed context, so that they cannot
684    /// capture the arguments' variables), or to the projection determining
685    /// them.
686    pub(super) fn applies_at<'vir, E: TaskEncoder + 'vir + ?Sized>(
687        vcx: &'vir vir::VirCtxt<'vir>,
688        deps: &mut TaskEncoderDependencies<'vir, E>,
689        impl_did: DefId,
690        trait_tys: &[vir::ExprTyVal<'vir>],
691        trait_consts: &[vir::ExprCSnap<'vir>],
692    ) -> Result<vir::ExprBool<'vir>, EncodeFullError<'vir, E>> {
693        let impl_ctx = GParams::from(impl_did).with_suffix("impl");
694        let impl_params = deps.require_dep::<GenericParamsEnc>(impl_ctx)?;
695        let trait_ref = vcx
696            .tcx()
697            .impl_trait_ref(impl_did)
698            .instantiate_identity()
699            .skip_normalization();
700
701        let mut bind_points = FxIndexMap::default();
702        for (&expr, ty) in std::iter::zip(trait_tys, trait_ref.args.types()) {
703            Self::discover_bind_points(deps, &mut bind_points, impl_ctx, expr, ty)?;
704        }
705        for (&expr, const_) in std::iter::zip(trait_consts, trait_ref.args.consts()) {
706            if let ty::ConstKind::Param(p) = const_.kind() {
707                bind_points.entry(p.index).or_insert(expr.upcast_ty());
708            }
709        }
710
711        let args = deps.require_dep::<GArgsTyEnc>(GArgs::new(impl_ctx, trait_ref.args))?;
712        let mut checks = Vec::new();
713        for (&arg, &impl_arg) in std::iter::zip(trait_tys, args.get_ty()) {
714            checks.push(vcx.mk_eq_expr(arg, impl_arg));
715        }
716        for (&arg, &impl_arg) in std::iter::zip(trait_consts, args.get_const()) {
717            checks.push(vcx.mk_eq_expr(arg, impl_arg));
718        }
719        checks.extend(Self::context_bounds(
720            vcx,
721            deps,
722            impl_ctx,
723            Some(&mut bind_points),
724        )?);
725
726        let checks = vcx.mk_conj(&checks);
727        Ok(bind_points.iter().rfold(checks, |acc, (&idx, &expr)| {
728            let decl = match impl_params.map_idx(idx) {
729                Ok(idx) => impl_params.ty_decls()[idx].upcast_ty(),
730                Err(idx) => impl_params.const_decls()[idx].upcast_ty(),
731            };
732            vcx.mk_let_expr(decl, expr, acc)
733        }))
734    }
735}

737/// The impls of a trait that make it hold (see [`implements_trait`]).
738pub(super) fn implementing_impls(
739    tcx: ty::TyCtxt<'_>,
740    trait_did: DefId,
741) -> impl Iterator<Item = DefId> {
742    tcx.all_impls(trait_did)
743        .filter(move |&impl_did| implements_trait(tcx, impl_did))
744}
745
746/// Whether the impl makes its trait ref hold, i.e. whether its condition
747/// and items may be assumed. Negative impls (`impl !Tr for X`) and
748/// reservation impls (`#[rustc_reservation_impl]`) are not used by trait
749/// selection, and a partial `default impl` only provides items for the impls
750/// specializing it.
751pub(super) fn implements_trait(tcx: ty::TyCtxt<'_>, impl_did: DefId) -> bool {
752    tcx.impl_polarity(impl_did) == ty::ImplPolarity::Positive
753        && !tcx.defaultness(impl_did).is_default()
754}
755
756/// The definition of `trait_item` that Rust uses wherever the impl applies,
757/// if it is final: the impl's own item, or one it inherits from an impl it
758/// specializes or from the trait (an inherited item cannot be specialized
759/// further, E0520). A `default` item may be overridden by a specializing
760/// impl, possibly in a crate this one cannot see; like rustc, which only
761/// resolves such an item once all types are known, it stays opaque.
762///
763/// Only for impls that implement the trait: `leaf_def` counts an item that
764/// the starting impl does not mention as finalized by it, which does not
765/// hold for a partial `default impl`.
766pub(super) fn final_leaf_def(
767    tcx: ty::TyCtxt<'_>,
768    impl_did: DefId,
769    trait_item: DefId,
770) -> Option<specialization_graph::LeafDef> {
771    debug_assert!(implements_trait(tcx, impl_did));
772    let trait_did = tcx.impl_trait_ref(impl_did).skip_binder().def_id;
773    specialization_graph::ancestors(tcx, trait_did, impl_did)
774        .ok()?
775        .leaf_def(tcx, trait_item)
776        .filter(specialization_graph::LeafDef::is_final)
777}

779/// The Viper name of an impl. `idx` is only unique within a crate, so foreign
780/// impls need the crate name for disambiguation.
781fn impl_name<'vir>(vcx: &'vir vir::VirCtxt<'vir>, impl_did: DefId) -> &'vir str {
782    let tcx = vcx.tcx();
783    let all_impls = tcx.trait_impls_in_crate(impl_did.krate);
784    let idx = all_impls.iter().position(|did| *did == impl_did).unwrap();
785    let krate = tcx.crate_name(impl_did.krate);
786    let trait_did = tcx.impl_trait_ref(impl_did).skip_binder().def_id;
787    let trait_name = ViperIdent::from_def_id(vcx, trait_did);
788    let implementing_ty = tcx
789        .type_of(impl_did)
790        .instantiate_identity()
791        .skip_normalization();
792    let implementing_ty = RustTyDecomposition::from_ty(implementing_ty, GParams::from(impl_did));
793    let implementing_ty = implementing_ty.ty.name();
794    vir::vir_format!(vcx, "{trait_name}_impl_{krate}_{implementing_ty}_{idx}")
795}

797/// The constructor keys gating the inclusion of an impl's content. An
798/// `impl<T> MyTrait<ArgType> for MyType<T, OtherType>` is only relevant if
799/// a) it is in the current crate (in which case it will be encoded by
800/// [`TraitImplEnc`]) or b) the `TyConstructorEnc` has been called with
801/// `MyType`, `OtherType` and `ArgType` (callers guard on the conjunction of
802/// the returned keys). Gating on the trait args (not just the self type) is
803/// fail-closed: any site that states a dependence on this impl does so by
804/// applying `impl_fun` to type expressions for the full trait ref, and
805/// building those requests exactly the constructors gated on here. Without
806/// the arg gate, encoding one impl's condition constructs its arg types as a
807/// side effect, unlocking further impls transitively (e.g. `Array` alone
808/// would pull in every `core::arch` <-> `Simd` `From` impl).
809pub(super) fn impl_unlock_keys<'vir>(impl_did: DefId) -> Vec<RustTy<'vir>> {
810    vir::with_vcx(|vcx| {
811        let tcx = vcx.tcx();
812        let impl_trait_ref = tcx
813            .impl_trait_ref(impl_did)
814            .instantiate_identity()
815            .skip_normalization();
816        fn collect_ctor_keys<'vir>(ty: ty::Ty<'vir>, ctx: DefId, out: &mut Vec<RustTy<'vir>>) {
817            let decomp = RustTyDecomposition::from_ty(ty, ctx);
818            if decomp.ty.specifics.is_param() {
819                return;
820            }
821            out.push(decomp.ty);
822            for inner in decomp.args.args().iter().filter_map(|arg| arg.as_type()) {
823                collect_ctor_keys(inner, ctx, out);
824            }
825        }
826        let mut keys = Vec::new();
827        for ty in impl_trait_ref.args.iter().filter_map(|arg| arg.as_type()) {
828            collect_ctor_keys(ty, impl_did, &mut keys);
829        }
830        keys
831    })
832}

834/// Encodes the assumable content of a single trait-impl item, each pulled in
835/// individually and only when needed. For an associated fn: the axioms
836/// bridging the trait's abstract pre/post functions (see `TraitFnEnc`) to
837/// this item's concrete specification, instantiated at the impl's trait ref
838/// (triggered per called function by `TraitFnEnc`). For an associated type:
839/// the axiom resolving the trait's type function to the impl's concrete type
840/// (triggered alongside the impl condition by `TraitEnc`).
841///
842/// An item that the impl inherits from an impl it specializes is encoded in
843/// the context of the latter, additionally guarded by the inheriting impl
844/// applying (see [`TraitImplEnc::applies_at`]).
845pub struct TraitImplItemEnc;
846
847impl TaskEncoder for TraitImplItemEnc {
848    task_encoder::encoder_cache!(TraitImplItemEnc);
849    const ENCODER_NAME: &'static str = "trait impl item encoder";
850
851    /// The impl and its final definition of an item (see [`final_leaf_def`]):
852    /// its own item, or one it inherits from an impl it specializes.
853    type TaskDescription<'vir> = (DefId, DefId);
854    type OutputFullLocal<'vir> = Domain<'vir>;
855
856    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
857        *task
858    }
859
860    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
861        for domain in Self::all_outputs_local(program) {
862            program.add_domain(domain);
863        }
864    }
865
866    fn do_encode_full<'vir>(
867        task_key: &Self::TaskKey<'vir>,
868        deps: &mut TaskEncoderDependencies<'vir, Self>,
869    ) -> EncodeFullResult<'vir, Self> {
870        deps.emit_output_ref(*task_key, ())?;
871
872        vir::with_vcx(|vcx| {
873            let tcx = vcx.tcx();
874
875            let (applying_impl, impl_item_def_id) = *task_key;
876            let impl_did = tcx.impl_of_assoc(impl_item_def_id).unwrap();
877            let impl_item = tcx.associated_item(impl_item_def_id);
878            let trait_item_def_id = impl_item.trait_item_def_id().unwrap();
879            let impl_span = tcx.def_span(impl_item_def_id);
880            let item_name = ViperIdent::from_def_id(vcx, impl_item_def_id);
881            let impl_name = impl_name(vcx, applying_impl);
882
883            let impl_context = GParams::from(impl_did);
884            let impl_params = deps.require_dep::<GenericParamsEnc>(impl_context)?;
885            let trait_ref = tcx
886                .impl_trait_ref(impl_did)
887                .instantiate_identity()
888                .skip_normalization();
889
890            let impl_item_context = GParams::from(impl_item_def_id);
891            let impl_item_params = deps.require_dep::<GenericParamsEnc>(impl_item_context)?;
892
893            // The ty and const decls of the trait items are the decls of the
894            // item itself prefixed by the decls of the impl itself (so the
895            // impl's where-clauses can guard the item's axioms).
896            assert_eq!(
897                impl_params.ty_decls(),
898                &impl_item_params.ty_decls()[..impl_params.ty_decls().len()]
899            );
900            assert_eq!(
901                impl_params.const_decls(),
902                &impl_item_params.const_decls()[..impl_params.const_decls().len()]
903            );
904
905            // Combine the args to the trait in the impl and the identity args
906            // for the item itself. That is, for:
907            // ```
908            // trait MyTrait<'a, A> { fn foo<'b, B>() {} }
909            // impl<T> MyTrait<'static, (T, bool)> for MyType {
910            //     fn foo<'b, B>() {}
911            // }
912            // ```
913            // The `impl_item_context` of `foo` is `<T, 'b, B>` and
914            // `impl_context` is `<T>`. We take the suffix of the impl item's
915            // params that are specific to the item (i.e. not inherited from
916            // the impl) and combine them with the trait args to get:
917            // `<MyType, 'static, (T, bool), 'b, B>`. We use
918            // `impl_item_context` rather than the trait item's context because
919            // the parameter indices must match the `impl_item_context` used in
920            // `GArgs::new` below.
921            let trait_item_context = GParams::from(trait_item_def_id);
922            let args = impl_item_context
923                .rust_params()
924                .rebase_onto(tcx, impl_did, trait_ref.args);
925            let impl_item_args = GArgs::new(impl_item_context, args);
926            let args = deps.require_dep::<GArgsTyEnc>(impl_item_args)?;
927
928            let trait_tys = args.get_ty();
929            let trait_consts = args.get_const();
930
931            // An inherited item describes only the inheriting impl's trait
932            // refs: other impls specializing the defining one may override it.
933            let inherited_guard = if applying_impl == impl_did {
934                None
935            } else {
936                Some(TraitImplEnc::applies_at(
937                    vcx,
938                    deps,
939                    applying_impl,
940                    trait_tys,
941                    trait_consts,
942                )?)
943            };
944            let guard_inherited = |body: vir::ExprBool<'vir>| match inherited_guard {
945                Some(applies) => vir::expr! { (applies) ==> (body) },
946                None => body,
947            };
948
949            let mut axioms = Vec::new();
950
951            match impl_item.kind {
952                ty::AssocKind::Type { .. } => {
953                    let trait_data = deps.require_ref::<TraitEnc>(trait_ref.def_id)?;
954                    let assoc_type = trait_data.assoc_types[&trait_item_def_id];
955
956                    // the type we want to resolve the type alias to
957                    let assoc_type_expr = impl_item_params.ty_expr(
958                        deps,
959                        RustTyDecomposition::from_ty(
960                            tcx.type_of(impl_item_def_id)
961                                .instantiate_identity()
962                                .skip_normalization(),
963                            impl_item_context,
964                        ),
965                    )?;
966                    // Guarded by the impl's where-clauses: a blanket impl and
967                    // a more specific one that Rust keeps apart by them would
968                    // otherwise resolve the same projection to two distinct
969                    // types, which is false and makes everything provable.
970                    let projection = assoc_type(trait_tys, trait_consts);
971                    let equation = vcx.mk_eq_expr(projection, assoc_type_expr);
972                    axioms.push(vcx.mk_domain_axiom(
973                        vir_format_identifier!(vcx, "{impl_name}_assoc_type_{item_name}"),
974                        TraitImplEnc::guarded_forall(
975                            vcx,
976                            deps,
977                            impl_did,
978                            impl_item_def_id,
979                            &[],
980                            projection.upcast_ty(),
981                            guard_inherited(equation),
982                        )?,
983                    ));
984                }
985                ty::AssocKind::Fn { .. } => {
986                    let assoc_fn = deps.require_ref::<TraitFnEnc>(trait_item_def_id)?;
987                    let local_defs =
988                        deps.require_dep::<MirLocalDefEnc>(MirLocalDefEncTask::Local {
989                            def_id: impl_item_def_id,
990                            all_locals: false,
991                        })?;
992                    let func_args = local_defs.local_decl_args().collect::<Vec<_>>();
993                    let func_ret = local_defs.local_decl_ret();
994
995                    let impl_item_is_pure = crate::encoders::is_function_pure(
996                        impl_item_def_id,
997                        GArgs::new(impl_item_context, impl_item_context.rust_params()),
998                    );
999
1000                    let impl_item_spec = deps.require_dep_spanned::<MirSpecEnc>(
1001                        (
1002                            impl_item_def_id,
1003                            impl_item_def_id,
1004                            MirSpecEncMode::PureWithoutResult,
1005                        ),
1006                        impl_span,
1007                    )?;
1008                    let pres = vcx.mk_conj(&impl_item_spec.pre_exprs().collect::<Vec<_>>());
1009
1010                    let signature = RustSignature::new(trait_item_def_id);
1011
1012                    // TODO: clean up: this kind of casting also happens in
1013                    //   `FunctionCallEncOutput::call_pure`.
1014                    let casted_args = func_args
1015                        .iter()
1016                        .zip(signature.inputs)
1017                        .map(|(arg, ty)| {
1018                            let normalized =
1019                                ty.decompose_compare_normalize(trait_item_context, impl_item_args);
1020                            let caster =
1021                                deps.require_dep::<GArgsCastEnc<Pure>>(normalized).unwrap();
1022                            caster.cast_to_callee_ctx(vcx.mk_local_ex(arg))
1023                        })
1024                        .collect::<Vec<_>>();
1025                    let casted_args_slice = vcx.alloc_slice(&casted_args);
1026                    let pre_func_call =
1027                        assoc_fn.pre_func.call()(casted_args_slice, trait_tys, trait_consts);
1028                    let arg_decls = func_args
1029                        .iter()
1030                        .map(|arg| arg.upcast_ty())
1031                        .collect::<Vec<vir::LocalDeclDyn>>();
1032                    // Skipped preconditions establish nothing: the impl's
1033                    // method may still assume them.
1034                    if !impl_item_spec.skipped {
1035                        axioms.push(vcx.mk_domain_axiom(
1036                            vir_format_identifier!(vcx, "{impl_name}_fn_pre_{item_name}"),
1037                            TraitImplEnc::guarded_forall(
1038                                vcx,
1039                                deps,
1040                                impl_did,
1041                                impl_item_def_id,
1042                                &arg_decls,
1043                                pre_func_call.upcast_ty(),
1044                                guard_inherited(vir::expr! { (pres) ==> (pre_func_call) }),
1045                            )?,
1046                        ));
1047                    }
1048                    let mut posts = impl_item_spec.post_exprs().collect::<Vec<_>>();
1049                    if impl_item_is_pure {
1050                        let pure_func = deps.require_dep::<FunctionCallEnc>(
1051                            CallTaskDescription::new(
1052                                impl_item_def_id,
1053                                impl_item_context.rust_params(),
1054                                impl_item_def_id,
1055                            )
1056                            .resolve_trait_calls(false),
1057                        )?;
1058                        let raw_args = func_args
1059                            .iter()
1060                            .map(|arg| vcx.mk_local_ex(arg))
1061                            .collect::<Vec<_>>();
1062                        let pure_func_app = pure_func.call_pure(raw_args);
1063                        posts.push(vir::expr! {
1064                            ([func_ret]) == ([pure_func_app])
1065                        });
1066                    }
1067                    let posts = vcx.mk_conj(&posts);
1068                    let post_func_call = assoc_fn.post_func.call()(
1069                        {
1070                            let normalized = signature
1071                                .output
1072                                .decompose_compare_normalize(trait_item_context, impl_item_args);
1073                            let caster =
1074                                deps.require_dep::<GArgsCastEnc<Pure>>(normalized).unwrap();
1075                            caster.cast_to_callee_ctx(vcx.mk_local_ex(func_ret))
1076                        },
1077                        casted_args_slice,
1078                        trait_tys,
1079                        trait_consts,
1080                    );
1081                    let ret_and_arg_decls = std::iter::once(func_ret.upcast_ty())
1082                        .chain(arg_decls)
1083                        .collect::<Vec<_>>();
1084                    axioms.push(vcx.mk_domain_axiom(
1085                        vir_format_identifier!(vcx, "{impl_name}_fn_post_{item_name}"),
1086                        TraitImplEnc::guarded_forall(
1087                            vcx,
1088                            deps,
1089                            impl_did,
1090                            impl_item_def_id,
1091                            &ret_and_arg_decls,
1092                            post_func_call.upcast_ty(),
1093                            guard_inherited(vir::expr! { (post_func_call) ==> (posts) }),
1094                        )?,
1095                    ));
1096                }
1097                ty::AssocKind::Const { .. } => (),
1098            }
1099
1100            let domain = vcx.mk_domain(
1101                vir_format_identifier!(vcx, "trait_{impl_name}_{item_name}"),
1102                &[],
1103                vcx.alloc_slice(&axioms),
1104                &[],
1105                None,
1106            );
1107            Ok((domain, ()))
1108        })
1109    }
1110}

1112/// Encodes, for an impl that inherits the default body of a pure trait fn,
1113/// the axiom that at the impl's trait refs the fn returns what the default
1114/// body does. The default body is not part of the trait's contract (impls
1115/// may override it with a different result, see `TraitFnEnc`), so it is
1116/// stated per inheriting impl, like the fn items of overriding impls (see
1117/// [`TraitImplItemEnc`]).
1118pub struct TraitImplDefaultFnEnc;
1119
1120impl TaskEncoder for TraitImplDefaultFnEnc {
1121    task_encoder::encoder_cache!(TraitImplDefaultFnEnc);
1122    const ENCODER_NAME: &'static str = "trait impl default fn encoder";
1123
1124    /// The impl and the trait fn whose default body it inherits.
1125    type TaskDescription<'vir> = (DefId, DefId);
1126    type OutputFullLocal<'vir> = Domain<'vir>;
1127
1128    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
1129        *task
1130    }
1131
1132    fn do_encode_full<'vir>(
1133        task_key: &Self::TaskKey<'vir>,
1134        deps: &mut TaskEncoderDependencies<'vir, Self>,
1135    ) -> EncodeFullResult<'vir, Self> {
1136        deps.emit_output_ref(*task_key, ())?;
1137
1138        vir::with_vcx(|vcx| {
1139            let (impl_did, trait_fn) = *task_key;
1140            let impl_name = impl_name(vcx, impl_did);
1141            let item_name = ViperIdent::from_def_id(vcx, trait_fn);
1142            let trait_ref = vcx
1143                .tcx()
1144                .impl_trait_ref(impl_did)
1145                .instantiate_identity()
1146                .skip_normalization();
1147
1148            // The trait fn's parameters: the trait's, then the fn's own.
1149            let item_params = GParams::from(trait_fn);
1150            let item_generics = deps.require_dep::<GenericParamsEnc>(item_params)?;
1151            let post_func = deps.require_ref::<TraitFnEnc>(trait_fn)?.post_func;
1152            let local_defs = deps.require_dep::<MirLocalDefEnc>(MirLocalDefEncTask::Local {
1153                def_id: trait_fn,
1154                all_locals: false,
1155            })?;
1156            let func_args = local_defs.local_decl_args().collect::<Vec<_>>();
1157            let func_arg_exprs = func_args
1158                .iter()
1159                .map(|arg| vcx.mk_local_ex(arg))
1160                .collect::<Vec<_>>();
1161            let func_ret = local_defs.local_decl_ret();
1162
1163            let default_body = deps.require_dep::<FunctionCallEnc>(
1164                CallTaskDescription::new(trait_fn, item_params.rust_params(), trait_fn)
1165                    .resolve_trait_calls(false),
1166            )?;
1167            let default_body_app = default_body.call_pure(func_arg_exprs.clone());
1168            let post_func_call = post_func.call()(
1169                vcx.mk_local_ex(func_ret),
1170                vcx.alloc_slice(&func_arg_exprs),
1171                item_generics.ty_exprs(),
1172                item_generics.const_exprs(),
1173            );
1174
1175            // Quantified over the impl's parameters (in a suffixed context, so
1176            // that they cannot capture the trait fn's) and the fn's own, and
1177            // triggered on the application at the impl's trait ref, so that
1178            // the axiom only fires for calls it can match. The trait's
1179            // parameters, which form the trait ref, are let-bound to it.
1180            let impl_ctx = GParams::from(impl_did).with_suffix("impl");
1181            let impl_args = deps.require_dep::<GArgsTyEnc>(GArgs::new(impl_ctx, trait_ref.args))?;
1182            let (impl_tys, impl_consts) = (impl_args.get_ty(), impl_args.get_const());
1183            let own_ty_decls = &item_generics.ty_decls()[impl_tys.len()..];
1184            let own_const_decls = &item_generics.const_decls()[impl_consts.len()..];
1185
1186            let trigger_tys = impl_tys
1187                .iter()
1188                .chain(&item_generics.ty_exprs()[impl_tys.len()..])
1189                .copied()
1190                .collect::<Vec<_>>();
1191            let trigger_consts = impl_consts
1192                .iter()
1193                .chain(&item_generics.const_exprs()[impl_consts.len()..])
1194                .copied()
1195                .collect::<Vec<_>>();
1196            let trigger = post_func.call()(
1197                vcx.mk_local_ex(func_ret),
1198                vcx.alloc_slice(&func_arg_exprs),
1199                vcx.alloc_slice(&trigger_tys),
1200                vcx.alloc_slice(&trigger_consts),
1201            );
1202
1203            let body = vir::expr! {
1204                (post_func_call) ==> (([func_ret]) == ([default_body_app]))
1205            };
1206            let trait_lets = std::iter::zip(item_generics.ty_decls(), impl_tys)
1207                .map(|(decl, &ty)| (decl.upcast_ty(), ty.upcast_ty()))
1208                .chain(
1209                    std::iter::zip(item_generics.const_decls(), impl_consts)
1210                        .map(|(decl, &const_)| (decl.upcast_ty(), const_.upcast_ty())),
1211                )
1212                .collect::<Vec<(vir::LocalDeclDyn, vir::ExprDyn)>>();
1213            let body = trait_lets
1214                .into_iter()
1215                .rfold(body, |acc, (decl, expr)| vcx.mk_let_expr(decl, expr, acc));
1216
1217            let extra = std::iter::once(func_ret.upcast_ty())
1218                .chain(func_args.iter().map(|arg| arg.upcast_ty()))
1219                .chain(own_ty_decls.iter().map(|decl| decl.upcast_ty()))
1220                .chain(own_const_decls.iter().map(|decl| decl.upcast_ty()))
1221                .collect::<Vec<_>>();
1222            let axiom = TraitImplEnc::guarded_forall_in(
1223                vcx,
1224                deps,
1225                impl_did,
1226                impl_ctx,
1227                impl_ctx,
1228                &extra,
1229                trigger.upcast_ty(),
1230                body,
1231            )?;
1232            let domain = vcx.mk_domain(
1233                vir_format_identifier!(vcx, "trait_{impl_name}_default_{item_name}"),
1234                &[],
1235                vcx.alloc_slice(&[vcx.mk_domain_axiom(
1236                    vir_format_identifier!(vcx, "{impl_name}_fn_default_{item_name}"),
1237                    axiom,
1238                )]),
1239                &[],
1240                None,
1241            );
1242            Ok((domain, ()))
1243        })
1244    }
1245
1246    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
1247        for domain in Self::all_outputs_local(program) {
1248            program.add_domain(domain);
1249        }
1250    }
