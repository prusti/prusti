8#[derive(Debug, Clone)]
9pub struct PairUse<'vir> {
10    pub ty: vir::TypePair<'vir>,
11    pub constructor: vir::FunctionIdn<'vir, vir::ManyDyn, vir::Pair>,
12    pub destructors: Vec<vir::AdtDestructor<'vir, vir::Pair, vir::Dyn>>,
13}
14
15impl<'vir> PairUse<'vir> {
16    pub fn mk_cons<Curr, Next, T: vir::CompType>(
17        &self,
18        elems: &[vir::ExprGen<'vir, Curr, Next, T>],
19    ) -> vir::ExprGenPair<'vir, Curr, Next> {
20        self.constructor.call()(elems.as_dyn())
21    }
22
23    pub fn mk_elem<Curr, Next>(
24        &self,
25        pair: vir::ExprGenPair<'vir, Curr, Next>,
26        elem: usize,
27    ) -> vir::ExprGenDyn<'vir, Curr, Next> {
28        self.destructors[elem].call()(pair)
29    }
30}

32impl TaskEncoder for PairUseEnc {
33    task_encoder::encoder_cache!(PairUseEnc);
34    const ENCODER_NAME: &'static str = "pair use encoder";
35    type TaskDescription<'vir> = Vec<vir::TypeDyn<'vir>>;
36    type OutputFullDependency<'vir> = PairUse<'vir>;
37
38    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
39        task.clone()
40    }
41
42    fn do_encode_full<'vir>(
43        task_key: &Self::TaskKey<'vir>,
44        deps: &mut task_encoder::TaskEncoderDependencies<'vir, Self>,
45    ) -> task_encoder::EncodeFullResult<'vir, Self> {
46        deps.emit_output_ref(task_key.clone(), ())?;
47        let tuple = deps.require_dep::<PairEnc>(task_key.len())?;
48
49        vir::with_vcx(|vcx| {
50            let params = vcx.alloc_slice(task_key);
51            let ty = (tuple.self_ty)(params);
52
53            let constructor =
54                vir::FunctionIdn::new(vir::ViperIdent::new(tuple.constructor.name), params, ty);
55
56            let destructors = tuple
57                .constructor
58                .args
59                .iter()
60                .zip(params)
61                .map(|(local, field)| vcx.mk_adt_destructor(local.name, ty, field))
62                .collect::<Vec<_>>();
63            Ok((
64                (),
65                PairUse {
66                    ty,
67                    constructor,
68                    destructors,
69                },
70            ))
71        })
72    }
73
74    /// `PairEnc` is private to this module, so its ADTs are emitted from
75    /// here, which `lib.rs` calls.
76    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
77        PairEnc::emit_outputs(program);
78    }
79}

82struct Pair<'vir> {
83    self_ty: vir::DomainIdnPair<'vir>,
84    constructor: vir::AdtConstructor<'vir>,

87/// Encodes, for the arity N given as the task, the generic ADT
88/// `PairN[Ty0, ..]` with one constructor `TN` of N elements. It groups values
89/// that stand for no Rust value, such as the pure encoder's merges of the
90/// locals that the branches of a conditional modify. Unlike the snapshots of
91/// Rust tuples, which box their elements into `s_Param`, it keeps the
92/// elements' types, and it never stands for a Rust tuple, so these groupings
93/// stay apart from the tuples the user writes.
94struct PairEnc;
95
96impl TaskEncoder for PairEnc {
97    task_encoder::encoder_cache!(PairEnc);
98    const ENCODER_NAME: &'static str = "pair encoder";
99
100    type TaskDescription<'vir> = usize;
101    type OutputFullDependency<'vir> = Pair<'vir>;
102    type OutputFullLocal<'vir> = vir::Adt<'vir>;
103
104    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
105        *task
106    }
107
108    fn do_encode_full<'vir>(
109        task_key: &Self::TaskKey<'vir>,
110        deps: &mut task_encoder::TaskEncoderDependencies<'vir, Self>,
111    ) -> task_encoder::EncodeFullResult<'vir, Self> {
112        deps.emit_output_ref(*task_key, ())?;
113        vir::with_vcx(|vcx| {
114            let idn = vir::ViperIdent::new(vir::vir_format!(vcx, "Pair{}", *task_key));
115            let self_ty = vir::DomainIdn::new(idn, *task_key);
116            let typarams = (0..*task_key)
117                .map(|i| {
118                    vcx.alloc(vir::DomainParamData {
119                        name: vir::vir_format!(vcx, "Ty{}", i),
120                        index: i,
121                    })
122                })
123                .collect::<Vec<_>>();
124            let typarams = vcx.alloc_slice(&typarams);
125            let locals = typarams
126                .iter()
127                .enumerate()
128                .map(|(i, dpd)| {
129                    let dt = vir::TypeKind::DomainTypeParam(**dpd);
130                    let ty = vcx.alloc(vir::TypeData::<vir::Dyn>::new(dt));
131                    vcx.mk_local_decl(vir::vir_format!(vcx, "_{task_key}_{i}"), ty)
132                })
133                .collect::<Vec<_>>();
134            let locals = vcx.alloc_slice(&locals);
135            let constructor =
136                vcx.mk_adt_constructor(vir::vir_format!(vcx, "T{}", *task_key), locals);
137            let constructors = vcx.alloc_array(&[constructor]);
138            let adt = vcx.mk_adt(idn, typarams, constructors);
139
140            Ok((
141                adt,
142                Pair {
143                    self_ty,
144                    constructor,
145                },
146            ))
147        })
148    }
149
150    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
151        let outputs = Self::all_outputs_local(program);
152        for output in outputs {
153            program.add_adt(output);
154        }
155    }
156}
