28/// The cause of an encoding error, reported when a function or method body, a
29/// contract or one of its specs cannot be encoded (e.g. an unsupported
30/// feature): the link of the error's chain that holds the actual error, and
31/// the innermost span of the chain.
32pub(crate) struct DepError {
33    pub cause: DependencyErrorLink,
34    pub span: Option<Span>,
35}
36
37impl DepError {
38    /// The error to report at `span` when `what` cannot be encoded. A mistake
39    /// in the user's program is reported as an invalid-specification error
40    /// stating only the cause; a failure caused by an internal error (e.g. a
41    /// panic) as an internal error, which, unlike an unsupported feature, is
42    /// not downgraded to a warning when unsupported features are skipped;
43    /// anything else is an unsupported-feature error. The latter two also
44    /// name `what`.
45    pub(crate) fn report(self, what: String, span: Span) -> PrustiError {
46        let cause = self.cause;
47        let error = match cause.kind {
48            ErrorKind::Incorrect => PrustiError::incorrect(cause.message, span.into()),
49            ErrorKind::Internal => PrustiError::internal(
50                format!("cannot encode {what}: {}", cause.message),
51                span.into(),
52            ),
53            ErrorKind::Unsupported => PrustiError::unsupported(
54                format!("cannot encode {what}: {}", cause.message),
55                span.into(),
56            ),
57        };
58        match cause.help {
59            Some(help) => error.set_help(help),
60            None => error,
61        }
62    }
63}

65/// Reports the encoding error `err` with the diagnostic that `report` builds
66/// from its cause and returns the proof that it was reported. If `err` was
67/// already reported, reports nothing and returns that report's proof.
68pub(crate) fn report_dep_error<'vir, E: TaskEncoder + ?Sized>(
69    vcx: &'vir vir::VirCtxt<'vir>,
70    err: &EncodeFullError<'vir, E>,
71    report: impl FnOnce(DepError) -> PrustiError,
72) -> vir::Reported {
73    match err {
74        EncodeFullError::Reported(reported) => *reported,
75        err => {
76            let reported = vcx.emit_early_error(report(dep_error(err)));
77            err.mark_reported(reported);
78            reported
79        }
80    }
81}

83/// Extracts the cause of an encoding error. For a dependency error, the cause
84/// is the root (the last link of the chain), which holds the actual error.
85fn dep_error<'vir, E: TaskEncoder + ?Sized>(err: &EncodeFullError<'vir, E>) -> DepError {
86    match err {
87        EncodeFullError::DependencyError(chain) => DepError {
88            cause: chain.last().cloned().unwrap_or_else(|| {
89                DependencyErrorLink::new(
90                    E::ENCODER_NAME,
91                    "encoding dependency error".to_string(),
92                    Vec::new(),
93                    ErrorKind::Unsupported,
94                )
95            }),
96            span: chain
97                .iter()
98                .rev()
99                .find_map(|link| link.spans.first().copied()),
100        },
101        other => DepError {
102            cause: DependencyErrorLink::new(
103                E::ENCODER_NAME,
104                format!("{other:?}"),
105                Vec::new(),
106                ErrorKind::Unsupported,
107            ),
108            span: None,
109        },
110    }
111}

113#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
114pub struct CallTaskDescription<'tcx> {
115    gargs: GArgs<'tcx>,
116    callee: DefId,
117    resolve_trait_calls: bool,
118}
119
120impl<'tcx> CallTaskDescription<'tcx> {
121    pub fn new(
122        context: impl Into<GParams<'tcx>>,
123        args: &'tcx [ty::GenericArg<'tcx>],
124        callee: DefId,
125    ) -> Self {
126        Self {
127            gargs: GArgs::new(context, args),
128            callee,
129            resolve_trait_calls: true,
130        }
131    }
132
133    pub fn resolve_trait_calls(self, resolve_trait_calls: bool) -> Self {
134        Self {
135            resolve_trait_calls,
136            ..self
137        }
138    }
139
140    fn trait_call<'vir, E: TaskEncoder>(
141        &self,
142        deps: &mut TaskEncoderDependencies<'vir, E>,
143    ) -> Result<(DefId, Option<TraitFnEncOutputRef<'vir>>), EncodeFullError<'vir, E>> {
144        let tcx = vir::with_vcx(|vcx| vcx.tcx());
145        if self.resolve_trait_calls
146            && let Some(assoc_item) = tcx.opt_associated_item(self.callee)
147            && assoc_item.trait_container(tcx).is_some()
148        {
149            let assoc_enc = deps.require_ref::<TraitFnEnc>(assoc_item.def_id)?;
150            Ok((assoc_item.def_id, Some(assoc_enc)))
151        } else {
152            Ok((self.callee, None))
153        }
154    }
155}

157// TODO: this should be a "crate" encoder, which will deps.require all the methods in the crate
158pub fn encode_all_in_crate<'tcx>(tcx: ty::TyCtxt<'tcx>) {
159    for def_id in tcx.hir_body_owners() {
160        tracing::debug!("test_entrypoint item: {def_id:?}");
161        match tcx.def_kind(def_id) {
162            // Closure bodies are verified like `fn` bodies, whether or not
163            // they carry a `closure!` specification; only the closures of
164            // specifications themselves are exempt.
165            hir::def::DefKind::Fn | hir::def::DefKind::AssocFn | hir::def::DefKind::Closure => {
166                let def_id = def_id.to_def_id();
167                if prusti_interface::specs::is_spec_item(tcx, def_id) {
168                    continue;
169                }
170                // A closure inside a trusted function, including one inside an
171                // anonymous constant in its body (e.g. an array length), is part
172                // of its (unencoded) body.
173                if tcx.is_closure_like(def_id)
174                    && crate::encoders::is_function_trusted(
175                        tcx.hir_get_parent_item(tcx.local_def_id_to_hir_id(def_id.expect_local()))
176                            .to_def_id(),
177                    )
178                {
179                    continue;
180                }
181                // Extern-spec stubs are macro-generated forwarding bodies whose
182                // spec is transplanted onto the foreign target: verifying the
183                // stub against it would be circular, and by this point the stub
184                // has no spec of its own left to verify against.
185                if prusti_interface::utils::get_prusti_attr_of(tcx, def_id, "extern_spec").is_some()
186                {
187                    continue;
188                }
189
190                let (is_pure, is_trusted) = crate::encoders::with_proc_spec(
191                    SpecQuery::GetProcKind(def_id, ty::List::identity_for_item(tcx, def_id)),
192                    |proc_spec| {
193                        // Report an invalid trait-to-impl kind refinement once,
194                        // here, rather than on every purity query.
195                        crate::encoders::report_kind_refinement_error(def_id, &proc_spec.kind);
196                        let is_pure = crate::encoders::kind_is_pure(&proc_spec.kind);
197                        let is_trusted = crate::encoders::spec_is_trusted(proc_spec, def_id);
198                        (is_pure, is_trusted)
199                    },
200                )
201                .unwrap_or_default();
202
203                if !(is_trusted && is_pure) {
204                    let _ = method::MethodEnc::encode(def_id, false, tcx.def_span(def_id));
205                }
206            }
207            unsupported_item_kind => {
208                tracing::debug!("unsupported item: {unsupported_item_kind:?}");
209            }
210        }
211    }
212
213    // This creates the impl encoding for all traits in the crate
214    // To iterate over all _visible_ impl blocks,
215    // use tcx.visible_traits and tcx.all_impls(trait_id)
216    for def_id in tcx.hir_crate_items(()).definitions() {
217        // A failed encoding stays in the encoder's cache, from which its
218        // `emit_outputs` reports it at the first non-dummy span of the item's
219        // requests.
220        match tcx.def_kind(def_id) {
221            hir::def::DefKind::Trait => {
222                let _ = TraitEnc::encode(def_id.to_def_id(), false, tcx.def_span(def_id));
223            }
224            hir::def::DefKind::Impl { of_trait: true } => {
225                let _ = TraitImplEnc::encode(def_id.to_def_id(), false, tcx.def_span(def_id));
226            }
227            _ => (),
228        }
229    }
