38impl<'vir: 'a, 'a, 'enc, E: TaskEncoder> ImpureEncVisitor<'vir, 'enc, E> {
39    /// Calculate invariant at loop head, and the places whose permission it
40    /// holds, which are the places the body invariants may read. A place
41    /// behind a shared reference is held, and read, through the first shared
42    /// reference on its path.
43    pub(crate) fn get_loop_inv(
44        &mut self,
45        cfpcs: &PcgBasicBlock<'_, 'vir>,
46        loop_place_usages: &PlaceUsages<'vir>,
47        ctxt: impl HasCompilerCtxt<'a, 'vir>,
48    ) -> EncodeResult<
49        'vir,
50        (
51            Vec<vir::ExprBool<'vir>>,
52            FxHashMap<Place<'vir>, CapabilityKind>,
53        ),
54        E,
55    > {
56        let mut inv = Vec::new();
57        let start = &cfpcs.statements[0];
58        let state = &start.states[EvalStmtPhase::PreOperands];
59        let loop_invariant_place_capabilities =
60            cfpcs.loop_invariant_place_capabilities(loop_place_usages, ctxt);
61
62        let mut held_places = FxHashMap::default();
63        for (place, capability) in loop_invariant_place_capabilities.iter() {
64            if capability.is_write() {
65                continue; // No permissions are encoded for places with write capabilities currently
66            }
67            // A place behind a shared reference has no predicate of its own: its value is
68            // part of the snapshot of the first shared reference on its path from the local.
69            // The invariant holds the predicate of that reference instead, only once even if
70            // the loop reads several places behind it.
71            let place = place
72                .iter_projections()
73                .find_map(|(base, _)| {
74                    let base = Place::from(base);
75                    base.is_shared_ref(ctxt).then_some(base)
76                })
77                .unwrap_or(*place);
78            if held_places.insert(place, *capability).is_some() {
79                continue;
80            }
81            let place = place.with_inherent_region(ctxt);
82            let (place_res, _snap, ty, ty_out) = self.encode_place_with_snap(place)?;
83            let task = RustTyDecomposition::from_ty(ty.ty, self.def_id);
84            let pred = ty_out.ref_to_pred(self.vcx, place_res.expr.expect_predicate(), None);
85            inv.push(pred);
86            // gets all the lifetime projections of the place
87            // for example, if we add an invariant for x and x is a reference, then we also want to add
88            // an invariant for *x
89            let projs = place.generalized_lifetime_projections(ctxt);
90            for proj in projs {
91                let region = proj.region(ctxt);
92                let indirect = self.indirect_predicates(task, region, ty.ty)?;
93                inv.extend(indirect.predicate_applications.iter().map(|p| {
94                    p.reify(
95                        self.vcx,
96                        ty_out.ref_to_snap(place_res.expr.expect_predicate()),
97                    )
98                }));
99            }
100        }
101
102        for edge in state.borrow_pcg().graph().coupled_edges() {
103            let mut let_bind = WandOldOuter::LetBind(Vec::new());
104            let (wand, _) = self.coupled_wand(edge.value(), &mut let_bind)?;
105            let mut wand = self.vcx.mk_wand_expr(wand);
106            let WandOldOuter::LetBind(let_bind) = let_bind else {
107                unreachable!()
108            };
109            for lb in let_bind {
110                wand = lb.map_or_else(
111                    |(d, e)| self.vcx.mk_let_expr(d, e, wand),
112                    |(d, e)| self.vcx.mk_let_expr(d, e, wand),
113                );
114            }
115            inv.push(wand);
116        }
117
118        Ok((inv, held_places))
119    }

121    pub(crate) fn encode_pcg_node<T: PcgLifetimeProjectionBaseLike<'vir>>(
122        &mut self,
123        node: &PcgNode<'vir, MaybeLabelledPlace<'vir>, T>,
124        wand_rhs: &mut Vec<vir::ExprBool<'vir>>,
125        old_outer: &mut WandOldOuter<'vir>,
126    ) -> EncodeResult<'vir, (), E> {
127        match node {
128            PcgNode::Place(place) => {
129                let p = place.place();
130                let ty = (*p).ty(self.local_decls, self.vcx.tcx());
131                let task = RustTyDecomposition::from_ty(ty.ty, self.def_id);
132                let ty_out = self.deps.require_dep::<TyUseImpureEnc>(task).unwrap();
133                let p = self.encode_place(p)?;
134                let p = self.configure_old((*place).into(), p.expr.expect_predicate(), old_outer);
135
136                let pred = ty_out.ref_to_pred(self.vcx, p, None);
137                wand_rhs.push(pred);
138            }
139            PcgNode::LifetimeProjection(r) => {
140                let exprs = self.encode_lifetime_projection(*r, old_outer)?;
141                wand_rhs.extend(exprs);
142            }
143        }
144        Ok(())
145    }

147    pub(super) fn encode_lifetime_projection<T: PcgLifetimeProjectionBaseLike<'vir>>(
148        &mut self,
149        r: LifetimeProjection<'vir, T>,
150        old_outer: &mut WandOldOuter<'vir>,
151    ) -> EncodeResult<'vir, Vec<vir::ExprBool<'vir>>, E> {
152        let place = r.base().to_pcg_lifetime_projection_base();
153        let (place_snap, ty, _) = match place {
154            PcgLifetimeProjectionBase::Place(p) => {
155                self.encode_maybe_remote_place_snap(p, old_outer)?
156            }
157            PcgLifetimeProjectionBase::Const(c) => todo!("{c:?}"),
158        };
159        let decomp = RustTyDecomposition::from_ty(ty.ty, self.def_id);
160        let region = r.with_base(place).region(self.pcg_ctxt());
161        let indirect = self.indirect_predicates(decomp, region, ty.ty)?;
162        Ok(indirect
163            .predicate_applications
164            .into_iter()
165            .map(|expr| expr.reify(self.vcx, place_snap))
166            .collect::<Vec<_>>())
167    }

169    /// Encodes the indirect predicates of the lifetime projection of `decomp`, the
170    /// decomposition of `ty`, at `region`. If `region` is not among the lifetimes of
171    /// `decomp` (e.g. because `ty` is an opaque type, a trait object or an unnormalized
172    /// alias), nothing behind it is encoded and there are no indirect predicates. The
173    /// exception is a type containing a closure, whose captures are encoded although the
174    /// closure type has no capture regions; for it an unsupported feature is reported.
175    fn indirect_predicates(
176        &mut self,
177        decomp: RustTyDecomposition<'vir>,
178        region: GeneralizedLifetime<'vir>,
179        ty: ty::Ty<'vir>,
180    ) -> EncodeResult<'vir, IndirectPredicatesEncOutputRef<'vir>, E> {
181        let Some(proj) = LifetimeProjection::new(decomp, region, None, PrustiPcgCtxt) else {
182            if !ty
183                .walk()
184                .any(|arg| arg.as_type().is_some_and(|ty| ty.is_closure()))
185            {
186                return Ok(IndirectPredicatesEncOutputRef::new(vec![]));
187            }
188            return Err(EncodeFullError::unsupported(
189                format!("a closure capturing a reference (`{ty}`)"),
190                vec![self.current_span()],
191            ));
192        };
193        self.deps.require_dep::<IndirectPredicatesEnc>(proj)
194    }

196    fn get_place(place: MaybeRemotePlace<'vir>) -> Place<'vir> {
197        match place {
198            MaybeRemotePlace::Local(MaybeLabelledPlace::Current(place)) => place,
199            MaybeRemotePlace::Local(MaybeLabelledPlace::Labelled(place)) => place.place(),
200            MaybeRemotePlace::Remote(r) => r.assigned_local().into(),
201        }
202    }

204    fn encode_maybe_remote_place_snap(
205        &mut self,
206        place: MaybeRemotePlace<'vir>,
207        old_outer: &mut WandOldOuter<'vir>,
208    ) -> EncodeResult<'vir, (vir::ExprSnap<'vir>, mir::PlaceTy<'vir>, TyUseImpure<'vir>), E> {
209        let p = Self::get_place(place);
210        let (_, place_snap, ty, ty_out) = self.encode_place_with_snap(p)?;
211        let place_snap = self.configure_old(place, place_snap, old_outer);
212        Ok((place_snap, ty, ty_out))
213    }
214
215    fn configure_old<T: SnapOrRef>(
216        &mut self,
217        place: MaybeRemotePlace,
218        expr: vir::Expr<'vir, T>,
219        old_outer: &mut WandOldOuter<'vir>,
220    ) -> vir::Expr<'vir, T> {
221        match place {
222            MaybeRemotePlace::Local(MaybeLabelledPlace::Current { .. }) => {
223                self.mk_wand_outer(expr, old_outer)
224            }
225            MaybeRemotePlace::Local(MaybeLabelledPlace::Labelled(place)) => {
226                let label = self.get_location_label(place.at());
227                self.vcx.mk_old(expr, label)
228            }
229            MaybeRemotePlace::Remote(_) => self.vcx.mk_old_expr(expr),
230        }
231    }
232
233    fn mk_wand_outer<T: SnapOrRef>(
234        &mut self,
235        expr: vir::Expr<'vir, T>,
236        old_outer: &mut WandOldOuter<'vir>,
237    ) -> vir::Expr<'vir, T> {
238        match old_outer {
239            WandOldOuter::LetBind(let_bind) => {
240                let ident = vir::vir_format!(self.vcx, "_lb{}", let_bind.len());
241                let decl = self.vcx.mk_local_decl(ident, expr.ty());
242                let_bind.push(T::as_result(decl, expr));
243                self.vcx.mk_local_ex(decl)
244            }
245            WandOldOuter::Label(label) => {
246                let label = *label.get_or_insert_with(|| self.new_label("outer_package"));
247                self.vcx.mk_local_labelled_old_expr(expr, label)
248            }
249        }
250    }
251}

253type LetBind<'vir> = Result<
254    (vir::LocalDeclSnap<'vir>, vir::ExprSnap<'vir>),
255    (vir::LocalDeclRef<'vir>, vir::ExprRef<'vir>),
256>;
257
258trait SnapOrRef: vir::CompType {
259    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir>;
260}
261
262impl SnapOrRef for vir::Snap {
263    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir> {
264        Ok((d, e))
265    }
266}

268impl SnapOrRef for vir::Ref {
269    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir> {
270        Err((d, e))
