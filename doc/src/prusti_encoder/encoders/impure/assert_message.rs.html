4/// The error reported when the condition of an `Assert` terminator with the
5/// message `msg` might not hold, or `None` if the condition is assumed rather
6/// than checked.
7pub(crate) fn assert_error_message(msg: &mir::AssertMessage<'_>) -> Option<&'static str> {
8    match *msg {
9        mir::AssertMessage::BoundsCheck { .. } => {
10            Some("the array or slice index may be out of bounds")
11        }
12        mir::AssertMessage::Overflow(..) | mir::AssertMessage::OverflowNeg(..)
13            if !config::check_overflows() =>
14        {
15            // If we are not checking for overflows, encode an overflow-checking
16            // assertion as an assume instead.
17            None
18        }
19        mir::AssertMessage::Overflow(op, ..) => Some(overflow_message(op)),
20        mir::AssertMessage::OverflowNeg(..) => {
21            Some("assertion might fail with \"attempt to negate with overflow\"")
22        }
23        mir::AssertMessage::DivisionByZero(..) => {
24            Some("assertion might fail with \"attempt to divide by zero\"")
25        }
26        mir::AssertMessage::RemainderByZero(..) => Some(
27            "assertion might fail with \"attempt to calculate the remainder with a divisor of zero\"",
28        ),
29        mir::AssertMessage::ResumedAfterReturn(..) => Some("execution may continue after return"),
30        mir::AssertMessage::ResumedAfterPanic(..) => Some("execution may continue after panic"),
31        mir::AssertMessage::MisalignedPointerDereference { .. } => {
32            Some("misaligned pointer may be dereferenced")
33        }
34        mir::AssertKind::ResumedAfterDrop(..) => Some("execution may continue after drop"),
35        mir::AssertKind::NullPointerDereference => Some("null pointer may be dereferenced"),
36        mir::AssertKind::InvalidEnumConstruction(..) => Some("invalid enum construction may occur"),
37        mir::AssertKind::NullReferenceConstructed => Some("null reference may be constructed"),
38    }
39}

41/// The error for an overflow of `op`. MIR building checks for overflow the
42/// arithmetic `Add`, `Sub` and `Mul`, shifts by at least the bit width (`Shl`,
43/// `Shr`), and the signed `MIN / -1` (`Div`, `Rem`); no other operation is
44/// checked for overflow.
45fn overflow_message(op: mir::BinOp) -> &'static str {
46    match op {
47        mir::BinOp::Add => "assertion might fail with \"attempt to add with overflow\"",
48        mir::BinOp::Sub => "assertion might fail with \"attempt to subtract with overflow\"",
49        mir::BinOp::Mul => "assertion might fail with \"attempt to multiply with overflow\"",
50        mir::BinOp::Div => "assertion might fail with \"attempt to divide with overflow\"",
51        mir::BinOp::Rem => {
52            "assertion might fail with \"attempt to calculate the remainder with overflow\""
53        }
54        mir::BinOp::Shl => "assertion might fail with \"attempt to shift left with overflow\"",
55        mir::BinOp::Shr => "assertion might fail with \"attempt to shift right with overflow\"",
56        mir::BinOp::AddUnchecked
57        | mir::BinOp::AddWithOverflow
58        | mir::BinOp::SubUnchecked
59        | mir::BinOp::SubWithOverflow
60        | mir::BinOp::MulUnchecked
61        | mir::BinOp::MulWithOverflow
62        | mir::BinOp::BitXor
63        | mir::BinOp::BitAnd
64        | mir::BinOp::BitOr
65        | mir::BinOp::ShlUnchecked
66        | mir::BinOp::ShrUnchecked
67        | mir::BinOp::Eq
68        | mir::BinOp::Lt
69        | mir::BinOp::Le
70        | mir::BinOp::Ne
71        | mir::BinOp::Ge
72        | mir::BinOp::Gt
73        | mir::BinOp::Cmp
74        | mir::BinOp::Offset => unreachable!("`{op:?}` is not checked for overflow"),
75    }
76}
