5impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
6    /// Encodes the closure argument of a quantifier/spec-block builtin: the
7    /// closure's arguments become the returned quantified variable
8    /// declarations (empty for `spec_block`) and its body the returned,
9    /// recursively encoded expression. `name` is the builtin's
10    /// `prusti_contracts` name, for error reporting.
11    ///
12    /// The closure body is part of this specification, so when it reads a
13    /// capture in some modes, it reads the operand this specification built
14    /// the capture from, in those modes (a quantified `old(arr.lookup(i))`
15    /// reads the reborrow `&*arr` in the pre-state). Those operands are
16    /// encoded here, at `curr_ver` in `block`, which builds the closure and
17    /// calls the builtin, and passed to the closure body as its captures (see
18    /// `ExprInput`).
19    fn encode_spec_closure(
20        &mut self,
21        name: &str,
22        closure_ty: ty::Ty<'vir>,
23        block: mir::BasicBlock,
24        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
25    ) -> EncodeResult<'vir, SpecClosure<'vir>, MirPureEnc> {
26        let (qvar_tys, cl_kind, cl_def_id, cl_args) = match closure_ty.kind() {
27            TyKind::Closure(cl_def_id, cl_args) => (
28                match cl_args.as_closure().sig().skip_binder().inputs()[0].kind() {
29                    TyKind::Tuple(list) => list,
30                    _ => unreachable!(),
31                },
32                cl_args.as_closure().kind(),
33                *cl_def_id,
34                *cl_args,
35            ),
36            other => panic!("illegal prusti::{name}: expected closure, got {other:?}"),
37        };
38        // The builtins' signatures should enforce `Fn`-only closures.
39        assert_eq!(cl_kind, ty::ClosureKind::Fn);
40
41        let qvars = self.vcx.alloc_slice(
42            &qvar_tys
43                .iter()
44                .enumerate()
45                .map(|(idx, qvar_ty)| {
46                    let ty_out = self.ty_use(qvar_ty);
47                    self.vcx.mk_local_decl(
48                        vir::vir_format!(self.vcx, "qvar_{}_{idx}", self.encoding_depth),
49                        ty_out.snapshot,
50                    )
51                })
52                .collect::<Vec<_>>(),
53        );
54
55        let mut reify_args = FxHashMap::default();
56        // The arguments to the closure are the closure itself, which gets no
57        // value as the body reads it only through its captures, and the
58        // qvars.
59        reify_args.extend(
60            qvars
61                .iter()
62                .enumerate()
63                .map(|(idx, qvar)| ((idx + 2).into(), self.vcx.mk_local_ex(qvar))),
64        );
65
66        use vir::Reify;
67        let out = self.deps.require_dep::<MirPureEnc>(MirPureEncTask {
68            encoding_depth: self.encoding_depth + 1,
69            kind: PureKind::Closure {
70                context: self.context_def_id,
71            },
72            parent_def_id: cl_def_id,
73            // The body becomes part of this expression, so it is encoded
74            // in this context, in which `cl_args` are expressed.
75            gargs: GArgs::new(self.context, cl_args),
76        })?;
77        // The specification macros build the closure in the `block` that
78        // calls the builtin, after which the block assigns only temporaries
79        // (borrows of the closure, the triggers); captured places are rooted
80        // at variables, so they read at `curr_ver` what the closure captured.
81        // Other closures are not supported. A closure without captures needs
82        // no aggregate: its borrow may be promoted to a constant.
83        let statements = &self.body[block].statements;
84        let built = statements
85            .iter()
86            .enumerate()
87            .rev()
88            .find_map(|(index, stmt)| match &stmt.kind {
89                mir::StatementKind::Assign(box (
90                    _,
91                    mir::Rvalue::Aggregate(box mir::AggregateKind::Closure(def_id, _), operands),
92                )) if *def_id == cl_def_id => Some((index, operands)),
93                _ => None,
94            });
95        let assigns_variable = |stmt: &mir::Statement<'vir>| match &stmt.kind {
96            mir::StatementKind::Assign(box (dest, _)) => self.is_variable(dest.local),
97            _ => false,
98        };
99        let operands = match built {
100            Some((index, operands)) if !statements[index + 1..].iter().any(assigns_variable) => {
101                operands.as_slice()
102            }
103            None if out.captures.is_empty() => IndexSlice::empty(),
104            _ => {
105                return Err(EncodeFullError::unsupported(
106                    format!("a `{name}` whose closure is not built right before it"),
107                    vec![self.current_span()],
108                ));
109            }
110        };
111        let mut captures = IndexVec::<abi::FieldIdx, _>::with_capacity(operands.len());
112        for (capture, reads) in operands.iter().zip(&out.captures) {
113            let mut values = FxHashMap::default();
114            for &modes in reads {
115                values.insert(modes, self.encode_capture(capture, modes, curr_ver)?);
116            }
117            captures.push(values);
118        }
119        let captures = self.vcx.alloc(captures);
120        let body = out.expr;
121        let reify_args = self.vcx.alloc(reify_args);
122        let body = self.vcx.mk_lazy_expr(
123            vir::vir_format!(self.vcx, "spec closure body ({name})"),
124            body.ty(),
125            Box::new(move |vcx, lctx: ExprInput<'vir>| {
126                let captures = captures
127                    .iter()
128                    .map(|values| {
129                        values
130                            .iter()
131                            .map(|(&modes, value)| (modes, value.reify(vcx, lctx)))
132                            .collect()
133                    })
134                    .collect::<IndexVec<_, _>>();
135                body.reify(vcx, (cl_def_id, reify_args, lctx.2, vcx.alloc(captures)))
136                    .kind
137            }),
138        );
139        Ok((qvars, body.downcast_ty::<vir::Bool>()))
140    }

142    /// The assignment statements of this body, as destination and value.
143    fn assignments(
144        &self,
145    ) -> impl Iterator<Item = (&'enc mir::Place<'vir>, &'enc mir::Rvalue<'vir>)> {
146        self.body
147            .basic_blocks
148            .iter()
149            .flat_map(|block| &block.statements)
150            .filter_map(|stmt| match &stmt.kind {
151                mir::StatementKind::Assign(box (dest, rvalue)) => Some((dest, rvalue)),
152                _ => None,
153            })
154    }

156    /// Whether `local` is a variable or an argument rather than a temporary:
157    /// MIR building gives debug info to every variable, to every argument
158    /// bound to a name, and to no temporary. (An argument bound to a pattern
159    /// counts as a temporary; it is never captured, as the pattern's bindings
160    /// are variables of their own. `LocalDecl::is_user_variable` would panic
161    /// in bodies of other crates, where the local info it reads is cleared.)
162    fn is_variable(&self, local: mir::Local) -> bool {
163        self.body.var_debug_info.iter().any(|info| {
164            matches!(info.value, mir::VarDebugInfoContents::Place(place)
165                if place.as_local() == Some(local))
166        })
167    }

169    /// A capture of a spec closure, passed to the closure as `capture` and
170    /// read in `modes`, encoded in those modes and the modes this body is in
171    /// at the builtin. MIR building passes a capture by reference as a
172    /// temporary that it assigns the reference, such as `&*arr`; the capture is
173    /// encoded from that value, so that the modes reach the read of `arr`
174    /// inside it. A capture by value is the captured place itself.
175    fn encode_capture(
176        &mut self,
177        capture: &mir::Operand<'vir>,
178        modes: Modes,
179        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
180    ) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
181        let temp = capture
182            .place()
183            .and_then(|place| place.as_local())
184            .filter(|&local| !self.is_variable(local));
185        let value = temp.and_then(|temp| {
186            self.assignments()
187                .find_map(|(dest, value)| (dest.as_local() == Some(temp)).then_some(value))
188        });
189
190        let outer_modes = self.modes;
191        self.modes = outer_modes | modes;
192        let res = match value {
193            Some(value) => self.encode_rvalue(curr_ver, value, self.current_span()),
194            None => self.encode_operand_snap(capture, curr_ver),
195        };
196        self.modes = outer_modes;
197        res
198    }

200    /// Encodes the pure-only `prusti_contracts` builtins (quantifiers, spec
201    /// blocks, and the old/rel/before-expiry mode markers), which need this
202    /// encoder's own state. The operand-based builtins are handled by the
203    /// shared [`PureRvalueEnc::encode_prusti_builtin`].
204    pub(super) fn encode_pure_only_builtin(
205        &mut self,
206        builtin: SpecBuiltin,
207        arg_tys: ty::GenericArgsRef<'vir>,
208        args: &[Spanned<mir::Operand<'vir>>],
209        block: mir::BasicBlock,
210        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
211    ) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
212        let mk_bool = |prim: vir::ExprGenBool<'vir, _, _>| prim.upcast_ty::<vir::CSnap>();
213        Ok(match builtin {
214            SpecBuiltin::Forall | SpecBuiltin::Exists => {
215                assert_eq!(arg_tys.len(), 3);
216                // The operands are the triggers and the closure.
217                assert_eq!(args.len(), 2);
218
219                let name = if builtin == SpecBuiltin::Forall {
220                    "forall"
221                } else {
222                    "exists"
223                };
224                let (qvars, body) =
225                    self.encode_spec_closure(name, arg_tys[2].expect_ty(), block, curr_ver)?;
226                // TODO: triggers
227                let res = if builtin == SpecBuiltin::Forall {
228                    self.vcx.mk_forall_expr(qvars, &[], body)
229                } else {
230                    self.vcx.mk_exists_expr(qvars, &[], body)
231                };
232                mk_bool(res)
233            }
234            SpecBuiltin::SpecBlock => {
235                assert_eq!(arg_tys.len(), 2);
236                assert_eq!(args.len(), 1);
237
238                let (qvars, body) = self.encode_spec_closure(
239                    "spec_block",
240                    arg_tys[1].expect_ty(),
241                    block,
242                    curr_ver,
243                )?;
244                assert!(qvars.is_empty(), "`spec_block` closures take no arguments");
245                mk_bool(body)
246            }
247            SpecBuiltin::ModeStart(mode) => {
248                match mode {
249                    Mode::Old => {
250                        assert!(!self.modes.contains(Modes::OLD));
251                        self.modes.insert(Modes::OLD);
252                    }
253                    Mode::Rel(0) => {
254                        assert!(!self.modes.contains(Modes::REL0));
255                        assert!(!self.modes.contains(Modes::REL1));
256                        self.modes.insert(Modes::REL0);
257                    }
258                    Mode::Rel(_) => {
259                        assert!(!self.modes.contains(Modes::REL1));
260                        assert!(!self.modes.contains(Modes::REL0));
261                        self.modes.insert(Modes::REL1);
262                    }
263                    Mode::BeforeExpiry => {
264                        assert!(!self.modes.contains(Modes::BEFORE_EXPIRY));
265                        self.modes.insert(Modes::BEFORE_EXPIRY);
266                    }
267                }
268                mk_bool(self.vcx.mk_bool::<true>().lazy()) // TODO: what value do we return?
269            }
270            SpecBuiltin::ModeEnd(mode) => {
271                match mode {
272                    Mode::Old => {
273                        assert!(self.modes.contains(Modes::OLD));
274                        self.modes.remove(Modes::OLD);
275                    }
276                    Mode::Rel(0) => {
277                        assert!(self.modes.contains(Modes::REL0));
278                        assert!(!self.modes.contains(Modes::REL1));
279                        self.modes.remove(Modes::REL0);
280                    }
281                    Mode::Rel(_) => {
282                        assert!(!self.modes.contains(Modes::REL0));
283                        assert!(self.modes.contains(Modes::REL1));
284                        self.modes.remove(Modes::REL1);
285                    }
286                    Mode::BeforeExpiry => {
287                        assert!(self.modes.contains(Modes::BEFORE_EXPIRY));
288                        self.modes.remove(Modes::BEFORE_EXPIRY);
289                    }
290                }
291                mk_bool(self.vcx.mk_bool::<true>().lazy()) // TODO: what value do we return?
292            }
293        }
294        .upcast_ty())
295    }
