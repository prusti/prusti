1//! The triggers of quantifiers (see `PureKind::Trigger` for how their terms
2//! are encoded), and the checks that they are valid. The encoding of a
3//! trigger closure rejects the conditionals and the calls that never return
4//! in it (`Enc::encode_cfg`), and its quantifiers, specification blocks and
5//! states other than `old` (`Enc::encode_pure_only_builtin`). The encoded
6//! trigger is then checked for Viper's own operators and for its outermost
7//! operation (`Enc::check_trigger`), and each trigger set for the bound
8//! variables its triggers mention (`Enc::encode_trigger_sets`).
9//!
10//! Rust's operators on primitive types are encoded as applications of
11//! functions, so a trigger may use them; the snapshot equality `===` and the
12//! operations of `Int` and `Real` are encoded as Viper's own operators, so it
13//! may not.
14

24/// What the outermost operation of a trigger can be.
25const HEADS: &str = "a call, an operator, a constructor, a field access, an indexing or a cast";
26
27impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
28    /// Encodes the trigger sets of a quantifier builtin over `qvars`, whose
29    /// type `sets_ty` is a tuple of trigger sets, each itself a tuple of
30    /// references to spec closures whose arguments are the quantified
31    /// variables and whose body is one trigger expression. Like the body
32    /// closure, the trigger closures get their captures from their
33    /// constructions among `closures` (see [`Self::encode_spec_closure`]).
34    /// Reports a set that is empty, an expression that reads no quantified
35    /// variable, and a set whose expressions do not read every quantified
36    /// variable; the rest of each expression is checked where it is encoded
37    /// (see [`Self::check_trigger`]).
38    pub(super) fn encode_trigger_sets(
39        &mut self,
40        name: &str,
41        sets_ty: ty::Ty<'vir>,
42        qvars: &'vir [vir::LocalDeclSnap<'vir>],
43        closures: &FxHashMap<DefId, ClosureConstruction<'vir>>,
44    ) -> EncodeResult<'vir, &'vir [TriggerRet<'vir>], MirPureEnc> {
45        let TyKind::Tuple(set_tys) = sets_ty.kind() else {
46            panic!("illegal prusti::{name}: expected a trigger-set tuple, got {sets_ty:?}")
47        };
48        let mut sets = Vec::with_capacity(set_tys.len());
49        for set_ty in set_tys.iter() {
50            let TyKind::Tuple(cl_tys) = set_ty.kind() else {
51                panic!("illegal prusti::{name}: expected a trigger tuple, got {set_ty:?}")
52            };
53            let mut exprs = Vec::with_capacity(cl_tys.len());
54            let mut def_ids = Vec::with_capacity(cl_tys.len());
55            let mut mentioned = vec![false; qvars.len()];
56            for cl_ty in cl_tys.iter() {
57                let &TyKind::Ref(_, cl_ty, _) = cl_ty.kind() else {
58                    panic!("illegal prusti::{name}: expected a trigger closure, got {cl_ty:?}")
59                };
60                let &TyKind::Closure(cl_def_id, cl_args) = cl_ty.kind() else {
61                    panic!("illegal prusti::{name}: expected a trigger closure, got {cl_ty:?}")
62                };
63                let cl_span = self.vcx.tcx().def_span(cl_def_id);
64                let kind = PureKind::Trigger {
65                    context: self.context_def_id,
66                };
67                let body =
68                    self.encode_closure_body(name, kind, cl_def_id, cl_args, qvars, closures)?;
69                if !body.reads.contains(&true) {
70                    return Err(EncodeFullError::incorrect(
71                        "every expression of a trigger set must mention a bound variable of the \
72                         quantifier"
73                            .to_string(),
74                        vec![cl_span],
75                    ));
76                }
77                for (mentioned, &read) in mentioned.iter_mut().zip(body.reads) {
78                    *mentioned |= read;
79                }
80                // Created under the trigger's span, so that the reified term
81                // carries it and the verifier's warnings about the trigger
82                // point at it.
83                let trigger = self.vcx.with_span(cl_span, |vcx| {
84                    body.embed(vcx, vir::vir_format!(vcx, "trigger ({name})"))
85                });
86                exprs.push(trigger);
87                def_ids.push(cl_def_id);
88            }
89            let (Some(&first), Some(&last)) = (def_ids.first(), def_ids.last()) else {
90                return Err(EncodeFullError::incorrect(
91                    "a trigger set must contain at least one expression".to_string(),
92                    vec![self.current_span()],
93                ));
94            };
95            if let Some(missing) = mentioned.iter().position(|&mentioned| !mentioned) {
96                let message = match bound_name(self.vcx, first, missing) {
97                    Some(name) => {
98                        format!("this trigger set does not mention the bound variable `{name}`")
99                    }
100                    None => "this trigger set does not mention every bound variable of the \
101                             quantifier"
102                        .to_string(),
103                };
104                let span = self
105                    .vcx
106                    .tcx()
107                    .def_span(first)
108                    .to(self.vcx.tcx().def_span(last));
109                return Err(EncodeFullError::incorrect(message, vec![span]));
110            }
111            sets.push(self.vcx.mk_trigger(&exprs));
112        }
113        Ok(self.vcx.alloc_slice(&sets))
114    }
115
116    /// Checks that `expr`, the encoding of this trigger closure's body, is a
117    /// valid trigger term. Its outermost operation must be one the user
118    /// writes (see [`Self::check_trigger_head`]), and it must contain no
119    /// Viper operator; the encoding has already rejected the conditionals and
120    /// quantifiers in it, so the operators are all this check finds. The
121    /// captures and the quantified variables are holes in `expr`, which the
122    /// check does not look into.
123    pub(super) fn check_trigger(&self, expr: ExprRet<'vir>) -> EncodeResult<'vir, (), MirPureEnc> {
124        let span = self.vcx.tcx().def_span(self.body.source.def_id());
125        if vir::has_forbidden_trigger_node(expr.as_dyn()) {
126            return Err(EncodeFullError::incorrect(
127                "a trigger cannot contain the snapshot equality `===` or an operation on \
128                 Prusti's mathematical integers and reals (`Int`, `Real`)"
129                    .to_string(),
130                vec![span],
131            ));
132        }
133        self.check_trigger_head(span)
134    }
135
136    /// Checks that the outermost operation of the trigger that this body
137    /// computes is one the user writes (see [`HEADS`]), reported at `span`.
138    /// The simplifier folds away the borrows and dereferences that rustc and
139    /// the encoding insert, but no application of what the user writes (see
140    /// `task_encoder::Program::add_user_adt`), so only the latter is sure to
141    /// remain a term the trigger can match. Dereferences are looked through:
142    /// rustc inserts them for reads through references and for overloaded
143    /// indexing, a call to `Index::index` whose result it dereferences.
144    fn check_trigger_head(&self, span: Span) -> EncodeResult<'vir, (), MirPureEnc> {
145        let error = |message: String| Err(EncodeFullError::incorrect(message, vec![span]));
146        let variable = || {
147            error(format!(
148                "a trigger must be {HEADS}, not a bare variable or constant"
149            ))
150        };
151        let mut definitions = Definitions(IndexVec::from_elem(Set1::Empty, &self.body.local_decls));
152        definitions.visit_body(self.body);
153        // The value of `local`, if one assignment defines it.
154        let assigned = |local: mir::Local| match definitions.0[local] {
155            Set1::One(location) => match &self.body.stmt_at(location).left()?.kind {
156                mir::StatementKind::Assign(box (_, rvalue)) => Some(rvalue),
157                _ => None,
158            },
159            Set1::Empty | Set1::Many => None,
160        };
161        let mut place = mir::Place::from(mir::RETURN_PLACE);
162        loop {
163            let mut projection = place.projection.as_slice();
164            while let [rest @ .., mir::ProjectionElem::Deref] = projection {
165                projection = rest;
166            }
167            match projection {
168                // A capture, i.e. a field of the closure: a captured variable,
169                // or a field of one that the closure captures by itself (a
170                // disjoint field capture).
171                [mir::ProjectionElem::Deref, mir::ProjectionElem::Field(..)]
172                    if place.local == ty::CAPTURE_STRUCT_LOCAL =>
173                {
174                    return error(format!(
175                        "a trigger must be {HEADS}, not a value from outside the quantifier"
176                    ));
177                }
178                // A field of a tuple or a closure that the trigger builds,
179                // which the simplifier folds away with the constructor (see
180                // below).
181                [mir::ProjectionElem::Field(..), ..] => {
182                    return match assigned(place.local) {
183                        Some(mir::Rvalue::Aggregate(
184                            box (mir::AggregateKind::Tuple | mir::AggregateKind::Closure(..)),
185                            _,
186                        )) => error(format!(
187                            "a trigger cannot be a field of a tuple or a closure it builds: it \
188                             must be {HEADS}"
189                        )),
190                        _ => Ok(()),
191                    };
192                }
193                [_, ..] => return Ok(()),
194                [] => {}
195            }
196            let Some(rvalue) = assigned(place.local) else {
197                // An argument (the closure itself or a bound variable), which
198                // nothing assigns; or else the result of a call, a local
199                // assigned several times, or the result left unassigned
200                // because the body ends in a call that never returns, which
201                // the encoding reports.
202                let argument = (1..=self.body.arg_count).contains(&place.local.as_usize());
203                return if argument && matches!(definitions.0[place.local], Set1::Empty) {
204                    variable()
205                } else {
206                    Ok(())
207                };
208            };
209            place = match rvalue {
210                mir::Rvalue::Use(mir::Operand::Copy(p) | mir::Operand::Move(p), _)
211                | mir::Rvalue::CopyForDeref(p) => *p,
212                // A reborrow `&*x` is inserted by rustc.
213                mir::Rvalue::Ref(_, _, p)
214                    if p.projection.as_slice() == [mir::ProjectionElem::Deref] =>
215                {
216                    mir::Place::from(p.local)
217                }
218                // So is a pointer coercion, e.g. unsizing `&[T; N]` to `&[T]`.
219                mir::Rvalue::Cast(
220                    mir::CastKind::PointerCoercion(..),
221                    mir::Operand::Copy(p) | mir::Operand::Move(p),
222                    _,
223                ) => *p,
224                mir::Rvalue::Use(mir::Operand::Constant(_) | mir::Operand::RuntimeChecks(_), _) => {
225                    return variable();
226                }
227                mir::Rvalue::Ref(..) | mir::Rvalue::RawPtr(..) => {
228                    return error(format!(
229                        "a trigger cannot be a borrow: it must be {HEADS} (a borrow can be the \
230                         argument of a call)"
231                    ));
232                }
233                // Rustc and Prusti build tuples and closures themselves (e.g.
234                // the arguments of a call to a closure), so the simplifier folds
235                // their constructors away too.
236                mir::Rvalue::Aggregate(
237                    box (mir::AggregateKind::Tuple | mir::AggregateKind::Closure(..)),
238                    _,
239                ) => {
240                    return error(format!(
241                        "a trigger cannot be a tuple or a closure: it must be {HEADS}"
242                    ));
243                }
244                _ => return Ok(()),
245            };
246        }
247    }
248}

250/// The locations that define each local of a body, as rustc's SSA analysis
251/// finds them: an assignment of the whole local, or the call it receives the
252/// result of. Any other mutation, such as an assignment of one of its fields,
253/// makes its value no single definition's.
254struct Definitions(IndexVec<mir::Local, Set1<mir::Location>>);
255
256impl<'tcx> Visitor<'tcx> for Definitions {
257    fn visit_local(&mut self, local: mir::Local, context: PlaceContext, location: mir::Location) {
258        if context.is_place_assignment() {
259            self.0[local].insert(location);
260        } else if let PlaceContext::MutatingUse(_) = context {
261            self.0[local] = Set1::Many;
262        }
263    }
264}

266/// The name of the quantified variable at position `idx` of the trigger
267/// closure `cl_def_id`, for error messages. Crate metadata records the
268/// parameter names of functions only, so a closure of another crate has none.
269fn bound_name<'vir>(vcx: &'vir vir::VirCtxt<'vir>, cl_def_id: DefId, idx: usize) -> Option<Symbol> {
270    if !cl_def_id.is_local() {
271        return None;
272    }
273    vcx.tcx()
274        .fn_arg_idents(cl_def_id)
275        .get(idx)
276        .copied()
277        .flatten()
278        .map(|ident| ident.name)
279}
