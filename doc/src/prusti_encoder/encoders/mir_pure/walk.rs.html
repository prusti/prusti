5impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
6    fn encode_common(
7        &mut self,
8        start: mir::BasicBlock,
9        end: mir::BasicBlock,
10        local_count: usize,
11        result_local: mir::Local,
12    ) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
13        let mut init = Update::new();
14        let v0 = Version::default();
15        // TODO: what about locals which never have StorageLive (i.e. always_live)?
16        init.versions.insert(mir::RETURN_PLACE, v0);
17        for local in 1..local_count {
18            let local_ex = self.vcx.mk_lazy_expr(
19                vir::vir_format!(self.vcx, "pure in _{local}"),
20                self.get_ty_for_local(local.into()),
21                Box::new(move |_vcx, lctx: ExprInput<'vir>| lctx.1[&local.into()].kind),
22            );
23            // check that `local` and `expr` type correspond
24            self.bump_version_init(&mut init, local.into(), local_ex, v0.location);
25        }
26
27        let update = self.encode_cfg(&init.versions, start, end)?;
28
29        // The body may unconditionally terminate with `unreachable` (e.g. a
30        // `match` on an uninhabited type): its result is the unreachable snapshot.
31        let Some(res) = init.merge(update) else {
32            let result_ty = self.body.local_decls[result_local].ty;
33            return Ok(self.ty_use(result_ty).unreachable_to_snap());
34        };
35        let ret_version = res.versions.get(&result_local).copied().unwrap_or(v0);
36        self.uses.read(result_local, ret_version.index);
37
38        let ex = self.mk_local_ex(result_local, ret_version);
39        Ok(self.reify_binds(res, ex))
40    }

42    pub(super) fn encode_body(&mut self) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
43        assert!(
44            !graph::is_cyclic(&self.body.basic_blocks),
45            "MIR pure encoding does not support loops"
46        );
47        self.encode_common(
48            mir::START_BLOCK,
49            self.rev_doms.end,
50            self.body.arg_count + 1,
51            mir::RETURN_PLACE,
52        )
53    }

55    pub(super) fn encode_spec_block(
56        &mut self,
57        block: mir::BasicBlock,
58    ) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
59        let Some(mir::TerminatorKind::Call { destination, .. }) = self.body.basic_blocks[block]
60            .terminator
61            .as_ref()
62            .map(|t| &t.kind)
63        else {
64            unreachable!("malformed spec-only block: should end in a call terminator");
65        };
66        assert!(destination.projection.is_empty());
67        self.encode_common(
68            block,
69            self.body.basic_blocks.successors(block).next().unwrap(),
70            self.body.local_decls.len(),
71            destination.local,
72        )
73    }

75    fn encode_cfg(
76        &mut self,
77        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
78        curr: mir::BasicBlock,
79        join_point: mir::BasicBlock,
80    ) -> EncodeResult<'vir, Option<Update<'vir>>, MirPureEnc> {
81        if curr == join_point {
82            // We are done with the current fragment of the CFG, the rest is
83            // handled in a parent call.
84            return Ok(Some(Update::new()));
85        }
86
87        // walk block statements first
88        let mut new_curr_ver = curr_ver.clone();
89        let mut stmt_update = Update::new();
90        for (statement_index, stmt) in self.body[curr].statements.iter().enumerate() {
91            let location = mir::Location {
92                block: curr,
93                statement_index,
94            };
95            let newer = self.encode_stmt(&new_curr_ver, stmt, location)?;
96            newer.add_to_map(&mut new_curr_ver);
97            stmt_update = stmt_update.merge_inner(newer);
98        }
99
100        // then walk terminator. Its operands and call are encoded under its
101        // span, like statements in `encode_stmt`; the walks of its successors
102        // are not, as they encode other blocks.
103        let term = self.body[curr].terminator.as_ref().unwrap();
104        let location = mir::Location {
105            block: curr,
106            statement_index: self.body[curr].statements.len(),
107        };
108        match &term.kind {
109            &mir::TerminatorKind::Goto { target }
110            | &mir::TerminatorKind::FalseEdge {
111                real_target: target,
112                ..
113            }
114            | &mir::TerminatorKind::Drop { target, .. } => {
115                let rest_update = self.encode_cfg(&new_curr_ver, target, join_point)?;
116                Ok(stmt_update.merge(rest_update))
117            }
118
119            mir::TerminatorKind::SwitchInt { discr, targets } => {
120                // The `if false` switch of a `ghost!` block or a `closure!`
121                // spec marker: continue straight into the live target (the
122                // inline ghost body, or the continuation); the other arm is
123                // skipped.
124                if let Some(live) = self.spec_arms.switches.get(&curr) {
125                    let rest_update = self.encode_cfg(&new_curr_ver, *live, join_point)?;
126                    return Ok(stmt_update.merge(rest_update));
127                }
128
129                // encode the discriminant operand
130                let discr_expr = self
131                    .vcx
132                    .with_span(term.source_info.span, |_| {
133                        self.encode_operand_snap(discr, &new_curr_ver)
134                    })?
135                    .downcast_ty();
136                let discr_ty = discr.ty(self.body, self.vcx.tcx());
137                let discr_ty_out = self.ty_use(discr_ty).expect_primitive();
138
139                // walk `curr` -> `targets[i]` -> `join` for each target. The
140                // join point the bb which is an immediate reverse dominator of
141                // the branch point.
142                // TODO: indexvec?
143                let new_join_point = self.rev_doms.immediate_dominator(curr);
144                let mut updates = targets
145                    .all_targets()
146                    .iter()
147                    .map(|target| self.encode_cfg(&new_curr_ver, *target, new_join_point))
148                    .collect::<Result<Vec<_>, _>>()?;
149
150                // find locals updated in any of the results, which were also
151                // defined before the branch
152                let mut mod_locals = updates
153                    .iter()
154                    .flat_map(|update| update.iter().flat_map(|v| v.versions.keys()))
155                    .filter(|local| new_curr_ver.contains_key(local))
156                    .copied()
157                    .collect::<Vec<_>>();
158                mod_locals.sort();
159                mod_locals.dedup();
160                let mod_tys = mod_locals
161                    .iter()
162                    .map(|l| self.get_ty_for_local(*l).as_dyn())
163                    .collect();
164
165                // for each branch, create a Viper tuple of the updated locals
166                let tuple_ref = self.deps.require_dep::<PairUseEnc>(mod_tys).unwrap();
167                let otherwise_update = updates.pop().unwrap();
168                let phi_expr = targets.iter().zip(updates).fold(
169                    self.reify_branch(&tuple_ref, &mod_locals, &new_curr_ver, otherwise_update),
170                    |expr, ((cond_val, _target), branch_update)| {
171                        self.vcx.mk_ternary_expr(
172                            self.vcx.mk_eq_expr(
173                                discr_ty_out.snap_to_prim(discr_expr),
174                                discr_ty_out.expr_from_bits(discr_ty, cond_val).lazy(),
175                            ),
176                            self.reify_branch(
177                                &tuple_ref,
178                                &mod_locals,
179                                &new_curr_ver,
180                                branch_update,
181                            ),
182                            expr,
183                        )
184                    },
185                );
186
187                // assign tuple into a `phi` variable
188                let phi_idx = Version {
189                    index: self.phi_ctr,
190                    location,
191                    initialised: None,
192                };
193                let phi = self.vcx.mk_local_decl(self.mk_phi(phi_idx), tuple_ref.ty);
194                self.phi_ctr += 1;
195                let mut phi_update = Update::new();
196                phi_update.binds.push(UpdateBind::Phi(phi, phi_expr));
197
198                // update locals by destructuring `phi` variable
199                // TODO: maybe this is unnecessary, we could instead use tuple
200                //   access directly instead of the locals going forward?
201                for (elem_idx, local) in mod_locals.iter().enumerate() {
202                    let expr = tuple_ref
203                        .mk_elem(self.vcx.mk_local_ex(phi), elem_idx)
204                        .downcast_ty();
205                    self.bump_version(&mut phi_update, *local, expr, location);
206                    new_curr_ver.insert(*local, phi_update.versions[local]);
207                }
208
209                // walk `join` -> `end`
210                let end_update = self.encode_cfg(&new_curr_ver, new_join_point, join_point)?;
211                Ok(stmt_update.merge(phi_update.merge(end_update)))
212            }
213
214            // TODO: there is some code duplication between here and SwitchInt
215            mir::TerminatorKind::Assert {
216                cond,
217                expected,
218                target,
219                ..
220            } => {
221                // encode the condition operand
222                let cond_ty = cond.ty(self.body, self.vcx.tcx());
223                assert_eq!(*cond_ty.kind(), TyKind::Bool);
224                let cond_expr = self
225                    .vcx
226                    .with_span(term.source_info.span, |_| {
227                        self.encode_operand_snap(cond, &new_curr_ver)
228                    })?
229                    .downcast_ty::<vir::Bool>();
230
231                // if cond == expected: walk the rest of the CFG
232                let ok_update = self.encode_cfg(&new_curr_ver, *target, join_point)?;
233
234                // find locals updated in the "ok" branch, which were also
235                // defined before the branch
236                // TODO: is the unwrap here ok? can ok_update also be None?
237                let mut mod_locals = ok_update
238                    .as_ref()
239                    .unwrap()
240                    .versions
241                    .keys()
242                    .filter(|local| new_curr_ver.contains_key(local))
243                    .copied()
244                    .collect::<Vec<_>>();
245                mod_locals.sort();
246                mod_locals.dedup();
247                let mod_tys = mod_locals
248                    .iter()
249                    .map(|l| self.get_ty_for_local(*l).as_dyn())
250                    .collect();
251
252                // create a Viper tuple of the updated locals
253                let tuple_ref = self.deps.require_dep::<PairUseEnc>(mod_tys).unwrap();
254                let phi_expr = self.vcx.mk_ternary_expr(
255                    self.vcx.mk_eq_expr(
256                        cond_expr,
257                        self.vcx
258                            .mk_const_expr(vir::ConstData::Bool(*expected))
259                            .downcast_ty(),
260                    ),
261                    self.reify_branch(&tuple_ref, &mod_locals, &new_curr_ver, ok_update),
262                    self.reify_branch(&tuple_ref, &mod_locals, &new_curr_ver, None),
263                );
264
265                // assign tuple into a `phi` variable
266                let phi_idx = Version {
267                    index: self.phi_ctr,
268                    location,
269                    initialised: None,
270                };
271                let phi = self.vcx.mk_local_decl(self.mk_phi(phi_idx), tuple_ref.ty);
272                self.phi_ctr += 1;
273                let mut phi_update = Update::new();
274                phi_update.binds.push(UpdateBind::Phi(phi, phi_expr));
275
276                // update locals by destructuring `phi` variable
277                // TODO: maybe this is unnecessary, we could instead use tuple
278                //   access directly instead of the locals going forward?
279                for (elem_idx, local) in mod_locals.iter().enumerate() {
280                    let expr = tuple_ref
281                        .mk_elem(self.vcx.mk_local_ex(phi), elem_idx)
282                        .downcast_ty();
283                    self.bump_version(&mut phi_update, *local, expr, location);
284                    new_curr_ver.insert(*local, phi_update.versions[local]);
285                }
286
287                Ok(stmt_update.merge(Some(phi_update)))
288            }
289
290            mir::TerminatorKind::Return => Ok(Some(stmt_update)),
291
292            mir::TerminatorKind::Unreachable => {
293                // will result in generating an unreachable phi merge
294                Ok(None)
295            }
296
297            mir::TerminatorKind::Call {
298                func,
299                args,
300                destination,
301                target,
302                ..
303            } => {
304                let Some(target) = *target else {
305                    // target is none if the call diverges (i.e. returns !), in
306                    // this case treat it just like an unreachable terminator.
307                    return Ok(None);
308                };
309
310                let func_ty = func.ty(self.body, self.vcx.tcx());
311                let (def_id, arg_tys) = RustSignature::get_def_id_and_caller_substs(func_ty);
312                let expr = self.vcx.with_span(term.source_info.span, |_| {
313                    // Diverging calls were handled above; every other call
314                    // must be to an intrinsic, a Prusti builtin or a pure
315                    // function.
316                    let is_pure = crate::encoders::is_function_pure(def_id, self.gargs(arg_tys));
317
318                    // The bodiless `ptr_metadata` intrinsic is only lowered to
319                    // `UnOp::PtrMetadata` in optimized MIR; do the lowering here.
320                    let intrinsic = self.vcx.tcx().intrinsic(def_id);
321                    let intrinsic = intrinsic.and_then(RustcIntrinsic::from_intrinsic);
322                    if let Some(intrinsic) = intrinsic {
323                        self.encode_intrinsic(intrinsic, arg_tys, args, &new_curr_ver)
324                    } else if let Some(builtin) = PrustiBuiltin::new(def_id, self.gargs(arg_tys)) {
325                        match self.encode_prusti_builtin(
326                            builtin,
327                            def_id,
328                            self.gargs(arg_tys),
329                            args,
330                            term.source_info.span,
331                            &new_curr_ver,
332                        )? {
333                            Some(expr) => Ok(expr),
334                            // The pure-only builtins (quantifiers, spec blocks,
335                            // mode markers).
336                            None => {
337                                let PrustiBuiltin::Spec(builtin) = builtin else {
338                                    // Operand-based builtins are handled by
339                                    // `PureRvalueEnc::encode_prusti_builtin` before this is reached.
340                                    unreachable!(
341                                        "operand-based builtin in the pure-only handler: {builtin:?}"
342                                    )
343                                };
344                                self.encode_pure_only_builtin(
345                                    builtin,
346                                    arg_tys,
347                                    args,
348                                    curr,
349                                    &new_curr_ver,
350                                )
351                            }
352                        }
353                    } else if is_pure {
354                        let pure_func = self.deps.require_dep_spanned::<FunctionCallEnc>(
355                            CallTaskDescription::new(self.context, arg_tys, def_id),
356                            term.source_info.span,
357                        )?;
358                        let snap_args = args
359                            .iter()
360                            .map(|arg| self.encode_operand_snap(&arg.node, &new_curr_ver))
361                            .collect::<Result<Vec<_>, _>>()?;
362                        Ok(pure_func.call_pure(snap_args))
363                    } else {
364                        Err(EncodeFullError::DependencyError(vec![
365                            DependencyErrorLink::new(
366                                MirPureEnc::ENCODER_NAME,
367                                format!(
368                                    "use of impure function `{}` in pure code is not allowed",
369                                    self.vcx.tcx().def_path_str(def_id)
370                                ),
371                                vec![term.source_info.span],
372                                ErrorKind::Incorrect,
373                            ),
374                        ]))
375                    }
376                });
377
378                let expr = self.encode_place_update(
379                    &new_curr_ver,
380                    *destination,
381                    expr?,
382                    term.source_info.span,
383                )?;
384                let mut term_update = Update::new();
385                self.bump_version(&mut term_update, destination.local, expr, location);
386                term_update.add_to_map(&mut new_curr_ver);
387
388                // walk rest of CFG
389                let end_update = self.encode_cfg(&new_curr_ver, target, join_point)?;
390
391                Ok(stmt_update.merge_inner(term_update).merge(end_update))
392            }
393
394            k => todo!("terminator kind {k:?}"),
395        }
396    }

398    fn encode_stmt(
399        &mut self,
400        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
401        stmt: &mir::Statement<'vir>,
402        location: mir::Location,
403    ) -> EncodeResult<'vir, Update<'vir>, MirPureEnc> {
404        // Encoded under the statement's span: expressions pick up the ambient
405        // span at creation, which makes error positions inside pure/spec
406        // bodies point at the originating source (e.g. the failing conjunct
407        // of a spec).
408        self.vcx.with_span(stmt.source_info.span, |_| {
409            let mut update = Update::new();
410            match &stmt.kind {
411                &mir::StatementKind::StorageLive(local) => {
412                    let new_version = self.bump_version_no_assign(local, location);
413                    update.versions.insert(local, new_version);
414                }
415                mir::StatementKind::StorageDead(..)
416                | mir::StatementKind::FakeRead(..)
417                | mir::StatementKind::AscribeUserType(..)
418                | mir::StatementKind::PlaceMention(..)
419                | mir::StatementKind::Coverage(..)
420                | mir::StatementKind::ConstEvalCounter
421                | mir::StatementKind::Nop
422                | mir::StatementKind::BackwardIncompatibleDropHint { .. } => {} // nop
423                mir::StatementKind::Assign(box (dest, rvalue)) => {
424                    let span = stmt.source_info.span;
425                    self.uses.start_assignment();
426                    let expr = self
427                        .encode_rvalue(curr_ver, rvalue, span)
428                        .and_then(|expr| self.encode_place_update(curr_ver, *dest, expr, span));
429                    let reads = self.uses.end_assignment();
430                    self.bump_version(&mut update, dest.local, expr?, location);
431                    let version = update.versions[&dest.local].index;
432                    self.uses.assigned(dest.local, version, reads);
433                }
434                k => todo!("statement kind {k:?}"),
435            }
436            Ok(update)
437        })
438    }
