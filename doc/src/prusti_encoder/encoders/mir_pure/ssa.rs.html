1//! The versions of locals and the `let`s that bind them.
2
3use super::*;
4
5impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
6    pub(super) fn mk_local_ex(
7        &mut self,
8        _local: mir::Local,
9        version: Version<'vir>,
10    ) -> ExprRet<'vir> {
11        self.vcx.mk_local_ex(version.initialised.unwrap())
12    }
13
14    pub(super) fn mk_phi(&self, version: Version<'vir>) -> &'vir str {
15        vir::vir_format!(self.vcx, "_{}_phi_{}", self.encoding_depth, version)
16    }
17
18    pub(super) fn bump_version(
19        &mut self,
20        update: &mut Update<'vir>,
21        local: mir::Local,
22        expr: ExprRet<'vir>,
23        location: mir::Location,
24    ) {
25        if self.version_ctr[local] == 0 {
26            // Version 0 signifies "inputs into the expression", as checked in
27            // `do_encode_full` to figure out which locals should be encoded
28            // and provided by the caller/user of the encoded expression. In
29            // most cases, version 0 is not assigned to for non-input locals,
30            // because a preceding `StorageLive` bumps the version before the
31            // assignment takes place, and so any later reads of that local use
32            // at least version 1. However, there are some operations, such as
33            // checked binary operations, that do not cause the compiler to
34            // emit a `StorageLive` for their target place. This workaround
35            // checks for such a case and bumps the version twice.
36            //
37            // see: https://github.com/rust-lang/rust/issues/67400
38            self.bump_version_no_assign(local, location);
39        }
40
41        let new_version = self.bump_version_no_assign(local, location);
42        // check that `local` and `expr` type correspond
43        update.assign(self.vcx, self.encoding_depth, local, new_version, expr);
44    }
45
46    pub(super) fn bump_version_init(
47        &mut self,
48        update: &mut Update<'vir>,
49        local: mir::Local,
50        expr: ExprRet<'vir>,
51        location: mir::Location,
52    ) {
53        let new_version = self.bump_version_no_assign(local, location);
54        // check that `local` and `expr` type correspond
55        update.assign(self.vcx, self.encoding_depth, local, new_version, expr);
56    }
57
58    pub(super) fn bump_version_no_assign(
59        &mut self,
60        local: mir::Local,
61        location: mir::Location,
62    ) -> Version<'vir> {
63        let index = self.version_ctr[local];
64        self.version_ctr[local] += 1;
65        Version {
66            index,
67            location,
68            initialised: None,
69        }
70    }
71
72    pub(super) fn reify_binds<T: CompType>(
73        &self,
74        update: Update<'vir>,
75        expr: ExprRetAny<'vir, T>,
76    ) -> ExprRetAny<'vir, T> {
77        update.binds.iter().rfold(expr, |expr, bind| match bind {
78            UpdateBind::Local(local, version, val) => {
79                // skip bindings which were not used
80                // TODO: this might optimise away some function calls which
81                //   can act as triggers for quantifiers
82                if !self.uses.is_read(*local, version.index) {
83                    expr
84                } else {
85                    let decl = version.initialised.unwrap();
86                    self.vcx.mk_let_expr(decl, val, expr)
87                }
88            }
89            UpdateBind::Phi(phi, val) => self.vcx.mk_let_expr(*phi, val, expr),
90        })
91    }
92
93    pub(super) fn reify_branch(
94        &mut self,
95        tuple_ref: &PairUse<'vir>,
96        mod_locals: &[mir::Local],
97        curr_ver: &FxHashMap<mir::Local, Version<'vir>>,
98        update: Option<Update<'vir>>,
99    ) -> ExprRetAny<'vir, vir::Pair> {
100        update
101            .map(|update| {
102                let tuple_args = mod_locals
103                    .iter()
104                    .map(|local| {
105                        let version = update.versions.get(local).copied().unwrap_or_else(|| {
106                            // TODO: remove (debug)
107                            if !curr_ver.contains_key(local) {
108                                tracing::error!("unknown version of local! {}", local.as_usize());
109                                return Version {
110                                    index: 0xff,
111                                    ..Default::default()
112                                };
113                            }
114                            curr_ver[local]
115                        });
116
117                        self.uses.read(*local, version.index);
118                        self.mk_local_ex(*local, version)
119                    })
120                    .collect::<Vec<_>>();
121                self.reify_binds(update, tuple_ref.mk_cons(&tuple_args))
122            })
123            .unwrap_or_else(|| {
124                // A branch whose update is `None` cannot complete (it ends in
125                // `unreachable` or a diverging call, or is the failing side
126                // of an `Assert`). It merges the unreachable snapshots of the
127                // locals' types, which assume `false` where they are
128                // evaluated; with no locals to merge, nothing is assumed.
129                let unreachable = mod_locals
130                    .iter()
131                    .map(|local| {
132                        self.ty_use(self.body.local_decls[*local].ty)
133                            .unreachable_to_snap()
134                    })
135                    .collect::<Vec<_>>();
136                tuple_ref.mk_cons(&unreachable)
137            })
138    }
139}
