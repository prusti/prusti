3use super::*;
4
5impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
6    pub(super) fn encode_rvalue(
7        &mut self,
8        curr_ver: &Versions<'vir>,
9        rvalue: &mir::Rvalue<'vir>,
10        span: Span,
11    ) -> EncodeResult<'vir, ExprRet<'vir>, MirPureEnc> {
12        let rvalue_ty = rvalue.ty(self.body, self.vcx.tcx());
13        match rvalue {
14            mir::Rvalue::Use(op, _) => self.encode_operand_snap(op, curr_ver),
15            mir::Rvalue::Ref(_, kind, place) => {
16                // Reading a mutable reference's referent needs an address
17                // holding a predicate, which pure-created mutrefs don't have.
18                if kind.mutability().is_mut() {
19                    return Err(EncodeFullError::unsupported(
20                        format!(
21                            "mutable borrow of `{}` in a specification: take a shared \
22                             reference (`&`) instead",
23                            place.ty(self.body, self.vcx.tcx()).ty
24                        ),
25                        vec![self.current_span()],
26                    ));
27                }
28                let (value, place_ref, metadata) =
29                    self.encode_borrowed_place(curr_ver, (*place).into(), rvalue_ty)?;
30                let e_rvalue_ty = self.ty_use(rvalue_ty)?.expect_immref();
31                Ok(e_rvalue_ty
32                    .prim_to_snap(place_ref, metadata, value)
33                    .upcast_ty())
34            }
35            mir::Rvalue::BinaryOp(op, box (l, r)) => {
36                self.encode_binop_snap(rvalue_ty, *op, l, r, curr_ver, span)
37            }
38            mir::Rvalue::UnaryOp(unop, operand) => {
39                self.encode_unary_op_snap(rvalue_ty, *unop, operand, curr_ver)
40            }
41            mir::Rvalue::Aggregate(
42                box kind @ (mir::AggregateKind::Adt(..)
43                | mir::AggregateKind::Tuple
44                | mir::AggregateKind::Closure(..)),
45                fields,
46            ) => self.encode_aggregate_snap(rvalue_ty, kind, fields, curr_ver),
47            mir::Rvalue::Aggregate(box mir::AggregateKind::Array(..), elements) => {
48                self.encode_array_snap(rvalue_ty, elements, curr_ver)
49            }
50            mir::Rvalue::Repeat(operand, len) => {
51                self.encode_repeat_snap(rvalue_ty, operand, *len, curr_ver)
52            }
53            mir::Rvalue::Discriminant(place) => {
54                let place_ty = place.ty(self.body, self.vcx.tcx());
55                let ty = self.ty_use(place_ty.ty)?;
56                if ty.specifics.is_opaque() {
57                    return Err(unsupported_opaque(place_ty.ty, span));
58                }
59                let discr = match ty
60                    .get_enumlike()
61                    .filter(|_| place_ty.variant_index.is_none())
62                {
63                    Some(ty) => ty.snap_to_discr_snap(
64                        self.encode_place_snap((*place).into(), curr_ver)?
65                            .downcast_ty(),
66                    ),
67                    None => {
68                        let e_rvalue_ty = self.ty_use(rvalue_ty)?.expect_primitive();
69                        // mir::Rvalue::Discriminant documents "Returns zero for types without discriminant"
70                        let zero = self.vcx.mk_uint::<0>();
71                        e_rvalue_ty.prim_to_snap(zero.upcast_ty()).lazy()
72                    }
73                };
74                Ok(discr.upcast_ty())
75            }
76            mir::Rvalue::Cast(kind, operand, ty) => {
77                assert_eq!(*ty, rvalue_ty);
78                // `encode_cast_snap` also returns a statement, which moves the
79                // referent's predicate in a `&mut` unsize coercion. Pure code
80                // holds no predicates, so only the value (`.1`) is used.
81                Ok(self
82                    .encode_cast_snap(rvalue_ty, *kind, operand, curr_ver)?
83                    .1)
84            }
85            mir::Rvalue::RawPtr(_, place) => {
86                let (_, place_ref, metadata) =
87                    self.encode_borrowed_place(curr_ver, (*place).into(), rvalue_ty)?;
88                let raw_ty = self.ty_use(rvalue_ty)?;
89                Ok(raw_ty
90                    .expect_raw()
91                    .prim_to_snap(place_ref, metadata)
92                    .upcast_ty())
93            }
94            _ => Err(EncodeFullError::unsupported(
95                format!("unsupported rvalue {rvalue:?} in pure code"),
96                vec![span],
97            )),
98        }
99    }
100}
