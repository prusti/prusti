47/// The values of an encoded body's inputs.
48pub type Inputs<'vir> = &'vir FxHashMap<mir::Local, vir::ExprSnap<'vir>>;
49/// The values of a spec closure's captures, in the modes its body reads them
50/// in (see `MirPureEncOutput::captures`).
51type Captures<'vir> = &'vir IndexSlice<abi::FieldIdx, FxHashMap<Modes, vir::ExprSnap<'vir>>>;
52/// What an encoded body is instantiated with: the body's `DefId`, the values
53/// of its inputs, the state `old` refers to, and, for a spec closure, the
54/// values of its captures.
55type ExprInput<'vir> = (DefId, Inputs<'vir>, vir::OldLabel<'vir>, Captures<'vir>);
56type ExprRet<'vir> = vir::ExprGenSnap<'vir, ExprInput<'vir>, vir::ExprKind<'vir>>;
57type ExprRetRef<'vir> = vir::ExprGenRef<'vir, ExprInput<'vir>, vir::ExprKind<'vir>>;
58type ExprRetAny<'vir, T> = vir::ExprGen<'vir, ExprInput<'vir>, vir::ExprKind<'vir>, T>;
59/// An encoded spec closure (see `Enc::encode_spec_closure`): the quantified
60/// variables derived from the closure's arguments, and the closure's body.
61type SpecClosure<'vir> = (
62    &'vir [vir::LocalDeclSnap<'vir>],
63    ExprRetAny<'vir, vir::Bool>,
64);

77impl<'vir> MirPureEncOutput<'vir> {
78    /// The encoded body for the given values of its inputs, with `old`
79    /// referring to the pre-state of the method.
80    pub fn reify(&self, inputs: Inputs<'vir>) -> vir::ExprSnap<'vir> {
81        self.reify_at(inputs, vir::OldLabel::None)
82    }
83
84    /// Like `reify`, with `old(..)` in the body referring to the state `old`.
85    pub fn reify_at(&self, inputs: Inputs<'vir>, old: vir::OldLabel<'vir>) -> vir::ExprSnap<'vir> {
86        use vir::Reify;
87        let input = (self.def_id, inputs, old, IndexSlice::empty());
88        vir::with_vcx(|vcx| self.expr.reify(vcx, input))
89    }

93    /// The modes a specification reads places in. A mode only changes reads
94    /// rooted at an argument or the result: `OLD` and `BEFORE_EXPIRY` the
95    /// heap read of a `&mut` dereference, `REL0`/`REL1` the whole read.
96    #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
97    struct Modes: u8 {
98        const OLD = 1 << 0;
99        const REL0 = 1 << 1;
100        const REL1 = 1 << 2;
101        const BEFORE_EXPIRY = 1 << 3;
102    }

106pub enum PureKind {
107    /// The closure passed to a quantifier or to `spec_block`, encoded as part
108    /// of the expression that contains it, in the generic context of the item
109    /// `context`.

114    /// not the specified item itself when an `impl` refines the

121        /// snapshots are deep (see `Enc::impure_context`).

126    SpecBlock(mir::BasicBlock),

168    fn do_encode_full<'vir>(
169        task_key: &Self::TaskKey<'vir>,
170        deps: &mut TaskEncoderDependencies<'vir, Self>,
171    ) -> EncodeFullResult<'vir, Self> {
172        deps.emit_output_ref(*task_key, ())?;
173
174        let (_, kind, def_id, gargs) = *task_key;
175
176        tracing::debug!("encoding {def_id:?}");
177        let (inputs, expr, captures) = vir::with_vcx(move |vcx| {
178            // A specification is instantiated with the arguments of the
179            // encoding, which differ from its own when a trait's specification
180            // is encoded for an `impl` of that trait. So is the closure of a
181            // quantifier or `spec_block`, which is encoded in the context of
182            // the expression it is part of. The other bodies are encoded
183            // generically, so they are taken as stored.
184            let body = match kind {
185                PureKind::Closure { context } => {
186                    let substs = vcx.tcx().mk_args(gargs.args());
187                    vcx.body_mut().get_closure_body(def_id, substs, context)
188                }
189                PureKind::Spec { context, .. } => {
190                    let substs = vcx.tcx().mk_args(gargs.args());
191                    vcx.body_mut().get_spec_body(def_id, substs, context)
192                }
193                PureKind::Pure => crate::encoders::pure_body(def_id),
194                PureKind::Constant(promoted) => {
195                    vcx.body_mut().get_promoted_constant_body(def_id, promoted)
196                }
197                PureKind::SpecBlock(_) => crate::encoders::impure_body(def_id)
198                    .unwrap_or_else(|| panic!("no body to encode for {def_id:?}")),
199            };
200
201            // Encode under the body's span: expression nodes pick up the
202            // ambient span at creation, so everything in a pure/spec body
203            // defaults to the body's source (e.g. the spec attribute for a
204            // spec closure), refined per statement and terminator. This is
205            // what error positions inside the body backtranslate to. The MIR
206            // span of a body deserialized from another crate is dummy; the
207            // definition's span survives in the crate metadata.
208            let body_span = if body.span.is_dummy() {
209                vcx.tcx().def_span(def_id)
210            } else {
211                body.span
212            };
213            let mut enc = Enc::new(vcx, task_key.0, def_id, kind, gargs, &body, deps);
214            let expr_inner = vcx.with_span(body_span, |_| {
215                if let PureKind::SpecBlock(block) = kind {
216                    enc.encode_spec_block(block)
217                } else {
218                    enc.encode_body()
219                }
220            })?;
221            let captures = enc.captures.take().unwrap_or_default();
222            let inputs = std::mem::take(&mut enc.uses)
223                .read
224                .into_iter()
225                .filter(|(l, v)| *l != mir::RETURN_PLACE && *v == 0)
226                .map(|(l, _v)| l)
227                .unique()
228                .sorted()
229                .collect::<Vec<_>>();
230            let inputs_expected = inputs.len();
231
232            // We wrap the expression with an additional lazy that will perform
233            // some sanity checks. These requirements cannot be expressed using
234            // only the type system.
235            let snapshot = if let PureKind::SpecBlock(..) = kind {
236                vir::TYPE_BOOL.upcast_ty()
237            } else {
238                let ret = RustTyDecomposition::from_ty(body.return_ty(), enc.context);
239                deps.require_ref::<TyUsePureEnc>(ret)?.snapshot
240            };
241            // Created under the body's span: this wrapper is the node whose
242            // span survives reification as the top of the expression (`span`
243            // is passed through by `reify`), so it is what error positions
244            // pointing at the whole body (e.g. a failing precondition
245            // conjunct) backtranslate to.
246            let expr = vcx.with_span(body_span, |vcx| {
247                vcx.mk_lazy_expr(
248                    vir::vir_format!(vcx, "pure body {def_id:?}"),
249                    snapshot,
250                    Box::new(move |vcx, lctx: ExprInput<'_>| {
251                        // check: are we actually providing inputs for the
252                        //   correct `DefId`?
253                        assert_eq!(lctx.0, def_id);
254
255                        // check: are we providing the expected number of inputs?
256                        // TODO: check that the expected inputs are present; this
257                        //   check is not precise
258                        assert!(lctx.1.len() >= inputs_expected);
259
260                        use vir::Reify;
261                        expr_inner.kind.reify(vcx, lctx)
262                    }),
263                )
264            });
265            add_debug_note!(expr.debug_info, "Inner expr: {}", expr_inner.debug_info);
266            Ok((inputs, expr, captures))
267        })?;
268        tracing::debug!("finished {def_id:?}");
269
270        let output = MirPureEncOutput {
271            inputs,
272            def_id,
273            expr,
274            captures,
275        };
276        Ok(((), output))
277    }
278}

292impl<'vir> Update<'vir> {
293    fn new() -> Self {
294        Self::default()
295    }
296
297    fn mk_local(
298        vcx: &'vir vir::VirCtxt<'vir>,
299        encoding_depth: usize,
300        local: mir::Local,
301        version: Version<'vir>,
302    ) -> &'vir str {
303        vir::vir_format!(vcx, "_{}_{}s_{}", encoding_depth, local.as_usize(), version)
304    }
305
306    fn assign(
307        &mut self,
308        vcx: &'vir vir::VirCtxt<'vir>,
309        encoding_depth: usize,
310        local: mir::Local,
311        mut version: Version<'vir>,
312        expr: ExprRet<'vir>,
313    ) {
314        let decl = vcx.mk_local_decl(
315            Self::mk_local(vcx, encoding_depth, local, version),
316            expr.ty(),
317        );
318        version.initialised = Some(decl);
319        self.binds.push(UpdateBind::Local(local, version, expr));
320        self.versions.insert(local, version);
321    }
322
323    fn merge(self, newer: Option<Self>) -> Option<Self> {
324        Some(self.merge_inner(newer?))
325    }
326
327    fn merge_inner(self, newer: Self) -> Self {
328        Self {
329            binds: self.binds.into_iter().chain(newer.binds).collect(),
330            versions: self.versions.into_iter().chain(newer.versions).collect(),
331        }
332    }
333
334    fn add_to_map(&self, curr_ver: &mut FxHashMap<mir::Local, Version<'vir>>) {
335        for (local, ver) in &self.versions {
336            curr_ver.insert(*local, *ver);
337        }
338    }

341/// The versions of locals that the encoded expression reads, directly or
342/// through the value of a version it reads. Only the bindings of these
343/// versions are emitted, so that a binding nothing reads emits nothing its
344/// value reads either.
345#[derive(Default)]
346struct Uses {
347    /// The versions read. Closed under `pending`: once a version is read, so
348    /// are the versions its value reads.
349    read: FxHashSet<(mir::Local, usize)>,
350    /// For each assigned version not read yet, the versions its value reads.
351    pending: FxHashMap<(mir::Local, usize), Vec<(mir::Local, usize)>>,
352    /// While an assignment is encoded, the versions its value reads so far.
353    assignment: Option<Vec<(mir::Local, usize)>>,
354}
355
356impl Uses {
357    /// Records a read of version `index` of `local`: by the value of the
358    /// assignment being encoded, if any, or else by the expression itself.
359    fn read(&mut self, local: mir::Local, index: usize) {
360        if let Some(reads) = &mut self.assignment {
361            reads.push((local, index));
362        } else if self.read.insert((local, index))
363            && let Some(reads) = self.pending.remove(&(local, index))
364        {
365            for (local, index) in reads {
366                self.read(local, index);
367            }
368        }
369    }
370
371    /// Starts recording the reads of an assignment's value.
372    fn start_assignment(&mut self) {
373        debug_assert!(self.assignment.is_none(), "assignments do not nest");
374        self.assignment = Some(Vec::new());
375    }
376
377    /// Stops recording the reads of an assignment's value, returning them.
378    fn end_assignment(&mut self) -> Vec<(mir::Local, usize)> {
379        self.assignment.take().unwrap()
380    }
381
382    /// Records that version `index` of `local` is assigned a value that
383    /// reads `reads`; they count as read once that version is.
384    fn assigned(&mut self, local: mir::Local, index: usize, reads: Vec<(mir::Local, usize)>) {
385        self.pending.insert((local, index), reads);
386    }
387
388    fn is_read(&self, local: mir::Local, index: usize) -> bool {
389        self.read.contains(&(local, index))
390    }
391}

486impl<'vir: 'enc, 'enc> Enc<'vir, 'enc> {
487    fn new(
488        vcx: &'vir vir::VirCtxt<'vir>,
489        encoding_depth: usize,
490        def_id: DefId,
491        kind: PureKind,
492        gargs: GArgs<'vir>,
493        body: &'enc mir::Body<'vir>,
494        deps: &'enc mut TaskEncoderDependencies<'vir, MirPureEnc>,
495    ) -> Self {
496        let rev_doms = rev_doms::ReverseDominators::new(&body.basic_blocks);
497        Self {
498            vcx,
499            encoding_depth,
500            context: gargs.context(),
501            context_def_id: match kind {
502                PureKind::Closure { context } | PureKind::Spec { context, .. } => context,
503                PureKind::Pure | PureKind::Constant(_) | PureKind::SpecBlock(_) => def_id,
504            },
505            body,
506            rev_doms,
507            spec_arms: SpecArms::for_body(def_id, body),
508            deps,
509            // visited: IndexVec::from_elem_n(false, body.basic_blocks.len()),
510            version_ctr: IndexVec::from_elem_n(0, body.local_decls.len()),
511            uses: Default::default(),
512            phi_ctr: 0,
513            modes: Modes::empty(),
514            // An impure method's specs and contained spec blocks get shallow
515            // argument snapshots (and can take a snapshot of the predicates
516            // behind `&mut`s). Pure functions currently also only get shallow
517            // snapshots but cannot get the value behind `&mut`s.
518            impure_context: matches!(
519                kind,
520                PureKind::Spec {
521                    mode: MirSpecEncMode::Impure,
522                    ..
523                } | PureKind::SpecBlock(..)
524            ),
525            captures: matches!(kind, PureKind::Closure { .. }).then(|| {
526                let closure_ty = body.local_decls[ty::CAPTURE_STRUCT_LOCAL].ty.peel_refs();
527                let TyKind::Closure(_, args) = closure_ty.kind() else {
528                    unreachable!("a closure body takes its closure first");
529                };
530                IndexVec::from_elem_n(FxHashSet::default(), args.as_closure().upvar_tys().len())
531            }),
532        }
533    }

543        self.deps

545            .unwrap()
546            .snapshot

550mod rev_doms {
551    /// Identical to `body.basic_blocks.dominators()` except in reverse. Since
552    /// there may be multiple `Return`/`Unreachable`/etc. terminators, we add a
553    /// special end block index which is invalid in `basic_blocks` but pretends
554    /// to be the successor of all these no-successor blocks.
555    pub struct ReverseDominators {
556        pub dom: dominators::Dominators<mir::BasicBlock>,
557        pub end: mir::BasicBlock,
558    }
559    impl ReverseDominators {
560        #[allow(clippy::needless_lifetimes)]
561        pub fn new<'a, 'vir>(blocks: &'a mir::BasicBlocks<'vir>) -> Self {
562            let no_succ_blocks = blocks
563                .iter_enumerated()
564                .filter(|(_, data)| {
565                    /* The commented line below prevents e.g. a SwitchInt with a branch
566                    going to an Unreachable terminator from having a join point, so
567                    do not treat such terminators as return blocks. */
568                    // data.terminator().successors().next().is_none()
569                    matches!(
570                        data.terminator().kind,
571                        mir::TerminatorKind::Return | mir::TerminatorKind::UnwindResume
572                    )
573                })
574                .map(|(bb, _)| bb)
575                .collect();
576            let rbb = RevBasicBlocks(blocks, no_succ_blocks);
577            Self {
578                dom: dominators::dominators(&rbb),
579                end: rbb.start_node(),
580            }
581        }
582        pub fn immediate_dominator(&self, bb: mir::BasicBlock) -> mir::BasicBlock {
583            // This unwrap should never fail since all blocks can reach `end`
584            self.dom.immediate_dominator(bb).unwrap()
585        }
586    }
587
588    use super::*;
589    use prusti_rustc_interface::data_structures::graph::*;

591    /// A wrapper around `mir::BasicBlocks` which reverses the direction of the
592    /// edges. Implements `ControlFlowGraph` such that we can call `dominators`.
593    struct RevBasicBlocks<'a, 'vir>(&'a mir::BasicBlocks<'vir>, Vec<mir::BasicBlock>);
594    impl DirectedGraph for RevBasicBlocks<'_, '_> {
595        type Node = mir::BasicBlock;
596
597        fn num_nodes(&self) -> usize {
598            self.0.len() + 1
599        }
600    }
601    impl StartNode for RevBasicBlocks<'_, '_> {
602        fn start_node(&self) -> Self::Node {
603            self.0.next_index()
604        }
605    }
606
607    impl Predecessors for RevBasicBlocks<'_, '_> {
608        fn predecessors(&self, node: Self::Node) -> impl Iterator<Item = Self::Node> {
609            if node == self.start_node() {
610                Box::new([].into_iter())
611            } else if self.1.contains(&node) {
612                Box::new([self.start_node()].into_iter())
613            } else {
614                Box::new(self.0.successors(node)) as Box<dyn Iterator<Item = _>>
615            }
616        }
617    }
618
619    impl Successors for RevBasicBlocks<'_, '_> {
620        fn successors(&self, node: Self::Node) -> impl Iterator<Item = Self::Node> {
621            if node == self.start_node() {
622                Box::new(self.1.iter().copied()) as Box<dyn Iterator<Item = _>>
623            } else {
624                Box::new((&self.0).predecessors(node)) as Box<dyn Iterator<Item = _>>
625            }
626        }
627    }
628}

631struct Version<'vir> {
632    index: usize,
633    location: mir::Location,
634    initialised: Option<vir::LocalDeclSnap<'vir>>,
635}
