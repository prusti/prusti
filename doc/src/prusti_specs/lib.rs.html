57/// The argument of an attribute in inner position, i.e. its tokens with the
58/// surrounding delimiters dropped, or no tokens if it has no argument. These
59/// are the tokens rustc passes to an attribute macro in outer position.
60fn unwrap_argument(attr: &syn::Attribute) -> syn::Result<TokenStream> {
61    match &attr.meta {
62        syn::Meta::Path(_) => Ok(TokenStream::new()),
63        syn::Meta::List(list) => Ok(list.tokens.clone()),
64        syn::Meta::NameValue(_) => {
65            let name = &attr.path().segments.last().unwrap().ident;
66            Err(syn::Error::new(
67                attr.span(),
68                format!("`#[{name} = ...]` is not supported"),
69            ))
70        }
71    }
72}

74/// The kind of a Prusti attribute written as `#[kind]` or
75/// `#[prusti_contracts::kind]`, or `None` for any other attribute.
76fn prusti_attribute_kind(attr: &syn::Attribute) -> Option<SpecAttributeKind> {
77    let segments = &attr.path().segments;
78    let is_prusti_path =
79        segments.len() == 1 || (segments.len() == 2 && segments[0].ident == "prusti_contracts");
80    if !is_prusti_path {
81        return None;
82    }
83    segments.last()?.ident.to_string().try_into().ok()
84}

86/// Removes the Prusti attributes that apply to functions from `item`, as
87/// [extract_applicable_attributes] describes.
88fn extract_prusti_attributes(
89    item: &mut untyped::AnyFnItem,
90) -> syn::Result<Vec<(SpecAttributeKind, Span, TokenStream)>> {
91    extract_applicable_attributes(item.attrs_mut(), |kind| match kind {
92        SpecAttributeKind::Requires
93        | SpecAttributeKind::Ensures
94        | SpecAttributeKind::AfterExpiry
95        | SpecAttributeKind::AssertOnExpiry
96        | SpecAttributeKind::RefineSpec
97        | SpecAttributeKind::Terminates
98        | SpecAttributeKind::Pure
99        | SpecAttributeKind::Trusted
100        | SpecAttributeKind::Verified => true,
101        SpecAttributeKind::Invariant
102        | SpecAttributeKind::Model
103        | SpecAttributeKind::PrintCounterexample => false,
104    })
105}

128/// Removes the Prusti attributes whose kind `applies` to the item from
129/// `attrs`, returning each with its span and argument. The others, such as
130/// `#[invariant]` on a function, are left for the compiler to expand: a
131/// Prusti attribute then rejects the item it does not apply to, and an
132/// attribute of another crate that has the same name keeps its meaning.
133fn extract_applicable_attributes(
134    attrs: &mut Vec<syn::Attribute>,
135    applies: impl Fn(SpecAttributeKind) -> bool,
136) -> syn::Result<Vec<(SpecAttributeKind, Span, TokenStream)>> {
137    let mut prusti_attributes = Vec::new();
138    let mut regular_attributes = Vec::new();
139    for attr in attrs.drain(0..) {
140        let attr_kind = match prusti_attribute_kind(&attr) {
141            Some(attr_kind) if applies(attr_kind) => attr_kind,
142            _ => {
143                regular_attributes.push(attr);
144                continue;
145            }
146        };
147        // The span of the annotation itself, so diagnostics can point
148        // at the specific attribute rather than the whole item.
149        let attr_span = attr.path().span();
150        prusti_attributes.push((attr_kind, attr_span, unwrap_argument(&attr)?));
151    }
152    *attrs = regular_attributes;
153    Ok(prusti_attributes)
154}

156/// Rejects the empty argument of a Prusti attribute that needs one, which
157/// would otherwise be reported as an unexpected end of input at the
158/// outermost attribute.
159fn require_argument(span: Span, argument: &TokenStream) -> syn::Result<()> {
160    if argument.is_empty() {
161        Err(syn::Error::new(span, "this attribute expects an argument"))
162    } else {
163        Ok(())
164    }

167/// Rewrite an item as required by *all* its specification attributes.
168///
169/// The first attribute (the outer one) needs to be passed via `attr_kind` and `attr` because
170/// the compiler executes it as as a procedural macro attribute.
171pub fn rewrite_prusti_attributes(
172    outer_attr_kind: SpecAttributeKind,
173    outer_attr_tokens: TokenStream,
174    item_tokens: TokenStream,
175) -> TokenStream {
176    let mut item: untyped::AnyFnItem = match syn::parse2(item_tokens) {
177        Ok(item) => item,
178        Err(err) => return not_parsable(outer_attr_kind, err).to_compile_error(),
179    };
180
181    // Start with the outer attribute. It is the one the compiler ran as the
182    // procedural macro, so its span is the call site.
183    let mut prusti_attributes = vec![(outer_attr_kind, Span::call_site(), outer_attr_tokens)];
184
185    // Collect the remaining Prusti attributes, removing them from `item`.
186    prusti_attributes.extend(handle_result!(extract_prusti_attributes(&mut item)));
187
188    let (generated_spec_items, generated_attributes) =
189        handle_result!(generate_spec_and_assertions(prusti_attributes, &item));
190
191    quote_spanned! {item.span()=>
192        #(#generated_spec_items)*
193        #(#generated_attributes)*
194        #[prusti::specs_version = #SPECS_VERSION]
195        #item
196    }
197}

201/// Generate spec items and attributes for `item` from the Prusti attributes
202fn generate_spec_and_assertions(
203    mut prusti_attributes: Vec<(SpecAttributeKind, Span, TokenStream)>,
204    item: &untyped::AnyFnItem,
205) -> GeneratedResult {
206    let mut generated_items = vec![];
207    let mut generated_attributes = vec![];
208
209    for (attr_kind, attr_span, attr_tokens) in prusti_attributes.drain(..) {
210        // These attributes need an argument; the argument of `terminates` is
211        // optional.
212        let needs_argument = matches!(
213            attr_kind,
214            SpecAttributeKind::Requires
215                | SpecAttributeKind::Ensures
216                | SpecAttributeKind::AfterExpiry
217                | SpecAttributeKind::AssertOnExpiry
218                | SpecAttributeKind::RefineSpec
219        );
220        if needs_argument {
221            require_argument(attr_span, &attr_tokens)?;
222        }
223        let rewriting_result = match attr_kind {
224            SpecAttributeKind::Requires => generate_for_requires(attr_tokens, attr_span, item),
225            SpecAttributeKind::Ensures => generate_for_ensures(attr_tokens, attr_span, item),
226            SpecAttributeKind::AfterExpiry => {
227                generate_for_after_expiry(attr_tokens, attr_span, item)
228            }
229            SpecAttributeKind::AssertOnExpiry => {
230                generate_for_assert_on_expiry(attr_tokens, attr_span, item)
231            }
232            SpecAttributeKind::Pure => generate_marker_attribute("pure", attr_tokens, attr_span),
233            SpecAttributeKind::Verified => {
234                generate_marker_attribute("verified", attr_tokens, attr_span)
235            }
236            SpecAttributeKind::Terminates => generate_for_terminates(attr_tokens, attr_span, item),
237            SpecAttributeKind::Trusted => {
238                generate_marker_attribute("trusted", attr_tokens, attr_span)
239            }
240            SpecAttributeKind::Invariant => unreachable!(),
241            SpecAttributeKind::RefineSpec => type_cond_specs::generate(attr_tokens, item),
242            SpecAttributeKind::Model => unreachable!(),
243            SpecAttributeKind::PrintCounterexample => unreachable!(),
244        };
245        let (new_items, new_attributes) = rewriting_result?;
246        generated_items.extend(new_items);
247        generated_attributes.extend(new_attributes);
248    }
249
250    Ok((generated_items, generated_attributes))
251}

286/// Generate spec items and attributes to typecheck and later retrieve "after_expiry" annotations.
287fn generate_for_assert_on_expiry(
288    attr: TokenStream,
289    span: Span,
290    item: &untyped::AnyFnItem,
291) -> GeneratedResult {
292    let mut rewriter = rewriter::AstRewriter::new();
293    let spec_id_lhs = rewriter.generate_spec_id();
294    let spec_id_lhs_str = spec_id_lhs.to_string();
295    let spec_id_rhs = rewriter.generate_spec_id();
296    let spec_id_rhs_str = spec_id_rhs.to_string();

298        rewriter.process_assert_pledge(spec_id_lhs, spec_id_rhs, attr, item)?;
299    Ok((
300        vec![spec_item_lhs, spec_item_rhs],
301        vec![
302            parse_quote_spanned! {span=>
303                #[prusti::assert_pledge_spec_id_ref_lhs = #spec_id_lhs_str]
304            },
305            parse_quote_spanned! {span=>
306                #[prusti::assert_pledge_spec_id_ref_rhs = #spec_id_rhs_str]
307            },
308        ],
309    ))

312/// Generate spec items and attributes to typecheck and later retrieve "terminates" annotations.
313fn generate_for_terminates(
314    mut attr: TokenStream,
315    span: Span,
316    item: &untyped::AnyFnItem,
317) -> GeneratedResult {
318    if attr.is_empty() {
319        attr = quote! { Int::from(1) };
320    } else {
321        let mut attr_iter = attr.clone().into_iter();
322        let first = attr_iter.next();
323        if let Some(TokenTree::Ident(ident)) = first {
324            if attr_iter.next().is_none() && ident == "trusted" {
325                attr = quote! { prusti_terminates_trusted() }
326            }
327        }
328    }
329
330    generate_spec_item("terminates_spec_id_ref", span, |rewriter, spec_id| {
331        rewriter.process_assertion(rewriter::SpecItemType::Termination, spec_id, attr, item)
332    })
333}

335/// Generates the spec item built by `process`, which typechecks a
336/// specification, and the attribute `#[prusti::<ref_attr> = "<spec id>"]`,
337/// with which the spec collection of `prusti-interface` later finds it.
338fn generate_spec_item(
339    ref_attr: &str,
340    span: Span,
341    process: impl FnOnce(&mut AstRewriter, SpecificationId) -> syn::Result<syn::Item>,
342) -> GeneratedResult {
343    let mut rewriter = rewriter::AstRewriter::new();
344    let spec_id = rewriter.generate_spec_id();
345    let spec_id_str = spec_id.to_string();
346    let spec_item = process(&mut rewriter, spec_id)?;
347    let ref_attr = syn::Ident::new(ref_attr, span);
348    Ok((
349        vec![spec_item],
350        vec![parse_quote_spanned! {span=>
351            #[prusti::#ref_attr = #spec_id_str]
352        }],
353    ))
354}

426/// Generates the spec function of a type specification, built by `spec_fn`
427/// from its name and spec id, in an inherent impl of the type `item`. The

472/// Generates the TokenStream encoding an expression using prusti syntax
473/// Used for body invariants, assertions, and assumptions
474fn generate_expression_closure(
475    fun: &dyn Fn(&mut AstRewriter, SpecificationId, TokenStream) -> syn::Result<TokenStream>,
476    tokens: TokenStream,
477) -> TokenStream {
478    let mut rewriter = rewriter::AstRewriter::new();
479    let spec_id = rewriter.generate_spec_id();
480    let closure = handle_result!(fun(&mut rewriter, spec_id, tokens));
481    let callsite_span = Span::call_site();
482    quote_spanned! {callsite_span=>
483        #[allow(unused_must_use, unused_variables, unused_braces, unused_parens)]
484        #[prusti::specs_version = #SPECS_VERSION]
485        if false {
486            #closure
487        }
488    }
489}

491/// Rejects `result` as a binding anywhere in a closure parameter pattern: it
492/// would collide with the postcondition closure's `result` parameter.
493fn check_no_result_binding(pat: &syn::Pat) -> syn::Result<()> {
494    struct FindResult(Option<Span>);
495    impl<'ast> syn::visit::Visit<'ast> for FindResult {
496        fn visit_pat_ident(&mut self, pat: &'ast syn::PatIdent) {
497            if self.0.is_none() && pat.ident == "result" {
498                self.0 = Some(pat.ident.span());
499            }
500            syn::visit::visit_pat_ident(self, pat);
501        }
502    }
503    let mut visitor = FindResult(None);
504    syn::visit::Visit::visit_pat(&mut visitor, pat);
505    match visitor.0 {
506        Some(span) => Err(syn::Error::new(
507            span,
508            "closure parameter bindings may not be named `result`",
509        )),
510        None => Ok(()),
511    }
512}

514/// The value of a closure parameter, rebuilt from its pattern's bindings for
515/// the `closure_spec_*` marker tuples. The value sits in dead code and only
516/// pins the spec closures' parameter types, but it must typecheck: a pattern
517/// without `..` and without `ref` bindings binds everything it matches, so
518/// its value can be rebuilt structurally (moving the bindings, which the
519/// diverging `return` permits).
520///
521/// `_` binds nothing, and nothing can mention it either, but the stand-in
522/// value must still typecheck at its position. In a structural position
523/// (a tuple slot, behind `&`, or the parameter itself) the expected type is
524/// whatever we build, so `()` serves. In a `pinned` position (the field of
525/// a struct the pattern names, or an array's element type, unified with the
526/// sibling elements) no stand-in can be right: the field's type is fixed
527/// by a definition this macro cannot see, so `()` mismatches a concrete
528/// field, while an inference variable (`None.unwrap()`) is unresolvable
529/// when `_` is all that constrains a generic one. Such a `_` is therefore
530/// rejected; binding the value instead (`_x`) is exact in both cases.
531fn reconstruct_arg(pat: &syn::Pat, pinned: bool) -> syn::Result<TokenStream> {
532    let unsupported = |span, what: &str| {
533        syn::Error::new(
534            span,
535            format!("the parameters of a closure with specifications may not contain {what}"),
536        )
537    };
538    match pat {
539        syn::Pat::Ident(pat_ident) => {
540            if let Some(by_ref) = &pat_ident.by_ref {
541                return Err(unsupported(by_ref.span(), "`ref` bindings"));
542            }
543            // An `x @ subpattern` binding names the whole value, whatever
544            // the subpattern is.
545            let ident = &pat_ident.ident;
546            Ok(quote_spanned! {ident.span()=> #ident })
547        }
548        syn::Pat::Wild(wild) if pinned => Err(unsupported(
549            wild.span(),
550            "`_` inside a struct or array pattern (bind the value instead, e.g. `_x`)",
551        )),
552        syn::Pat::Wild(wild) => Ok(quote_spanned! {wild.span()=> () }),
553        // A tuple (or reference) type is structural: if the position of the
554        // whole is pinned, so is each slot, and vice versa.
555        syn::Pat::Tuple(tuple) => {
556            let elems = tuple
557                .elems
558                .iter()
559                .map(|elem| reconstruct_arg(elem, pinned))
560                .collect::<syn::Result<Vec<_>>>()?;
561            Ok(quote_spanned! {tuple.span()=> (#(#elems,)*) })
562        }
563        syn::Pat::Reference(reference) => {
564            let mutability = &reference.mutability;
565            let value = reconstruct_arg(&reference.pat, pinned)?;
566            Ok(quote_spanned! {reference.span()=> &#mutability #value })
567        }
568        // A constructor is a nominal signature: its fields' types are fixed
569        // by the struct definition, whatever the position of the whole.
570        syn::Pat::TupleStruct(tuple_struct) => {
571            if tuple_struct.qself.is_some() {
572                return Err(unsupported(tuple_struct.span(), "qualified paths"));
573            }
574            let path = &tuple_struct.path;
575            let elems = tuple_struct
576                .elems
577                .iter()
578                .map(|elem| reconstruct_arg(elem, true))
579                .collect::<syn::Result<Vec<_>>>()?;
580            Ok(quote_spanned! {tuple_struct.span()=> #path(#(#elems),*) })
581        }
582        syn::Pat::Struct(pat_struct) => {
583            if let Some(rest) = &pat_struct.rest {
584                return Err(unsupported(rest.span(), "`..`"));
585            }
586            if pat_struct.qself.is_some() {
587                return Err(unsupported(pat_struct.span(), "qualified paths"));
588            }
589            let path = &pat_struct.path;
590            let fields = pat_struct
591                .fields
592                .iter()
593                .map(|field| {
594                    let member = &field.member;
595                    let value = reconstruct_arg(&field.pat, true)?;
596                    Ok(quote_spanned! {field.span()=> #member: #value })
597                })
598                .collect::<syn::Result<Vec<_>>>()?;
599            Ok(quote_spanned! {pat_struct.span()=> #path { #(#fields),* } })
600        }
601        // A slice pattern in parameter position matches an array (on a slice
602        // it would be refutable, which rustc rejects on its own), so the
603        // value is an array literal. The elements share one type, so each
604        // slot is pinned by its siblings.
605        syn::Pat::Slice(slice) => {
606            let elems = slice
607                .elems
608                .iter()
609                .map(|elem| reconstruct_arg(elem, true))
610                .collect::<syn::Result<Vec<_>>>()?;
611            Ok(quote_spanned! {slice.span()=> [#(#elems),*] })
612        }
613        // A unit struct: the path is the value.
614        syn::Pat::Path(path) => Ok(path.to_token_stream()),
615        syn::Pat::Type(pat_type) => reconstruct_arg(&pat_type.pat, pinned),
616        syn::Pat::Paren(paren) => reconstruct_arg(&paren.pat, pinned),
617        syn::Pat::Rest(rest) => Err(unsupported(rest.span(), "`..`")),
618        other => Err(unsupported(other.span(), "this kind of pattern")),
619    }
620}

622/// Expands the `closure!` macro: returns the closure itself, with its specs
623/// embedded as `closure_spec_*` marker calls in dead `if false { return .. }`
624/// blocks inside the body (the `return` makes the argument moves diverge, so
625/// they cannot affect the body), one block per specification. For
626///
627/// ```ignore
628/// closure!(#[requires(P1)] #[requires(P2)] #[ensures(Q)] |a: A, b| body)
629/// ```
630///
631/// the macro expands to
632///
633/// ```ignore
634/// {
635///     #[prusti::closure]
636///     #[prusti::specs_version = "..."]
637///     let _prusti_closure = |a: A, b| {
638///         if false {
639///             return closure_spec_pre((a, b), #[prusti::spec_only] |a: _, b: _| -> bool { P1 });
640///         }
641///         if false {
642///             return closure_spec_pre((a, b), #[prusti::spec_only] |a: _, b: _| -> bool { P2 });
643///         }
644///         let prusti_closure_args = ::core::marker::PhantomData;    // mixed-site hygiene
645///         if false {
646///             return closure_spec_args((a, b, None.unwrap()), prusti_closure_args);
647///         }
648///         let result = body;
649///         if false {
650///             return closure_spec_post((None.unwrap(), None.unwrap(), result), prusti_closure_args, #[prusti::spec_only] |a: _, b: _, result| -> bool { Q });
651///         }
652///         result
653///     };
654///     _prusti_closure
655/// }
656/// ```
657///
658/// Each spec closure carries the span of its specification, so that
659/// diagnostics point at the violated clause.
660///
661/// The spec closures take the closure's parameters (plus `result` for
662/// postconditions) as a flat list of the parameters' own patterns; their
663/// types are deduced from the markers' `Fn<Args>` bound against the actual
664/// argument tuple, never from annotations (see [reconstruct_arg]: a `_`
665/// slot's stand-in value deliberately fakes its type, so a user annotation
666/// must not be forwarded). The argument tuple holds each parameter's value,
667/// rebuilt from the pattern's bindings for parameters that destructure.
668/// `closure_spec_args` unifies the `PhantomData` binding
669/// carrying the argument (and result) types from the closure entry to the
670/// postcondition; tuple slots whose values are unavailable at the
671/// respective marker are filled with `None.unwrap()`, which mints a fresh
672/// inference variable (a `!`-typed stand-in such as `unreachable!()` wouldn't
673/// work as it doesn't coerce at that point and would pin the slot to `!`).
674pub fn closure(tokens: TokenStream) -> TokenStream {
675    let cl_spec: ClosureWithSpec = handle_result!(syn::parse(tokens.into()));
676    let callsite_span = Span::call_site();
677
678    let syn::ExprClosure {
679        attrs,
680        lifetimes,
681        modifiers,
682        constness,
683        asyncness,
684        capture,
685        inputs_begin,
686        inputs,
687        inputs_end,
688        output,
689        body,
690    } = cl_spec.cl;
691    // The closure is rebuilt from its parts below. `modifiers` holds closure
692    // modifiers that syn does not parse yet; syn cannot print them on their
693    // own, so a closure with any is rejected rather than rebuilt without them.
694    handle_result!(modifiers.require_empty());
695
696    let mut attrs_ts = TokenStream::new();
697    for a in attrs {
698        let name = match prusti_attribute_kind(&a) {
699            Some(SpecAttributeKind::Pure) => "pure",
700            Some(SpecAttributeKind::Trusted) => "trusted",
701            // Reject other specification attributes rather than forwarding
702            // them (they would be silently ignored on the `let` binding).
703            Some(_) => {
704                let ident = &a.path().segments.last().unwrap().ident;
705                return syn::Error::new(
706                    a.span(),
707                    format!("`{ident}` is not supported on `closure!`"),
708                )
709                .to_compile_error();
710            }
711            None => {
712                attrs_ts.extend(a.into_token_stream());
713                continue;
714            }
715        };
716        let argument = handle_result!(unwrap_argument(&a));
717        let (_, marker) = handle_result!(generate_marker_attribute(name, argument, a.span()));
718        attrs_ts.extend(quote! { #(#marker)* });
719    }
720
721    // Each specification with its span, so that the spec closure carries the
722    // span of its expression and diagnostics point at that clause.
723    let parse_specs = |specs: Vec<TokenStream>| -> syn::Result<Vec<(TokenStream, Span)>> {
724        specs
725            .into_iter()
726            .map(|spec| {
727                let span = spec.span();
728                Ok((parse_prusti(spec)?, span))
729            })
730            .collect()
731    };
732    let pres = handle_result!(parse_specs(cl_spec.pres));
733    let posts = handle_result!(parse_specs(cl_spec.posts));
734
735    // The spec closures' parameters (the closure's own patterns) and the
736    // parameters' values for the marker tuples (rebuilt from the patterns'
737    // bindings). The parameter types are deliberately left to inference:
738    // each is pinned by its value, and for a pattern with `_` slots the
739    // value's type differs from the closure's (the slots hold `()`), so a
740    // user annotation must not be forwarded.
741    let mut spec_params: Vec<TokenStream> = vec![];
742    let mut arg_values: Vec<TokenStream> = vec![];
743    if !pres.is_empty() || !posts.is_empty() {
744        for input in &inputs {
745            let pat = match input {
746                syn::Pat::Type(pat_type) => &*pat_type.pat,
747                other => other,
748            };
749            handle_result!(check_no_result_binding(pat));
750            arg_values.push(handle_result!(reconstruct_arg(pat, false)));
751            spec_params.push(quote_spanned! {callsite_span=> #pat: _ });
752        }
753    }
754
755    let anys = arg_values
756        .iter()
757        .map(|_| quote_spanned! {callsite_span=> ::core::option::Option::None.unwrap() })
758        .collect::<Vec<_>>();
759    let result_param = match &output {
760        syn::ReturnType::Type(_, ty) => quote_spanned! {callsite_span=> result: #ty },
761        syn::ReturnType::Default => quote_spanned! {callsite_span=> result },
762    };
763
764    // Hygienic: user code inside the closure body cannot refer to this binding.
765    let args_ident = syn::Ident::new("prusti_closure_args", Span::mixed_site());
766
767    let pre_stmts = pres.iter().map(|(pre, span)| {
768        let spec_closure = quote_spanned! {*span=>
769            #[prusti::spec_only]
770            |#(#spec_params),*| -> bool { #pre }
771        };
772        quote_spanned! {callsite_span=>
773            #[allow(unused_must_use, unused_variables, unused_braces, unused_parens)]
774            if false {
775                return ::prusti_contracts::closure_spec_pre((#(#arg_values,)*), #spec_closure);
776            }
777        }
778    });
779
780    // The `PhantomData` tie is only needed (and only typechecks, its result
781    // slot being pinned by the post markers alone) when there are
782    // postconditions.
783    let body = if posts.is_empty() {
784        body.to_token_stream()
785    } else {
786        let post_stmts = posts.iter().map(|(post, span)| {
787            let spec_closure = quote_spanned! {*span=>
788                #[prusti::spec_only]
789                |#(#spec_params,)* #result_param| -> bool { #post }
790            };
791            quote_spanned! {callsite_span=>
792                #[allow(unused_must_use, unused_variables, unused_braces, unused_parens)]
793                if false {
794                    return ::prusti_contracts::closure_spec_post(
795                        (#(#anys,)* result,),
796                        #args_ident,
797                        #spec_closure,
798                    );
799                }
800            }
801        });
802        quote_spanned! {callsite_span=>
803            let #args_ident = ::core::marker::PhantomData;
804            #[allow(unused_must_use, unused_braces, unused_parens)]
805            if false {
806                return ::prusti_contracts::closure_spec_args(
807                    (#(#arg_values,)* ::core::option::Option::None.unwrap(),),
808                    #args_ident,
809                );
810            }
811            let result = #body;
812            #(#post_stmts)*
813            result
814        }
815    };
816    let new_body = quote_spanned! {callsite_span=>
817        {
818            #(#pre_stmts)*
819            #body
820        }
821    };
822
823    quote_spanned! {callsite_span=>
824        {
825            #[allow(unused_variables, unused_braces, unused_parens)]
826            #[prusti::closure]
827            #[prusti::specs_version = #SPECS_VERSION]
828            #attrs_ts
829            let _prusti_closure =
830                #lifetimes #constness #asyncness #capture
831                #inputs_begin #inputs #inputs_end #output
832                #new_body;
833            _prusti_closure
834        }
835    }
836}

838pub fn refine_trait_spec(_attr: TokenStream, tokens: TokenStream) -> TokenStream {
839    let mut impl_block: syn::ItemImpl = handle_result!(syn::parse2(tokens));
840    let impl_generics = &impl_block.generics;
841
842    let trait_path: syn::TypePath = match &impl_block.trait_ {
843        Some((trait_path, _)) => parse_quote_spanned!(trait_path.span()=>#trait_path),
844        None => handle_result!(Err(syn::Error::new(
845            impl_block.span(),
846            "Can refine trait specifications only on trait implementation blocks"
847        ))),
848    };
849
850    let self_type: &syn::Type = &impl_block.self_ty;
851
852    let mut new_items = Vec::new();
853    let mut generated_spec_items = Vec::new();
854    for item in impl_block.items {
855        match item {
856            syn::ImplItem::Fn(method) => {
857                let mut method_item = untyped::AnyFnItem::ImplMethod(method);
858                let prusti_attributes: Vec<_> =
859                    handle_result!(extract_prusti_attributes(&mut method_item));
860
861                let illegal_attribute_span = prusti_attributes
862                    .iter()
863                    .filter(|(kind, _, _)| kind == &SpecAttributeKind::RefineSpec)
864                    .map(|(_, span, _)| *span)
865                    .next();
866                if let Some(span) = illegal_attribute_span {
867                    let err = Err(syn::Error::new(
868                        span,
869                        "Type-conditional spec refinements in trait spec refinements not supported",
870                    ));
871                    handle_result!(err);
872                }
873
874                let (spec_items, generated_attributes) = handle_result!(
875                    generate_spec_and_assertions(prusti_attributes, &method_item)
876                );
877
878                spec_items
879                    .into_iter()
880                    .map(|spec_item| match spec_item {
881                        syn::Item::Fn(spec_item_fn) => spec_item_fn,
882                        x => unimplemented!("Unexpected variant: {:?}", x),
883                    })
884                    .for_each(|spec_item_fn| generated_spec_items.push(spec_item_fn));
885
886                let new_item = parse_quote_spanned! {method_item.span()=>
887                    #(#generated_attributes)*
888                    #method_item
889                };
890                new_items.push(new_item);
891            }
892            syn::ImplItem::Macro(makro) if is_predicate_macro(&makro) => {
893                let parsed_predicate =
894                    handle_result!(predicate::parse_predicate_in_impl(makro.mac.tokens.clone()));
895
896                let ParsedPredicate::Impl(predicate) = parsed_predicate else {
897                    unreachable!()
898                };
899
900                // Patch spec function: Rewrite self with _self: <SpecStruct>
901                let syn::Item::Fn(spec_function) = predicate.spec_function else {
902                    unreachable!()
903                };
904                generated_spec_items.push(spec_function);
905
906                // Add patched predicate function to new items
907                new_items.push(syn::ImplItem::Fn(predicate.patched_function));
908            }
909            _ => new_items.push(item),
910        }
911    }
912
913    // Patch the spec items (merge generics, handle associated types, rewrite receiver)
914    for generated_spec_item in generated_spec_items.iter_mut() {
915        merge_generics(&mut generated_spec_item.sig.generics, impl_generics);
916        generated_spec_item.rewrite_self_type(self_type, Some(&trait_path));
917        handle_result!(generated_spec_item.rewrite_receiver(self_type));
918    }
919
920    impl_block.items = new_items;
921    quote_spanned! {impl_block.span()=>
922        #(#generated_spec_items)*
923        #[prusti::refine_trait_spec]
924        #[prusti::specs_version = #SPECS_VERSION]
925        #impl_block
926    }
927}

929pub fn trusted(attr: TokenStream, tokens: TokenStream) -> TokenStream {
930    // `#[trusted]` can be applied to both types and to methods, figure out
931    // which one by trying to parse a `DeriveInput`.
932    if syn::parse2::<syn::DeriveInput>(tokens.clone()).is_ok() {
933        rewrite_prusti_attributes_for_types(SpecAttributeKind::Trusted, attr, tokens)
934    } else {
935        rewrite_prusti_attributes(SpecAttributeKind::Trusted, attr, tokens)
936    }
937}

956/// The error for the Prusti attribute of kind `kind`, written at `span`, on
957/// an item it does not apply to.
958fn not_applicable(kind: SpecAttributeKind, span: Span) -> syn::Error {
959    syn::Error::new(span, applicable_items(kind))
960}
961
962/// The error for the Prusti attribute of kind `kind` on an item that syn,
963/// whose error is `err`, cannot parse as one it applies to. Either the item
964/// is of another kind or it uses syntax that syn does not support, so the
965/// error says both and keeps syn's own error and its position.
966fn not_parsable(kind: SpecAttributeKind, err: syn::Error) -> syn::Error {
967    syn::Error::new(
968        err.span(),
969        format!(
970            "{}, and this item could not be parsed as one: {err}",
971            applicable_items(kind)
972        ),
973    )
974}

976/// The sentence saying which items the Prusti attribute of kind `kind`
977/// applies to.
978fn applicable_items(kind: SpecAttributeKind) -> String {
979    let (name, items) = match kind {
980        SpecAttributeKind::Requires => ("requires", "functions"),
981        SpecAttributeKind::Ensures => ("ensures", "functions"),
982        SpecAttributeKind::AfterExpiry => ("after_expiry", "functions"),
983        SpecAttributeKind::AssertOnExpiry => ("assert_on_expiry", "functions"),
984        SpecAttributeKind::RefineSpec => ("refine_spec", "functions"),
985        SpecAttributeKind::Pure => ("pure", "functions"),
986        SpecAttributeKind::Verified => ("verified", "functions"),
987        SpecAttributeKind::Terminates => ("terminates", "functions"),
988        SpecAttributeKind::Trusted => ("trusted", "functions, structs, enums and unions"),
989        SpecAttributeKind::Invariant => ("invariant", "structs, enums and unions"),
990        SpecAttributeKind::Model => ("model", "structs"),
991        SpecAttributeKind::PrintCounterexample => ("print_counterexample", "structs and enums"),
992    };
993    format!("the `#[{name}]` attribute can only be applied to {items}")
994}

996pub fn extern_spec(attr: TokenStream, tokens: TokenStream) -> TokenStream {
997    result_to_tokens!({
998        let item: syn::Item = syn::parse2(tokens)?;
999        let mod_path: syn::Path = Some(attr)
1000            .filter(|attr| !attr.is_empty())
1001            .map(syn::parse2)
1002            .transpose()?
1003            .unwrap_or_else(|| syn::Path {
1004                leading_colon: None,
1005                segments: syn::punctuated::Punctuated::new(),
1006            });
1007        match item {
1008            syn::Item::Impl(item_impl) => {
1009                if !mod_path.segments.is_empty() {
1010                    return Err(syn::Error::new(
1011                        mod_path.span(),
1012                        "extern_spec does not take a path argument for impls--you can qualify the involved types directly",
1013                    ));
1014                }
1015                extern_spec_rewriter::impls::rewrite_extern_spec(&item_impl)
1016            }
1017            syn::Item::Trait(item_trait) => {
1018                extern_spec_rewriter::traits::rewrite_extern_spec(&item_trait, mod_path)
1019            }
1020            syn::Item::Mod(item_mod) => {
1021                extern_spec_rewriter::mods::rewrite_mod(&item_mod, mod_path)
1022            }
1023            syn::Item::ForeignMod(item_foreign_mod) => {
1024                extern_spec_rewriter::foreign_mods::rewrite_extern_spec(
1025                    &item_foreign_mod,
1026                    &mod_path,
1027                )
1028            }
1029            // we're expecting function stubs, so they aren't represented as Item::Fn
1030            syn::Item::Verbatim(stub_tokens) => {
1031                extern_spec_rewriter::functions::rewrite_stub(&stub_tokens, &mod_path)
1032            }
1033            _ => Err(syn::Error::new(
1034                Span::call_site(), // this covers the entire macro invocation, unlike attr.span() which changes to only cover arguments if possible
1035                "Extern specs cannot be attached to this item",
1036            )),
1037        }
1038    })
1039}

1046pub fn rewrite_prusti_attributes_for_types(
1047    outer_attr_kind: SpecAttributeKind,
1048    outer_attr_tokens: TokenStream,
1049    item_tokens: TokenStream,
1050) -> TokenStream {
1051    let mut item: syn::DeriveInput = handle_result!(syn::parse2(item_tokens));
1052
1053    // Start with the outer attribute. It is the one the compiler ran as the
1054    // procedural macro, so its span is the call site.
1055    let mut prusti_attributes = vec![(outer_attr_kind, Span::call_site(), outer_attr_tokens)];
1056
1057    // Collect the remaining Prusti attributes, removing them from `item`.
1058    prusti_attributes.extend(handle_result!(extract_prusti_attributes_for_types(
1059        &mut item
1060    )));
1061
1062    // Each attribute's own kind of item is checked before any conflicts
1063    // between attributes.
1064    for (kind, span, _) in &prusti_attributes {
1065        let applies = match kind {
1066            SpecAttributeKind::Trusted | SpecAttributeKind::Invariant => true,
1067            SpecAttributeKind::Model => matches!(item.data, syn::Data::Struct(_)),
1068            SpecAttributeKind::PrintCounterexample => !matches!(item.data, syn::Data::Union(_)),
1069            SpecAttributeKind::Requires
1070            | SpecAttributeKind::Ensures
1071            | SpecAttributeKind::AfterExpiry
1072            | SpecAttributeKind::AssertOnExpiry
1073            | SpecAttributeKind::RefineSpec
1074            | SpecAttributeKind::Pure
1075            | SpecAttributeKind::Verified
1076            | SpecAttributeKind::Terminates => unreachable!("not a type attribute: {kind:?}"),
1077        };
1078        if !applies {
1079            return not_applicable(*kind, *span).to_compile_error();
1080        }
1081    }
1082
1083    // A model declaration describes the model of another type and is replaced
1084    // by a generated struct; attributes left on the declaration are lost. Of
1085    // the other type attributes, only `#[print_counterexample]` applies to it.
1086    // This is checked before the `#[trusted]` rule, the declaration itself
1087    // before its other attributes: on a model declaration, `#[model]` is the
1088    // attribute that the others conflict with.
1089    if let Some(model) = prusti_attributes
1090        .iter()
1091        .position(|(ak, _, _)| ak == &SpecAttributeKind::Model)
1092    {
1093        let (_, _, model_tokens) = &prusti_attributes[model];
1094        handle_result!(reject_parameters("model", model_tokens));
1095        if let Some((_, (kind, span, _))) =
1096            prusti_attributes
1097                .iter()
1098                .enumerate()
1099                .find(|(index, (ak, _, _))| {
1100                    *index != model && ak != &SpecAttributeKind::PrintCounterexample
1101                })
1102        {
1103            let message = if kind == &SpecAttributeKind::Model {
1104                "duplicate `#[model]` attribute"
1105            } else {
1106                "this attribute cannot be combined with `#[model]`"
1107            };
1108            return syn::Error::new(*span, message).to_compile_error();
1109        }
1110        // The attributes left on the declaration apply to functions only.
1111        if let Some((kind, span)) = item
1112            .attrs
1113            .iter()
1114            .find_map(|attr| Some((prusti_attribute_kind(attr)?, attr.span())))
1115        {
1116            return not_applicable(kind, span).to_compile_error();
1117        }
1118    }
1119
1120    // A trusted type is opaque to the encoder, so no other type attribute
1121    // has an effect on it. As for `#[model]`, the attribute's own problems
1122    // are reported first.
1123    if let Some(trusted) = prusti_attributes
1124        .iter()
1125        .position(|(ak, _, _)| ak == &SpecAttributeKind::Trusted)
1126    {
1127        let (_, _, trusted_tokens) = &prusti_attributes[trusted];
1128        handle_result!(reject_parameters("trusted", trusted_tokens));
1129        if let Some((_, (kind, span, _))) = prusti_attributes
1130            .iter()
1131            .enumerate()
1132            .find(|(index, _)| *index != trusted)
1133        {
1134            let message = if kind == &SpecAttributeKind::Trusted {
1135                "duplicate `#[trusted]` attribute"
1136            } else {
1137                "this attribute has no effect on a `#[trusted]` type"
1138            };
1139            return syn::Error::new(*span, message).to_compile_error();
1140        }
1141    }
1142
1143    // we order the attributes to ensure a model attribute is processed first
1144    prusti_attributes.sort_by_key(|(ak1, _, _)| *ak1);
1145
1146    let (generated_spec_items, generated_attributes) = handle_result!(
1147        generate_spec_and_assertions_for_types(prusti_attributes, &mut item)
1148    );
1149
1150    quote_spanned! {item.span()=>
1151        #(#generated_attributes)*
1152        #[prusti::specs_version = #SPECS_VERSION]
1153        #item
1154        #(#generated_spec_items)*
1155    }
1156}

1158/// Generate spec items and attributes for `item` from the Prusti attributes
1159fn generate_spec_and_assertions_for_types(
1160    mut prusti_attributes: Vec<(SpecAttributeKind, Span, TokenStream)>,
1161    item: &mut syn::DeriveInput,
1162) -> GeneratedResult {
1163    let mut generated_items = vec![];
1164    let mut generated_attributes = vec![];
1165
1166    for (attr_kind, attr_span, attr_tokens) in prusti_attributes.drain(..) {
1167        let rewriting_result = match attr_kind {
1168            SpecAttributeKind::Requires => unreachable!(),
1169            SpecAttributeKind::Ensures => unreachable!(),
1170            SpecAttributeKind::AfterExpiry => unreachable!(),
1171            SpecAttributeKind::AssertOnExpiry => unreachable!(),
1172            SpecAttributeKind::Pure => unreachable!(),
1173            SpecAttributeKind::Verified => unreachable!(),
1174            SpecAttributeKind::Invariant => generate_for_invariant(attr_tokens, attr_span, item),
1175            SpecAttributeKind::RefineSpec => unreachable!(),
1176            SpecAttributeKind::Terminates => unreachable!(),
1177            SpecAttributeKind::Trusted => generate_for_trusted_for_types(item),
1178            SpecAttributeKind::Model => generate_for_model(item),
1179            SpecAttributeKind::PrintCounterexample => {
1180                generate_for_print_counterexample(attr_tokens, item)
1181            }
1182        };
1183        let (new_items, new_attributes) = rewriting_result?;
1184        generated_items.extend(new_items);
1185        generated_attributes.extend(new_attributes);
1186    }
1187
1188    Ok((generated_items, generated_attributes))
1189}

1191/// Generate spec items and attributes to typecheck and later retrieve "model" annotations.
1192fn generate_for_model(item: &mut syn::DeriveInput) -> GeneratedResult {
1193    match syn::Item::from(item.clone()) {
1194        syn::Item::Struct(item_struct) => {
1195            match type_model::rewrite(item_struct) {
1196                Ok(result) => {
1197                    match result.first() {
1198                        Some(syn::Item::Struct(new_item)) => {
1199                            *item = syn::DeriveInput::from(new_item.clone()); //the internal model replaces the original struct
1200                            Ok((vec![result[1].clone(), result[2].clone()], vec![]))
1201                        }
1202                        _ => unreachable!(),
1203                    }
1204                }
1205                Err(err) => Err(err),
1206            }
1207        }
1208        _ => unreachable!("`rewrite_prusti_attributes_for_types` rejects models of other items"),
1209    }
1210}

1251pub fn ghost(tokens: TokenStream) -> TokenStream {
1252    let mut rewriter = rewriter::AstRewriter::new();
1253    let callsite_span = Span::call_site();
1254
1255    let spec_id = rewriter.generate_spec_id();
1256    let spec_id_str = spec_id.to_string();
1257
1258    /// Finds the expressions that leave the ghost block.
1259    struct Visitor {
1260        /// The loops and blocks around the visited expression, inside the
1261        /// ghost block: their labels, and whether each is a loop.
1262        targets: Vec<(Option<syn::Ident>, bool)>,
1263        exits: Vec<Span>,
1264    }
1265
1266    impl Visitor {
1267        fn visit_target(
1268            &mut self,
1269            label: &Option<syn::Label>,
1270            is_loop: bool,
1271            visit: impl FnOnce(&mut Self),
1272        ) {
1273            let label = label.as_ref().map(|label| label.name.ident.clone());
1274            self.targets.push((label, is_loop));
1275            visit(self);
1276            self.targets.pop();
1277        }
1278
1279        /// Records a `break` or `continue` that leaves the ghost block: one
1280        /// whose label, or for an unlabeled one any loop, is not around it.
1281        fn visit_jump(&mut self, label: &Option<syn::Lifetime>, span: Span) {
1282            let stays = self.targets.iter().any(|(target, is_loop)| match label {
1283                Some(label) => target.as_ref() == Some(&label.ident),
1284                None => *is_loop,
1285            });
1286            if !stays {
1287                self.exits.push(span);
1288            }
1289        }
1290    }
1291
1292    impl<'ast> Visit<'ast> for Visitor {
1293        fn visit_expr_for_loop(&mut self, ex: &'ast syn::ExprForLoop) {
1294            // The iterator is evaluated outside the loop.
1295            self.visit_pat(&ex.pat);
1296            self.visit_expr(&ex.expr);
1297            self.visit_target(&ex.label, true, |v| v.visit_block(&ex.body));
1298        }
1299        fn visit_expr_while(&mut self, ex: &'ast syn::ExprWhile) {
1300            self.visit_target(&ex.label, true, |v| syn::visit::visit_expr_while(v, ex));
1301        }
1302        fn visit_expr_loop(&mut self, ex: &'ast syn::ExprLoop) {
1303            self.visit_target(&ex.label, true, |v| syn::visit::visit_expr_loop(v, ex));
1304        }
1305        fn visit_expr_block(&mut self, ex: &'ast syn::ExprBlock) {
1306            self.visit_target(&ex.label, false, |v| syn::visit::visit_expr_block(v, ex));
1307        }
1308        fn visit_expr_continue(&mut self, ex: &'ast syn::ExprContinue) {
1309            self.visit_jump(&ex.label, ex.span());
1310            syn::visit::visit_expr_continue(self, ex);
1311        }
1312        fn visit_expr_break(&mut self, ex: &'ast syn::ExprBreak) {
1313            self.visit_jump(&ex.label, ex.span());
1314            syn::visit::visit_expr_break(self, ex);
1315        }
1316        fn visit_expr_return(&mut self, e: &'ast syn::ExprReturn) {
1317            self.exits.push(e.span());
1318        }
1319        // `?` exits the enclosing function like `return` does.
1320        fn visit_expr_try(&mut self, ex: &'ast syn::ExprTry) {
1321            self.exits.push(ex.span());
1322            syn::visit::visit_expr_try(self, ex);
1323        }
1324        // Do not descend into nested closures, async blocks or items:
1325        // `return`/`?` (and loops) inside them are local to them and do not
1326        // leave the ghost block.
1327        fn visit_expr_closure(&mut self, _: &'ast syn::ExprClosure) {}
1328        fn visit_expr_async(&mut self, _: &'ast syn::ExprAsync) {}
1329        fn visit_item(&mut self, _: &'ast syn::Item) {}
1330    }
1331
1332    let mut visitor = Visitor {
1333        targets: vec![],
1334        exits: vec![],
1335    };
1336
1337    let tokens = quote! {
1338        {#tokens}
1339    };
1340
1341    let input: syn::Block = handle_result!(syn::parse2(tokens.clone()));
1342
1343    visitor.visit_block(&input);
1344
1345    let exit_errors = visitor.exits;
1346
1347    // The ghost body lives in dead code, so it is pruned before codegen; the
1348    // live path merely constructs the `Ghost` ZST. The body appears twice:
1349    // inline (encoded for verification) and duplicated into a never-called
1350    // closure so that the compiler checks it complies with `Fn` capture rules
1351    // (no mutation or consumption of outer variables); `ghost_call`'s
1352    // signature unifies the types of the two copies. The `if`/`else` unifies
1353    // the types of the two arms, giving `ghost_erased()` the body's type.
1354    if exit_errors.is_empty() {
1355        quote_spanned! {callsite_span=>
1356            #[allow(unused_must_use, unused_variables, unused_braces, unused_parens)]
1357            if false {
1358                ::prusti_contracts::ghost_call(
1359                    &(
1360                        #[prusti::spec_only]
1361                        #[prusti::spec_id = #spec_id_str]
1362                        #[prusti::specs_version = #SPECS_VERSION]
1363                        || {#tokens}
1364                    ),
1365                    {#tokens},
1366                )
1367            } else {
1368                ::prusti_contracts::ghost_erased()
1369            }
1370        }
1371    } else {
1372        let mut syn_errors = quote! {};
1373        for error in exit_errors {
1374            let error =
1375                syn::Error::new(error, "Can't leave the ghost block early").to_compile_error();
1376            syn_errors = quote! {
1377                #syn_errors
1378                #error
1379            }
1380        }
1381        syn_errors
1382    }
1383}
