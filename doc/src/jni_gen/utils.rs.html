20pub fn generate_jni_type(signature: &str) -> String {
21    match signature.chars().next().unwrap() {
22        '[' | 'L' => "JObject".to_string(),
23        'B' => "jbyte".to_string(),
24        'C' => "jchar".to_string(),
25        'S' => "jshort".to_string(),
26        'I' => "jint".to_string(),
27        'J' => "jlong".to_string(),
28        'Z' => "bool".to_string(),
29        'F' => "jfloat".to_string(),
30        'D' => "jdouble".to_string(),
31        'V' => "()".to_string(),
32        x => unreachable!("{}", x),
33    }
34}

60    match signature.chars().next().unwrap() {

62        'B' => format!("JValue::Byte({par})"),
63        'C' => format!("JValue::Char({par})"),
64        'S' => format!("JValue::Short({par})"),
65        'I' => format!("JValue::Int({par})"),
66        'J' => format!("JValue::Long({par})"),
67        'Z' => format!("JValue::Bool({par} as jboolean)"),
68        'F' => format!("JValue::Float({par})"),
69        'D' => format!("JValue::Double({par})"),
70        x => unreachable!("{}", x),
71    }

85/// Returns the JNI type descriptor of the `java.lang.Class` `class`, e.g. `I` or
86/// `Ljava/lang/String;`.
87pub fn class_descriptor<'a>(env: &JNIEnv<'a>, class: JObject<'a>) -> Result<String> {
88    let method_type = env
89        .call_static_method(
90            "java/lang/invoke/MethodType",
91            "methodType",
92            "(Ljava/lang/Class;)Ljava/lang/invoke/MethodType;",
93            &[JValue::Object(class)],
94        )?
95        .l()?;
96    // The descriptor of a method without parameters that returns `class`.
97    let method_descriptor = method_type_descriptor(env, method_type)?;
98    Ok(method_descriptor["()".len()..].to_string())
99}

101/// Returns the JNI descriptor of the `java.lang.reflect.Method` `method`.
102pub fn method_descriptor<'a>(env: &JNIEnv<'a>, method: JObject<'a>) -> Result<String> {
103    let return_type = env
104        .call_method(method, "getReturnType", "()Ljava/lang/Class;", &[])?
105        .l()?;
106    executable_descriptor(env, method, return_type)
107}

109/// Returns the JNI descriptor of the `java.lang.reflect.Constructor` `constructor`.
110pub fn constructor_descriptor<'a>(env: &JNIEnv<'a>, constructor: JObject<'a>) -> Result<String> {
111    let void = env
112        .get_static_field("java/lang/Void", "TYPE", "Ljava/lang/Class;")?
113        .l()?;
114    executable_descriptor(env, constructor, void)
115}

117/// Returns the JNI descriptor of the method or constructor `executable`, given its return type.
118fn executable_descriptor<'a>(
119    env: &JNIEnv<'a>,
120    executable: JObject<'a>,
121    return_type: JObject<'a>,
122) -> Result<String> {
123    let parameter_types = env
124        .call_method(executable, "getParameterTypes", "()[Ljava/lang/Class;", &[])?
125        .l()?;
126    let method_type = env
127        .call_static_method(
128            "java/lang/invoke/MethodType",
129            "methodType",
130            "(Ljava/lang/Class;[Ljava/lang/Class;)Ljava/lang/invoke/MethodType;",
131            &[JValue::Object(return_type), JValue::Object(parameter_types)],
132        )?
133        .l()?;
134    method_type_descriptor(env, method_type)
135}
136
137fn method_type_descriptor<'a>(env: &JNIEnv<'a>, method_type: JObject<'a>) -> Result<String> {
138    let descriptor = env
139        .call_method(
140            method_type,
141            "toMethodDescriptorString",
142            "()Ljava/lang/String;",
143            &[],
144        )?
145        .l()?;
146    java_str_to_string(&env.get_string(descriptor.into())?)
147}

149pub fn java_identifier_to_rust(name: &str) -> String {
150    // identifier can only be composed of [a-zA-Z&_] characters - https://docs.oracle.com/javase/specs/jls/se7/html/jls-3.html#jls-3.8
151    name.replace('_', "__").replace('$', "_dollar_")
152}

159/// Generates a runtime check that the object is of the expected class - applies only to object types (starting with L)
160/// If this value is also the returned value (result), unwrap it first (since it is of type Result<T>)
161fn generate_type_check(variable_name: &str, variable_type_name: &str, is_result: bool) -> String {
162    let mut type_check: Vec<String> = vec![];
163    type_check.push("    debug_assert!(".to_string());
164    type_check.push(format!(
165        "        self.env.is_instance_of({variable_name}, self.env.find_class({variable_type_name})?)?,"
166    ));
167    type_check.push(format!("        \"Java binding type failure. Expected object of class {{{variable_type_name}}}, but got {{}} instead\", get_jobject_class_name(self.env, {variable_name})"));
168    type_check.push("    );".to_string());
169
170    if is_result {
171        let indented_type_check = type_check
172            .iter()
173            .map(|x| format!("    {x}"))
174            .collect::<Vec<_>>()
175            .join("\n");
176
177        vec![
178            "    if let Ok(result) = result {".to_string(),
179            indented_type_check,
180            "    }".to_string(),
181        ]
182    } else {
183        type_check
184    }
185    .join("\n")
186        + "\n"
187}

189/// Generates dynamic type checks to assert that the result (named result and wrapped in the Result enum) object is of an expected class.
190/// Only to be called on object of java class types (having signature starting with L)
191/// class_name_variable_name - needs to be a name of a variable in the generated code that holds the '/'-separated class name
192pub fn generate_result_type_check(class_name_variable_name: &str) -> String {
193    generate_type_check("result", class_name_variable_name, true)
194}

196/// Generates dynamic type checks to assert that an object (receiver or parameter) is of an expected class.
197/// Only to be called on object of java class types (having signature starting with L)
198/// class_name_variable_name - needs to be a name of a variable in the generated code that holds the '/'-separated class name
199pub fn generate_variable_type_check(variable_name: &str, class_name_variable_name: &str) -> String {
200    generate_type_check(variable_name, class_name_variable_name, false)
