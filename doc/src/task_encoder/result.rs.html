24/// Records whether the failure of a task has been reported to the user.
25///
26/// The dependency error that passes the failure on to a dependent refers to
27/// the record. A dependent that reports a dependency error marks the records
28/// it refers to (see [EncodeFullError::mark_reported]). The final error
29/// collection reports each failure whose record is not marked (see
30/// [TaskEncoder::all_outputs_local]), so a failure is only left out
31/// once it was reported, and a dependent that drops the error cannot hide it.
32#[derive(Clone, Debug, Default)]
33pub struct ErrorReport(std::rc::Rc<std::cell::OnceCell<vir::Reported>>);
34
35impl ErrorReport {
36    /// The record of a failure that was reported when it occurred.
37    pub(crate) fn reported(reported: vir::Reported) -> Self {
38        let report = Self::default();
39        report.mark(reported);
40        report
41    }
42
43    fn mark(&self, reported: vir::Reported) {
44        // Marking it once is enough to prove that the failure was reported.
45        let _ = self.0.set(reported);
46    }
47
48    pub(crate) fn is_marked(&self) -> bool {
49        self.0.get().is_some()
50    }
51}

53/// What kind of failure a dependency error is.
54#[derive(Clone, Copy, Debug, PartialEq, Eq)]
55pub enum ErrorKind {
56    /// A limitation of the encoder, e.g. an unsupported feature.
57    Unsupported,
58    /// A mistake in the user's program, such as a specification that calls an
59    /// impure function.
60    Incorrect,
61    /// An internal error (a panic, a cycle or an enqueueing error), which is
62    /// reported as such, also when unsupported features are only warnings.
63    Internal,
64}
65
66/// One link of a dependency error: the encoder, the task it was encoding (or,
67/// for the last link, the error itself), the spans involved, the kind of the
68/// failure that the chain passes on, and optionally advice for the user on
69/// how to avoid the error. A link that names a failed dependency refers to
70/// the record of that dependency's failure.
71#[derive(Clone)]
72pub struct DependencyErrorLink {
73    pub encoder: &'static str,
74    pub message: String,
75    pub spans: Vec<Span>,
76    pub kind: ErrorKind,
77    pub help: Option<String>,
78    pub(crate) report: Option<ErrorReport>,
79}

81// Manual implementation: errors are shown to users through `Debug` (e.g. the
82// encoder errors of the final collection, or a panic on an error), and whether
83// the record is marked depends on what has been reported so far, so it is left
84// out to keep these messages independent of the reporting order.
85impl std::fmt::Debug for DependencyErrorLink {
86    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
87        f.debug_struct("DependencyErrorLink")
88            .field("encoder", &self.encoder)
89            .field("message", &self.message)
90            .field("spans", &self.spans)
91            .field("kind", &self.kind)
92            .field("help", &self.help)
93            .finish_non_exhaustive()
94    }
95}

97impl DependencyErrorLink {
98    pub fn new(encoder: &'static str, message: String, spans: Vec<Span>, kind: ErrorKind) -> Self {
99        Self {
100            encoder,
101            message,
102            spans,
103            kind,
104            help: None,
105            report: None,
106        }
107    }
108
109    /// The link for a task whose encoding failed with the dependency error
110    /// `chain`, which it passes on: it has the kind of the chain's root.
111    pub(crate) fn passing_on(
112        encoder: &'static str,
113        task: String,
114        spans: Vec<Span>,
115        chain: &[Self],
116    ) -> Self {
117        let kind = chain
118            .last()
119            .map_or(ErrorKind::Unsupported, |root| root.kind);
120        Self::new(encoder, task, spans, kind)
121    }
122
123    #[must_use]
124    pub fn with_help(mut self, help: String) -> Self {
125        self.help = Some(help);
126        self
127    }
128}

130/// An unsuccessful result occurring in `do_encode_full`.
131pub enum EncodeFullError<'vir, E: TaskEncoder + 'vir + ?Sized> {
132    /// Indicates that the current task has already been encoded. This can
133    /// occur when there are cyclic dependencies between multiple encoders.
134    /// This error is specifically returned when one encoder depends on
135    /// another encoder (using e.g. `TaskEncoderDependencies::require_ref`),
136    /// that latter encoder then depending on the former again, causing the
137    /// former encoder to complete its full encoding in the inner invocation.
138    /// The outer invocation remains on the stack, but will be aborted early
139    /// as soon as the control flow returns to it.
140    AlreadyEncoded,
141
142    /// An actual error occurred during encoding.
143    EncodingError(
144        <E as TaskEncoder>::EncodingError,
145        Option<Box<E::OutputFullDependency<'vir>>>,
146    ),
147
148    DependencyError(Vec<DependencyErrorLink>),
149
150    /// The task cannot be encoded, and this was already reported to the user.
151    /// It is passed on to the encoders that depend on the task, so that they
152    /// do not report it again.
153    Reported(vir::Reported),
154}
155
156impl<'vir, E: TaskEncoder + 'vir + ?Sized> EncodeFullError<'vir, E> {
157    /// The error for a feature that `E` does not support, which `message`
158    /// describes, at `spans`.
159    pub fn unsupported(message: String, spans: Vec<Span>) -> Self {
160        Self::root(message, spans, ErrorKind::Unsupported)
161    }
162
163    /// The error for an internal error of `E` (e.g. an unexpected analysis
164    /// result), which `message` describes, at `spans`.
165    pub fn internal(message: String, spans: Vec<Span>) -> Self {
166        Self::root(message, spans, ErrorKind::Internal)
167    }
168
169    /// The error for a mistake in the user's program, such as a specification
170    /// that calls an impure function, which `message` describes, at `spans`.
171    pub fn incorrect(message: String, spans: Vec<Span>) -> Self {
172        Self::root(message, spans, ErrorKind::Incorrect)
173    }
174
175    fn root(message: String, spans: Vec<Span>, kind: ErrorKind) -> Self {
176        Self::DependencyError(vec![DependencyErrorLink::new(
177            E::ENCODER_NAME,
178            message,
179            spans,
180            kind,
181        )])
182    }
183
184    /// Records that this error has been reported, as `reported` proves: the
185    /// failures of the dependencies it contains are then not reported again.
186    pub fn mark_reported(&self, reported: vir::Reported) {
187        if let Self::DependencyError(chain) = self {
188            for report in chain.iter().filter_map(|link| link.report.as_ref()) {
189                report.mark(reported);
190            }
191        }
192    }
193}

204                .finish(),
205            Self::DependencyError(contents) => write!(f, "DependencyError({contents:?})"),
206            Self::Reported(reported) => write!(f, "Reported({reported:?})"),
207        }
208    }

211pub enum TaskEncoderError<E: TaskEncoder + ?Sized> {
212    EnqueueingError(<E as TaskEncoder>::EnqueueingError),
213    EncodingError(<E as TaskEncoder>::EncodingError),

215    CyclicError,
216    PanicError(String),
217    Reported(vir::Reported),

220impl<E: TaskEncoder + ?Sized> std::fmt::Debug for TaskEncoderError<E>
221where
222    <E as TaskEncoder>::EncodingError: std::fmt::Debug,
223{
224    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
225        let mut helper = f.debug_struct("TaskEncoderError");
226        match self {
227            Self::EncodingError(err) => helper.field("EncodingError", err),
228            Self::EnqueueingError(err) => helper.field("EnqueueingError", err),
229            Self::DependencyError(stack) => helper.field("DependencyError", stack),
230            Self::CyclicError => helper.field("CyclicError", &""),
231            Self::PanicError(msg) => helper.field("PanicError", msg),
232            Self::Reported(reported) => helper.field("Reported", reported),
233        };
234        helper.finish()
235    }
236}
