3pub enum TaskEncoderCacheState<'vir, E: TaskEncoder + 'vir + ?Sized> {
4    // None, // indicated by absence in the cache
5    /// Task's encoding is running but has not yet emitted an output
6    /// reference. Requesting the task re-runs the encoding (see `Started`).
7    Encoding,
8
9    /// Task is being re-encoded while its first encoding (which has not yet
10    /// emitted an output reference, see `Encoding`) is suspended. Requesting
11    /// the task in this state is a cyclic dependency error: a third run
12    /// would only repeat the re-run's requests.
13    ReEncoding,
14
15    /// Task is currently being encoded. The output reference is available.
16    /// Full encoding is not available yet: querying for it re-runs the
17    /// encoding, which succeeds when the re-run makes progress (e.g. because
18    /// its dependencies now find this task's output reference or the outputs
19    /// cached by the first run), in which case the earlier run adopts its
20    /// result (`EncodeFullError::AlreadyEncoded`).
21    Started {
22        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
23    },
24
25    /// Task is being re-encoded while its first encoding is suspended (see
26    /// `Started`). Requesting the task in this state is a cyclic dependency
27    /// error: a third run would only repeat the re-run's requests.
28    Restarted {
29        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
30    },
31
32    /// Task was successfully encoded.
33    /// TODO: can still collect errors?
34    Encoded {
35        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
36        deps: TaskEncoderDependencies<'vir, E>,
37        output_local: <E as TaskEncoder>::OutputFullLocal<'vir>,
38        output_dep: <E as TaskEncoder>::OutputFullDependency<'vir>,
39    },
40
41    /// The task's encoding failed before emitting an output reference.
42    ErrorBeforeRef { failure: TaskFailure<E> },
43
44    /// An error occurred when encoding the task. The full "local" encoding is
45    /// not available. However, tasks which depend on this task may still
46    /// succeed, so the encoding for dependents may be present.
47    ///
48    /// As an example, encoding a method may fail, but it may still be possible
49    /// to encode its signature, to be included in dependents' programs.
50    ErrorEncode {
51        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
52        deps: TaskEncoderDependencies<'vir, E>,
53        output_dep: Option<<E as TaskEncoder>::OutputFullDependency<'vir>>,
54        failure: TaskFailure<E>,
55    },
56}

66impl<'vir, E: TaskEncoder + 'vir + ?Sized> TaskEncoderCacheState<'vir, E> {
67    /// The task's output reference, if its encoding (or a re-run of it)
68    /// emitted one.
69    pub fn output_ref(&self) -> Option<&<E as TaskEncoder>::OutputRef<'vir>> {
70        match self {
71            Self::Started { output_ref }
72            | Self::Restarted { output_ref }
73            | Self::Encoded { output_ref, .. }
74            | Self::ErrorEncode { output_ref, .. } => Some(output_ref),
75            Self::Encoding | Self::ReEncoding | Self::ErrorBeforeRef { .. } => None,
76        }
77    }
78
79    /// The task's failure, if its encoding failed.
80    pub fn failure(&self) -> Option<&TaskFailure<E>> {
81        match self {
82            Self::ErrorBeforeRef { failure } | Self::ErrorEncode { failure, .. } => Some(failure),
83            Self::Encoding
84            | Self::ReEncoding
85            | Self::Started { .. }
86            | Self::Restarted { .. }
87            | Self::Encoded { .. } => None,
88        }
89    }
90
91    /// See [Self::failure].
92    pub fn failure_mut(&mut self) -> Option<&mut TaskFailure<E>> {
93        match self {
94            Self::ErrorBeforeRef { failure } | Self::ErrorEncode { failure, .. } => Some(failure),
95            Self::Encoding
96            | Self::ReEncoding
97            | Self::Started { .. }
98            | Self::Restarted { .. }
99            | Self::Encoded { .. } => None,
100        }
101    }
102
103    /// Whether the task's encoding has finished, successfully or not.
104    pub fn is_finished(&self) -> bool {
105        match self {
106            Self::Encoded { .. } | Self::ErrorBeforeRef { .. } | Self::ErrorEncode { .. } => true,
107            Self::Encoding | Self::ReEncoding | Self::Started { .. } | Self::Restarted { .. } => {
108                false
109            }
110        }
111    }
112}

114/// Cache for a task encoder. See `TaskEncoderCacheState` for a description of
115/// the possible values in the encoding process.
116pub type Cache<'vir, E> =
117    FxIndexMap<<E as TaskEncoder>::TaskKey<'vir>, TaskEncoderCacheState<'vir, E>>;
118pub type CacheRef<'vir, E> = RefCell<Cache<'vir, E>>;
119
120pub type CacheStatic<E> =
121    FxIndexMap<<E as TaskEncoder>::TaskKey<'static>, TaskEncoderCacheState<'static, E>>;
122pub type CacheStaticRef<E> = RefCell<CacheStatic<E>>;

124/// Create the cache storage (a static `RefCell`) and a `with_cache`
125/// implementation within a `TaskEncoder` `impl` block. This should always be
126/// placed at the beginning of the `impl` block for consistency.
127///
128/// (Implementation notes: the implementation is always the same. However, it
129/// cannot be a method provided by the trait, because such an implementation
130/// would only create a single static; each cache storage must syntactically
131/// differ. A supertrait of `TaskEncoder` which only contains the cache and
132/// has a derive macro *might* work, but the `CacheRef` etc types make this a
133/// bit difficult without introducing a cyclic dependency in the two traits.)
134#[macro_export]
135macro_rules! encoder_cache {
136    ($encoder: ty) => {
137        fn with_cache<'vir, F, R>(f: F) -> R
138            where F: FnOnce(&'vir $crate::CacheRef<'vir, $encoder>) -> R,
139        {
140            ::std::thread_local! {
141                static CACHE: $crate::CacheStaticRef<$encoder> = ::std::cell::RefCell::new(Default::default());
142            }
143            CACHE.with(|cache| {
144                // SAFETY: the 'vir and 'tcx given to this function will always be
145                //   the same (or shorter) than the lifetimes of the VIR arena and
146                //   the rustc type context, respectively
147                let cache = unsafe { ::std::mem::transmute(cache) };
148                f(cache)
149            })
150        }
151
152        fn with_watchers<'vir, F, R>(f: F) -> R
153            where F: FnOnce(&'vir $crate::WatchersRef<'vir, $encoder>) -> R,
154        {
155            ::std::thread_local! {
156                static WATCHERS: $crate::WatchersStaticRef<$encoder> = ::std::cell::RefCell::new(Default::default());
157            }
158            WATCHERS.with(|watchers| {
159                // SAFETY: as for the cache above
160                let watchers = unsafe { ::std::mem::transmute(watchers) };
161                f(watchers)
162            })
163        }
164    };
165}
