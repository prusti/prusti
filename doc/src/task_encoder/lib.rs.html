59/// An operator with a body and no pre- or postcondition, as a domain function
60/// and the axiom that defines it (see [`Program::add_user_operator`]).
61#[derive(Debug)]
62struct Operator<'vir> {
63    function: vir::DomainFunction<'vir>,
64    definition: vir::DomainAxiom<'vir>,
65}
66
67impl<'vir> Operator<'vir> {
68    /// `function`, whose body is `body`, as a domain function with the same
69    /// name, arguments and `no_auto_trigger`, defined by the axiom
70    /// `forall args :: { function(args) } function(args) == body`.
71    fn new(function: vir::Function<'vir>, body: vir::ExprDyn<'vir>) -> Self {
72        vir::with_vcx(|vcx| {
73            let name = vir::ViperIdent::new(function.name);
74            let tys = vcx.alloc_slice(&function.args.iter().map(|arg| arg.ty).collect::<Vec<_>>());
75            let idn = vir::FunctionIdn::<vir::ManyDyn, _>::new(name, tys, function.ret);
76            let args = function
77                .args
78                .iter()
79                .map(|&arg| vcx.mk_local_ex(arg))
80                .collect::<Vec<vir::ExprDyn<'vir>>>();
81            let app = idn.call()(&args);
82            let definition = vcx.mk_domain_axiom(
83                vir::ViperIdent::new(vcx.alloc_str(&format!("{}_definition", function.name))),
84                vcx.mk_forall_expr(
85                    function.args,
86                    vcx.alloc_slice(&[vcx.mk_trigger(&[app])]),
87                    vcx.mk_eq_expr(app, body),
88                ),
89            );
90            let function = vcx.alloc(vir::DomainFunctionData {
91                unique: false,
92                name,
93                args: tys,
94                ret: function.ret,
95                interpretation: None,
96                no_auto_trigger: function.no_auto_trigger,
97            });
98            Operator {
99                function,
100                definition,
101            }
102        })
103    }
104}

129    /// Adds an adt whose constructors and fields encode what the user writes,
130    /// such as the snapshot of a Rust struct. [`Self::simplify`] never folds
131    /// away an application of such a user-level adt, domain or function, so
132    /// that a trigger of a quantifier the user wrote still finds the terms the
133    /// user wrote. What rustc adds to the MIR, or the encoding introduces, is
134    /// not user-level and may be folded away.
135    ///
136    /// - User-level: the snapshots of structs, enums, arrays and slices, pure
137    ///   functions, operators, constants, ghost collection operations and
138    ///   value casts, which come from `as`.
139    /// - Not user-level: references, boxes, raw pointers, closures and
140    ///   tuples, which rustc and Prusti build and take apart themselves (e.g.
141    ///   auto-refs, the arguments of a closure call, the trigger sets of a
142    ///   quantifier); the snapshots of primitives and generic parameters,
143    ///   which wrap values for the encoding; and pointer coercions, which are
144    ///   mostly implicit (one written with `as` is encoded as the same
145    ///   function).
146    pub fn add_user_adt(&mut self, adt: vir::Adt<'vir>) {
147        self.user_level.insert(adt.name);
148        self.add_adt(adt);
149    }

151    /// Adds a domain whose functions are user-level (see
152    /// [`Self::add_user_adt`]).
153    pub fn add_user_domain(&mut self, domain: vir::Domain<'vir>) {
154        self.user_level.insert(domain.name);
155        self.add_domain(domain);
156    }

158    /// Adds a user-level function (see [`Self::add_user_adt`]).
159    pub fn add_user_function(&mut self, function: vir::Function<'vir>) {
160        self.user_level.insert(function.name);
161        self.add_function(function);
162    }

164    /// Adds a user-level function that encodes an operator or a cast (see
165    /// [`Self::add_user_adt`]), marked `no_auto_trigger`: a trigger on the
166    /// result of `i - 1` matches the subtraction that a recursive function
167    /// performs, and can instantiate its quantifier without end.
168    ///
169    /// An operator with a body but no pre- or postcondition is added instead
170    /// as a domain function of the user-level domain [`OPERATORS_DOMAIN`],
171    /// with an axiom for its body (see [`Operator::new`]). Silicon applies the body of
172    /// a function only to the applications it has evaluated, which in a
173    /// quantifier are those a trigger matches; an axiom applies to every
174    /// application.
175    pub fn add_user_operator(&mut self, function: vir::Function<'vir>) {
176        vir::with_vcx(|vcx| {
177            let function = vcx.alloc(vir::FunctionGenData {
178                no_auto_trigger: true,
179                ..*function
180            });
181            let (Some(body), [], []) = (function.expr, function.pres, function.posts) else {
182                self.add_user_function(function);
183                return;
184            };
185            if self.operators.is_empty() {
186                self.user_level.insert(OPERATORS_DOMAIN);
187                self.code.push(CodeItem::Operators);
188            }
189            self.operators.push(Operator::new(function, body));
190        });
191    }
192
193    /// The domain [`OPERATORS_DOMAIN`], if an operator was added to it.
194    fn operators_domain(&self) -> Option<vir::Domain<'vir>> {
195        if self.operators.is_empty() {
196            return None;
197        }
198        let (functions, definitions): (Vec<_>, Vec<_>) = self
199            .operators
200            .iter()
201            .map(|op| (op.function, op.definition))
202            .unzip();
203        Some(vir::with_vcx(|vcx| {
204            vcx.mk_domain(
205                vir::ViperIdent::new(OPERATORS_DOMAIN),
206                &[],
207                vcx.alloc_slice(&definitions),
208                vcx.alloc_slice(&functions),
209                None,
210            )
211        }))
212    }
213

217        self.domains
218            .iter()
219            .copied()
220            .chain(self.operators_domain())

244    /// The program in Viper syntax, with section headers.
245    pub fn code(&self) -> String {
246        let mut code = String::new();
247        for item in &self.code {
248            match item {
249                CodeItem::Header(title) => {
250                    code.push_str("// -----------------------------\n");
251                    code.push_str(&format!("// {title}\n"));
252                    code.push_str("// -----------------------------\n");
253                }
254                CodeItem::Field(idx) => code.push_str(&format!("{:?}\n", self.fields[*idx])),
255                CodeItem::Adt(idx) => code.push_str(&format!("{:?}\n", self.adts[*idx])),
256                CodeItem::Domain(idx) => code.push_str(&format!("{:?}\n", self.domains[*idx])),
257                CodeItem::Predicate(idx) => {
258                    code.push_str(&format!("{:?}\n", self.predicates[*idx]))
259                }
260                CodeItem::Function(idx) => code.push_str(&format!("{:?}\n", self.functions[*idx])),
261                CodeItem::Method(idx) => code.push_str(&format!("{:?}\n", self.methods[*idx])),
262                CodeItem::Operators => {
263                    if let Some(domain) = self.operators_domain() {
264                        code.push_str(&format!("{domain:?}\n"));
265                    }
266                }
267            }
268        }
269        code
270    }

278    /// Simplifies the expressions of all domain axioms, predicates, functions
279    /// and methods added so far (see [`vir::simplify`]).
280    pub fn simplify(&mut self) {
281        vir::with_vcx(|vcx| {
282            let ctx = vir::simplify::SimplifyCtx::new(
283                &self.adts,
284                &self.all_domains(),
285                &self.functions,
286                &self.literal_inverses,
287                &self.user_level,
288            );
289            self.fold(&mut vir::simplify::Simplify::new(vcx, &ctx));
290        });
291    }

293    /// Folds the items added so far with `f`, in the order of
294    /// [`Self::mk_program`].
295    fn fold(&mut self, f: &mut impl vir::Folder<'vir, (), !>) {
296        fn fold_all<'a, 'vir, T: vir::Foldable<'vir, (), !> + 'a>(
297            items: impl IntoIterator<Item = &'a mut T>,
298            f: &mut impl vir::Folder<'vir, (), !>,
299        ) {
300            for item in items {
301                if let Some(folded) = item.fold_with(f) {
302                    *item = folded;
303                }
304            }
305        }
306        fold_all(&mut self.adts, f);
307        fold_all(&mut self.domains, f);
308        fold_all(self.operators.iter_mut().map(|op| &mut op.definition), f);
309        fold_all(&mut self.predicates, f);
310        fold_all(&mut self.functions, f);
311        fold_all(&mut self.methods, f);
312    }

321                vcx.alloc_slice(&self.functions),
322                vcx.alloc_slice(&self.methods),

338pub trait TaskEncoder {
339    /// Description of a task to be performed. Should be easily obtained by
340    /// clients of this encoder.
341    type TaskDescription<'vir>: std::hash::Hash + Eq + Clone + std::fmt::Debug;
342
343    /// Cache key for a task to be performed. May differ from `TaskDescription`,
344    /// for example if the description should be normalised or some non-trivial
345    /// resolution needs to happen. In other words, multiple descriptions may
346    /// lead to the same key and hence the same output.
347    type TaskKey<'vir>: std::hash::Hash + Eq + Clone + std::fmt::Debug =
348        Self::TaskDescription<'vir>;
349
350    /// A reference to an encoded item. Should be non-unit for tasks which can
351    /// be "referred" to from other parts of a program, as opposed to tasks
352    /// where only the full output matters.
353    type OutputRef<'vir>: Clone + std::fmt::Debug + OutputRefAny
354        = ()
355    where
356        Self: 'vir;
357
358    /// Fully encoded output for this task. When encoding items which can be
359    /// dependencies (such as methods), this output should only be emitted in
360    /// one Viper program.
361    type OutputFullLocal<'vir>: Clone
362        = ()
363    where
364        Self: 'vir;
365
366    /// Fully encoded output for this task for dependents. When encoding items
367    /// which can be dependencies (such as methods), this output should be
368    /// emitted in each Viper program that depends on this task.
369    type OutputFullDependency<'vir>: Clone
370        = ()
371    where
372        Self: 'vir;

374    type EnqueueingError: Clone + std::fmt::Debug = NeverError;
375    type EncodingError: Clone + std::fmt::Debug = NeverError;

378    const ENCODER_NAME: &'static str;

381        format!("{task:?}")
382    }

388    /// Enters the given function with a reference to the cache for this
389    /// encoder.
390    fn with_cache<'vir, F, R>(f: F) -> R
391    where
392        Self: 'vir,
393        F: FnOnce(&'vir CacheRef<'vir, Self>) -> R;
394
395    /// Enters the given function with a reference to the trigger watchers for
396    /// this encoder. These are continuations registered via [`Self::on_task_requested`],
397    /// waiting for a task of this encoder to be requested.
398    fn with_watchers<'vir, F, R>(f: F) -> R
399    where
400        Self: 'vir,
401        F: FnOnce(&'vir WatchersRef<'vir, Self>) -> R;
402
403    /// Queues `f` to run in [`drain_triggers`] once the given task of this
404    /// encoder has been requested (enqueued, started, or encoded - not
405    /// necessarily successfully). If it already has been, `f` is queued
406    /// immediately. Continuations of tasks that are never requested do not
407    /// run.
408    ///
409    /// Triggers must be registered during the encoding phase:
410    /// [`drain_triggers`] runs before outputs are emitted, so continuations
411    /// registered during `emit_outputs` would never run. From within an
412    /// in-flight encoder that already knows the task is needed, use a plain
413    /// `deps.require_*` instead.
414    fn on_task_requested<'vir>(key: Self::TaskKey<'vir>, f: impl FnOnce() + 'vir)
415    where
416        Self: Sized + 'vir,
417    {
418        let f = mk_continuation(f);
419        let requested = Self::with_cache(|cache| cache.borrow().contains_key(&key));
420        if requested {
421            triggers::queue(f);
422        } else {
423            Self::with_watchers(move |watchers| {
424                watchers.borrow_mut().entry(key).or_default().push(f)
425            });
426        }
427    }
428
429    /// Queues `f` to run once *every* task in `keys` has been requested (the
430    /// conjunction of [`Self::on_task_requested`]). An empty `keys` runs `f`
431    /// unconditionally.
432    fn on_all_requested<'vir>(mut keys: Vec<Self::TaskKey<'vir>>, f: impl FnOnce() + 'vir)
433    where
434        Self: Sized + 'vir,
435    {
436        match keys.pop() {
437            None => queue(mk_continuation(f)),
438            Some(key) => Self::on_task_requested(key, move || Self::on_all_requested(keys, f)),
439        }
440    }
441
442    /// Queues `f` to run once *any* task in `keys` has been requested (the
443    /// disjunction of [`Self::on_task_requested`]). `f` runs at most once, on
444    /// the first such task; an empty `keys` never runs `f`.
445    fn on_any_requested<'vir>(keys: Vec<Self::TaskKey<'vir>>, f: impl FnOnce() + 'vir)
446    where
447        Self: Sized + 'vir,
448    {
449        // The continuations share ownership of `f` and race to take it, so that
450        // it runs exactly once regardless of how many keys become requested.
451        let slot: std::rc::Rc<std::cell::Cell<Option<triggers::Continuation>>> =
452            std::rc::Rc::new(std::cell::Cell::new(Some(mk_continuation(f))));
453        for key in keys {
454            let slot = slot.clone();
455            Self::on_task_requested(key, move || {
456                if let Some(f) = slot.take() {
457                    f();
458                }
459            });
460        }
461    }

491    fn encode_ref<'vir>(
492        task: Self::TaskDescription<'vir>,
493        span: Span,
494    ) -> Result<Self::OutputRef<'vir>, TaskEncoderError<Self>>
495    where
496        Self: 'vir,
497    {
498        let task_key = Self::task_to_key(&task);
499
500        // is there an output ref available already?
501        if let Some(output_ref) = cached_output_ref::<Self>(&task_key) {
502            return Ok(output_ref);
503        }
504
505        // Otherwise, we need to start the encoding. Note that this is done
506        // even if the encoding was started previously, i.e. if the same task
507        // was (recursively) requested from the same encoder before its first
508        // invocation reached a call to `emit_output_ref` (see
509        // `TaskEncoderCacheState::Encoding`).
510        match Self::encode(task, false, span) {
511            // A successful encoding has emitted the output ref.
512            Ok(_) => {
513                Ok(cached_output_ref::<Self>(&task_key)
514                    .expect("output ref not found after encoding"))
515            }
516            // The output ref may have been emitted before the error. A
517            // requester then gets it, as later requesters do from the cache,
518            // and fails only if it needs the part that could not be encoded.
519            Err(err @ (TaskEncoderError::DependencyError(..) | TaskEncoderError::Reported(..))) => {
520                cached_output_ref::<Self>(&task_key).ok_or(err)
521            }
522            Err(err) => Err(err),
523        }
524    }
525
526    fn encode<'vir>(
527        task: Self::TaskDescription<'vir>,
528        need_output: bool,
529        span: Span,
530    ) -> EncodeResult<'vir, Self>
531    where
532        Self: 'vir,
533    {
534        let task_key = Self::task_to_key(&task);
535
536        let in_cache = Self::with_cache(|cache| {
537            let mut cache = cache.borrow_mut();
538
539            let new_state = match cache.get(&task_key) {
540                Some(
541                    TaskEncoderCacheState::ErrorBeforeRef { failure }
542                    | TaskEncoderCacheState::ErrorEncode { failure, .. },
543                ) => return Some(Err(failure.error.clone())),
544                Some(TaskEncoderCacheState::Encoded {
545                    output_ref,
546                    output_local,
547                    output_dep,
548                    ..
549                }) => {
550                    return if need_output {
551                        Some(Ok(Some((
552                            output_ref.clone(),
553                            output_local.clone(),
554                            output_dep.clone(),
555                        ))))
556                    } else {
557                        Some(Ok(None))
558                    };
559                }
560                // A re-run is already in progress; another one could not make
561                // progress over it (see `TaskEncoderCacheState`).
562                Some(
563                    TaskEncoderCacheState::ReEncoding | TaskEncoderCacheState::Restarted { .. },
564                ) => return Some(Err(TaskEncoderError::CyclicError)),
565                // Start the first run.
566                None => {
567                    triggers::fire_watchers::<Self>(&task_key);
568                    TaskEncoderCacheState::Encoding
569                }
570                // In progress: re-run the encoding (see
571                // `TaskEncoderCacheState::Started`).
572                Some(TaskEncoderCacheState::Encoding) => TaskEncoderCacheState::ReEncoding,
573                Some(TaskEncoderCacheState::Started { output_ref }) => {
574                    TaskEncoderCacheState::Restarted {
575                        output_ref: output_ref.clone(),
576                    }
577                }
578            };
579            cache.insert(task_key.clone(), new_state);
580            None
581        });
582        if let Some(in_cache) = in_cache {
583            return in_cache;
584        }
585
586        let value = task_key.clone();
587        // The span stack is isolated across the encoder-context boundary: the
588        // encoding is demand-driven and cached, so its spans must not chain
589        // up to whatever span is ambient at the (first) demand site.
590        let catch_result = vir::with_vcx(|vcx| {
591            vcx.with_span_stack_isolated(|| {
592                std::panic::catch_unwind(std::panic::AssertUnwindSafe(move || {
593                    let mut deps = TaskEncoderDependencies::new();
594                    let encode_result = Self::do_encode_full(&value, &mut deps);
595                    (encode_result, deps)
596                }))
597            })
598        });
599
600        let (encode_result, deps) = catch_result.map_err(|panic_payload| {
601            // There was a panic within the encoder. We want to report it
602            // and return an error to the caller.
603            let msg = if let Some(s) = panic_payload.downcast_ref::<&str>() {
604                s.to_string()
605            } else if let Some(s) = panic_payload.downcast_ref::<String>() {
606                s.clone()
607            } else {
608                "<unknown panic>".to_string()
609            };
610            let error = TaskEncoderError::PanicError(msg);
611            record_error::<Self>(
612                task_key.clone(),
613                error.clone(),
614                TaskEncoderDependencies::new(),
615                None,
616                span,
617            );
618            error
619        })?;
620
621        match encode_result {
622            Ok((output_local, output_dep)) => {
623                let output_ref = cached_output_ref::<Self>(&task_key).unwrap_or_else(|| {
624                    panic!("encoder did not provide output ref for task {task_key:?}")
625                });
626                if need_output {
627                    Self::with_cache(|cache| {
628                        cache.borrow_mut().insert(
629                            task_key,
630                            TaskEncoderCacheState::Encoded {
631                                output_ref: output_ref.clone(),
632                                deps,
633                                output_local: output_local.clone(),
634                                output_dep: output_dep.clone(),
635                            },
636                        )
637                    });
638                    Ok(Some((output_ref, output_local, output_dep)))
639                } else {
640                    Self::with_cache(|cache| {
641                        cache.borrow_mut().insert(
642                            task_key,
643                            TaskEncoderCacheState::Encoded {
644                                output_ref,
645                                deps,
646                                output_local,
647                                output_dep,
648                            },
649                        )
650                    });
651                    Ok(None)
652                }
653            }
654            Err(EncodeFullError::AlreadyEncoded) => {
655                Self::with_cache(|cache| match cache.borrow().get(&task_key).unwrap() {
656                    TaskEncoderCacheState::Encoded {
657                        output_ref,
658                        output_local,
659                        output_dep,
660                        ..
661                    } => {
662                        if need_output {
663                            Ok(Some((
664                                // TODO: does it even make sense for an encoder to request the full encoding
665                                //   when a cycle can occur?
666                                output_ref.clone(),
667                                output_local.clone(),
668                                output_dep.clone(),
669                            )))
670                        } else {
671                            Ok(None)
672                        }
673                    }
674                    TaskEncoderCacheState::ErrorBeforeRef { failure }
675                    | TaskEncoderCacheState::ErrorEncode { failure, .. } => {
676                        Err(failure.error.clone())
677                    }
678                    TaskEncoderCacheState::Encoding
679                    | TaskEncoderCacheState::ReEncoding
680                    | TaskEncoderCacheState::Started { .. }
681                    | TaskEncoderCacheState::Restarted { .. } => {
682                        panic!("encoder did not finish for task {task_key:?}")
683                    }
684                })
685            }
686            Err(EncodeFullError::DependencyError(stack)) => {
687                let owned_stack = std::iter::once(DependencyErrorLink::passing_on(
688                    Self::ENCODER_NAME,
689                    Self::describe_task(task),
690                    Vec::new(),
691                    &stack,
692                ))
693                .chain(stack)
694                .collect::<Vec<_>>();
695                record_error::<Self>(
696                    task_key,
697                    TaskEncoderError::DependencyError(owned_stack.clone()),
698                    deps,
699                    None,
700                    span,
701                );
702                Err(TaskEncoderError::DependencyError(owned_stack))
703            }
704            Err(EncodeFullError::EncodingError(err, maybe_output_dep)) => {
705                record_error::<Self>(
706                    task_key,
707                    TaskEncoderError::EncodingError(err.clone()),
708                    deps,
709                    maybe_output_dep.map(|dep| *dep),
710                    span,
711                );
712                Err(TaskEncoderError::EncodingError(err))
713            }
714            Err(EncodeFullError::Reported(reported)) => {
715                record_error::<Self>(
716                    task_key,
717                    TaskEncoderError::Reported(reported),
718                    deps,
719                    None,
720                    span,
721                );
722                Err(TaskEncoderError::Reported(reported))
723            }
724        }
725    }

839    /// The local outputs of the successfully encoded tasks of this encoder.
840    /// The failures of the other tasks that have not been reported (see
841    /// [ErrorReport]) are added to `program` as encoder errors.
842    fn all_outputs_local<'vir>(program: &mut Program<'vir>) -> Vec<Self::OutputFullLocal<'vir>>
843    where
844        Self: 'vir,
845    {
846        Self::with_cache(|cache| {
847            let mut outputs = Vec::new();
848            for (key, cache_state) in cache.borrow().iter() {
849                if let TaskEncoderCacheState::Encoded { output_local, .. } = cache_state {
850                    outputs.push(output_local.clone());
851                    continue;
852                }
853                let failure = cache_state
854                    .failure()
855                    .unwrap_or_else(|| panic!("task encoder not completed: {key:?}"));
856                if failure.report.is_marked() {
857                    continue;
858                }
859                let span = failure
860                    .spans
861                    .iter()
862                    .copied()
863                    .find(|span| !span.is_dummy())
864                    .unwrap_or(prusti_rustc_interface::span::DUMMY_SP);
865                let msg = match &failure.error {
866                    TaskEncoderError::EncodingError(err) => Self::describe_error(err.clone()),
867                    other => format!(
868                        "encoder '{}' failed to encode {:?}:\n {:?}",
869                        Self::ENCODER_NAME,
870                        key,
871                        other
872                    ),
873                };
874                program.encoder_errors.push((msg, span));
875            }
876            outputs
877        })
878    }

883/// The task's output ref, if its encoding (or a re-run of it) emitted one.
884fn cached_output_ref<'vir, E: TaskEncoder + 'vir + ?Sized>(
885    task_key: &E::TaskKey<'vir>,
886) -> Option<E::OutputRef<'vir>> {
887    E::with_cache(|cache| {
888        let cache = cache.borrow();
889        cache
890            .get(task_key)
891            .and_then(TaskEncoderCacheState::output_ref)
892            .cloned()
893    })
894}

896/// Records that encoding the task failed with `error`. If the encoding (or a
897/// re-run of it) emitted an output ref, the ref stays available to dependents
898/// (`ErrorEncode`); otherwise the task has none (`ErrorBeforeRef`). The spans
899/// of an error recorded earlier for the task are kept.
900fn record_error<'vir, E: TaskEncoder + 'vir + ?Sized>(
901    task_key: E::TaskKey<'vir>,
902    error: TaskEncoderError<E>,
903    deps: TaskEncoderDependencies<'vir, E>,
904    output_dep: Option<E::OutputFullDependency<'vir>>,
905    span: Span,
906) {
907    let output_ref = cached_output_ref::<E>(&task_key);
908    E::with_cache(|cache| {
909        let mut cache = cache.borrow_mut();
910        let mut spans = cache
911            .get(&task_key)
912            .and_then(TaskEncoderCacheState::failure)
913            .map_or_else(Vec::new, |failure| failure.spans.clone());
914        spans.push(span);
915        let report = match error {
916            TaskEncoderError::Reported(reported) => ErrorReport::reported(reported),
917            _ => ErrorReport::default(),
918        };
919        let failure = TaskFailure {
920            error,
921            spans,
922            report,
923        };
924        let new_state = match output_ref {
925            Some(output_ref) => TaskEncoderCacheState::ErrorEncode {
926                output_ref,
927                deps,
928                output_dep,
929                failure,
930            },
931            None => TaskEncoderCacheState::ErrorBeforeRef { failure },
932        };
933        cache.insert(task_key, new_state);
934    });
