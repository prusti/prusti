14/// A `let`-bound local in scope.
15#[derive(Clone, Copy)]
16pub(super) struct Binding<'vir> {
17    pub(super) val: ExprDyn<'vir>,
18    /// The locals `val` mentions.
19    locals: &'vir [&'vir str],
20    /// Whether `val` is a local or constant: it means the same in any state
21    /// and costs nothing to copy.
22    trivial: bool,
23    /// How often the `let` body mentions the local.
24    uses: usize,
25}

27pub(super) struct Simplifier<'a, 'vir, 'tcx> {
28    pub(super) vcx: &'vir VirCtxt<'tcx>,
29    pub(super) ctx: &'a SimplifyCtx<'vir>,
30    /// The bindings whose values mean the same at the current node as at
31    /// their `let`: a binder hides the bindings it rebinds or whose values
32    /// mention a local it binds, and `old(..)` and `unfolding` hide those
33    /// with a non-trivial value.
34    pub(super) env: FxHashMap<&'vir str, Binding<'vir>>,
35    /// The node being rewritten, whose span and debug information the nodes
36    /// built for it take.
37    pub(super) orig: Option<ExprDyn<'vir>>,
38    /// Identifies the bindings in scope: a fresh number for every scope
39    /// entered.
40    scope: usize,
41    scopes: usize,
42    /// Memoized [`Self::simplify`], by node and scope: an expression shared
43    /// in a tree is simplified once.
44    simplified: FxHashMap<(Ptr<'vir>, usize), Option<ExprDyn<'vir>>>,
45    /// Memoized [`Self::is_impure`].
46    impure: FxHashMap<Ptr<'vir>, bool>,
47}

49/// Magic wands are left untouched, wherever they occur: Viper matches
50/// packaged wand instances syntactically, and evaluates the heap-dependent
51/// subterms of a wand only when applying it, in the order of its conjuncts,
52/// so moving them around inside a wand changes which permissions are there.
53impl<'vir> Folder<'vir, (), !> for Simplifier<'_, 'vir, '_> {
54    fn fold_expr(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
55        self.simplify(e)
56    }
57
58    fn fold_wand(&mut self, _: Wand<'vir>) -> Option<Wand<'vir>> {
59        None
60    }
61}

63impl<'a, 'vir, 'tcx> Simplifier<'a, 'vir, 'tcx> {
64    pub(super) fn new(vcx: &'vir VirCtxt<'tcx>, ctx: &'a SimplifyCtx<'vir>) -> Self {
65        Self {
66            vcx,
67            ctx,
68            env: FxHashMap::default(),
69            orig: None,
70            scope: 0,
71            scopes: 0,
72            simplified: FxHashMap::default(),
73            impure: FxHashMap::default(),
74        }
75    }

77    /// Simplifies the root of an expression tree, keeping the root's span.
78    /// Viper blames e.g. a failing postcondition on the clause's root node,
79    /// and the error handler is registered on the span of that node only (see
80    /// `realloc_span`); replacing the root by a subexpression would make the
81    /// error impossible to backtranslate.
82    pub(super) fn root(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
83        let simplified = self.simplify(e)?;
84        // A root without a span has no handler; keep the result's own.
85        Some(match e.span {
86            None => simplified,
87            Some(_) => self.node_of_kind(e, simplified.kind),
88        })
89    }
90
91    /// The simplified `e`, `None` if nothing about it changes.
92    pub(super) fn simplify(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
93        let key = (std::ptr::from_ref(e), self.scope);
94        if let Some(&simplified) = self.simplified.get(&key) {
95            return simplified;
96        }
97        let simplified = self.simplify_node(e);
98        self.simplified.insert(key, simplified);
99        simplified
100    }
101
102    /// The simplified `e`. A local bound to a trivial value, or used only
103    /// once, is replaced by its value; the `let` is then dropped by the
104    /// equations, unless a use the traversal does not substitute (in a wand,
105    /// a trigger set kept as encoded, or where the binding is hidden)
106    /// remains.
107    fn simplify_node(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
108        let children = match e.kind {
109            ExprKindGenData::Local(l) => {
110                let b = self.env.get(l.name)?;
111                return (b.trivial || b.uses == 1).then_some(b.val);
112            }
113            ExprKindGenData::Let(l) => {
114                let new_val = self.simplify(l.val);
115                let val = new_val.unwrap_or(l.val);
116                let binding = Binding::new(self.vcx, val, l.name, l.expr);
117                let mut body = self.with_binding(l.name, binding, |s| s.simplify(l.expr));
118                // Folding may leave a single use of a local used more than
119                // once before; a second pass substitutes it.
120                if let Some(b) = body.filter(|b| binding.uses > 1 && occurrences(l.name, b) == 1) {
121                    let binding = Binding::new(self.vcx, val, l.name, b);
122                    body = self
123                        .with_binding(l.name, binding, |s| s.simplify(b))
124                        .or(body);
125                }
126                (new_val.is_some() || body.is_some()).then(|| {
127                    let expr = body.unwrap_or(l.expr);
128                    let l = LetGenData {
129                        name: l.name,
130                        val,
131                        expr,
132                    };
133                    self.node_at(e, ExprKindGenData::Let(self.vcx.alloc(l)))
134                })
135            }
136            // `old(..)` evaluates in another heap and the body of `unfolding`
137            // with other permissions, so only trivial values move into them.
138            ExprKindGenData::Old(_) => self.scoped(|_, b| b.trivial, |s| e.super_fold_with(s)),
139            ExprKindGenData::Unfolding(u) => {
140                let target = u.target.fold_with(self);
141                let body = self.scoped(|_, b| b.trivial, |s| s.simplify(u.expr));
142                (target.is_some() || body.is_some()).then(|| {
143                    let u = UnfoldingGenData {
144                        target: target.unwrap_or(u.target),
145                        expr: body.unwrap_or(u.expr),
146                    };
147                    self.node_at(e, ExprKindGenData::Unfolding(self.vcx.alloc(u)))
148                })
149            }
150            // The triggers are simplified like the body, so that a trigger
151            // becomes the same term as the body contains.
152            ExprKindGenData::Forall(ForallGenData {
153                qvars,
154                triggers,
155                body,
156            })
157            | ExprKindGenData::Exists(ExistsGenData {
158                qvars,
159                triggers,
160                body,
161            }) => {
162                let (new_triggers, new_body) = self.in_quantifier(qvars, |s| {
163                    (
164                        s.simplify_triggers(qvars, triggers),
165                        s.simplify(body.as_dyn()),
166                    )
167                });
168                (new_triggers.is_some() || new_body.is_some()).then(|| {
169                    let qvars = *qvars;
170                    let triggers = new_triggers.unwrap_or(*triggers);
171                    let body = new_body.unwrap_or(body.as_dyn()).inner_cast_ty();
172                    let kind = match e.kind {
173                        ExprKindGenData::Forall(_) => {
174                            let q = ForallGenData {
175                                qvars,
176                                triggers,
177                                body,
178                            };
179                            ExprKindGenData::Forall(self.vcx.alloc(q))
180                        }
181                        _ => {
182                            let q = ExistsGenData {
183                                qvars,
184                                triggers,
185                                body,
186                            };
187                            ExprKindGenData::Exists(self.vcx.alloc(q))
188                        }
189                    };
190                    self.node_at(e, kind)
191                })
192            }
193            _ => e.super_fold_with(self),
194        };
195        let node = children.unwrap_or(e);
196        let outer = self.orig.replace(node);
197        let folded = self.fold(node);
198        self.orig = outer;
199        folded.or(children)
200    }

202    /// Runs `f` in the body of a `let` binding `local`. A value that mentions
203    /// the local it is bound to means the outer local, which the body cannot
204    /// see, so such a binding is not entered.
205    pub(super) fn with_binding<R>(
206        &mut self,
207        local: &'vir str,
208        binding: Binding<'vir>,
209        f: impl FnOnce(&mut Self) -> R,
210    ) -> R {
211        self.scoped(
212            |name, b| name != local && !b.mentions(local),
213            |s| {
214                if !binding.mentions(local) {
215                    s.env.insert(local, binding);
216                }
217                f(s)
218            },
219        )
220    }

222    /// The simplified `triggers` of a quantifier over `qvars`, `None` if
223    /// nothing about them changes. A trigger set whose simplified terms are not
224    /// a valid trigger set stays as encoded: a value substituted for a `let` may
225    /// contain what a trigger cannot, such as a conditional, and a term may
226    /// fold to a variable, such as `(i, 0).0` to `i`.
227    fn simplify_triggers(
228        &mut self,
229        qvars: &[LocalDeclDyn<'vir>],
230        triggers: &'vir [Trigger<'vir>],
231    ) -> Option<&'vir [Trigger<'vir>]> {
232        let mut changed = false;
233        let mut sets = Vec::with_capacity(triggers.len());
234        for &set in triggers {
235            let terms = set
236                .exprs
237                .iter()
238                .map(|&t| self.simplify(t))
239                .collect::<Vec<_>>();
240            if terms.iter().all(Option::is_none) {
241                sets.push(set);
242                continue;
243            }
244            let terms = terms
245                .into_iter()
246                .zip(set.exprs)
247                .map(|(term, &orig)| term.unwrap_or(orig))
248                .collect::<Vec<_>>();
249            if crate::is_trigger_set(qvars, &terms) {
250                changed = true;
251                sets.push(self.vcx.mk_trigger(&terms));
252            } else {
253                sets.push(set);
254            }
255        }
256        changed.then(|| self.vcx.alloc_slice(&sets))
257    }

259    /// Runs `f` in the body of a quantifier over `qvars`.
260    fn in_quantifier<R>(
261        &mut self,
262        qvars: &[LocalDeclDyn<'vir>],
263        f: impl FnOnce(&mut Self) -> R,
264    ) -> R {
265        self.scoped(
266            |name, b| qvars.iter().all(|q| q.name != name && !b.mentions(q.name)),
267            f,
268        )
269    }
270
271    /// Runs `f` with only the bindings that `visible` keeps in scope.
272    fn scoped<R>(
273        &mut self,
274        visible: impl Fn(&str, &Binding<'vir>) -> bool,
275        f: impl FnOnce(&mut Self) -> R,
276    ) -> R {
277        let saved = self.env.clone();
278        self.env.retain(|name, b| visible(name, b));
279        self.scopes += 1;
280        let outer = std::mem::replace(&mut self.scope, self.scopes);
281        let result = f(self);
282        self.scope = outer;
283        self.env = saved;
284        result
285    }

287    /// A node of kind `kind` with the span and debug information of `e`.
288    pub(super) fn node_at(
289        &self,
290        e: ExprDyn<'vir>,
291        kind: ExprKindGenData<'vir, (), !>,
292    ) -> ExprDyn<'vir> {
293        self.node_of_kind(e, self.vcx.alloc(kind))
294    }

296    /// A node of the allocated kind `kind` with the span and debug
297    /// information of `e`.
298    pub(super) fn node_of_kind(
299        &self,
300        e: ExprDyn<'vir>,
301        kind: &'vir ExprKindGenData<'vir, (), !>,
302    ) -> ExprDyn<'vir> {
303        self.vcx.alloc(ExprGenData::new_inner(
304            kind,
305            e.debug_info,
306            e.span,
307            kind.ty(),
308        ))
309    }

311    /// Whether `e` contains an assertion (an access predicate, magic wand or
312    /// inhale-exhale expression).
313    pub(super) fn is_impure(&mut self, e: ExprDyn<'vir>) -> bool {
314        if let Some(&impure) = self.impure.get(&std::ptr::from_ref(e)) {
315            return impure;
316        }
317        let impure = match e.kind {
318            ExprKindGenData::AccField(_)
319            | ExprKindGenData::PredicateApp(_)
320            | ExprKindGenData::Wand(_)
321            | ExprKindGenData::InhaleExhale(_) => true,
322            _ => {
323                let mut children = Children(Vec::new());
324                e.super_visit_with(&mut children);
325                children.0.into_iter().any(|child| self.is_impure(child))
326            }
327        };
328        self.impure.insert(std::ptr::from_ref(e), impure);
329        impure
330    }

333impl<'vir> Binding<'vir> {
334    /// The binding of `name` to `val` in `body`.
335    pub(super) fn new(
336        vcx: &'vir VirCtxt<'_>,
337        val: ExprDyn<'vir>,
338        name: &str,
339        body: ExprDyn<'vir>,
340    ) -> Self {
341        let mut locals = FxHashSet::default();
342        collect_locals(val, &mut locals);
343        Self {
344            val,
345            locals: vcx.alloc_slice(&locals.into_iter().collect::<Vec<_>>()),
346            trivial: val.is_local_or_const(),
347            uses: occurrences(name, body),
348        }
349    }
350
351    fn mentions(&self, name: &str) -> bool {
352        self.locals.contains(&name)
353    }
354}
355
356/// How often `e` mentions the local `name`, also where it is rebound.
357pub(super) fn occurrences(name: &str, e: ExprDyn<'_>) -> usize {
358    struct Count<'n>(&'n str, usize);
359    impl<'vir> Visitor<'vir, (), !> for Count<'_> {
360        fn visit_expr(&mut self, e: ExprDyn<'vir>) {
361            if matches!(e.kind, ExprKindGenData::Local(l) if l.name == self.0) {
362                self.1 += 1;
363            }
364            e.super_visit_with(self);
365        }
366    }
367    let mut count = Count(name, 0);
368    e.visit_with(&mut count);
369    count.1
370}

372/// Collects the direct subexpressions of an expression.
373struct Children<'vir>(Vec<ExprDyn<'vir>>);
374
375impl<'vir> Visitor<'vir, (), !> for Children<'vir> {
376    fn visit_expr(&mut self, e: ExprDyn<'vir>) {
377        self.0.push(e);
378    }
