1//! Peephole simplification of hole-free expressions, run over the final
2//! program (domain axioms, predicates, functions and methods).
3//!
4//! Composing independently encoded snapshot operations produces wrap/unwrap
5//! round trips: reference snapshots built and immediately dereferenced,
6//! generic casts cancelled out, snapshot constructors compared field by
7//! field. The pass undoes them with [equations](equations), listed by the
8//! kind of node they rewrite, each of which can be checked on its own. The
9//! [traversal](walk) applies them bottom-up and takes care of everything
10//! that depends on the context of a node:
11//!  - `let`-bound locals are replaced by their values where that means the
12//!    same: inside an `old(..)` or `unfolding` only locals and constants
13//!    (other values may depend on the heap or the permissions held), and
14//!    not values whose locals a binder (a quantifier or `let`) rebinds,
15//!  - the span of the root of an expression tree is kept, since Prusti
16//!    backtranslates Viper errors through it,
17//!  - quantifier triggers are simplified like the body, so that they become
18//!    the same terms as the body contains, unless that would make them
19//!    invalid triggers,
20//!  - magic wands are left as encoded.
21//!
22//! Equations may drop subexpressions together with their well-definedness
23//! checks: Prusti checks all side conditions explicitly in impure code.
24

34/// A declaration that `outer(inner(k))` is `k` for every integer literal `k`
35/// in one of the inclusive `ranges` (pairs of literal bounds), such as for the
36/// value and constructor of a primitive snapshot domain, whose axiom only
37/// holds for the values of the type.
38#[derive(Clone, Copy, Debug)]
39pub struct LiteralInverse<'vir> {
40    pub outer: ViperIdent<'vir>,
41    pub inner: ViperIdent<'vir>,
42    pub ranges: &'vir [(ExprInt<'vir>, ExprInt<'vir>)],
43}

45/// What the simplification knows about the program: its adt constructors and
46/// destructors, its total functions and the literal inverses its encoder
47/// declares, used to recognize the foldable applications. The definitions
48/// that the encoder declares user-level, which encode what the user writes,
49/// are left out: folding away an application of one would remove a term
50/// that a trigger the user wrote can match, and the solver finds matches
51/// only among the terms that exist.
52pub struct SimplifyCtx<'vir> {
53    /// Constructor name to its ordered field (destructor) declarations.
54    constructors: FxHashMap<&'vir str, &'vir [crate::LocalDeclDyn<'vir>]>,
55    /// Destructor name to its constructor's name and field index.
56    destructors: FxHashMap<&'vir str, (&'vir str, usize)>,
57    /// The constructors of single-constructor adts.
58    eta: FxHashSet<&'vir str>,
59    /// Names of the total functions that are not user-level: domain functions
60    /// and functions without preconditions (which therefore cannot read the
61    /// heap either).
62    total_functions: FxHashSet<&'vir str>,
63    /// The declared literal inverses, by the name of `outer`.
64    literal_inverses: FxHashMap<&'vir str, LiteralInverse<'vir>>,
65}
66
67impl<'vir> SimplifyCtx<'vir> {
68    pub fn new(
69        adts: &[Adt<'vir>],
70        domains: &[Domain<'vir>],
71        functions: &[Function<'vir>],
72        literal_inverses: &[LiteralInverse<'vir>],
73        user_level: &FxHashSet<&'vir str>,
74    ) -> Self {
75        let foldable = |name: &str| !user_level.contains(name);
76        let total_functions = domains
77            .iter()
78            .filter(|d| foldable(d.name))
79            .flat_map(|d| d.functions.iter().map(|f| f.name.to_str()))
80            .chain(
81                functions
82                    .iter()
83                    .filter(|f| f.pres.is_empty() && foldable(f.name))
84                    .map(|f| f.name),
85            )
86            .collect();
87        let mut constructors = FxHashMap::default();
88        let mut destructors = FxHashMap::default();
89        let mut eta = FxHashSet::default();
90        for adt in adts.iter().filter(|adt| foldable(adt.name)) {
91            let eta_adt = adt.constructors.len() == 1;
92            for cons in adt.constructors {
93                constructors.insert(cons.name, cons.args);
94                for (idx, field) in cons.args.iter().enumerate() {
95                    destructors.insert(field.name, (cons.name, idx));
96                }
97                if eta_adt {
98                    eta.insert(cons.name);
99                }
100            }
101        }
102        Self {
103            constructors,
104            destructors,
105            eta,
106            total_functions,
107            literal_inverses: literal_inverses
108                .iter()
109                .map(|inverse| (inverse.outer.to_str(), *inverse))
110                .collect(),
111        }
112    }
113}

115/// Simplifies the expressions of the items it folds (a program, or its
116/// domains, predicates, functions and methods): hands the roots of their
117/// expression trees (contract clauses, invariants, bodies, axioms and
118/// statement operands) to `Simplifier::root`.
119pub struct Simplify<'a, 'vir, 'tcx>(walk::Simplifier<'a, 'vir, 'tcx>);
120
121impl<'a, 'vir, 'tcx> Simplify<'a, 'vir, 'tcx> {
122    pub fn new(vcx: &'vir VirCtxt<'tcx>, ctx: &'a SimplifyCtx<'vir>) -> Self {
123        Self(walk::Simplifier::new(vcx, ctx))
124    }
125}
126
127impl<'vir> Folder<'vir, (), !> for Simplify<'_, 'vir, '_> {
128    fn fold_expr(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
129        self.0.root(e)
130    }
131
132    fn fold_wand(&mut self, _: Wand<'vir>) -> Option<Wand<'vir>> {
133        None
134    }
