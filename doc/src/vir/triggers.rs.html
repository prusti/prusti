1//! The rules for the terms of a quantifier's trigger, on VIR expressions. A
2//! term is an application of a function, constructor or destructor, a field
3//! access or a collection operation at its top, and contains none of Viper's
4//! own operators, conditionals or ADT discriminators: these are Viper's rules,
5//! which Viper checks only when it parses a program from text. A term also
6//! contains no quantifier, which Viper allows but a pattern of the SMT solver
7//! cannot contain.
8//!
9//! The encoder's triggers contain `let`s, which [`InlineTriggerLets`]
10//! inlines in the whole program before the simplifier runs: Silicon looks a
11//! trigger up among the terms of the quantifier's body, and evaluates it with
12//! the body's path conditions, only if it has no `let`s. The rules apply to
13//! the terms with the `let`s inlined.
14

21/// `e` with its `let`s inlined like a trigger term (see
22/// [`InlineTriggerLets`]).
23#[cfg(test)]
24pub(crate) fn inline_lets<'vir>(vcx: &'vir VirCtxt<'_>, e: ExprDyn<'vir>) -> ExprDyn<'vir> {
25    let mut inline = Inline {
26        vcx,
27        bindings: Vec::new(),
28    };
29    e.fold_with(&mut inline).unwrap_or(e)
30}

32/// Folds an item into the item with the values of the `let`s in its triggers
33/// in place of their locals, and without the `let`s. Under an `old(..)`, only
34/// values that are locals or constants are inlined, and the `let` of a local
35/// still used there stays. The encoder gives every local it binds a fresh
36/// name, so a value never ends up under a `let` that rebinds one of its own
37/// locals.
38pub struct InlineTriggerLets<'vir, 'tcx> {
39    vcx: &'vir VirCtxt<'tcx>,
40}
41
42impl<'vir, 'tcx> InlineTriggerLets<'vir, 'tcx> {
43    pub fn new(vcx: &'vir VirCtxt<'tcx>) -> Self {
44        Self { vcx }
45    }
46}

50        t.super_fold_with(&mut Inline {
51            vcx: self.vcx,
52            bindings: Vec::new(),
53        })
54    }

57/// Replaces the locals that the `let`s of a term bind by their values (see
58/// [`InlineTriggerLets`]).
59struct Inline<'vir, 'tcx> {
60    vcx: &'vir VirCtxt<'tcx>,
61    bindings: Vec<(&'vir str, ExprDyn<'vir>)>,
62}
63
64impl<'vir> Folder<'vir, (), !> for Inline<'vir, '_> {
65    fn fold_expr(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
66        match e.kind {
67            ExprKindGenData::Let(l) => {
68                let folded_val = l.val.fold_with(self);
69                let val = folded_val.unwrap_or(l.val);
70                self.bindings.push((l.name, val));
71                let folded_body = l.expr.fold_with(self);
72                let body = folded_body.unwrap_or(l.expr);
73                self.bindings.pop();
74                let mut locals = FxHashSet::default();
75                collect_locals(body, &mut locals);
76                let used = locals.contains(l.name);
77                if used && folded_val.is_none() && folded_body.is_none() {
78                    return None;
79                }
80                // The `let` stays if its local is still used, under an
81                // `old(..)`; Silicon then evaluates the trigger on its own
82                // instead of finding it in the body. Otherwise the body takes
83                // its place, and keeps the `let`'s span: the verifier's
84                // warnings about a trigger point at the span of its root.
85                let (kind, debug_info) = if used {
86                    let l = LetGenData {
87                        name: l.name,
88                        val,
89                        expr: body,
90                    };
91                    let kind = ExprKindGenData::Let(self.vcx.alloc(l));
92                    (self.vcx.alloc(kind), e.debug_info)
93                } else {
94                    (body.kind, body.debug_info)
95                };
96                Some(
97                    self.vcx
98                        .alloc(ExprGenData::new_inner(kind, debug_info, e.span, body.ty())),
99                )
100            }
101            // A local keeps its value under `old(..)`, but a heap read in the
102            // value would read the old heap there. As in the simplifier, only
103            // the values that are locals or constants are inlined under it.
104            ExprKindGenData::Old(_) => {
105                let outer = self.bindings.clone();
106                self.bindings.retain(|b| b.1.is_local_or_const());
107                let folded = e.super_fold_with(self);
108                self.bindings = outer;
109                folded
110            }
111            ExprKindGenData::Local(local) => self
112                .bindings
113                .iter()
114                .rev()
115                .find(|b| b.0 == local.name)
116                .map(|b| b.1),
117            _ => e.super_fold_with(self),
118        }
119    }
120}

122/// Whether `terms` are a valid trigger set of a quantifier over `qvars`: each
123/// is a valid trigger term that mentions a quantified variable, and together
124/// they mention all of them.
125pub(crate) fn is_trigger_set<'vir>(qvars: &[LocalDeclDyn<'vir>], terms: &[ExprDyn<'vir>]) -> bool {
126    let shapes = terms
127        .iter()
128        .map(|&term| trigger_term_shape(term))
129        .collect::<Vec<_>>();
130    let mentions =
131        |shape: &TriggerTermShape<'_>, qvar: &LocalDeclDyn<'_>| shape.locals.contains(qvar.name);
132    shapes.iter().all(|shape| {
133        !shape.forbidden && shape.valid_head && qvars.iter().any(|q| mentions(shape, q))
134    }) && qvars
135        .iter()
136        .all(|q| shapes.iter().any(|shape| mentions(shape, q)))
137}

139/// A trigger term with its `let`s inlined (see [`InlineTriggerLets`]), as
140/// the rules see it.
141pub(crate) struct TriggerTermShape<'vir> {
142    /// Whether the term is an application or an access at its top, possibly
143    /// in `old(..)`s and in the `let`s that the inlining keeps.
144    pub(crate) valid_head: bool,
145    /// Whether the term contains a Viper operator, an ADT discriminator, a
146    /// conditional or a quantifier, none of which a trigger term may contain
147    /// anywhere.
148    pub(crate) forbidden: bool,
149    /// The locals the term mentions.
150    pub(crate) locals: FxHashSet<&'vir str>,
151}
152
153/// The shape of the trigger term `e` (see [`TriggerTermShape`]), whose `let`s
154/// are expected to be inlined (see [`InlineTriggerLets`]).
155pub(crate) fn trigger_term_shape<'vir>(e: ExprDyn<'vir>) -> TriggerTermShape<'vir> {
156    let mut head = e;
157    loop {
158        head = match head.kind {
159            ExprKindGenData::Old(old) => old.expr,
160            ExprKindGenData::Let(l) => l.expr,
161            _ => break,
162        };
163    }
164    let valid_head = matches!(
165        head.kind,
166        ExprKindGenData::FuncApp(_)
167            | ExprKindGenData::AdtDestructor(..)
168            | ExprKindGenData::Field(..)
169            | ExprKindGenData::CollectionBinOp(_)
170            | ExprKindGenData::CollectionLiteral(_)
171            | ExprKindGenData::CollectionUpdate(_)
172            | ExprKindGenData::CollectionLen(_)
173            | ExprKindGenData::MapDomain(_)
174            | ExprKindGenData::MapRange(_)
175    );
176    let mut locals = FxHashSet::default();
177    collect_locals(e, &mut locals);
178    TriggerTermShape {
179        valid_head,
180        forbidden: has_forbidden_trigger_node(e),
181        locals,
182    }
183}

185/// Whether `e` contains a Viper operator, an ADT discriminator, a
186/// conditional or a quantifier, none of which a trigger term may contain
187/// anywhere. The `Lazy` nodes of an expression that is not yet instantiated
188/// are not looked into.
189pub fn has_forbidden_trigger_node<'vir, Curr, Next>(e: ExprGenDyn<'vir, Curr, Next>) -> bool {
190    let mut forbidden = Forbidden(false);
191    e.visit_with(&mut forbidden);
192    forbidden.0
193}
194
195/// Finds a node that no trigger term may contain.
196struct Forbidden(bool);
197
198impl<'vir, Curr, Next> Visitor<'vir, Curr, Next> for Forbidden {
199    fn visit_expr(&mut self, e: ExprGenDyn<'vir, Curr, Next>) {
200        // A negative integer literal is encoded as the negation of a
201        // constant, which the SMT solver reads as a literal.
202        let negative_literal = matches!(
203            e.kind,
204            ExprKindGenData::UnOp(op)
205                if op.kind == crate::UnOpKind::Neg
206                    && matches!(op.expr.kind, ExprKindGenData::Const(_))
207        );
208        self.0 |= !negative_literal
209            && matches!(
210                e.kind,
211                ExprKindGenData::UnOp(_)
212                    | ExprKindGenData::BinOp(_)
213                    | ExprKindGenData::AdtDiscriminator(..)
214                    | ExprKindGenData::Ternary(_)
215                    | ExprKindGenData::Forall(_)
216                    | ExprKindGenData::Exists(_)
217            );
218        e.super_visit_with(self);
219    }
220}
