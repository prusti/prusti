20impl<'vir> ViperIdent<'vir> {
21    pub fn new(ident: &'vir str) -> ViperIdent<'vir> {
22        assert!(is_valid_identifier(ident));
23        ViperIdent(ident)
24    }

26    pub fn sanitize(vcx: &'vir VirCtxt<'_>, ident: &str) -> ViperIdent<'vir> {
27        let ident = sanitize_str(ident);
28        // Just a sanity check, if this fails there is a problem in `sanitize`
29        assert!(is_valid_identifier(ident.as_str()));
30        ViperIdent(vcx.alloc_str(&ident))
31    }

33    /// The identifier fragment naming the Rust item `def_id`. Callers combine
34    /// this with a role-specific prefix, e.g.
35    /// `vir_format_identifier!(vcx, "m_{}", ViperIdent::from_def_id(vcx, def_id))`.
36    pub fn from_def_id(vcx: &'vir VirCtxt<'_>, def_id: DefId) -> ViperIdent<'vir> {
37        let name = match vcx.ident_style {
38            IdentStyle::DefPath => def_path_name(vcx, def_id),
39            IdentStyle::ItemName => short_name(vcx, def_id),
40        };
41        Self::sanitize(vcx, &name)
42    }
43
44    pub fn to_str(&self) -> &'vir str {
45        self.0
46    }
47}

49/// The def path of `def_id`, unique per item. `def_path_str` alone is not:
50/// it omits the disambiguators that tell apart same-named items, e.g. types
51/// defined in two `const _: () = { .. };` blocks of one module (as
52/// `bitflags!` generates), which would then share one Viper identity. Those
53/// disambiguators are appended (closures and impls already print theirs, or
54/// are told apart by what they print). It also omits the name of the local
55/// crate, which is prepended so that a local item's name always has a path
56/// separator and so cannot coincide with a name that is not a def path, such
57/// as a built-in type's `Array`.
58///
59/// Crate names are not unique either: two versions of one dependency share a
60/// name, as may a dependency and the local crate. An item of such a crate gets
61/// the crate's `StableCrateId` appended. A crate name may also start with a
62/// digit, which is escaped since some Viper identifiers start with the name.
63fn def_path_name(vcx: &VirCtxt<'_>, def_id: DefId) -> String {
64    let mut name = vcx.tcx().def_path_str(def_id);
65    if def_id.is_local() {
66        name = format!("{}::{name}", vcx.tcx().crate_name(def_id.krate));
67    }
68    for (idx, component) in vcx.tcx().def_path(def_id).data.iter().enumerate() {
69        if component.disambiguator != 0
70            && matches!(
71                component.data,
72                DefPathData::TypeNs(_) | DefPathData::ValueNs(_) | DefPathData::MacroNs(_)
73            )
74        {
75            name.push_str(&format!("#{idx}_{}", component.disambiguator));
76        }
77    }
78    if crate_name_is_shared(vcx, def_id.krate) {
79        let stable_id = vcx.tcx().stable_crate_id(def_id.krate).as_u64();
80        name.push_str(&format!("#crate_{stable_id:x}"));
81    }
82    if name.starts_with(|c: char| c.is_ascii_digit()) {
83        name.insert(0, '$');
84    }
85    name
86}

88/// Whether another crate, local or external, has the same name as `krate`.
89fn crate_name_is_shared(vcx: &VirCtxt<'_>, krate: CrateNum) -> bool {
90    let name = vcx.tcx().crate_name(krate);
91    std::iter::once(LOCAL_CRATE)
92        .chain(vcx.tcx().crates(()).iter().copied())
93        .any(|other| other != krate && vcx.tcx().crate_name(other) == name)
94}

96/// Asking for the `item_name` of a closure triggers an ICE in the compiler, so
97/// a closure is named after its nearest non-closure ancestor (closures can
98/// nest, e.g. a quantifier's closure inside an assertion's closure).
99fn short_name(vcx: &VirCtxt<'_>, mut def_id: DefId) -> String {
100    let mut key = vcx.tcx().def_key(def_id);
101    let mut suffix = String::new();
102    while let DefPathData::Closure = key.disambiguated_data.data {
103        suffix = format!("_Closure_{}{suffix}", key.disambiguated_data.disambiguator);
104        def_id.index = key.parent.unwrap();
105        key = vcx.tcx().def_key(def_id);
106    }
107    format!("{}{suffix}", vcx.tcx().item_name(def_id).to_ident_string())
108}

110fn sanitize_char(c: char) -> Option<String> {
111    match c {
112        '<' => Some("$lt$".to_string()),
113        '>' => Some("$gt$".to_string()),
114        ' ' => Some("$sp$".to_string()),
115        ',' => Some("$com$".to_string()),
116        ':' => Some("$col$".to_string()),
117        '\'' => Some("$sq$".to_string()),
118        '&' => Some("$amp$".to_string()),
119        '-' => Some("$hyp$".to_string()),
120        '(' => Some("$lp$".to_string()),
121        ')' => Some("$rp$".to_string()),
122        '[' => Some("$lb$".to_string()),
123        ']' => Some("$rb$".to_string()),
124        '{' => Some("$lc$".to_string()),
125        '}' => Some("$rc$".to_string()),
126        '?' => Some("$qm$".to_string()),
127        ';' => Some("$sc$".to_string()),
128        '#' => Some("$oc$".to_string()),
129        '/' => Some("$fs$".to_string()),
130        '*' => Some("$as$".to_string()),
131        '=' => Some("$eq$".to_string()),
132        '+' => Some("$pl$".to_string()),
133        '!' => Some("$ex$".to_string()),
134        _ => None,
135    }
136}
137
138fn sanitize_str(s: &str) -> String {
139    s.chars()
140        .map(|c| sanitize_char(c).unwrap_or_else(|| c.to_string()))
141        .collect()
142}
143
144fn is_valid_identifier(s: &str) -> bool {
145    s.chars().all(|c| sanitize_char(c).is_none())
146}

152    /// Call sites compose sanitized fragments into larger identifiers with

156        let path = "<core::ops::Range<T> as Foo>::bar::{closure#0}";
157        let once = sanitize_str(path);
158        assert!(is_valid_identifier(&once));
159        assert_eq!(sanitize_str(&once), once);
