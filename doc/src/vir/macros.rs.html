261/// The operand types of `+`, `-`, `*` and unary `-` in [`expr!`]: `Int`, or
262/// `Perm` for arithmetic on reals.
263pub trait Arith: CompType {
264    const ADD: crate::BinOpKind;
265    const SUB: crate::BinOpKind;
266    const MUL: crate::BinOpKind;
267    const NEG: crate::UnOpKind;
268}
269impl Arith for crate::Int {
270    const ADD: crate::BinOpKind = crate::BinOpKind::Add;
271    const SUB: crate::BinOpKind = crate::BinOpKind::Sub;
272    const MUL: crate::BinOpKind = crate::BinOpKind::Mul;
273    const NEG: crate::UnOpKind = crate::UnOpKind::Neg;
274}
275impl Arith for crate::Perm {
276    const ADD: crate::BinOpKind = crate::BinOpKind::PermAdd;
277    const SUB: crate::BinOpKind = crate::BinOpKind::PermSub;
278    const MUL: crate::BinOpKind = crate::BinOpKind::PermMul;
279    const NEG: crate::UnOpKind = crate::UnOpKind::PermNeg;
280}
281
282/// A binary arithmetic operator of [`expr!`].
283pub enum ArithOp {
284    Add,
285    Sub,
286    Mul,
287}
288
289/// `lhs op rhs`, as built by the binary arithmetic operators of [`expr!`].
290pub fn arith<'vir, T: Arith>(
291    vcx: &'vir crate::VirCtxt,
292    op: ArithOp,
293    lhs: crate::Expr<'vir, T>,
294    rhs: crate::Expr<'vir, T>,
295) -> crate::Expr<'vir, T>
296where
297    crate::Prim: crate::TransmuteFrom<T>,
298{
299    let kind = match op {
300        ArithOp::Add => T::ADD,
301        ArithOp::Sub => T::SUB,
302        ArithOp::Mul => T::MUL,
303    };
304    crate::CastType::downcast_ty(vcx.mk_bin_op_expr(kind, lhs, rhs))
305}
306
307/// `-e`, as built by the unary `-` of [`expr!`].
308pub fn neg<'vir, T: Arith>(
309    vcx: &'vir crate::VirCtxt,
310    e: crate::Expr<'vir, T>,
311) -> crate::Expr<'vir, T>
312where
313    crate::Prim: crate::TransmuteFrom<T>,
314{
315    crate::CastType::downcast_ty(vcx.mk_unary_op_expr(T::NEG, crate::CastType::upcast_ty(e)))
316}

319macro_rules! expr {
320    ($vcx:tt; $($tokens:tt)*) => { {
321        use $crate::macros::{ExprApply, ExprQuote};
322        let vcx = $vcx; #[allow(unused)] macro_rules! vcx { () => { vcx }; }
323        $crate::expr_inner!(@expr_one; $($tokens)*)
324    } };
325    ($($tokens:tt)+) => { vir::with_vcx(|vcx| $crate::expr!(vcx; $($tokens)*)) };
326    () => { compile_error!(concat!("VIR malformed empty")) };
327}

330macro_rules! expr_inner {
331    (@expr_one; ($($e:tt)+) as $ty:ident) => {
332        $crate::CastType::inner_cast_ty::<$crate::$ty>($crate::expr_inner!(@expr_one; $($e)*))
333    };
334    (@expr_one; unfolding ( [ $outer:expr ] ( $($args:tt)* ) ) in ( $($rhs:tt)+ ) ) => { vcx!().mk_unfolding_expr(
335        $outer.call_once($crate::expr_inner!(@expr_args; $($args)*))(None),
336        $crate::expr_inner!(@expr_one; $($rhs)*),
337    ) };
338    (@expr_one; acc( [ $outer:expr ] ( $($args:tt)* ) ) ) => { {
339        let pred: $crate::PredicateIdn<_> = $outer;
340        pred.expr_apply(
341            vcx!(),
342            $crate::expr_inner!(@expr_args; $($args)*),
343        )
344    } };
345    (@expr_one; acc( ( $($args:tt)+ ).[ $outer:expr ] ) ) => { vcx!().mk_acc_field_expr(
346        $crate::expr_inner!(@expr_one; $($args)*),
347        $outer,
348        None,
349    ) };
350    (@expr_one; [ $outer:expr ] ( $($args:tt)* ) ) => { $outer.expr_apply(
351        vcx!(),
352        $crate::expr_inner!(@expr_args; $($args)*),
353    ) };
354    (@expr_one; [ $outer:expr ] ) => { $outer.expr(vcx!()) };
355    (@expr_one; ( $($lhs:tt)+ ) == ( $($rhs:tt)+ )) => { vcx!().mk_eq_expr(
356        $crate::expr_inner!(@expr_one; $($lhs)*),
357        $crate::expr_inner!(@expr_one; $($rhs)*),
358    ) };
359    (@expr_one; ( $($cond:tt)+ ) ? ( $($e_then:tt)+ ) : ( $($e_else:tt)+ )) => { vcx!().mk_ternary_expr(
360        $crate::expr_inner!(@expr_one; $($cond)*),
361        $crate::expr_inner!(@expr_one; $($e_then)*),
362        $crate::expr_inner!(@expr_one; $($e_else)*),
363    ) };
364    (@expr_one; ( $($lhs:tt)+ ) && ( $($rhs:tt)+ )) => { vcx!().mk_conj(&[
365        $crate::expr_inner!(@expr_one; $($lhs)*),
366        $crate::expr_inner!(@expr_one; $($rhs)*),
367    ]) };
368    (@expr_one; ( $($lhs:tt)+ ) || ( $($rhs:tt)+ )) => { vcx!().mk_disj(&[
369        $crate::expr_inner!(@expr_one; $($lhs)*),
370        $crate::expr_inner!(@expr_one; $($rhs)*),
371    ]) };
372    (@expr_one; ( $($lhs:tt)+ ) == > ( $($rhs:tt)+ )) => { $crate::CastType::downcast_ty::<$crate::Bool>(vcx!().mk_bin_op_expr(
373        $crate::BinOpKind::Implies,
374        $crate::expr_inner!(@expr_one; $($lhs)*),
375        $crate::expr_inner!(@expr_one; $($rhs)*),
376    )) };
377    (@expr_one; ( $($lhs:tt)+ ) <= ( $($rhs:tt)+ )) => { $crate::CastType::downcast_ty::<$crate::Bool>(vcx!().mk_bin_op_expr(
378        $crate::BinOpKind::CmpLe,
379        $crate::expr_inner!(@expr_one; $($lhs)*),
380        $crate::expr_inner!(@expr_one; $($rhs)*),
381    )) };
382    (@expr_one; ( $($lhs:tt)+ ) < ( $($rhs:tt)+ )) => { $crate::CastType::downcast_ty::<$crate::Bool>(vcx!().mk_bin_op_expr(
383        $crate::BinOpKind::CmpLt,
384        $crate::expr_inner!(@expr_one; $($lhs)*),
385        $crate::expr_inner!(@expr_one; $($rhs)*),
386    )) };
387    (@expr_one; ( $($lhs:tt)+ ) >= ( $($rhs:tt)+ )) => { $crate::CastType::downcast_ty::<$crate::Bool>(vcx!().mk_bin_op_expr(
388        $crate::BinOpKind::CmpGe,
389        $crate::expr_inner!(@expr_one; $($lhs)*),
390        $crate::expr_inner!(@expr_one; $($rhs)*),
391    )) };
392    (@expr_one; ( $($lhs:tt)+ ) > ( $($rhs:tt)+ )) => { $crate::CastType::downcast_ty::<$crate::Bool>(vcx!().mk_bin_op_expr(
393        $crate::BinOpKind::CmpGt,
394        $crate::expr_inner!(@expr_one; $($lhs)*),
395        $crate::expr_inner!(@expr_one; $($rhs)*),
396    )) };
397    (@expr_one; ( $($lhs:tt)+ ) + ( $($rhs:tt)+ )) => { $crate::macros::arith(
398        vcx!(),

401        $crate::expr_inner!(@expr_one; $($rhs)*),
402    ) };
403    (@expr_one; ( $($lhs:tt)+ ) - ( $($rhs:tt)+ )) => { $crate::macros::arith(

407        $crate::expr_inner!(@expr_one; $($rhs)*),
408    ) };

412        $crate::expr_inner!(@expr_one; $($lhs)*),
413        $crate::expr_inner!(@expr_one; $($rhs)*),
414    ) };

418    ) };
419    (@expr_one; ( $($lhs:tt)+ ) in ( $($rhs:tt)+ )) => { vcx!().mk_contains_expr(
420        $crate::expr_inner!(@expr_one; $($lhs)*),
421        $crate::expr_inner!(@expr_one; $($rhs)*),
422    ) };
423    (@expr_one; ( $($lhs:tt)+ ) setminus ( $($rhs:tt)+ )) => { vcx!().mk_set_difference_expr(

425        $crate::expr_inner!(@expr_one; $($rhs)*),
426    ) };

430    ) };

433            $crate::UnOpKind::Not,

444    ) };

447    ) };

449    (@expr_one; true) => { vcx!().mk_bool::<true>() };
450    (@expr_one; false) => { vcx!().mk_bool::<false>() };

459

463    (@expr_tuple; ..[ $args:expr ]) => {

465    };

470    };

479    };
480    (@expr_args; $one:tt , $two:tt , $three:tt , $four:tt , $($tail:tt)+ ) => {
481        ($crate::expr_inner!(@expr_tuple; $one), $crate::expr_inner!(@expr_tuple; $two), $crate::expr_inner!(@expr_tuple; $three), $crate::expr_inner!(@expr_tuple; $four), $crate::expr_inner!(@expr_tuple; $($tail)*),)
482    };
483    (@expr_args; $one:tt , $two:tt , $three:tt , $($tail:tt)+ ) => {
484        ($crate::expr_inner!(@expr_tuple; $one), $crate::expr_inner!(@expr_tuple; $two), $crate::expr_inner!(@expr_tuple; $three), $crate::expr_inner!(@expr_tuple; $($tail)*),)
485    };
486    (@expr_args; $one:tt , $two:tt , $($tail:tt)+ ) => {
487        ($crate::expr_inner!(@expr_tuple; $one), $crate::expr_inner!(@expr_tuple; $two), $crate::expr_inner!(@expr_tuple; $($tail)*),)
488    };
489    (@expr_args; $one:tt , $($tail:tt)+ ) => {
490        ($crate::expr_inner!(@expr_tuple; $one), $crate::expr_inner!(@expr_tuple; $($tail)*),)
491    };
492    (@expr_args; $($tail:tt)+ ) => {
493        ($crate::expr_inner!(@expr_tuple; $($tail)*),)
494    };
495    (@expr_args; ) => { () };
496    (@expr_args; $tokens:tt*) => {
497        compile_error!(concat!("VIR malformed arg: `" , stringify!($($tokens)*), "`. Expected `ident/(...)` for single arg or `[...]` for many args"))
498    };
499
500    (@expr_list; ) => { Vec::<$crate::Expr<_>>::new().as_slice() };
501    (@expr_list; $($args:tt)+ ) => {
502        $crate::expr_inner!(@expr_iter; , $($args)*).collect::<Vec<$crate::Expr<_>>>().as_slice()
503    };
504
505    (@expr_iter; ) => { [].into_iter() };
506    (@expr_iter; , ( $($first:tt)+ ) $($rest:tt)*) => { [$crate::expr_inner!(@expr_one; $($first)*)].into_iter().chain($crate::expr_inner!(@expr_iter; $($rest)*)) };
507    (@expr_iter; ,   $ident:ident    $($rest:tt)*) => { [$crate::expr_inner!(@expr_one; $ident)].into_iter().chain($crate::expr_inner!(@expr_iter; $($rest)*)) };
508    (@expr_iter; , ..[ $outer:expr ] ) => { $outer.iter().map(|e| e.expr(vcx!())) };
509    (@expr_iter; , $($last:tt)+) => { [$crate::expr_inner!(@expr_one; $($last)*)].into_iter() };
510    (@expr_iter; $($tokens:tt)+) => { compile_error!(concat!("VIR malformed expression list: `" , stringify!($($tokens)*), "`")) };
511
512    // The trigger sets are collected one `{ .. }` group at a time, since a
513    // repetition of them directly followed by the body is ambiguous. A body
514    // never starts with `{`.
515    (@forall_qvars($qvars:ident); :: $($tokens:tt)+) => {
516        $crate::expr_inner!(@forall_triggers($qvars;); $($tokens)*)
517    };
518    (@forall_qvars($qvars:ident); :: $($tokens:tt)*) => { compile_error!(concat!("VIR missing triggers or body: `" , stringify!($($tokens)*), "`")) };
519    (@forall_triggers($qvars:ident; $($done:tt)*); { $($trigger:tt)+ } $($tokens:tt)+) => {
520        $crate::expr_inner!(@forall_triggers($qvars; $($done)* { $($trigger)* }); $($tokens)*)
521    };
522    (@forall_triggers($qvars:ident; $({ $($triggers:tt)+ })+); $($tokens:tt)+) => { vcx!().mk_forall_expr(
523        vcx!().alloc_slice($qvars.as_slice()),
524        vcx!().alloc_slice(
525            [$($crate::expr_inner!(@expr_list; $($triggers)*)),*].into_iter()
526                .map(|e: &[$crate::Expr<_>]| vcx!().mk_trigger(e))
527                .collect::<Vec<_>>().as_slice()
528        ),
529        $crate::expr_inner!(@expr_one; $($tokens)*),
530    ) };
531    (@forall_triggers($qvars:ident; $($done:tt)*); $($tokens:tt)*) => { compile_error!(concat!("VIR missing triggers or body: `" , stringify!($($tokens)*), "`")) };
532
533    (@forall_qvars($qvars:ident); , ..[$outer_decls:expr] $($tokens:tt)*) => { {
534        $qvars.extend($outer_decls.iter().map(|local| $crate::CastType::as_dyn(local.clone())));
535        $crate::expr_inner!(@forall_qvars($qvars); $($tokens)*)
536    } };
537    (@forall_qvars($qvars:ident); , [$outer_decl:expr] $($tokens:tt)*) => { {
538        $qvars.push($crate::CastType::as_dyn($outer_decl));
539        $crate::expr_inner!(@forall_qvars($qvars); $($tokens)*)
540    } };
541    (@forall_qvars($qvars:ident); , $qvar:ident : $qtype:tt $($tokens:tt)* ) => { {
542        let local = vcx!().mk_local_decl(stringify!($qvar), $crate::vir_type!(vcx!(); $qtype));
543        $qvars.push($crate::CastType::as_dyn(local));
544        let $qvar: $crate::Expr<_> = vcx!().mk_local_ex(local);
545        $crate::expr_inner!(@forall_qvars($qvars); $($tokens)*)
546    } };
547    (@forall_qvars($qvars:ident); $($tokens:tt)*) => { compile_error!(concat!("VIR malformed quantifier: `" , stringify!($($tokens)*), "`")) };
548}
