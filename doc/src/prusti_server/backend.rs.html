10pub enum Backend<'a> {
11    Viper(
12        viper::Verifier<'a>,
13        &'a VerificationContext<'a>,
14        jni::objects::GlobalRef,
15    ),
16}
17
18impl<'a> Backend<'a> {
19    /// Verifies the program, and returns the result together with the
20    /// warnings the verifier reported.
21    pub fn verify(
22        &mut self,
23        procedures: prusti_rustc_interface::data_structures::fx::FxHashSet<String>,
24        sender: mpsc::Sender<ServerMessage>,
25    ) -> (VerificationResultKind, Vec<VerifierWarning>) {
26        // Convert FxHashSet to std HashSet for viper compatibility
27        #[allow(clippy::disallowed_types)]
28        let procedures_hashset: std::collections::HashSet<String> =
29            procedures.iter().cloned().collect();
30
31        match self {
32            Backend::Viper(ref mut verifier, context, viper_program_ref) => {
33                let ast_utils = context.new_ast_utils();
34
35                ast_utils.with_local_frame(16, || {
36                    let viper_program = viper::Program::new(viper_program_ref.as_obj());
37
38                    // If report_viper_messages is activated, we need to check if we only run consistency checks here.
39                    // Otherwise, we will keep waiting for the messages to come in
40                    if prusti_utils::config::report_viper_messages() && verifier.is_verify() {
41                        verify_and_poll_msgs(
42                            verifier,
43                            context,
44                            viper_program,
45                            procedures_hashset,
46                            sender,
47                        )
48                    } else {
49                        let kind = verifier.verify(viper_program, None);
50                        (kind, verifier.take_warnings())
51                    }
52                })
53            }
54        }
55    }
56}

58#[allow(clippy::disallowed_types)]
59fn verify_and_poll_msgs(
60    verifier: &mut viper::Verifier<'_>,
61    verification_context: &viper::VerificationContext,
62    viper_program: viper::Program,
63    procedures: std::collections::HashSet<String>,
64    sender: mpsc::Sender<ServerMessage>,
65) -> (VerificationResultKind, Vec<VerifierWarning>) {
66    let mut kind = VerificationResultKind::Success;
67    let mut warnings = vec![];
68
69    // get the reporter global reference outside of the thread scope because it needs to
70    // be dropped by thread attached to the jvm. This is also why we pass it as reference
71    // and not per value
72    let env = &verification_context.env();
73    let jni = JniUtils::new(env);
74    let verifier_wrapper = silver::verifier::Verifier::with(env);
75    let reporter = jni.unwrap_result(verifier_wrapper.call_reporter(*verifier.verifier_instance()));
76    let rep_glob_ref = env.new_global_ref(reporter).unwrap();
77
78    debug!("Starting viper message polling thread");
79
80    // start thread for polling messages
81    let (stop, stopped) = mpsc::channel::<()>();
82    thread::scope(|scope| {
83        let polling_thread = scope
84            .spawn(|| polling_function(&rep_glob_ref, procedures, sender, stopped, &mut warnings));
85        kind = verifier.verify(viper_program, Some(polling_thread));
86        drop(stop);
87    });
88    debug!("Viper message polling thread terminated");
89    // A run that times out stops the polling thread without waiting for the
90    // termination message, so warnings can still be queued.
91    warnings.extend(verifier.take_warnings());
92    (kind, warnings)
93}

95#[allow(clippy::disallowed_types)]
96fn polling_function(
97    rep_glob_ref: &jni::objects::GlobalRef,
98    procedures: std::collections::HashSet<String>,
99    sender: mpsc::Sender<ServerMessage>,
100    stopped: mpsc::Receiver<()>,
101    warnings: &mut Vec<VerifierWarning>,
102) -> std::collections::HashSet<u64> {
103    debug!("attach polling thread to JVM.");
104    let verification_context = VIPER
105        .get()
106        .expect("Viper was not instantiated before polling")
107        .attach_current_thread();
108    let env = verification_context.env();
109    let jni = JniUtils::new(env);
110    let reporter_instance = rep_glob_ref.as_obj();
111    let reporter_wrapper = silver::reporter::PollingReporter::with(env);
112
113    let mut error_hashes: std::collections::HashSet<u64> = std::collections::HashSet::new();
114    loop {
115        while reporter_wrapper
116            .call_hasNewMessage(reporter_instance)
117            .unwrap()
118        {
119            let msg = reporter_wrapper
120                .call_getNewMessage(reporter_instance)
121                .unwrap();
122            debug!("Polling thread received {}", jni.class_name(msg).as_str());
123            match jni.class_name(msg).as_str() {
124                "viper.silver.reporter.QuantifierInstantiationsMessage" => {
125                    let msg_wrapper = silver::reporter::QuantifierInstantiationsMessage::with(env);
126                    let q_name =
127                        jni.get_string(jni.unwrap_result(msg_wrapper.call_quantifier(msg)));
128                    let q_inst = jni.unwrap_result(msg_wrapper.call_instantiations(msg));
129                    // debug!("QuantifierInstantiationsMessage: {} {}", q_name, q_inst);
130                    // also matches the "-aux" and "_precondition" quantifiers generated
131                    // we encoded the position id in the line and column number since this is not used by
132                    // prusti either way
133                    if q_name.starts_with("prog.l") {
134                        let no_pref = q_name.strip_prefix("prog.l").unwrap();
135                        let stripped = no_pref
136                            .strip_suffix("-aux")
137                            .or(no_pref.strip_suffix("_precondition"))
138                            .unwrap_or(no_pref);
139                        let parsed = stripped.parse::<usize>();
140                        match parsed {
141                            Ok(pos_id) => {
142                                sender
143                                    .send(ServerMessage::QuantifierInstantiation {
144                                        q_name,
145                                        insts: u64::try_from(q_inst).unwrap(),
146                                        pos_id,
147                                    })
148                                    .unwrap();
149                            }
150                            _ => info!("Unexpected quantifier name {q_name}"),
151                        }
152                    }
153                }
154                "viper.silver.reporter.QuantifierChosenTriggersMessage" => {
155                    let obj_wrapper = java::lang::Object::with(env);
156                    let positioned_wrapper = silver::ast::Positioned::with(env);
157                    let msg_wrapper = silver::reporter::QuantifierChosenTriggersMessage::with(env);
158
159                    let viper_quant = jni.unwrap_result(msg_wrapper.call_quantifier(msg));
160                    let viper_quant_str =
161                        jni.get_string(jni.unwrap_result(obj_wrapper.call_toString(viper_quant)));
162                    // we encoded the position id in the line and column number since this is not used by
163                    // prusti either way
164                    let pos = jni.unwrap_result(positioned_wrapper.call_pos(viper_quant));
165                    let pos_string =
166                        jni.get_string(jni.unwrap_result(obj_wrapper.call_toString(pos)));
167                    if let Some(pos_id_index) = pos_string.rfind('.') {
168                        let pos_id = pos_string[pos_id_index + 1..].parse::<usize>().unwrap();
169
170                        let viper_triggers = jni
171                            .get_string(jni.unwrap_result(msg_wrapper.call_triggers__string(msg)));
172                        debug!(
173                            "QuantifierChosenTriggersMessage: {viper_quant_str} {viper_triggers} {pos_id}"
174                        );
175                        sender
176                            .send(ServerMessage::QuantifierChosenTriggers {
177                                viper_quant: viper_quant_str,
178                                triggers: viper_triggers,
179                                pos_id,
180                            })
181                            .unwrap();
182                    } else {
183                        debug!("quantifier chosen trigger for {viper_quant_str} had no position.");
184                    }
185                }
186                "viper.silver.reporter.WarningsDuringVerification" => {
187                    warnings.extend(viper::extract_warnings(&jni, env, msg))
188                }
189                "viper.silver.reporter.VerificationTerminationMessage" => return error_hashes,
190                "viper.silver.reporter.EntitySuccessMessage" => {
191                    let msg_wrapper = silver::reporter::EntitySuccessMessage::with(env);
192                    let concerning = jni.unwrap_result(msg_wrapper.call_concerning(msg));
193                    if jni.is_instance_of(concerning, "viper/silver/ast/Method") {
194                        let method_wrapper = silver::ast::Method::with(env);
195                        let method_name =
196                            jni.get_string(jni.unwrap_result(method_wrapper.call_name(concerning)));
197                        debug!("Entity success for method: {method_name}");
198                        // this should only match local methods
199                        if procedures.contains(&method_name) {
200                            let verification_time =
201                                jni.unwrap_result(msg_wrapper.call_verificationTime(msg));
202                            // verification_time is a long -> i64. but we are using u128
203                            if verification_time >= 0 {
204                                let verification_time_u128 = verification_time as u64 as u128;
205                                sender
206                                    .send(ServerMessage::MethodTermination {
207                                        viper_method_name: method_name.to_string(),
208                                        result: VerificationResultKind::Success,
209                                        verification_time: verification_time_u128,
210                                    })
211                                    .unwrap();
212                            } else {
213                                debug!(
214                                      "EntitySuccessMessage for {method_name} had negative verification time {verification_time}"
215                                  );
216                            }
217                        }
218                    } else {
219                        debug!("Entity is not a method");
220                    }
221                }
222                "viper.silver.reporter.EntityFailureMessage" => {
223                    let msg_wrapper = silver::reporter::EntityFailureMessage::with(env);
224                    let concerning = jni.unwrap_result(msg_wrapper.call_concerning(msg));
225                    if jni.is_instance_of(concerning, "viper/silver/ast/Method") {
226                        let method_wrapper = silver::ast::Method::with(env);
227                        let method_name =
228                            jni.get_string(jni.unwrap_result(method_wrapper.call_name(concerning)));
229                        debug!("Entity failure for method: {method_name}");
230                        // this should only match local methods
231                        if procedures.contains(&method_name) {
232                            let verification_time =
233                                jni.unwrap_result(msg_wrapper.call_verificationTime(msg));
234                            // verification_time is a long -> i64. but we are using u128
235                            if verification_time >= 0 {
236                                let viper_result = jni.unwrap_result(msg_wrapper.call_result(msg));
237                                let result = viper::extract_errors(
238                                    &jni,
239                                    env,
240                                    viper_result,
241                                    Some(&mut error_hashes),
242                                );
243                                let verification_time_u128 = verification_time as u64 as u128;
244                                sender
245                                    .send(ServerMessage::MethodTermination {
246                                        viper_method_name: method_name.to_string(),
247                                        result,
248                                        verification_time: verification_time_u128,
249                                    })
250                                    .unwrap();
251                            } else {
252                                debug!(
253                                      "EntityFailureMessage for {method_name} had negative verification time {verification_time}"
254                                  );
255                            }
256                        }
257                    } else {
258                        debug!("Received entity was not a method");
259                    }
260                }
261                _ => (),
262            }
263        }
264        if !matches!(
265            stopped.recv_timeout(time::Duration::from_millis(10)),
266            Err(RecvTimeoutError::Timeout)
267        ) {
268            return error_hashes;
269        }
270    }
