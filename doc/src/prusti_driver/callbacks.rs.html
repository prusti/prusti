28#[allow(clippy::needless_lifetimes)]
29#[tracing::instrument(level = "debug", skip(tcx))]
30fn mir_borrowck<'tcx>(tcx: TyCtxt<'tcx>, def_id: LocalDefId) -> MirBorrowck<'tcx> {
31    // Like rustc's own `mir_borrowck`, skip trivial consts: they have no `mir_promoted`
32    // body to borrow-check, and no closures whose bodies Prusti would need.
33    if !tcx.is_trivial_const(def_id) {
34        // Computing the Polonius input facts can be very slow, so they are computed only where
35        // the PCG may need them: not for spec functions, and not in crates that are not verified.
36        let consumer_opts =
37            if is_spec_fn(tcx, def_id.to_def_id()) || config::verify_mode().is_compile_only() {
38                consumers::ConsumerOptions::RegionInferenceContext
39            } else {
40                consumers::ConsumerOptions::PoloniusInputFacts
41            };
42        let body_with_facts = consumers::get_bodies_with_borrowck_facts(tcx, def_id, consumer_opts);
43        for (def_id, body) in body_with_facts {
44            // SAFETY: This is safe because we are feeding in the same `tcx` that is
45            // going to be used as a witness when pulling out the data.
46            unsafe {
47                mir_storage::store_mir_body(tcx, def_id, body);
48            }
49        }
50    }
51    let mut providers = Providers::default();
52    prusti_rustc_interface::borrowck::provide(&mut providers);
53    let original_mir_borrowck = providers.mir_borrowck;
54    original_mir_borrowck(tcx, def_id)
55}

62    fn config(&mut self, config: &mut Config) {
63        // With incremental compilation, rustc does not run the `mir_borrowck` provider below for
64        // a body that has not changed since the last session, so it would not store its MIR.
65        config.opts.incremental = None;
66        assert!(config.override_queries.is_none());
67        config.override_queries = Some(|_session: &Session, providers| {
68            providers.queries.mir_borrowck = mir_borrowck;
69        });
70    }

72    fn after_crate_root_parsing(&mut self, compiler: &Compiler, krate: &mut Crate) -> Compilation {
73        // `prusti-rustc` will pass a `--extern` option indicating where to find
74        // the `prusti-contracts` crate. This crate must be loaded in order to
75        // include the default extern specs (for e.g build-in `assert!`
76        // statements). However, if the user does not explicitly import the
77        // crate in the code or use an `extern crate` statement, then the
78        // compiler will not load the crate.
79        //
80        // To work around this, we add a dummy `extern crate` statement to the
81        // crate if one does not already exist.
82        //
83        // Ideally we could load the prusti-contracts crate directly in the
84        // TyCtxt rather than modifying the crate AST.  However, it does not
85        // seem possible to load the crate directly in the TyCtxt in the
86        // `after_expansion` or `after_analysis` callback as the `cstore` object
87        // appears to be frozen at that point.
88
89        if compiler.sess.opts.externs.get("prusti_contracts").is_some() {
90            let prusti_contracts_symbol = Symbol::intern("prusti_contracts");
91
92            if !krate.items.iter().any(|item| match item.kind {
93                ItemKind::ExternCrate(Some(original_ident), _) => {
94                    original_ident == prusti_contracts_symbol
95                }
96                ItemKind::ExternCrate(None, ident) => ident.name == prusti_contracts_symbol,
97                _ => false,
98            }) {
99                krate.items.push(Box::new(Item {
100                    attrs: AttrVec::new(),
101                    id: DUMMY_NODE_ID,
102                    span: DUMMY_SP,
103                    vis: Visibility {
104                        kind: VisibilityKind::Public,
105                        span: DUMMY_SP,
106                    },
107                    kind: ItemKind::ExternCrate(
108                        None,
109                        Ident::new(prusti_contracts_symbol, DUMMY_SP),
110                    ),
111                    tokens: None,
112                }));
113            }
114        }
115        Compilation::Continue
116    }

118    #[tracing::instrument(level = "debug", skip_all)]
119    fn after_expansion<'tcx>(&mut self, compiler: &Compiler, tcx: TyCtxt<'tcx>) -> Compilation {
120        if compiler.sess.is_rust_2015() {
121            compiler
122                .sess
123                .dcx()
124                .struct_warn(
125                    "Prusti specifications are supported only from 2018 edition. Please \
126                    specify the edition with adding a command line argument `--edition=2018` or \
127                    `--edition=2021`.",
128                )
129                .emit();
130        }
131        compiler.sess.dcx().abort_if_errors();
132        if config::print_desugared_specs() {
133            // based on the implementation of rustc_driver::pretty::print_after_parsing
134            let sess = &compiler.sess;
135            let krate = &tcx.resolver_for_lowering().1.borrow();
136            let src_name = sess.io.input.file_name(sess);
137            let src = sess
138                .source_map()
139                .get_source_file(&src_name)
140                .expect("get_source_file")
141                .src
142                .as_ref()
143                .expect("src")
144                .to_string();
145            print!(
146                "{}",
147                prusti_rustc_interface::ast_pretty::pprust::print_crate(
148                    sess.source_map(),
149                    krate,
150                    src_name,
151                    src,
152                    &NoAnn,
153                    false,
154                    sess.edition(),
155                    &sess.psess.attr_id_generator,
156                )
157            );
158        }
159        Compilation::Continue
160    }

162    #[tracing::instrument(level = "debug", skip_all)]
163    fn after_analysis<'tcx>(&mut self, compiler: &Compiler, tcx: TyCtxt<'tcx>) -> Compilation {
164        compiler.sess.dcx().abort_if_errors();
165        let mut env = Environment::new(tcx, env!("CARGO_PKG_VERSION"));
166        let spec_checker = specs::checker::SpecChecker::new();
167        spec_checker.check(&env);
168        compiler.sess.dcx().abort_if_errors();
169
170        let mut spec_collector = specs::SpecCollector::new(&mut env);
171        spec_collector.collect_specs();
172
173        let mut def_spec = spec_collector.build_def_specs();
174        // Do print_typeckd_specs prior to importing cross crate
175        if config::print_typeckd_specs() {
176            for value in def_spec.all_values_debug(config::hide_uuids()) {
177                println!("{value}");
178            }
179        }
180        CrossCrateSpecs::import_export_cross_crate(&mut env, &mut def_spec);
181
182        // TODO: can we replace `get_annotated_procedures` with information
183        // that is already in `def_spec`?
184        let annotated_procedures = env.get_annotated_procedures();
185
186        if config::show_ide_info() && !config::verify_mode().is_compile_only() {
187            let compiler_info = IdeInfo::collect(&env, &annotated_procedures, &def_spec);
188            let out = serde_json::to_string(&compiler_info).unwrap();
189            PrustiError::message(format!("compilerInfo{out}"), DUMMY_SP.into())
190                .emit(&env.diagnostic);
191        }
192        // as long as we have to throw a fake error we need to check this
193        // note that is_single_file will still be false if this is run through x.py.
194        // it will find a manifest in prusti-launch but CARGO_PRIMARY_PACKAGE will still
195        // not be ok. meaning that selective verification can't currently be tested through
196        // x.py if the target is a single-file program.
197        let is_single_file = std::env::var("CARGO_MANIFEST_DIR").is_err();
198        let is_primary_package = is_single_file || std::env::var("CARGO_PRIMARY_PACKAGE").is_ok();
199
200        // collect and output Information used by IDE:
201        if !config::verify_mode().is_compile_only() && !config::skip_verification() {
202            let target_def_paths = config::verify_only_defpaths();
203            debug!(
204                "Received def paths: {target_def_paths:?}. Package is primary: {is_primary_package}, Package is single-file: {is_single_file}"
205            );
206            if !target_def_paths.is_empty() {
207                // if we do selective verification, then definitely only
208                // for methods of the primary package. Check needed because
209                // of the fake_error, otherwise verification stops early
210                // with local dependencies
211                if is_primary_package {
212                    let procedures = annotated_procedures
213                        .into_iter()
214                        .filter(|x| target_def_paths.contains(&env.name.get_unique_item_name(*x)))
215                        .collect::<Vec<_>>();
216                    if !procedures.is_empty() {
217                        let selective_task = VerificationTask {
218                            procedures,
219                            selective: true,
220                        };
221                        verify(env, def_spec, selective_task);
222                    } else {
223                        info!("Passed selective defpaths were empty or had no matching procedures - skipping verification.");
224                    }
225                    // fake_error(&env_diagnostic);
226                }
227            } else {
228                let verification_task = VerificationTask {
229                    procedures: annotated_procedures,
230                    selective: false,
231                };
232                verify(env, def_spec, verification_task);
233            }
234        } else if config::skip_verification()
235            && !config::verify_mode().is_compile_only()
236            && is_primary_package
237        {
238            // add a fake error, reason explained in issue #1261
239            fake_error(&env.diagnostic);
240        }
241
242        compiler.sess.dcx().abort_if_errors();
243        if config::full_compilation() {
244            Compilation::Continue
245        } else {
246            Compilation::Stop
247        }
248    }
