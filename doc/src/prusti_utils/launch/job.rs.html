12//! # Job management on Windows:
13//!
14//! Most of the time when you're running cargo you expect Ctrl-C to actually
15//! terminate the entire tree of processes in play, not just the one at the top
16//! (cargo). This currently works "by default" on Unix platforms because Ctrl-C
17//! actually sends a signal to the *process group* rather than the parent
18//! process, so everything will get torn down. On Windows, however, this does
19//! not happen and Ctrl-C just kills cargo.
20//!
21//! To achieve the same semantics on Windows we use Job Objects to ensure that
22//! all processes die at the same time. Job objects have a mode of operation
23//! where when all handles to the object are closed it causes all child
24//! processes associated with the object to be terminated immediately.
25//! Conveniently whenever a process in the job object spawns a new process the
26//! child will be associated with the job object as well. This means if we add
27//! ourselves to the job object we create then everything will get torn down!
28//!
29//! # Job management on Unix:
30//!
31//! A launcher and everything it starts stay in the caller's process group, so
32//! signals that the terminal, a shell or `timeout` send to that group reach
33//! all of them, and the terminal treats them as one job. A SIGINT, SIGTERM or
34//! SIGHUP sent to the launcher alone is passed on to every process still among
35//! its descendants, found in the process table, so `kill` stops the whole
36//! tree but no other member of the group, such as the other commands of a
37//! pipeline. The launcher passes on the signal it received itself, so that
38//! launchers among its descendants pass it on in turn and the drivers' JVMs
39//! shut down cleanly.
40//! The launcher then terminates with that signal, once the child it runs with
41//! `status` has exited, or at once if there is no such child (e.g. during the
42//! `java` and `rustup` queries that precede it). If one of these signals or
43//! SIGQUIT terminated the child, the launcher terminates with it too. While
44//! that child runs, the launcher ignores SIGQUIT, which Ctrl-\ sends to the
45//! child too: the driver's JVM then prints a thread dump and continues.
46//! A signal that is ignored when the launcher starts (e.g. SIGHUP under
47//! `nohup`) stays ignored. SIGKILL cannot be handled, so sent to the launcher
48//! alone, it leaves the processes the launcher started running.
49

63/// Runs `cmd` like `Command::status`. On Unix, a SIGINT, SIGTERM or SIGHUP
64/// that arrives while the child runs terminates the launcher only after the
65/// child has exited, and so does one of these signals or SIGQUIT that
66/// terminated the child. Only one call may run at a time.
67pub fn status(cmd: &mut Command) -> io::Result<ExitStatus> {
68    imp::status(cmd)
69}

88    struct State {
89        /// The SIGINT, SIGTERM or SIGHUP received while `status` waits.
90        received: Option<libc::c_int>,
91        /// The child that `status` waits for. While there is one, `status`,
92        /// not the signal thread, terminates the launcher.
93        child: Option<libc::pid_t>,
94    }

96    /// The signal thread holds the lock while handling a signal, and `status`
97    /// while spawning, so a signal either reaches the new child or terminates
98    /// the launcher before it is spawned.
99    static STATE: Mutex<State> = Mutex::new(State {
100        received: None,
101        child: None,
102    });
103
104    fn state() -> MutexGuard<'static, State> {
105        STATE.lock().unwrap_or_else(PoisonError::into_inner)
106    }

108    /// Terminates the launcher by the default action of `signal`, so that its
109    /// caller sees which signal ended it.
110    fn terminate(signal: libc::c_int) -> ! {
111        unsafe { libc::signal(signal, libc::SIG_DFL) };
112        let _ = signal_hook::low_level::raise(signal);
113        // The default action does not terminate e.g. the init process of a pid namespace.
114        process::exit(128 + signal)
115    }
116
117    fn is_ignored(signal: libc::c_int) -> bool {
118        let mut action: libc::sigaction = unsafe { mem::zeroed() };
119        // With a null new action, `sigaction` only reads the current one.
120        let read = unsafe { libc::sigaction(signal, ptr::null(), &mut action) };
121        read == 0 && action.sa_sigaction == libc::SIG_IGN
122    }
123
124    /// The signals that the launcher handles, unless they are ignored at start.
125    const HANDLED: [libc::c_int; 4] = [libc::SIGINT, libc::SIGTERM, libc::SIGHUP, libc::SIGQUIT];
126
127    pub(super) unsafe fn setup() -> Option<Setup> {
128        let signals = HANDLED.into_iter().filter(|&signal| !is_ignored(signal));
129        // signal-hook's handler only wakes the thread spawned below, so that thread may allocate
130        // and read the process table.
131        let mut signals = Signals::new(signals).expect("Error setting signal handlers");
132        thread::spawn(move || {
133            for signal in signals.forever() {
134                let mut state = state();
135                if signal == libc::SIGQUIT {
136                    // Ctrl-\ also reaches the child, whose JVM prints a thread dump and
137                    // continues, so the launcher keeps waiting for it.
138                    if state.child.is_none() {
139                        terminate(signal);
140                    }
141                    continue;
142                }
143                signal_descendants(signal, state.child);
144                if state.child.is_none() {
145                    terminate(signal);
146                }
147                state.received = Some(signal);
148            }
149        });
150        Some(false)
151    }

153    pub(super) fn status(cmd: &mut Command) -> io::Result<ExitStatus> {
154        let mut child = {
155            let mut state = state();
156            let child = cmd.spawn()?;
157            state.child = Some(child.id() as libc::pid_t);
158            child
159        };
160        let status = child.wait();
161        let mut state = state();
162        state.child = None;
163        // A signal sent to the whole group, such as Ctrl-C, can terminate the child before the
164        // signal thread has handled it, so a child terminated by a handled signal counts too. A
165        // child that catches the signal and exits with a status instead, like the JVM, is not
166        // recognised this way, but it normally exits only after the signal thread has recorded
167        // the signal.
168        let child_signal = status.as_ref().ok().and_then(|status| status.signal());
169        if let Some(signal) = state
170            .received
171            .or(child_signal.filter(|signal| HANDLED.contains(signal)))
172        {
173            terminate(signal);
174        }
175        status
176    }

178    /// Sends `signal` to `child`, if any, and every other descendant of this
179    /// process, parents first. The descendants are looked up before any of them
180    /// is signalled, since the children of a process that terminates are
181    /// reparented and so leave the tree. `child` is signalled even if the
182    /// process table cannot be read. A second lookup catches processes that the
183    /// recipients started meanwhile.
184    fn signal_descendants(signal: libc::c_int, child: Option<libc::pid_t>) {
185        let root = process::id() as libc::pid_t;
186        let descendants_before = descendants(root);
187        let mut signalled = FxHashSet::default();
188        let mut send = |pid| {
189            if signalled.insert(pid) {
190                unsafe { libc::kill(pid, signal) };
191            }
192        };
193        child.into_iter().for_each(&mut send);
194        descendants_before.into_iter().for_each(&mut send);
195        descendants(root).into_iter().for_each(&mut send);
196    }
197
198    /// The descendants of `root` in the process table, in breadth-first order.
199    fn descendants(root: libc::pid_t) -> Vec<libc::pid_t> {
200        let mut children: FxHashMap<libc::pid_t, Vec<libc::pid_t>> = FxHashMap::default();
201        for (pid, parent) in process_table() {
202            children.entry(parent).or_default().push(pid);
203        }
204        let mut result = children.remove(&root).unwrap_or_default();
205        let mut next = 0;
206        while let Some(&pid) = result.get(next) {
207            result.extend(children.remove(&pid).unwrap_or_default());
208            next += 1;
209        }
210        result
211    }

213    /// The (pid, parent pid) of every process that can be inspected.
214    #[cfg(target_os = "linux")]
215    fn process_table() -> Vec<(libc::pid_t, libc::pid_t)> {
216        let Ok(entries) = std::fs::read_dir("/proc") else {
217            return Vec::new();
218        };
219        entries
220            .filter_map(|entry| {
221                let entry = entry.ok()?;
222                let pid = entry.file_name().to_str()?.parse().ok()?;
223                // The parent pid is the second field after the parenthesised command name,
224                // which may itself contain spaces and parentheses.
225                let stat = std::fs::read_to_string(entry.path().join("stat")).ok()?;
226                let parent = stat.rsplit_once(')')?.1.split_whitespace().nth(1)?;
227                Some((pid, parent.parse().ok()?))
228            })
229            .collect()
230    }
231
232    /// The (pid, parent pid) of every process that can be inspected.
233    #[cfg(target_vendor = "apple")]
234    fn process_table() -> Vec<(libc::pid_t, libc::pid_t)> {
235        // Called without a buffer, `proc_listallpids` returns the number of processes. With a
236        // buffer, it returns the number of pids it stored, which equals the buffer's capacity if
237        // the buffer was too small.
238        let mut capacity = unsafe { libc::proc_listallpids(ptr::null_mut(), 0) }.max(0) as usize;
239        let mut pids: Vec<libc::pid_t> = Vec::new();
240        loop {
241            capacity += 64;
242            pids.resize(capacity, 0);
243            let size = (capacity * mem::size_of::<libc::pid_t>()) as libc::c_int;
244            let count = unsafe { libc::proc_listallpids(pids.as_mut_ptr().cast(), size) };
245            let count = count.max(0) as usize;
246            if count < capacity {
247                pids.truncate(count);
248                break;
249            }
250        }
251        pids.into_iter()
252            .filter_map(|pid| {
253                let mut info: libc::proc_bsdinfo = unsafe { mem::zeroed() };
254                let size = mem::size_of::<libc::proc_bsdinfo>() as libc::c_int;
255                let pointer = (&mut info as *mut libc::proc_bsdinfo).cast();
256                let filled =
257                    unsafe { libc::proc_pidinfo(pid, libc::PROC_PIDTBSDINFO, 0, pointer, size) };
258                (filled == size).then_some((pid, info.pbi_ppid as libc::pid_t))
259            })
260            .collect()
261    }

294    pub(super) unsafe fn setup() -> Option<Setup> {
295        // Creates a new job object for us to use and then adds ourselves to it.
296        // Note that all errors are basically ignored in this function,
297        // intentionally. Job objects are "relatively new" in Windows,
298        // particularly the ability to support nested job objects. Older
299        // Windows installs don't support this ability. We probably don't want
300        // to force Cargo to abort in this situation or force others to *not*
301        // use job objects, so we instead just ignore errors and assume that
302        // we're otherwise part of someone else's job object in this case.
303
304        let job = CreateJobObjectW(ptr::null_mut(), ptr::null());
305        if job.is_null() {
306            return None;
307        }
308        let job = Handle { inner: job };
309
310        // Indicate that when all handles to the job object are gone that all
311        // process in the object should be killed. Note that this includes our
312        // entire process tree by default because we've added ourselves and
313        // our children will reside in the job once we spawn a process.
314        let mut info: JOBOBJECT_EXTENDED_LIMIT_INFORMATION;
315        info = mem::zeroed();
316        info.BasicLimitInformation.LimitFlags = JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE;
317        let r = SetInformationJobObject(
318            job.inner,
319            JobObjectExtendedLimitInformation,
320            &mut info as *mut _ as LPVOID,
321            mem::size_of_val(&info) as DWORD,
322        );
323        if r == 0 {
324            return None;
325        }
326
327        // Assign our process to this job object, meaning that our children will
328        // now live or die based on our existence.
329        let me = GetCurrentProcess();
330        let r = AssignProcessToJobObject(job.inner, me);
331        if r == 0 {
332            return None;
333        }
334
335        Some(Setup { job })
336    }

343    impl Drop for Setup {
344        fn drop(&mut self) {
345            // On normal exits (not ctrl-c), we don't want to kill any child
346            // processes. The destructor here configures our job object to
347            // *not* kill everything on close, then closes the job object.
348            unsafe {
349                let mut info: JOBOBJECT_EXTENDED_LIMIT_INFORMATION;
350                info = mem::zeroed();
351                let r = SetInformationJobObject(
352                    self.job.inner,
353                    JobObjectExtendedLimitInformation,
354                    &mut info as *mut _ as LPVOID,
355                    mem::size_of_val(&info) as DWORD,
356                );
357                if r == 0 {
358                    log::info!("failed to configure job object to defaults: {}", last_err());
359                }
360            }
361        }
362    }

364    impl Drop for Handle {
365        fn drop(&mut self) {
366            unsafe {
367                CloseHandle(self.inner);
368            }
369        }
