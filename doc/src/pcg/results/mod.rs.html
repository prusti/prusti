106    /// Returns the free pcs for the location `exp_loc` and iterates the cursor
107    /// to the *end* of that location.
108    ///
109    /// This function may return `None` if the PCG did not analyze this block.
110    /// This could happen, for example, if the block would only be reached when unwinding from a panic.
111    fn next(&mut self, exp_loc: Location) -> Result<Option<PcgLocation<'a, 'tcx>>, PcgError> {
112        let location = self.curr_stmt.unwrap();
113        assert_eq!(location, exp_loc);
114        assert!(location < self.end_stmt.unwrap());
115
116        self.cursor.seek_after_primary_effect(location);
117
118        let state = self.cursor.get().expect_results_or_error()?;
119
120        let result = PcgLocation {
121            location,
122            actions: state.data.actions.clone(),
123            states: state.data.pcg.states.to_owned(),
124        };
125
126        self.curr_stmt = Some(location.successor_within_block());
127
128        Ok(Some(result))
129    }
130    pub(crate) fn terminator<'slf>(&'slf mut self) -> Result<PcgTerminator<'a, 'tcx>, PcgError> {
131        let location = self.curr_stmt.unwrap();
132        assert_eq!(location, self.end_stmt.unwrap());
133        self.curr_stmt = None;
134        self.end_stmt = None;
135
136        let state = self.cursor.get().expect_results_or_error()?;
137        let from_pcg = &state.data.pcg;
138        let from_post_main = from_pcg.states[EvalStmtPhase::PostMain].clone();
139        let states = state.data.pcg.states.to_owned();
140        let self_abstraction_edges = from_post_main
141            .borrow
142            .graph()
143            .abstraction_edges()
144            .collect::<FxHashSet<_>>();
145
146        let ctxt: ResultsCtxt<'a, 'tcx> = self.results_ctxt();
147        let block = &self.body()[location.block];
148
149        let succ_blocks = successor_blocks(block.terminator())
150            .into_iter()
151            .filter(|succ| {
152                self.cursor
153                    .analysis()
154                    .0
155                    .reachable_blocks
156                    .borrow()
157                    .contains(succ.index())
158            })
159            .collect::<Vec<_>>();
160        let succs = succ_blocks
161            .into_iter()
162            .map(|succ| {
163                self.cursor.seek_to_block_start(succ);
164                let to = self
165                    .cursor
166                    .get()
167                    .expect_results_or_error()?
168                    .data
169                    .pcg
170                    .clone();
171
172                let mut actions =
173                    from_post_main.bridge(&to.entry_state, location.block, succ, ctxt)?;
174
175                let mut borrow_actions = BorrowPcgActions::new();
176                for abstraction in to.entry_state.borrow.graph().abstraction_edges() {
177                    if !self_abstraction_edges.contains(&abstraction) {
178                        borrow_actions.push(
179                            BorrowPcgAction::add_edge(
180                                BorrowPcgEdge::new(
181                                    abstraction.value.clone().into(),
182                                    abstraction.conditions,
183                                ),
184                                "terminator",
185                            ),
186                            self.engine(),
187                        );
188                    }
189                }
190
191                actions.extend(borrow_actions.into());
192
193                Ok(PcgSuccessor::new(
194                    succ,
195                    actions,
196                    to.entry_state.borrow.clone().into(),
197                ))
198            })
199            .collect::<Result<Vec<_>, PcgError>>()?;
200        Ok(PcgTerminator { succs, states })
201    }

203    /// Obtains the results of the dataflow analysis for all blocks.
204    ///
205    /// This is rather expensive to compute and may take a lot of memory. You
206    /// may want to consider using `get_all_for_bb` instead.
207    pub fn results_for_all_blocks(&mut self) -> Result<PcgBasicBlocks<'a, 'tcx>, PcgError> {
208        let mut result = IndexVec::new();
209        for block in self.body().basic_blocks.indices() {
210            let pcg_block = self.get_all_for_bb(block)?;
211            result.push(pcg_block);
212        }
213        Ok(PcgBasicBlocks(result))
214    }

226    /// Recommended interface.
227    /// Does *not* require that one calls `analysis_for_bb` first
228    /// This function may return `None` if the PCG did not analyze this block.
229    /// This could happen, for example, if the block would only be reached when unwinding from a panic.
230    pub fn get_all_for_bb<'slf>(
231        &'slf mut self,
232        block: BasicBlock,
233    ) -> Result<Option<PcgBasicBlock<'a, 'tcx>>, PcgError> {
234        let mut statements: Vec<PcgLocation<'a, 'tcx>> = Vec::new();
235        if !self.for_each_statement_in_bb(block, |stmt| statements.push(stmt))? {
236            return Ok(None);
237        }
238        let terminator = self.terminator()?;
239        Ok(Some(PcgBasicBlock {
240            statements,
241            terminator,
242            block,
243        }))
244    }
245
246    /// Passes the results of each statement of `block` (not its terminator),
247    /// in order, to `f`, so that the caller need not hold the results of the
248    /// whole block at once. Returns `false` if the PCG did not analyze `block`.
249    /// On an error, `f` has seen the statements before the failing one.
250    pub(crate) fn for_each_statement_in_bb(
251        &mut self,
252        block: BasicBlock,
253        mut f: impl FnMut(PcgLocation<'a, 'tcx>),
254    ) -> Result<bool, PcgError> {
255        if !self
256            .analysis()
257            .reachable_blocks
258            .borrow()
259            .contains(block.index())
260        {
261            return Ok(false);
262        }
263        self.analysis_for_bb(block);
264        let end_stmt = self.end_stmt.unwrap();
265        while self.curr_stmt.unwrap() != end_stmt {
266            let Some(stmt) = self.next(self.curr_stmt.unwrap())? else {
267                return Ok(false);
268            };
269            f(stmt);
270        }
271        Ok(true)
272    }
273}

290        &self,
291        f: impl Fn(&PcgLocation<'_, 'tcx>) -> FxHashSet<T>,
292    ) -> FxHashSet<T> {
293        let mut result = FxHashSet::default();
294        for block in &self.0 {

297                    result.extend(f(stmt));
298                }
299            }
300        }

315/// The capabilities a loop requires at its head, as reported by
316/// [`PcgBasicBlock::loop_invariant_place_capabilities`].
317pub struct LoopInvariantCapabilities<'tcx> {
318    /// The capabilities of places.
319    pub places: BTreeMap<Place<'tcx>, CapabilityKind>,
320    /// The capabilities of the referents of a place's references of one
321    /// lifetime, keyed by the lifetime projection that holds them; reported
322    /// only for places in `places` whose own value the loop does not write.
323    pub lifetime_projections: BTreeMap<LifetimeProjection<'tcx, Place<'tcx>>, CapabilityKind>,
324}
325
326impl<'tcx> LoopInvariantCapabilities<'tcx> {
327    /// The places and then the lifetime projections, displayed as strings.
328    pub fn display_entries<'a, Ctxt: Copy + 'a>(
329        &'a self,
330        ctxt: Ctxt,
331    ) -> impl Iterator<Item = (String, CapabilityKind)> + 'a
332    where
333        Place<'tcx>: DisplayWithCtxt<Ctxt>,
334        LifetimeProjection<'tcx, Place<'tcx>>: DisplayWithCtxt<Ctxt>,
335    {
336        let places = self
337            .places
338            .iter()
339            .map(move |(place, &capability)| (place.display_string(ctxt), capability));
340        let lifetime_projections = self
341            .lifetime_projections
342            .iter()
343            .map(move |(projection, &capability)| (projection.display_string(ctxt), capability));
344        places.chain(lifetime_projections)
345    }
346}

364    /// The capabilities that the loop with this block as its head requires of
365    /// the places that hold its `place_usages` at the head.
366    ///
367    /// A usage is held by the place itself or its nearest ancestor with a
368    /// capability at the head; if the place is unpacked at the head, by the
369    /// places it is unpacked into. The capability of a holder is the minimum
370    /// of its capability at the head and the loop's usage of it, or the usage
371    /// if the two are incomparable: e.g. a mutable reference whose target is
372    /// unpacked (`ShallowExclusive`) and whose own value the loop only reads
373    /// is reported at `R`.
374    ///
375    /// A usage behind a mutable reference in its holder uses the referents of
376    /// the holder's lifetime projections rather than the holder's own value:
377    /// that of the last mutable reference the usage's path dereferences with
378    /// the usage, and those of the references before it only by reading them.
379    /// A usage also uses, with its kind, the lifetime projections of the
380    /// lifetimes in the type of the used place, whose referents it may read or
381    /// change; a read of the holder's own value reads all of them. Lifetimes
382    /// that are not the holder's (e.g. `'static`) are left out, since none of
383    /// its lifetime projections holds what their references refer to.
384    /// If the loop does not write the holder's own value, the holder is
385    /// reported at `R` and each of its lifetime projections separately, with
386    /// the capability of its usages limited by the holder's, or at `R` if the
387    /// loop does not use its referents. If the loop writes the holder's own
388    /// value, the holder's capability also covers its referents, and its
389    /// lifetime projections are not reported.
390    ///
391    /// Each reference with a capability at the head on the path to a reported
392    /// place is reported as well, separately from the places behind it: at
393    /// `R` if the loop does not use the reference itself, since the loop then
394    /// only reads it to reach the places behind it.
395    ///
396    /// The places are ordered by `Place`'s `Ord`, so a place comes after the
397    /// places it extends, through which it is reached.
398    pub fn loop_invariant_place_capabilities<'c>(
399        &self,
400        place_usages: &PlaceUsages<'tcx>,
401        ctxt: impl HasCompilerCtxt<'c, 'tcx>,
402    ) -> LoopInvariantCapabilities<'tcx>
403    where
404        'tcx: 'c,
405    {
406        /// The loop's usages held by one place: of the place's own value, and
407        /// of the referents of its lifetime projections. Multiple usages can
408        /// resolve to the same place or lifetime projection; their
409        /// requirements are joined before limiting the capability to what is
410        /// available at the head.
411        #[derive(Default)]
412        struct HolderUsages<'tcx> {
413            own: Option<PlaceUsageType>,
414            behind: HashMap<LifetimeProjection<'tcx, Place<'tcx>>, PlaceUsageType>,
415        }
416
417        let initial_capabilities =
418            self.statements[0].states[EvalStmtPhase::PreOperands].capabilities();
419        let mut usages: HashMap<Place<'tcx>, HolderUsages<'tcx>> = HashMap::default();
420        for place_usage in place_usages.iter() {
421            let holder = std::iter::successors(Some(place_usage.place), |p| p.parent_place())
422                .find(|place| initial_capabilities.get(*place, ctxt).is_some());
423            let holders = match holder {
424                Some(holder) => vec![holder],
425                None => initial_capabilities
426                    .iter()
427                    .map(|(place, _)| place)
428                    .filter(|place| place_usage.place.is_strict_prefix_of(*place))
429                    .collect(),
430            };
431            let usage = place_usage.usage;
432            for holder in holders {
433                let holder_usages = usages.entry(holder).or_default();
434                // A usage of a place behind the holder's mutable references
435                // uses the referents of the last of them on its path, and reads
436                // those before it. It may also use what the references in the
437                // used place refer to (e.g. through a shared borrow of it, or by
438                // assigning it). A usage of a place the holder extends, or of a
439                // place behind no mutable reference of one of the holder's
440                // lifetimes, counts as a usage of the holder's own value.
441                let behind = place_usage
442                    .place
443                    .mut_deref_projections_after(holder, ctxt)
444                    .and_then(|projections| {
445                        let (last, through) = projections.split_last()?;
446                        let reads = through
447                            .iter()
448                            .map(|&projection| (projection, PlaceUsageType::Read));
449                        let uses = place_usage
450                            .place
451                            .outer_lifetime_projections(holder, ctxt)
452                            .into_iter()
453                            .chain([*last])
454                            .map(|projection| (projection, usage));
455                        Some(reads.chain(uses).collect::<Vec<_>>())
456                    });
457                match behind {
458                    Some(behind) => {
459                        for (projection, usage) in behind {
460                            holder_usages
461                                .behind
462                                .entry(projection)
463                                .and_modify(|existing| *existing = existing.joined(usage))
464                                .or_insert(usage);
465                        }
466                    }
467                    None => {
468                        holder_usages.own =
469                            Some(holder_usages.own.map_or(usage, |own| own.joined(usage)));
470                    }
471                }
472            }
473        }
474        let capability_of = |holder: Place<'tcx>, usage: PlaceUsageType| {
475            let usage_capability = match usage {
476                PlaceUsageType::Read => CapabilityKind::Read,
477                PlaceUsageType::Write => CapabilityKind::Write,
478                PlaceUsageType::Exclusive => CapabilityKind::Exclusive,
479            };
480            initial_capabilities
481                .get(holder, ctxt)
482                .expect("a holder has a capability at the head")
483                .minimum(usage_capability)
484                .unwrap_or(usage_capability)
485        };
486        let mut capabilities: BTreeMap<Place<'tcx>, CapabilityKind> = BTreeMap::new();
487        let mut lifetime_projections = BTreeMap::new();
488        for (holder, HolderUsages { own, behind }) in usages {
489            if own.is_none_or(PlaceUsageType::is_read) {
490                capabilities.insert(holder, capability_of(holder, PlaceUsageType::Read));
491                // A read of the holder's own value may read its referents too.
492                for projection in holder.lifetime_projections(ctxt) {
493                    let usage = behind
494                        .get(&projection)
495                        .copied()
496                        .into_iter()
497                        .chain(own)
498                        .reduce(PlaceUsageType::joined)
499                        .unwrap_or(PlaceUsageType::Read);
500                    lifetime_projections.insert(projection, capability_of(holder, usage));
501                }
502            } else {
503                let usage = behind
504                    .into_values()
505                    .chain(own)
506                    .reduce(PlaceUsageType::joined)
507                    .expect("a holder holds a usage");
508                capabilities.insert(holder, capability_of(holder, usage));
509            }
510        }
511        let references_on_paths = capabilities
512            .keys()
513            .flat_map(|place| std::iter::successors(place.parent_place(), |p| p.parent_place()))
514            .filter(|place| {
515                place.is_ref(ctxt)
516                    && initial_capabilities.get(*place, ctxt).is_some()
517                    && !capabilities.contains_key(place)
518            })
519            .collect::<Vec<_>>();
520        for reference in references_on_paths {
521            capabilities.insert(reference, CapabilityKind::Read);
522        }
523        LoopInvariantCapabilities {
524            places: capabilities,
525            lifetime_projections,
526        }
527    }

529    #[must_use]
530    pub fn debug_lines<Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>>(
531        &self,
532        ctxt: Ctxt,
533    ) -> Vec<Cow<'static, str>> {
534        let mut result = Vec::new();
535        for stmt in &self.statements {
536            for phase in EvalStmtPhase::phases() {
537                for line in stmt.debug_lines(phase, ctxt) {
538                    result.push(format!("{:?} {}: {}", stmt.location, phase, line).into());
539                }
540            }
541        }
542        for term_succ in &self.terminator.succs {
543            for line in term_succ.debug_lines(ctxt) {
544                result
545                    .push(format!("{:?} -> {:?}: {}", self.block, term_succ.block(), line).into());
546            }
547        }
548        result
549    }

596    pub fn aliases<'mir>(
597        &self,
598        place: impl Into<Place<'tcx>>,
599        body: &'mir Body<'tcx>,
600        tcx: TyCtxt<'tcx>,
601    ) -> FxHashSet<mir::Place<'tcx>> {
602        let place: Place<'tcx> = place.into();
603        let ctxt = CompilerCtxt::new(body, tcx, ());
604        self.states[EvalStmtPhase::PostMain]
605            .borrow
606            .graph()
607            .aliases(place.into(), ctxt)
608            .into_iter()
609            .filter_map(|p| match p {
610                PcgNode::Place(p) => p.as_current_place(),
611                PcgNode::LifetimeProjection(p) => match p.base() {
612                    PlaceOrConst::Place(p) => {
613                        let assoc_place = p.related_local_place();
614                        assoc_place
615                            .is_ref(ctxt)
616                            .then(|| assoc_place.project_deref(ctxt))
617                    }
618                    _ => None,
619                },
620            })
621            .map(|p| p.to_rust_place(ctxt))
622            .collect()
623    }
624
625    pub(crate) fn debug_lines<Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>>(
626        &self,
627        phase: EvalStmtPhase,
628        ctxt: Ctxt,
629    ) -> Vec<Cow<'static, str>> {
630        let mut result = self.states[phase].debug_lines(ctxt);
631        for action in self.actions[phase].iter() {
632            result.push(action.action.debug_line(ctxt));
633        }
634        result
635    }
