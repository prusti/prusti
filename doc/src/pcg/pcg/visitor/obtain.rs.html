109    fn restore_place(&mut self, place: Place<'tcx>, context: &str) -> Result<(), PcgError> {
110        // The place to restore could come from a local that was conditionally
111        // allocated and therefore we can't get back to it, and certainly
112        // shouldn't give it any capability
113        // TODO: Perhaps the join should label such places?
114        if !self.pcg.owned.is_allocated(place.local) {
115            return Ok(());
116        }
117        let blocked_cap = self.pcg.capability_of(place, self.ctxt);
118
119        // TODO: If the place projects a shared ref, do we even need to restore a capability?
120        //
121        // For owned places we consult the initialisation state so we
122        // don't unsoundly restore to `Exclusive` when the place was
123        // moved out on some incoming path (#137): the join of the init
124        // trees would then leave the place as `Uninit`, which maps to
125        // `Write`.
126        let restore_cap = if place.place().projects_shared_ref(self.ctxt)
127            || self
128                .ctxt
129                .body_analysis()
130                .is_read_only_in_loop(self.location().block, place)
131        {
132            CapabilityKind::Read
133        } else if let Some(owned) = place.as_owned_place(self.ctxt)
134            && let Some(init_cap) = self.pcg.owned.owned_capability(owned)
135        {
136            init_cap
137        } else {
138            CapabilityKind::Exclusive
139        };
140
141        // The blocked capability would be None if the place was mutably
142        // borrowed. The capability would be ShallowExclusive if the place is
143        // a mutable reference (when dereferencing a mutable ref, the ref
144        // place retains shallow-exclusive capability).
145        if matches!(
146            blocked_cap,
147            None | Some(CapabilityKind::Write | CapabilityKind::ShallowExclusive)
148        ) && blocked_cap != Some(restore_cap)
149        {
150            self.record_and_apply_action(PcgAction::restore_capability(
151                place,
152                restore_cap,
153                context.to_owned(),
154                self.ctxt,
155            ))?;
156        }
157        for rp in place.lifetime_projections(self.ctxt) {
158            self.record_and_apply_action(
159                    BorrowPcgAction::remove_lifetime_projection_label(
160                        rp.with_placeholder_label(self.ctxt).into(),
161                        format!(
162                            "Place {} unblocked: remove placeholder label of rps of newly unblocked nodes",
163                            place.display_string(self.ctxt.bc_ctxt())
164                        ),
165                    )
166                    .into(),
167                )?;
168        }
169        Ok(())
170    }
171    fn update_unblocked_node_capabilities_and_remove_placeholder_projections(
172        &mut self,
173        edge: &BorrowPcgEdgeKind<'tcx>,
174    ) -> Result<(), PcgError> {
175        let graph = &self.pcg.borrow.graph;
176        let blocked_nodes = edge.blocked_nodes(self.ctxt.bc_ctxt());
177
178        // After removing an edge, some nodes may become accessible, their capabilities should be restored
179        let to_restore = blocked_nodes
180            .into_iter()
181            .filter(|node| {
182                graph
183                    .edges_blocking(*node, self.ctxt.bc_ctxt())
184                    .next()
185                    .is_none()
186            })
187            .collect::<Vec<_>>();
188
189        for node in to_restore {
190            if let Some(place) = node.as_current_place() {
191                // Only restore if the place is still tracked somewhere.  After
192                // a loop abstraction edge is removed, the blocked place (e.g.
193                // `(*variant).fields`) may no longer appear in either the owned
194                // PCG or the borrow graph; restoring a capability there would
195                // violate the invariant that every capability-holding place must
196                // be tracked.
197                if !self.pcg.borrow.graph.contains(place, self.ctxt.bc_ctxt())
198                    && !self.pcg.owned.contains_place(place, self.ctxt.bc_ctxt())
199                {
200                    continue;
201                }
202                self.restore_place(
203                    place,
204                    "update_unblocked_node_capabilities_and_remove_placeholder_projections",
205                )?;
206            }
207        }
208        Ok(())
209    }

211    /// If the following conditions apply:
212    /// 1. `expansion` is a dereference of a place `p`
213    /// 2. `*p` does not contain any borrows
214    /// 3. The target of this expansion is not labelled
215    ///
216    /// Then we perform an optimization where instead of connecting the blocked
217    /// lifetime projection to the current one, we instead remove the label of
218    /// the blocked lifetime projection.
219    ///
220    /// This is sound because the lifetime projection only contains the single
221    /// borrow that `p` refers to and therefore the set of borrows cannot be
222    /// changed. In other words, the set of borrows in the lifetime projection
223    /// at the point it was dereferenced is the same as the current set of
224    /// borrows in the lifetime projection.
225    ///
226    /// Note the third condition: if the expansion is labelled, that indicates
227    /// that the expansion occurred at a point where `p` had a different value
228    /// than the current one. We don't want to perform this optimization because
229    /// the it is referring to this different value.
230    /// For test case see rustls-pki-types@1.11.0 `server_name::parser::Parser::`<'`a>::read_char`
231    ///
232    /// TODO: In the above test case, should the parent place also be labelled?
233    fn unlabel_blocked_region_projections_if_applicable(
234        &mut self,
235        deref: &DerefEdge<'tcx>,
236        context: &str,
237    ) -> Result<(), PcgError> {
238        if deref.deref_place.is_current()
239            && deref.deref_place.lifetime_projections(self.ctxt).is_empty()
240        {
241            self.unlabel_source_lifetime_projections(deref, context)
242        } else {
243            Ok(())
244        }
245    }

247    pub(crate) fn remove_deref_edges_to(
248        &mut self,
249        deref_place: MaybeLabelledPlace<'tcx>,
250        edges: HashSet<Conditioned<DerefEdge<'tcx>>>,
251        context: &str,
252    ) -> Result<(), PcgError> {
253        for edge in edges {
254            let borrow_edge: BorrowPcgEdge<'tcx> =
255                BorrowPcgEdge::new(edge.value.into(), edge.conditions);
256            self.apply_action(
257                BorrowPcgAction::remove_edge(borrow_edge, context.to_owned()).into(),
258            )?;
259            self.unlabel_blocked_region_projections_if_applicable(&edge.value, context)?;
260        }
261        if let Some(deref_place) = deref_place.as_current_place() {
262            self.apply_action(
263                BorrowPcgAction::label_place_and_update_related_capabilities(
264                    deref_place,
265                    self.prev_snapshot_location(),
266                    LabelPlaceReason::Collapse,
267                )
268                .into(),
269            )?;
270            let ref_place = deref_place.parent_place().unwrap();
271            // Perhaps the place isn't yet a leaf(e.g. the ref itself is conditionally borrowed)
272            // In that case we shouldn't restore its caps
273            if self.pcg.is_leaf_place(ref_place, self.ctxt) {
274                self.restore_place(
275                    ref_place,
276                    &format!("{context}: remove_deref_edges_to: restore parent place"),
277                )?;
278            }
279        }
280        Ok(())
281    }

283    #[tracing::instrument(skip(self, edge))]
284    pub(crate) fn remove_edge_and_perform_associated_state_updates(
285        &mut self,
286        edge: &BorrowPcgEdge<'tcx>,
287        context: &str,
288    ) -> Result<(), PcgError> {
289        if let BorrowPcgEdgeKind::Deref(deref) = edge.kind() {
290            return self.remove_deref_edges_to(
291                deref.deref_place,
292                vec![Conditioned::new(*deref, edge.conditions.clone())]
293                    .into_iter()
294                    .collect(),
295                context,
296            );
297        }
298        self.record_and_apply_action(
299            BorrowPcgAction::remove_edge(edge.clone(), context.to_owned()).into(),
300        )?;
301
302        // This is true iff the expansion is for a place (not a region projection), and changes
303        // could have been made to the root place via the expansion
304        // We check that the base is place and either:
305        // - The base has no capability, meaning it was previously expanded mutably
306        // - The base has write capability, it is a mutable ref
307        let is_mutable_place_expansion = if let BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) =
308            edge.kind()
309            && let Some(place) = expansion.base().as_current_place()
310        {
311            matches!(
312                self.pcg.capability_of(place, self.ctxt),
313                Some(CapabilityKind::Write) | None
314            )
315        } else {
316            false
317        };
318
319        self.update_unblocked_node_capabilities_and_remove_placeholder_projections(&edge.value)?;
320
321        match &edge.value {
322            BorrowPcgEdgeKind::Deref(deref) => {
323                self.unlabel_blocked_region_projections_if_applicable(deref, context)?;
324                if deref.deref_place.is_current() {
325                    self.apply_action(
326                        BorrowPcgAction::label_place_and_update_related_capabilities(
327                            deref.deref_place.place(),
328                            self.prev_snapshot_location(),
329                            LabelPlaceReason::Collapse,
330                        )
331                        .into(),
332                    )?;
333                }
334            }
335            BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) => {
336                if is_mutable_place_expansion {
337                    // If the expansion contained region projections, we need to
338                    // label them, they will flow into the now unblocked
339                    // projection (i.e. the one obtained by removing the
340                    // placeholder label)
341
342                    // For example, if we a are packing *s.i into *s at l
343                    // we need to label *s.i|'s to  *s|'s at l
344                    // because we will remove the label from *s|'s at l'
345                    // to become *s|'s. Otherwise we'd have both *s|'s and *s.i|'s
346                    for exp_node in expansion.expansion() {
347                        if let PcgNode::Place(place) = exp_node {
348                            for rp in place.lifetime_projections(self.ctxt) {
349                                tracing::debug!(
350                                    "labeling region projection: {}",
351                                    rp.display_string(self.ctxt)
352                                );
353                                self.record_and_apply_action(
354                                    BorrowPcgAction::label_lifetime_projection(
355                                        LabelNodePredicate::equals_lifetime_projection(rp),
356                                        Some(self.prev_snapshot_location().into()),
357                                        format!(
358                                            "{}: {}",
359                                            context, "Label region projections of expansion"
360                                        ),
361                                    )
362                                    .into(),
363                                )?;
364                            }
365                        }
366                    }
367                }
368            }
369            _ => self.weaken_places_with_dead_lifetime_projections(&edge.value)?,
370        }
371        Ok(())
372    }

374    /// A place whose lifetime projection is dead can no longer be used, so once
375    /// the edge into that projection is removed, the place retains only
376    /// [`CapabilityKind::Write`].
377    ///
378    /// The place may still be dereferenced, reached by a dereference of its
379    /// parent, or the borrow assigned to it still be in the graph, in edges
380    /// that are removed only later; consumers of the actions may need the
381    /// place's value to apply those removals. Such a place is not weakened
382    /// while it is in use, see
383    /// [`Self::finish_weakening_places_with_dead_lifetime_projections`].
384    fn weaken_places_with_dead_lifetime_projections(
385        &mut self,
386        edge: &BorrowPcgEdgeKind<'tcx>,
387    ) -> Result<(), PcgError> {
388        for place in self.places_with_dead_lifetime_projections(edge) {
389            if !self.reference_in_use(place) {
390                self.weaken_place_with_dead_lifetime_projection(place)?;
391            }
392        }
393        Ok(())
394    }

409    /// The current places whose lifetime projections `edge` is blocked by and
410    /// are dead. Dereferences and expansions do not leave a place's lifetime
411    /// projection dead.
412    fn places_with_dead_lifetime_projections(
413        &self,
414        edge: &BorrowPcgEdgeKind<'tcx>,
415    ) -> HashSet<Place<'tcx>> {
416        if matches!(
417            edge,
418            BorrowPcgEdgeKind::Deref(_) | BorrowPcgEdgeKind::BorrowPcgExpansion(_)
419        ) {
420            return HashSet::default();
421        }
422        edge.blocked_by_nodes(self.ctxt.bc_ctxt())
423            .filter_map(|node| {
424                let PcgNode::LifetimeProjection(rp) = node else {
425                    return None;
426                };
427                let MaybeLabelledPlace::Current(place) = rp.base() else {
428                    return None;
429                };
430                self.ctxt
431                    .bc()
432                    .is_dead(node.into(), self.location())
433                    .then_some(place)
434            })
435            .collect()
436    }

458    /// Whether the graph still dereferences `place`, reaches it by a
459    /// dereference of its parent (whose removal packs `place` back into the
460    /// parent), or contains the borrow assigned to it.
461    fn reference_in_use(&self, place: Place<'tcx>) -> bool {
462        let place = MaybeLabelledPlace::Current(place);
463        self.pcg
464            .borrow
465            .graph()
466            .edges()
467            .any(|edge| match edge.kind() {
468                BorrowPcgEdgeKind::Deref(deref) => {
469                    deref.blocked_place == place || deref.deref_place == place
470                }
471                BorrowPcgEdgeKind::Borrow(borrow) => borrow.assigned_ref == place,
472                _ => false,
473            })
474    }

476    /// As an optimization, for expansions of the form {y, y|'y at l} -> *y,
477    /// if *y doesn't contain any borrows, we currently don't introduce placeholder
478    /// projections for y|'y: the set of borrows is guaranteed not to change as long as *y
479    /// is in the graph.
480    ///
481    /// Accordingly, when we want to remove *y in such cases, we just remove the
482    /// label rather than use the normal logic (of renaming the placeholder
483    /// projection to the current one).

485        &mut self,
486        deref: &DerefEdge<'tcx>,
487        context: &str,
488    ) -> Result<(), PcgError> {
489        self.record_and_apply_action(
490            BorrowPcgAction::remove_lifetime_projection_label(
491                deref.blocked_lifetime_projection,
492                format!("{context}: unlabel blocked_region_projections"),
493            )
494            .into(),
495        )?;
496        Ok(())
497    }

500        &mut self,
501        upgraded_place: Place<'tcx>,
502    ) -> Result<(), PcgError> {
503        let to_remove = self

520    pub(crate) fn upgrade_read_to_exclusive(&mut self, place: Place<'tcx>) -> Result<(), PcgError> {
521        tracing::debug!(
522            "upgrade_read_to_exclusive: {}",
523            place.display_string(self.ctxt.bc_ctxt())
524        );
525        self.record_and_apply_action(
526            BorrowPcgAction::restore_capability(
527                place,
528                CapabilityKind::Exclusive,
529                "upgrade_read_to_exclusive",
530            )
531            .into(),
532        )?;
533        self.remove_read_permission_downwards(place)?;
534        if let Some(parent) = place.parent_place() {
535            self.remove_read_permission_upwards_and_label_rps(
536                parent,
537                AdjustCapabilityReason::UpgradeReadToExclusive,
538            )?;
539        }
540        Ok(())

543    // This function is called if we want to obtain non-read capability to `place`
544    // If the closest ancestor of `place` has read capability, then we are allowed to
545    // upgrade the capability of the ancestor to `E` in exchange for downgrading all
546    // other pre / postfix places to None.
547    //
548    // This is sound because if we need to obtain non-read capability to
549    // `place`, and there are any ancestors of `place` in the graph with R
550    // capability, one such ancestor originally had E capability was
551    // subsequently downgraded. This function finds such an ancestor (if one
552    // exists), and performs the capability exchange.
553    pub(crate) fn upgrade_closest_read_ancestor_to_exclusive_and_update_rps(
554        &mut self,
555        place: Place<'tcx>,
556    ) -> Result<(), PcgError> {
557        let mut expand_root = place;
558        loop {
559            if let Some(cap) = self.pcg.capability_of(expand_root, self.ctxt) {
560                if cap.is_read() {
561                    self.upgrade_read_to_exclusive(expand_root)?;
562                }
563                return Ok(());
564            }
565            if let Some(parent) = expand_root.parent_place() {
566                expand_root = parent;
567            } else {
568                return Ok(());
569            }
570        }
571    }

573    pub(crate) fn record_and_apply_action(
574        &mut self,
575        action: PcgAction<'tcx>,
576    ) -> Result<(), PcgError>
577    where
578        Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
579    {
580        tracing::debug!("Applying Action: {}", action.debug_line(self.ctxt));
581        let result = match &action {
582            PcgAction::Borrow(action) => {
583                // An expansion guided by the index local outlives the local's
584                // value: its guide refers to the value at the label instead.
585                if let BorrowPcgActionKind::LabelPlace(label) = &action.kind
586                    && matches!(
587                        label.reason,
588                        LabelPlaceReason::StorageDead
589                            | LabelPlaceReason::Write
590                            | LabelPlaceReason::MoveOut
591                    )
592                    && let Some(index) = label.place.as_local()
593                {
594                    self.pcg.owned.label_index(index, label.location);
595                    self.pcg
596                        .borrow
597                        .graph
598                        .label_index(index, label.location, self.ctxt);
599                }
600                self.pcg
601                    .borrow
602                    .apply_action(action.clone(), self.pcg.place_capabilities, self.ctxt)
603            }
604            PcgAction::Owned(owned_action) => match owned_action.kind {
605                RepackOp::RegainLoanedCapability(regained_capability) => {
606                    self.pcg.place_capabilities.regain_loaned_capability(
607                        regained_capability.place,
608                        regained_capability.capability,
609                        self.pcg.borrow.as_mut_ref(),
610                        self.ctxt,
611                    );
612                    ApplyActionResult::changed_no_display()
613                }
614                RepackOp::Expand(expand) => {
615                    self.pcg.owned.perform_expand_action(
616                        expand,
617                        self.pcg.place_capabilities,
618                        self.ctxt,
619                    );
620                    ApplyActionResult::changed_no_display()
621                }
622                RepackOp::DerefShallowInit(from, to) => {
623                    let target_places = from.expand_one_level(to, self.ctxt)?.expansion();
624                    let capability_projections = self.pcg.owned[from.local].expansions_mut();
625                    capability_projections.insert_expansion(
626                        from,
627                        PlaceExpansion::from_places(target_places.clone(), self.ctxt),
628                    );
629                    for target_place in target_places {
630                        self.pcg.place_capabilities.insert(
631                            target_place,
632                            CapabilityKind::Read,
633                            self.ctxt,
634                        );
635                    }
636                    ApplyActionResult::changed_no_display()
637                }
638                RepackOp::Collapse(collapse) => {
639                    let capability_projections = self.pcg.owned[collapse.local()].expansions_mut();
640                    capability_projections.perform_collapse_action(
641                        collapse,
642                        self.pcg.place_capabilities,
643                        self.ctxt,
644                    );
645                    ApplyActionResult::changed_no_display()
646                }
647                RepackOp::Weaken(weaken) => {
648                    pcg_validity_assert!(
649                        self.pcg
650                            .place_capability_equals(weaken.place, weaken.from, self.ctxt)
651                            || weaken.place.contains_unsafe_deref(self.ctxt)
652                    );
653                    self.pcg
654                        .place_capabilities
655                        .insert(weaken.place, weaken.to, self.ctxt);
656                    ApplyActionResult::changed_no_display()
657                }
658                _ => unreachable!(),
659            },
660        };
661        #[cfg(feature = "visualization")]
662        let location = self.location();
663        #[cfg(feature = "visualization")]
664        let phase = self.phase();
665        if let Some(actions) = &mut self.actions {
666            #[cfg(feature = "visualization")]
667            if let Some(analysis_ctxt) = self.ctxt.try_into_analysis_ctxt()
668                && let Some(phase) = phase
669            {
670                let pcg_ref = PcgRef {
671                    borrow: self.pcg.borrow.as_ref(),
672                    place_capabilities: self.pcg.place_capabilities,
673                    owned: self.pcg.owned,
674                };
675                analysis_ctxt.generate_pcg_debug_visualization_graph(
676                    location,
677                    stmt_graphs::ToGraph::Action(phase, actions.len()),
678                    pcg_ref,
679                );
680            }
681            actions.push(AppliedAction::new(action, result));
682        }
683        Ok(())
684    }
685    pub(crate) fn phase(&self) -> Option<EvalStmtPhase> {
686        match self.prev_snapshot_location {
687            SnapshotLocation::Before(analysis_location) => Some(analysis_location.eval_stmt_phase),
688            _ => None,
689        }
690    }

701/// Whether the place must be collapsed before it can hold `obtain_cap`.
702fn requires_collapse(current_cap: Option<CapabilityKind>, obtain_cap: CapabilityKind) -> bool {
703    current_cap
704        .is_none_or(|cap| matches!(cap.partial_cmp(&obtain_cap), Some(Ordering::Less) | None))
705}

707impl<'state, 'a: 'state, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>>
708    PlaceObtainer<'state, 'a, 'tcx, Ctxt>
709{
710    /// Ensures that the place is expanded to the given place, with a certain
711    /// capability.
712    ///
713    /// This also handles corresponding lifetime projections of the place.
714    #[tracing::instrument(skip(self))]
715    /// Restores [`CapabilityKind::Write`] to the leaves at or below `place`
716    /// that have no capability, once `place` has been labelled for a write.
717    /// The label moves the loans of `place` to its old value, so nothing
718    /// blocks these leaves any more. For an owned `place` they are the owned
719    /// leaves at or below it (one can lack a capability after a join of a
720    /// [`CapabilityKind::Write`] leaf with a read-only expansion, or because
721    /// `place` itself was lent); a borrowed `place` has only borrowed places
722    /// below it, and is restored itself when it is a leaf of the borrow PCG.
723    /// Without this, a lent `place` would regain a capability only when its
724    /// loans end, which can be later: for `let r = &mut *x; let t = &**r;`,
725    /// the reborrow `t` keeps the loan of `*x` in the graph while `t` lives,
726    /// although `r` is dead.
727    fn restore_write_to_unblocked_leaves(&mut self, place: Place<'tcx>) -> Result<(), PcgError> {
728        let ctxt = self.ctxt;
729        if place.as_owned_place(ctxt).is_none() {
730            if self.pcg.capability_of(place, ctxt).is_none()
731                && self.pcg.borrow.graph.leaf_places(ctxt).contains(&place)
732            {
733                self.record_and_apply_action(PcgAction::restore_capability(
734                    place,
735                    CapabilityKind::Write,
736                    "restore write to unblocked borrowed place",
737                    ctxt,
738                ))?;
739            }
740            return Ok(());
741        }
742        if !self.pcg.owned.is_allocated(place.local) {
743            return Ok(());
744        }
745        // The leaves at or below `place` are leaves of its local.
746        let to_restore = self
747            .get_local_expansions(place.local)
748            .leaf_places(self.ctxt)
749            .into_iter()
750            .map(Place::from)
751            .filter(|p| place.is_prefix_of(*p) && self.pcg.capability_of(*p, self.ctxt).is_none())
752            .collect::<Vec<_>>();
753        if to_restore.is_empty() {
754            return Ok(());
755        }
756        let blocked = self.blocked_owned_places(self.ctxt.bc_ctxt());
757        for leaf in to_restore {
758            pcg_validity_assert!(
759                !blocked.contains(&leaf),
760                [ctxt],
761                "{} is blocked after labelling {} for a write",
762                leaf.display_string(ctxt),
763                place.display_string(ctxt)
764            );
765            self.record_and_apply_action(PcgAction::restore_capability(
766                leaf,
767                CapabilityKind::Write,
768                "restore write to unblocked owned leaf",
769                self.ctxt,
770            ))?;
771        }
772        Ok(())
773    }

775    pub(crate) fn obtain(
776        &mut self,
777        place: Place<'tcx>,
778        obtain_type: ObtainType,
779    ) -> Result<(), PcgError> {
780        let obtain_cap = obtain_type.capability(place, self.ctxt);
781
782        self.cut_delegations_through_loans_of(place, obtain_cap)?;
783
784        // Under `PackStrategy::Lazy` the expansions below `place` were not
785        // packed as soon as they became packable, so they are packed here,
786        // where a capability that `place` cannot hold while it is expanded is
787        // about to be required.
788        if self.ctxt.settings().pack_strategy == PackStrategy::Lazy
789            && requires_collapse(self.pcg.capability_of(place, self.ctxt), obtain_cap)
790        {
791            self.pack_expansions_for_obtain_of(place)?;
792        }
793
794        if obtain_cap.is_write() {
795            tracing::debug!(
796                "labeling and removing capabilities for deref projections of postfix places"
797            );
798            self.record_and_apply_action(
799                BorrowPcgAction::label_place_and_update_related_capabilities(
800                    place,
801                    self.prev_snapshot_location(),
802                    LabelPlaceReason::Write,
803                )
804                .into(),
805            )?;
806            self.record_and_apply_action(
807                BorrowPcgAction::label_lifetime_projection(
808                    LabelNodePredicate::And(vec![
809                        LabelNodePredicate::PlaceEquals(place),
810                        LabelNodePredicate::InSourceNodes,
811                        LabelNodePredicate::EdgeType(BorrowPcgEdgeType::BorrowFlow {
812                            future_edge_kind: Some(FutureEdgeKind::FromExpansion),
813                        }),
814                    ]),
815                    None,
816                    "label place and update related capabilities",
817                )
818                .into(),
819            )?;
820            self.restore_write_to_unblocked_leaves(place)?;
821            self.render_debug_graph(None, "after step 1");
822        }
823
824        let current_cap = self.pcg.capability_of(place, self.ctxt);
825        tracing::debug!(
826            "Obtain {:?} to place {} in phase {:?}: Current cap: {:?}, Obtain cap: {:?}",
827            obtain_type,
828            place.display_string(self.ctxt.bc_ctxt()),
829            self.phase(),
830            current_cap,
831            obtain_cap
832        );
833
834        // STEP 2
835        if requires_collapse(current_cap, obtain_cap) {
836            // If we want to get e.g. write permission but we currently have
837            // read permission, we will obtain read with the collapse and then
838            // upgrade in the subsequent step
839            let collapse_cap = if current_cap == Some(CapabilityKind::Read) {
840                CapabilityKind::Read
841            } else {
842                obtain_cap
843            };
844            // When collapsing to write capability (an overwrite or a
845            // StorageDead of e.g. a partially-moved local), leaves that still
846            // hold exclusive capability must be explicitly weakened first, so
847            // that consumers see the capability drop; see
848            // <https://github.com/prusti/pcg/issues/137>.
849            if collapse_cap.is_write() {
850                let to_weaken = self
851                    .pcg
852                    .place_capabilities
853                    .owned_capabilities(place.local, self.ctxt)
854                    .filter_map(|(p, k)| {
855                        (place.is_prefix_of(p) && (k.is_exclusive() || k.is_shallow_exclusive()))
856                            .then_some((p, *k))
857                    })
858                    .collect::<Vec<_>>();
859                for (p, from_cap) in to_weaken {
860                    let weaken = if obtain_type == ObtainType::ForStorageDead {
861                        RepackOp::weaken_for_storage_dead(p, from_cap, CapabilityKind::Write)
862                    } else {
863                        RepackOp::weaken(p, from_cap, CapabilityKind::Write)
864                    };
865                    self.record_and_apply_action(PcgAction::Owned(OwnedPcgAction::new(
866                        weaken, None,
867                    )))?;
868                }
869            }
870            tracing::debug!(
871                "Collapsing owned places to {}",
872                place.display_string(self.ctxt.bc_ctxt())
873            );
874            self.collapse_owned_places_and_lifetime_projections_to(
875                place,
876                collapse_cap,
877                format!("Obtain {}", place.display_string(self.ctxt.bc_ctxt())),
878                self.ctxt,
879            )?;
880            self.render_debug_graph(
881                None,
882                &format!(
883                    "after step 2 (collapse owned places and lifetime projections to {})",
884                    place.display_string(self.ctxt.bc_ctxt())
885                ),
886            );
887        }
888
889        // STEP 3
890        if !obtain_cap.is_read() {
891            tracing::debug!(
892                "Obtain {:?} to place {} in phase {:?}",
893                obtain_type,
894                place.display_string(self.ctxt.bc_ctxt()),
895                self.phase()
896            );
897            // It's possible that we want to obtain exclusive or write permission to
898            // a field that we currently only have read access for. For example,
899            // consider the following case:
900            //
901            // There is an existing shared borrow of (*c).f1
902            // Therefore we have read permission to *c, (*c).f1, and (*c).f2
903            // Then, we want to create a mutable borrow of (*c).f2
904            // This requires obtaining exclusive permission to (*c).f2
905            //
906            // We can upgrade capability of (*c).f2 from R to E by downgrading
907            // all other pre-and postfix places of (*c).f2 (in this case c will
908            // be downgraded to W and *c to None). In the example, (*c).f2 is
909            // actually the closest read ancestor, but this is not always the
910            // case (e.g. if we wanted to obtain (*c).f2.f3 instead)
911            //
912            // This also labels rps and adds placeholder projections
913            self.upgrade_closest_read_ancestor_to_exclusive_and_update_rps(place)?;
914            self.render_debug_graph(None, "after step 3");
915        }
916
917        // STEP 4
918        if obtain_cap.is_write() {
919            // If this place is a reference or contains references, reborrows of
920            // (postfixes of) place may have not yet expired, and therefore the borrowed
921            // caps are still in the PCG.
922            // This will remove them (since we're going to overwrite anyways)
923            self.pcg
924                .place_capabilities
925                .remove_all_strict_postfixes(place, self.ctxt);
926            self.render_debug_graph(None, "after step 4");
927        }
928
929        self.expand_to(place, obtain_type, self.ctxt)?;
930
931        if obtain_type == ObtainType::ForStorageDead
932            && let Some(cap @ (CapabilityKind::Exclusive | CapabilityKind::ShallowExclusive)) =
933                self.pcg.capability_of(place, self.ctxt)
934        {
935            // Temporary: mark as for_storage_dead until
936            // https://github.com/prusti/pcg/issues/137 is resolved.
937            self.record_and_apply_action(PcgAction::Owned(OwnedPcgAction::new(
938                RepackOp::weaken_for_storage_dead(place, cap, CapabilityKind::Write),
939                None,
940            )))?;
941        }
942
943        self.render_debug_graph(None, "after step 5");
944
945        Ok(())
946    }
947}
