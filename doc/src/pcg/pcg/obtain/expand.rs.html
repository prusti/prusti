66    #[tracing::instrument(skip(self, obtain_type, ctxt))]
67    fn expand_to(
68        &mut self,
69        place: Place<'tcx>,
70        obtain_type: ObtainType,
71        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
72    ) -> Result<(), PcgError> {
73        // Obtains at a loop head stand for the whole loop body, which may also
74        // change projections through borrows held in other locals.
75        let changed = if matches!(obtain_type, ObtainType::LoopInvariant { .. }) {
76            None
77        } else {
78            lifetimes_changed_through(place, ctxt)
79        };
80        let changed = changed.as_deref();
81        for (base, _) in place.iter_projections() {
82            let base: Place = base.into();
83            let base = base.with_inherent_region(ctxt);
84            let expansion = base.expand_one_level(place, ctxt)?;
85            if self.expand_place_one_level(base, &expansion, None, obtain_type, ctxt)? {
86                tracing::debug!(
87                    "expand region projections for {} one level",
88                    base.display_string(ctxt.ctxt())
89                );
90                self.expand_lifetime_projections_one_level(
91                    base,
92                    &expansion,
93                    changed,
94                    obtain_type,
95                    ctxt,
96                )?;
97            } else {
98                self.snapshot_unlabelled_expanded_lifetime_projections(
99                    base,
100                    changed,
101                    obtain_type,
102                    ctxt,
103                )?;
104            }
105        }
106        Ok(())
107    }

109    /// `base` is already expanded. Snapshots each of its projections that this
110    /// obtain may change and that is still the unlabelled base of its
111    /// expansion (an earlier obtain that could not change it left it
112    /// unlabelled). The snapshot's future is made of the expansion's parts.
113    fn snapshot_unlabelled_expanded_lifetime_projections(
114        &mut self,
115        base: Place<'tcx>,
116        changed: Option<&[GeneralizedLifetime<'tcx>]>,
117        obtain_type: ObtainType,
118        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
119    ) -> Result<(), PcgError> {
120        let mut to_snapshot = base
121            .lifetime_projections(ctxt)
122            .into_iter()
123            .filter(|rp| {
124                obtain_type.should_label_rp(rp.base, ctxt)
125                    && may_change(rp.region(ctxt.ctxt()), changed)
126            })
127            .map(|rp| (rp, vec![]))
128            .collect::<Vec<_>>();
129        if to_snapshot.is_empty() {
130            return Ok(());
131        }
132        let graph = self.borrows_graph();
133        for edge in graph.edges() {
134            if let BorrowPcgEdgeKind::BorrowPcgExpansion(BorrowPcgExpansion::LifetimeProjection(
135                expansion,
136            )) = edge.kind
137            {
138                for (rp, parts) in &mut to_snapshot {
139                    if expansion.base == (*rp).into() {
140                        // The expansion may hold earlier versions of the parts
141                        // (e.g. one labelled by a write). The snapshot's
142                        // future is made of each part's placeholder if the
143                        // graph has one, and otherwise of its current version.
144                        parts.extend(expansion.expansion().iter().map(|node| {
145                            let current = node
146                                .with_base(MaybeLabelledPlace::Current(node.base.place()))
147                                .with_label(None, ctxt);
148                            let future = current.with_placeholder_label(ctxt);
149                            if graph.contains(future, ctxt) {
150                                future
151                            } else {
152                                current
153                            }
154                            .to_pcg_node(ctxt.bc_ctxt())
155                            .expect_lifetime_projection()
156                        }));
157                    }
158                }
159            }
160        }
161        to_snapshot.retain(|(_, parts)| !parts.is_empty());
162        // Snapshotting a projection of `*base` acts like an exclusive
163        // dereference of `base`: when the snapshot's future expires, `base` gets
164        // its capability back, so it needs the capability such a dereference
165        // leaves it.
166        if base.is_mut_ref(ctxt) && !to_snapshot.is_empty() {
167            self.update_capabilities_for_exclusive_deref_of_expanded(base)?;
168        }
169        for (rp, parts) in to_snapshot {
170            self.snapshot_expanded_lifetime_projection(
171                base,
172                rp,
173                self.prev_snapshot_location(),
174                &parts,
175                ctxt,
176            )?;
177        }
178        Ok(())
179    }

181    /// `rp` is a lifetime projection of a place that is expanded on the way to
182    /// the place being obtained, and `changed` are the lifetimes that the
183    /// obtained capability may change (see [`lifetimes_changed_through`];
184    /// `None` for all of them). If `obtain_type` labels `rp` at all, `rp` gets
185    /// a new label only if the obtain may change it, or if the graph holds its
186    /// placeholder: then `rp` has no current version, and expanding it
187    /// unlabelled would bring one back.
188    fn label_for_rp(
189        &self,
190        rp: LifetimeProjection<'tcx, Place<'tcx>>,
191        changed: Option<&[GeneralizedLifetime<'tcx>]>,
192        obtain_type: ObtainType,
193        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
194    ) -> LabelForLifetimeProjection {
195        use LabelForLifetimeProjection::{
196            ExistingLabelOfTwoPhaseReservation, NewLabelAtCurrentLocation, NoLabel,
197        };
198        if obtain_type.should_label_rp(rp.base, ctxt) {
199            let rp_node: LocalLifetimeProjection<'tcx> = rp.into();
200            if may_change(rp.region(ctxt.ctxt()), changed)
201                || self
202                    .borrows_graph()
203                    .contains(rp_node.with_placeholder_label(ctxt), ctxt)
204            {
205                NewLabelAtCurrentLocation(self.prev_snapshot_location())
206            } else {
207                NoLabel
208            }
209        } else {
210            match self.label_for_shared_expansion_of_rp(rp, ctxt) {
211                Some(label) => ExistingLabelOfTwoPhaseReservation(label),
212                None => NoLabel,
213            }
214        }
215    }

217    /// If the base of `rp` is blocked by a two-phase borrow, we want to use the
218    /// existing label of its expansion
219    fn label_for_shared_expansion_of_rp(
220        &self,
221        rp: LifetimeProjection<'tcx, Place<'tcx>>,
222        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
223    ) -> Option<SnapshotLocation> {
224        ctxt.bc()
225            .borrows_blocking(rp.base, self.location(), ctxt.bc_ctxt())
226            .first()
227            .map(|borrow| {
228                let borrow_reserve_location = get_reserve_location(borrow);
229                SnapshotLocation::after_statement_at(borrow_reserve_location, ctxt)
230            })
231    }

240    fn expand_owned_place_one_level(
241        &mut self,
242        base: OwnedPlace<'tcx>,
243        expansion: &ShallowExpansion<'tcx>,
244        obtain_type: ObtainType,
245        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
246    ) -> Result<bool, PcgError> {
247        tracing::debug!(
248            "New owned expansion from {}",
249            base.display_string(ctxt.ctxt())
250        );
251        if expansion.kind.is_deref_box()
252            && obtain_type
253                .capability(base.place(), ctxt)
254                .is_shallow_exclusive()
255        {
256            self.apply_action(
257                OwnedPcgAction::new(
258                    RepackOp::DerefShallowInit(expansion.base_place(), expansion.target_place),
259                    None,
260                )
261                .into(),
262            )?;
263        } else if !base.is_raw_ptr(ctxt) {
264            self.apply_action(
265                OwnedPcgAction::new(
266                    RepackOp::expand(
267                        expansion.base_place(),
268                        expansion.guide(),
269                        self.capability_for_expand(expansion.base_place(), obtain_type, ctxt),
270                        ctxt,
271                    ),
272                    Some(format!("Expand owned place one level ({obtain_type:?})").into()),
273                )
274                .into(),
275            )?;
276        }
277        Ok(true)
278    }

280    /// Whether the PCG already contains `expansion` of `base`, in which case
281    /// [`Self::expand_place_one_level`] leaves the PCG unchanged.
282    fn contains_expansion(
283        &self,
284        base: Place<'tcx>,
285        expansion: &ShallowExpansion<'tcx>,
286        guide: Option<RepackGuide>,
287        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
288    ) -> Result<bool, PcgError> {
289        match expansion.kind {
290            ProjectionKind::DerefRef(_) => Ok(self
291                .borrows_graph()
292                .contains_deref_edge_to(base.project_deref(ctxt))),
293            ProjectionKind::DerefRawPtr(_) => Ok(false),
294            _ if base.as_owned_place(ctxt).is_some() => {
295                Ok(self.contains_owned_expansion_to(expansion.target_place))
296            }
297            _ => self
298                .borrows_graph()
299                .contains_borrow_pcg_expansion(
300                    &ExpandedPlace {
301                        place: base,
302                        expansion: PlaceExpansion::from_places(expansion.expansion(), ctxt)
303                            .with_guide(guide),
304                    },
305                    ctxt.bc_ctxt(),
306                )
307                .map_err(Into::into),
308        }
309    }

311    fn expand_place_one_level(
312        &mut self,
313        base: Place<'tcx>,
314        expansion: &ShallowExpansion<'tcx>,
315        guide: Option<RepackGuide>,
316        obtain_type: ObtainType,
317        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
318    ) -> Result<bool, PcgError> {
319        if self.contains_expansion(base, expansion, guide, ctxt)? {
320            return Ok(false);
321        }
322        let place_expansion =
323            PlaceExpansion::from_places(expansion.expansion(), ctxt).with_guide(guide);
324        if matches!(expansion.kind, ProjectionKind::DerefRef(_)) {
325            let blocked_lifetime_projection_label =
326                if base.is_mut_ref(ctxt) && obtain_type.should_label_rp(base, ctxt) {
327                    Some(self.prev_snapshot_location())
328                } else {
329                    self.label_for_shared_expansion_of_rp(
330                        base.base_lifetime_projection(ctxt).unwrap(),
331                        ctxt,
332                    )
333                };
334            let deref = DerefEdge::new(base, blocked_lifetime_projection_label, ctxt);
335            self.render_debug_graph(None, "expand_place_one_level: before apply action");
336            let action = BorrowPcgAction::add_edge(
337                BorrowPcgEdge::new(deref.into(), self.path_conditions()),
338                "expand_place_one_level: add deref edge",
339            );
340            self.apply_action(action.into())?;
341            self.render_debug_graph(None, "expand_place_one_level: after apply action");
342            self.update_capabilities_for_deref(
343                base,
344                obtain_type.capability(base, ctxt),
345                ctxt.bc_ctxt(),
346            );
347            self.render_debug_graph(
348                None,
349                "expand_place_one_level: after update_capabilities_for_deref",
350            );
351            if deref.blocked_lifetime_projection.label().is_some() {
352                self.apply_action(
353                    BorrowPcgAction::label_lifetime_projection(
354                        LabelNodePredicate::equals_lifetime_projection(
355                            deref.blocked_lifetime_projection.with_label(None, ctxt),
356                        ),
357                        deref.blocked_lifetime_projection.label(),
358                        "block deref",
359                    )
360                    .into(),
361                )?;
362            }
363            Ok(true)
364        } else if matches!(expansion.kind, ProjectionKind::DerefRawPtr(_)) {
365            let deref = RawPtrDerefEdge::new(base, ctxt);
366            let action = BorrowPcgAction::add_edge(
367                BorrowPcgEdge::new(deref.into(), self.path_conditions()),
368                "expand_place_one_level: add rawptr deref edge",
369            );
370            self.apply_action(action.into())?;
371            self.render_debug_graph(None, "expand_place_one_level: after apply action");
372            let _ = self.update_capabilities_for_deref(
373                base,
374                obtain_type.capability(base, ctxt),
375                ctxt.bc_ctxt(),
376            );
377            Ok(true)
378        } else if let Some(owned_base) = base.as_owned_place(ctxt) {
379            self.expand_owned_place_one_level(owned_base, expansion, obtain_type, ctxt)
380        } else {
381            self.add_borrow_pcg_expansion(base, place_expansion, obtain_type, ctxt)
382        }
383    }

391    fn add_borrow_pcg_expansion(
392        &mut self,
393        base: Place<'tcx>,
394        place_expansion: PlaceExpansion<'tcx>,
395        obtain_type: ObtainType,
396        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
397    ) -> Result<bool, PcgError>
398    where
399        'tcx: 'a,
400    {
401        let expanded_place = ExpandedPlace {
402            place: base,
403            expansion: place_expansion,
404        };
405        tracing::debug!(
406            "Create expansion from {}",
407            base.display_string(ctxt.bc_ctxt())
408        );
409        let block_type = expanded_place.expansion.block_type(base, obtain_type, ctxt);
410        tracing::debug!(
411            "Block type for {} is {:?}",
412            base.display_string(ctxt.bc_ctxt()),
413            block_type
414        );
415        if base.is_raw_ptr(ctxt) {
416            return Ok(true); // We have no aliasing information, PCG should not expand anything
417        }
418        let expansion: BorrowPcgExpansion<'tcx> =
419            BorrowPcgExpansion::new_place_expansion(base, &expanded_place.expansion, ctxt)?;
420
421        self.render_debug_graph(
422            None,
423            &format!(
424                "add_borrow_pcg_expansion: before update_capabilities_for_borrow_expansion {}",
425                expansion.display_string(ctxt.bc_ctxt())
426            ),
427        );
428        self.update_capabilities_for_borrow_expansion(&expansion, block_type, ctxt.bc_ctxt());
429        self.render_debug_graph(
430            None,
431            "add_borrow_pcg_expansion: after update_capabilities_for_borrow_expansion",
432        );
433        let action = BorrowPcgAction::add_edge(
434            BorrowPcgEdge::new(
435                BorrowPcgEdgeKind::BorrowPcgExpansion(expansion),
436                self.path_conditions(),
437            ),
438            "add_borrow_pcg_expansion",
439        );
440        self.apply_action(action.into())?;
441        Ok(true)
442    }

444    #[tracing::instrument(skip(self, base, expansion, ctxt))]
445    fn expand_lifetime_projections_one_level(
446        &mut self,
447        base: Place<'tcx>,
448        expansion: &ShallowExpansion<'tcx>,
449        changed: Option<&[GeneralizedLifetime<'tcx>]>,
450        obtain_type: ObtainType,
451        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
452    ) -> Result<(), PcgError> {
453        for base_rp in base.lifetime_projections(ctxt) {
454            if let Some(place_expansion) =
455                expansion.place_expansion_for_region(base_rp.region(ctxt.ctxt()), ctxt)
456            {
457                tracing::debug!("Expand {}", base_rp.display_string(ctxt.bc_ctxt()));
458                let mut expansion = BorrowPcgExpansion::new_lifetime_projection_expansion(
459                    base_rp,
460                    &place_expansion,
461                    ctxt,
462                )?;
463                let expansion_label = self.label_for_rp(base_rp, changed, obtain_type, ctxt);
464                if let Some(label) = expansion_label.label() {
465                    expansion.label_lifetime_projections(
466                        &LabelNodePredicate::equals_lifetime_projection(base_rp.into()),
467                        Some(label.into()),
468                        ctxt.bc_ctxt(),
469                    );
470                }
471                self.apply_action(
472                    BorrowPcgAction::add_edge(
473                        BorrowPcgEdge::new(
474                            BorrowPcgEdgeKind::BorrowPcgExpansion(expansion.clone()),
475                            self.path_conditions(),
476                        ),
477                        "expand_region_projections_one_level",
478                    )
479                    .into(),
480                )?;
481                if let LabelForLifetimeProjection::NewLabelAtCurrentLocation(label) =
482                    expansion_label
483                {
484                    let expansion_rps = expansion
485                        .expansion()
486                        .iter()
487                        .map(|node| {
488                            node.to_pcg_node(ctxt.bc_ctxt())
489                                .expect_lifetime_projection()
490                        })
491                        .collect::<Vec<_>>();
492                    self.snapshot_expanded_lifetime_projection(
493                        base,
494                        base_rp,
495                        label,
496                        &expansion_rps,
497                        ctxt,
498                    )?;
499                }
500            }
501        }
502        Ok(())
503    }

505    /// Labels `base_rp`, which is expanded into `expansion_rps`, as a snapshot
506    /// at `label`, and, for a borrowed place or a mutable reference, connects
507    /// the snapshot to the parts through placeholder edges.
508    fn snapshot_expanded_lifetime_projection(
509        &mut self,
510        base: Place<'tcx>,
511        base_rp: LifetimeProjection<'tcx, Place<'tcx>>,
512        label: SnapshotLocation,
513        expansion_rps: &[LifetimeProjection<'tcx>],
514        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
515    ) -> Result<(), PcgError> {
516        self.apply_action(
517            BorrowPcgAction::label_lifetime_projection(
518                LabelNodePredicate::equals_lifetime_projection(base_rp.into()),
519                Some(label.into()),
520                "snapshot an expanded lifetime projection",
521            )
522            .into(),
523        )?;
524
525        // Don't add placeholder edges for owned expansions, unless it's a deref
526        if base.as_borrowed_place(ctxt).is_some() || base.is_mut_ref(ctxt) {
527            let old_rp_base = base_rp.with_label(Some(label.into()), ctxt);
528            self.add_and_update_placeholder_edges(
529                old_rp_base.into(),
530                expansion_rps,
531                "obtain",
532                ctxt,
533            )?;
534        }
535        Ok(())
536    }

538    /// Performs bookkeeping for future nodes in the case where capability from
539    /// (generally labelled) `origin_rp` is temporarily transferred to
540    /// (unlabelled) `expansion_rps`, but will eventually be transferred back to
541    /// `origin_rp`.
542    ///
543    /// This happens in the case of borrows and expansions of borrowed places,
544    /// in which case `origin_rp` is labelled.
545    /// We also use this for constructing loop abstractions where `origin_rp`
546    /// is unlabelled (the labels are added in a later stage).
547    ///
548    /// The logic is as follows:
549    ///
550    /// Adds a node `future_rp` which is the future node for `origin_rp`.
551    ///
552    /// 1. Add a Future edge from `origin_rp` to `future_rp`
553    /// 2. Add Future edges from each `expansion_rp` to `future_rp`
554    /// 3. All Future edges with source `origin_rp` (except for the one
555    ///    created in step 1) are modified to now have source `future_rp`
556    ///
557    fn add_and_update_placeholder_edges(
558        &mut self,
559        origin_rp: LocalLifetimeProjection<'tcx>,
560        expansion_rps: &[LifetimeProjection<'tcx>],
561        context: &str,
562        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
563    ) -> Result<(), PcgError> {
564        if expansion_rps.is_empty() {
565            return Ok(());
566        }
567        let future_rp = origin_rp.with_placeholder_label(ctxt);
568
569        // Add edge {origin|r'a at l} -> {origin|r'a at FUTURE}
570        self.apply_action(
571            BorrowPcgAction::add_edge(
572                BorrowPcgEdge::new(
573                    BorrowFlowEdge::new(
574                        origin_rp.into(),
575                        future_rp,
576                        BorrowFlowEdgeKind::Future(FutureEdgeKind::ToFutureSelf),
577                    )
578                    .into(),
579                    self.path_conditions(),
580                ),
581                format!("{context}: placeholder bookkeeping"),
582            )
583            .into(),
584        )?;
585
586        // For each field F add edge {origin.F|'a} -> {origin|r'a at FUTURE}
587        for expansion_rp in expansion_rps {
588            self.apply_action(
589                BorrowPcgAction::add_edge(
590                    BorrowPcgEdge::new(
591                        BorrowFlowEdge::new(
592                            *expansion_rp,
593                            future_rp,
594                            BorrowFlowEdgeKind::Future(FutureEdgeKind::FromExpansion),
595                        )
596                        .into(),
597                        self.path_conditions(),
598                    ),
599                    format!("{context}: placeholder bookkeeping"),
600                )
601                .into(),
602            )?;
603        }
604        self.redirect_source_of_future_edges(origin_rp, future_rp, ctxt)
605    }

607    fn redirect_source_of_future_edges(
608        &mut self,
609        old_source: LocalLifetimeProjection<'tcx>,
610        new_source: LocalLifetimeProjection<'tcx>,
611        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
612    ) -> Result<(), PcgError> {
613        let to_replace = self
614            .borrows_graph()
615            .edges_blocking(old_source.into(), ctxt.bc_ctxt())
616            .filter_map(|edge| {
617                if let BorrowPcgEdgeKind::BorrowFlow(BorrowFlowEdge {
618                    kind: kind @ BorrowFlowEdgeKind::Future(_),
619                    target: short,
620                    ..
621                }) = edge.kind
622                    && *short != new_source
623                {
624                    return Some((
625                        edge.to_owned_edge(),
626                        BorrowPcgEdge::new(
627                            BorrowFlowEdge::new(new_source.into(), *short, *kind).into(),
628                            edge.conditions.clone(),
629                        ),
630                    ));
631                }
632                None
633            })
634            .collect::<Vec<_>>();
635        for (to_remove, to_insert) in to_replace {
636            self.apply_action(
637                BorrowPcgAction::remove_edge(to_remove, "placeholder bookkeeping").into(),
638            )?;
639            self.apply_action(
640                BorrowPcgAction::add_edge(to_insert, "placeholder bookkeeping").into(),
641            )?;
642        }
643        Ok(())
644    }
645}

647/// The lifetimes whose projections a capability to `place` may change: those
648/// in `place`'s type, and those of the references that `place` is reached
649/// through, since writing through a dereference changes what the reference's
650/// lifetime stands for. The other parts of the places that contain `place`
651/// cannot change through it, except in a union, whose fields share storage.
652///
653/// This compares lifetimes by region, which identifies a lifetime of `place`
654/// with that of a place containing it only if each place on the path has only
655/// lifetimes of the place it is a part of. That is not so for a field whose
656/// type contains an associated type that does not normalize to the
657/// container's regions, e.g. `I::Item` for a type parameter `I`. If a place on
658/// the path has another lifetime, or `place` is reached through a union, it
659/// returns `None`: every lifetime may change.
660fn lifetimes_changed_through<'a, 'tcx: 'a>(
661    place: Place<'tcx>,
662    ctxt: impl HasCompilerCtxt<'a, 'tcx>,
663) -> Option<Vec<GeneralizedLifetime<'tcx>>> {
664    let bases = place
665        .iter_projections()
666        .map(|(base, _)| Place::from(base))
667        .collect::<Vec<_>>();
668    if bases.iter().any(|base| base.ty(ctxt).ty.is_union()) {
669        return None;
670    }
671    let mut lifetimes = bases
672        .iter()
673        .chain([&place])
674        .map(|place| {
675            place
676                .lifetime_projections(ctxt)
677                .into_iter()
678                .map(|rp| rp.region(ctxt.ctxt()))
679                .collect::<Vec<_>>()
680        })
681        .collect::<Vec<_>>();
682    if lifetimes
683        .windows(2)
684        .any(|pair| !pair[1].iter().all(|lifetime| pair[0].contains(lifetime)))
685    {
686        return None;
687    }
688    let mut changed = lifetimes.pop().unwrap();
689    let of_dereferenced = place
690        .iter_projections()
691        .filter(|(_, elem)| matches!(elem, mir::ProjectionElem::Deref))
692        .filter_map(|(base, _)| Place::from(base).with_inherent_region(ctxt).ty_region(ctxt))
693        .map(GeneralizedLifetime::Region);
694    changed.extend(of_dereferenced);
695    Some(changed)
696}

698/// Whether an obtain that may change the lifetimes in `changed` (all of them
699/// if `None`) may change the projections with `lifetime`.
700fn may_change<'tcx>(
701    lifetime: GeneralizedLifetime<'tcx>,
702    changed: Option<&[GeneralizedLifetime<'tcx>]>,
703) -> bool {
704    changed.is_none_or(|changed| changed.contains(&lifetime))
705}
