28/// Insert `edge` into `edges`, joining validity conditions if an identical
29/// edge kind is already present. Edge mutations (e.g. unlabelling lifetime
30/// projections) can make two conditional edges identical; their conditions
31/// must be joined, not overwritten.
32fn insert_joining_conditions<EdgeKind: Eq + std::hash::Hash, VC, Ctxt: Copy>(
33    edges: &mut HashMap<EdgeKind, VC>,
34    edge: Conditioned<EdgeKind, VC>,
35    ctxt: Ctxt,
36) where
37    VC: ValidityConditionOps<Ctxt>,
38{
39    match edges.entry(edge.value) {
40        hash_map::Entry::Occupied(mut existing) => {
41            existing.get_mut().join(&edge.conditions, ctxt);
42        }
43        hash_map::Entry::Vacant(slot) => {
44            slot.insert(edge.conditions);
45        }
46    }
47}

49impl<'tcx, EdgeKind> BorrowsGraph<'tcx, EdgeKind> {
50    fn mut_edge_conditions(&mut self, mut f: impl FnMut(&mut ValidityConditions) -> bool) -> bool {
51        let mut changed = false;
52        for conditions in self.edges.values_mut() {
53            if f(conditions) {
54                changed = true;
55            }
56        }
57        changed
58    }
59
60    pub(crate) fn add_path_condition(
61        &mut self,
62        pc: PathCondition,
63        ctxt: CompilerCtxt<'_, 'tcx>,
64    ) -> bool {
65        self.mut_edge_conditions(|conditions| conditions.insert(pc, ctxt.body()))
66    }
67}

69impl<'tcx, EdgeKind, VC> BorrowsGraph<'tcx, EdgeKind, VC> {
70    pub(crate) fn mut_edges<Ctxt: Copy>(
71        &mut self,
72        mut f: impl FnMut(&mut Conditioned<EdgeKind, VC>) -> bool,
73        ctxt: Ctxt,
74    ) -> bool
75    where
76        EdgeKind: Eq + std::hash::Hash + PartialEq,
77        VC: ValidityConditionOps<Ctxt>,
78    {
79        let mut changed = false;
80        let mut new_edges = HashMap::default();
81        for (kind, conditions) in self.edges.drain() {
82            let mut edge = Conditioned::new(kind, conditions);
83            if f(&mut edge) {
84                changed = true;
85            }
86            insert_joining_conditions(&mut new_edges, edge, ctxt);
87        }
88        self.edges = new_edges;
89        changed
90    }
91
92    pub(crate) fn filter_mut_edges<Ctxt: Copy>(
93        &mut self,
94        mut f: impl FnMut(&mut Conditioned<EdgeKind, VC>) -> FilterMutResult,
95        ctxt: Ctxt,
96    ) -> bool
97    where
98        EdgeKind: Eq + std::hash::Hash + PartialEq,
99        VC: ValidityConditionOps<Ctxt>,
100    {
101        let mut changed = false;
102        let mut new_edges = HashMap::default();
103        for (kind, conditions) in self.edges.drain() {
104            let mut edge = Conditioned::new(kind, conditions);
105            match f(&mut edge) {
106                FilterMutResult::Changed => {
107                    changed = true;
108                    insert_joining_conditions(&mut new_edges, edge, ctxt);
109                }
110                FilterMutResult::Unchanged => {
111                    insert_joining_conditions(&mut new_edges, edge, ctxt);
112                }
113                FilterMutResult::Remove => {}
114            }
115        }
116        self.edges = new_edges;
117        changed
118    }
119    pub(crate) fn label_place<P: PcgPlace<'tcx, Ctxt>, Ctxt: Copy>(
120        &mut self,
121        place: P,
122        reason: LabelPlaceReason,
123        labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
124        ctxt: Ctxt,
125    ) -> HashSet<NodeReplacement<'tcx, P>>
126    where
127        EdgeKind: LabelEdgePlaces<'tcx, Ctxt, P> + Eq + std::hash::Hash,
128        VC: ValidityConditionOps<Ctxt>,
129    {
130        let mut all_replacements = HashSet::default();
131        self.mut_edges(
132            |edge| {
133                let replacements = reason.apply_to_edge(place, &mut edge.value, labeller, ctxt);
134                let changed = !replacements.is_empty();
135                all_replacements.extend(replacements);
136                changed
137            },
138            ctxt,
139        );
140        all_replacements
141    }
142}

144impl<'tcx, VC> BorrowsGraph<'tcx, BorrowPcgEdgeKind<'tcx>, VC> {
145    /// Labels the `Index` guides of `index` in the expansion edges; see
146    /// [`crate::owned_pcg::RepackGuide::label_index`].
147    pub(crate) fn label_index<Ctxt: Copy>(
148        &mut self,
149        index: Local,
150        label: SnapshotLocation,
151        ctxt: Ctxt,
152    ) -> bool
153    where
154        VC: ValidityConditionOps<Ctxt>,
155    {
156        let guided_by_index = |kind: &BorrowPcgEdgeKind<'tcx>| {
157            matches!(kind, BorrowPcgEdgeKind::BorrowPcgExpansion(expansion)
158                if expansion.guide().is_some_and(|guide| guide.is_unlabelled_index_of(index)))
159        };
160        if !self.edges.keys().any(guided_by_index) {
161            return false;
162        }
163        self.mut_edges(
164            |edge| match &mut edge.value {
165                BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) => {
166                    expansion.label_index(index, label)
167                }
168                _ => false,
169            },
170            ctxt,
171        )
172    }
